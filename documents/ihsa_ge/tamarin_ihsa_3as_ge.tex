\begin{tarif}[تذكير: القوانين التي نستعملها]
$(x_i;y_i)$ سلسلة إحصائية ذات متغيرين عدد نقطها $n$.
\begin{itemize}
\item \textbf{النقطة المتوسطة:} $G(\bar x;\bar y)$ حيث $\bar x=\dfrac{1}{n}\sum x_i$ و $\bar y=\dfrac{1}{n}\sum y_i$.
\item \textbf{التباين والانحراف المعياري:} $V(x)=\dfrac{1}{n}\sum x_i^{2}-\bar x^{2}$ ، \ $\sigma(x)=\sqrt{V(x)}$ (وبالمثل لـ $y$).
\item \textbf{التغاير:} $\operatorname{cov}(x;y)=\dfrac{1}{n}\sum x_iy_i-\bar x\,\bar y$.
\item \textbf{مستقيم الانحدار بالمربعات الدنيا} $y=ax+b$: \ $a=\dfrac{\operatorname{cov}(x;y)}{V(x)}$ \ و \ $b=\bar y-a\bar x$ (يشمل $G$).
\item \textbf{معامل الارتباط الخطي:} $r=\dfrac{\operatorname{cov}(x;y)}{\sigma(x)\,\sigma(y)}$ مع $-1\le r\le1$. كلما اقترب $|r|$ من $1$ كان الارتباط الخطي قويا (عمليا: $|r|\ge0{,}87$ يبرّر تعديلا خطيا).
\item \textbf{طريقة ماير:} نرتب النقط حسب $x_i$، نقسمها إلى مجموعتين متساويتين في العدد، نحسب النقطتين المتوسطتين $G_1$ و $G_2$؛ المستقيم $(G_1G_2)$ هو مستقيم ماير ويشمل $G$.
\item \textbf{تغيير المتغير:} إذا كان $z=\ln y$ و $z=ax+b$ فإن $y=e^{ax+b}=e^{b}\left(e^{a}\right)^{x}$، أي $y=k\,q^{x}$ مع $k=e^{b}$ و $q=e^{a}$.
\end{itemize}
\end{tarif}

\begin{tamrin}{مبيعات متجر -- تعديل بطريقة ماير}
يعطي الجدول التالي المبيعات الشهرية $y_i$ (بالألف دينار جزائري) لمتجر خلال الأشهر الستة الأولى من سنة 2025، حيث $x_i$ رتبة الشهر ($x_i=1$ يوافق جانفي).
\begin{center}\begin{LTR}\renewcommand{\arraystretch}{1.35}\small
\begin{tabular}{|c|c|c|c|c|c|c|}\hline
\rowcolor{cSec!12}\ar{رتبة الشهر} $x_i$ & $1$ & $2$ & $3$ & $4$ & $5$ & $6$\\\hline
\ar{المبيعات} $y_i$ & $420$ & $470$ & $490$ & $520$ & $590$ & $630$\\\hline
\end{tabular}\end{LTR}\end{center}
\QQ{1} مثّل سحابة النقط $M_i(x_i;y_i)$ في معلم متعامد (على محور الفواصل: $1\,\text{cm}$ لكل شهر؛ وعلى محور التراتيب: $1\,\text{cm}$ لكل $20$ ألف دينار، ونبدأ التدريج من $400$).
\QQ{2} احسب إحداثيي النقطة المتوسطة $G$ لهذه السلسلة ثم علّمها.
\QQ{3} هل يبرّر شكل السحابة إجراء تعديل خطي؟
\QQ{4} نقسم السلسلة إلى مجموعتين: الأولى تضم النقط الثلاث الأولى، والثانية تضم النقط الثلاث الأخيرة.
\begin{sq}
\qa احسب إحداثيي $G_1$ و $G_2$ النقطتين المتوسطتين للمجموعتين.
\qb عيّن معادلة مستقيم ماير $(G_1G_2)$ وتحقق أن $G$ تنتمي إليه، ثم ارسمه.
\end{sq}
\QQ{5} باستعمال هذا التعديل:
\begin{sq}
\qa قدّر مبيعات شهر سبتمبر 2025.
\qb ابتداء من أي شهر من سنة 2025 تتجاوز المبيعات $800$ ألف دينار (إذا استمر الاتجاه نفسه)؟
\end{sq}
\end{tamrin}
\begin{hal}
\QQ{1}{\bfseries و 2 و 4.} سحابة النقط (بالأزرق)، النقطة المتوسطة $G$ (بالأحمر)، النقطتان $G_1$ و $G_2$ (بالبرتقالي) ومستقيم ماير:
\begin{SC}{xmin=0,xmax=7.4,ymin=400,ymax=660,xtick={0,...,7},ytick={400,440,...,640},xlabel={$x_i$},ylabel={$y_i$}}
\addplot[ln,domain=0:7.2]{40*x+380};
\addplot[pt] coordinates {(1,420)(2,470)(3,490)(4,520)(5,590)(6,630)};
\addplot[gq] coordinates {(2,460)(5,580)};
\addplot[gp] coordinates {(3.5,520)};
\node[above left] at (axis cs:3.5,520){$G$};
\node[below right] at (axis cs:2,460){$G_1$};
\node[below right] at (axis cs:5,580){$G_2$};
\node[cThm,above left] at (axis cs:6.9,656){$(G_1G_2)$};
\end{SC}
\QQ{2} عدد النقط $n=6$:
\[\bar x=\frac{1+2+3+4+5+6}{6}=\frac{21}{6}=3{,}5\]\[\bar y=\frac{420+470+490+520+590+630}{6}=\frac{3120}{6}=520\]
إذن \bx{G(3{,}5\,;\,520)}

\QQ{3} نقط السحابة متقاربة من مستقيم (تتزايد المبيعات بانتظام تقريبا)، إذن يمكن إجراء \textbf{تعديل خطي}.

\QQ{4}\pa المجموعة الأولى: النقط ذات الفواصل $1$ ، $2$ ، $3$:
\[x_{G_1}=\frac{1+2+3}{3}=2\qquad y_{G_1}=\frac{420+470+490}{3}=\frac{1380}{3}=460\qquad\text{إذن}\qquad \bx{G_1(2\,;\,460)}\]
المجموعة الثانية: النقط ذات الفواصل $4$ ، $5$ ، $6$:
\[x_{G_2}=\frac{4+5+6}{3}=5\qquad y_{G_2}=\frac{520+590+630}{3}=\frac{1740}{3}=580\qquad\text{إذن}\qquad \bx{G_2(5\,;\,580)}\]
\pb معادلة $(G_1G_2)$ من الشكل $y=ax+b$ حيث المعامل الموجه:
\[a=\frac{y_{G_2}-y_{G_1}}{x_{G_2}-x_{G_1}}=\frac{580-460}{5-2}=\frac{120}{3}=40\]
و $G_1\in(G_1G_2)$ ومنه $460=40\times2+b$ أي $b=380$. إذن معادلة مستقيم ماير: \bx{y=40x+380}

التحقق: $40\times3{,}5+380=140+380=520=\bar y$، إذن \bx{G\in(G_1G_2)} \ok

\textbf{تفسير:} $a=40$ يعني أن المبيعات تزداد في المتوسط بـ $40$ ألف دينار كل شهر.

\QQ{5}\pa شهر سبتمبر رتبته $x=9$: \ $y=40\times9+380=740$. تقدَّر مبيعات سبتمبر بـ \bxt{$740$ ألف دينار} تقريبا.
\pb نحل المتراجحة: $40x+380>800$ \ أي \ $40x>420$ \ أي \ $x>10{,}5$. أصغر رتبة طبيعية هي $x=11$، إذن ابتداء من \bxt{شهر نوفمبر 2025} تتجاوز المبيعات $800$ ألف دينار.
\end{hal}

\begin{tamrin}{السعر والطلب -- مستقيم ماير والإيراد الأعظمي}
درست مؤسسة الطلب اليومي $y_i$ (عدد الوحدات المطلوبة) على منتوج بدلالة سعر الوحدة $x_i$ (بالدينار) فتحصلت على:

\end{tabular}\end{LTR}\end{center}
\QQ{1} مثّل سحابة النقط $M_i(x_i;y_i)$ (على محور الفواصل: $1\,\text{cm}$ لكل $10$ دنانير؛ وعلى محور التراتيب: $1\,\text{cm}$ لكل $20$ وحدة ابتداء من $200$).
\QQ{2} احسب إحداثيي النقطة المتوسطة $G$.
\QQ{3} باستعمال طريقة ماير (مجموعة أولى: النقط الثلاث الأولى) بيّن أن معادلة مستقيم التعديل هي: $y=-3{,}5x+486$.
\QQ{4} فسّر اقتصاديا المعامل $-3{,}5$.
\QQ{5}
\begin{sq}
\qa قدّر الطلب اليومي إذا كان سعر الوحدة $80$ دينارا.
\qb ما السعر الذي يوافق طلبا يوميا قدره $300$ وحدة؟ (دوّر إلى $10^{-2}$).
\end{sq}
\QQ{6} نعتبر أن الطلب معطى بـ $y=-3{,}5x+486$ من أجل $x\in[20;100]$. الإيراد اليومي هو $R(x)=x\cdot y$.
