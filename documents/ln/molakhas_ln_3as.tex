\section{التعريف والخواص الأساسية}
\begin{tarif}
الدالة الأسية مستمرة ومتزايدة تمامًا على $\R$، وتأخذ قيمها في $]0;+\infty[$؛ إذن من أجل كل عدد حقيقي $x>0$ توجد قيمة وحيدة $y$ تحقق $e^{y}=x$.\par
نسمي \textbf{الدالة اللوغاريتمية النيبيرية} الدالة التي ترفق بكل عدد حقيقي $x$ من المجال $]0;+\infty[$ العدد الحقيقي $y$ الذي يحقق $e^{y}=x$، ونرمز لها بالرمز $\ln$:
\[ y=\ln x \iff x=e^{y}\qquad (x>0,\ y\in\R) \]
\end{tarif}

\begin{khassia}[نتائج مباشرة]
\begin{itemize}
\item الدالة $\ln$ معرفة على $]0;+\infty[$ فقط.
\item $\ln 1=0$ و $\ln e=1$.
\item من أجل كل $x>0$: $e^{\ln x}=x$، ومن أجل كل $y\in\R$: $\ln\left(e^{y}\right)=y$.
\item الدالتان $\ln$ و $\exp$ متعاكستان: منحنياهما متناظران بالنسبة إلى المستقيم $y=x$.
\end{itemize}
\end{khassia}

\begin{tariqa}[تعيين مجموعة التعريف]
الدالة $x\mapsto \ln\big(u(x)\big)$ معرفة من أجل كل $x$ يحقق $u(x)>0$؛ والدالة $x\mapsto \ln|u(x)|$ معرفة من أجل كل $x$ يحقق $u(x)\neq 0$.
\end{tariqa}

\begin{mithal}
\begin{itemize}
\item $f(x)=\ln(3x-12)$: يجب $3x-12>0$ أي $x>4$، إذن \bx{D_f=]4;+\infty[}.
\item $g(x)=\ln\left(18-2x^{2}\right)$: يجب $18-2x^{2}>0$ أي $x^{2}<9$، إذن \bx{D_g=]-3;3[}.
\item $h(x)=\ln|x-2|$: يجب $x\neq 2$، إذن \bx{D_h=\R-\{2\}}.
\end{itemize}
\end{mithal}

\section{الخواص الجبرية}
\begin{khassia}
من أجل كل عددين حقيقيين موجبين تمامًا $a$ و $b$، ومن أجل كل عدد صحيح نسبي $n$:
\begin{align*}
\ln(ab)&=\ln a+\ln b & \ln\left(\frac{1}{a}\right)&=-\ln a & \ln\left(\frac{a}{b}\right)&=\ln a-\ln b\\
\ln\left(a^{n}\right)&=n\ln a & \ln\sqrt{a}&=\frac{1}{2}\ln a & \ln\left(e^{n}\right)&=n
\end{align*}
\end{khassia}

\begin{molahada}
من أجل كل عدد حقيقي $a\neq0$: $\ln\left(a^{2}\right)=2\ln|a|$ وليس $2\ln a$ (لأن $a$ قد يكون سالبًا).
\end{molahada}

\begin{mithal}[تبسيط عبارات]
\begin{itemize}
\item $A=\ln 12-\ln 3+2\ln\frac{1}{2}=\ln\frac{12}{3}-2\ln 2=\ln 4-2\ln 2=2\ln2-2\ln2=\bx{0}$.
\item $B=\ln\left(e^{2}\right)+\ln\frac{1}{e}-\ln\sqrt{e}=2-1-\frac{1}{2}=\bx{\frac{1}{2}}$.
\end{itemize}
\end{mithal}

\section{المعادلات والمتراجحات}
\begin{khassia}
من أجل كل عددين حقيقيين موجبين تمامًا $a$ و $b$:
\[ \ln a=\ln b \iff a=b \qquad\qquad \ln a<\ln b \iff a<b \]
وبالخصوص: $\ln x=0\iff x=1$ ، $\ln x>0\iff x>1$ ، $\ln x<0\iff 0<x<1$.
\end{khassia}

\begin{tariqa}[حل معادلة أو متراجحة]
\begin{enumerate}
\item نعيّن مجموعة الحلول الممكنة $D$ (شروط وجود اللوغاريتمات).
\item نكتب الطرفين على الشكل $\ln a=\ln b$ (أو $\ln a<\ln b$) باستعمال الخواص الجبرية.
\item نحل المعادلة (أو المتراجحة) $a=b$ (أو $a<b$).
\item نحتفظ بالحلول التي تنتمي إلى $D$.
\end{enumerate}
\end{tariqa}

\begin{mithal}[معادلات]
\begin{itemize}
\item $\ln(2x-1)=\ln(x+3)$: الشرط $x>\frac{1}{2}$؛ و $2x-1=x+3$ يعطي $x=4\in D$، إذن \bx{S=\{4\}}.
\item $\ln x+\ln(x-2)=\ln 3$: الشرط $x>2$؛ المعادلة تكافئ $\ln\big(x(x-2)\big)=\ln 3$ أي $x^{2}-2x-3=0$، حلاها $3$ و $-1$، ونرفض $-1$ لأنه لا ينتمي إلى $D$. إذن \bx{S=\{3\}}.
\item $(\ln x)^{2}-\ln x-2=0$: نضع $t=\ln x$ فنجد $t^{2}-t-2=0$ أي $t=2$ أو $t=-1$، ومنه $x=e^{2}$ أو $x=e^{-1}$. إذن \bx{S=\left\{e^{-1};e^{2}\right\}}.
\item $e^{2x}-3e^{x}+2=0$: نضع $X=e^{x}>0$ فنجد $X^{2}-3X+2=0$ أي $X=1$ أو $X=2$، ومنه $x=0$ أو $x=\ln 2$. إذن \bx{S=\{0;\ln2\}}.
\end{itemize}
\end{mithal}

\begin{mithal}[متراجحات]
\begin{itemize}
\item $\ln(x-1)<0$: الشرط $x>1$؛ و $\ln(x-1)<\ln 1$ يكافئ $x-1<1$ أي $x<2$. إذن \bx{S=]1;2[}.
\item $\ln(2x+1)\geqslant\ln(x+4)$: الشرط $x>-\frac{1}{2}$؛ و $2x+1\geqslant x+4$ يكافئ $x\geqslant 3$. إذن \bx{S=[3;+\infty[}.
\item $e^{x}>3$ تكافئ $x>\ln 3$، إذن \bx{S=]\ln3;+\infty[}.
\end{itemize}
\end{mithal}

\section{النهايات والنهايات الشهيرة}
\begin{khassia}[نهايات أساسية]
\[ \lim_{x\to0^{+}}\ln x=-\infty \qquad\qquad \lim_{x\to+\infty}\ln x=+\infty \]
\end{khassia}

\begin{khassia}[النهايات الشهيرة]
\begin{align*}
\lim_{x\to+\infty}\frac{\ln x}{x}&=0 & \lim_{x\to0^{+}}x\ln x&=0 & \lim_{x\to0}\frac{\ln(1+x)}{x}&=1\\
\lim_{x\to+\infty}\frac{\ln x}{x^{n}}&=0 & \lim_{x\to0^{+}}x^{n}\ln x&=0 & \lim_{x\to1}\frac{\ln x}{x-1}&=1
\end{align*}
($n$ عدد طبيعي غير معدوم). نقول: عند $+\infty$ «$x^{n}$ يتغلب على $\ln x$».
\end{khassia}

\begin{mithal}
\begin{itemize}
\item $\lim\limits_{x\to+\infty}(x-\ln x)$: حالة عدم تعيين $\infty-\infty$. نكتب $x-\ln x=x\left(1-\frac{\ln x}{x}\right)$، ولدينا $\frac{\ln x}{x}\to0$، إذن النهاية \bx{+\infty}.
\item $\lim\limits_{x\to0^{+}}(x-1)\ln x$: لدينا $x-1\to-1$ و $\ln x\to-\infty$، إذن النهاية \bx{+\infty}.
\item $\lim\limits_{x\to0}\frac{\ln(1+2x)}{x}=\lim\limits_{x\to0}2\times\frac{\ln(1+2x)}{2x}=2\times1=\bx{2}$ (نضع $X=2x\to0$).
\item $\lim\limits_{x\to+\infty}\big(\ln(x+1)-\ln x\big)=\lim\limits_{x\to+\infty}\ln\frac{x+1}{x}=\ln1=\bx{0}$.
\end{itemize}
\end{mithal}

\section{الاشتقاق والدالة $\ln u$}
\begin{khassia}
\begin{itemize}
\item الدالة $\ln$ قابلة للاشتقاق على $]0;+\infty[$ و $\left(\ln x\right)'=\dfrac{1}{x}$.
\item إذا كانت $u$ قابلة للاشتقاق وموجبة تمامًا على مجال $I$ فإن: $\big(\ln u\big)'=\dfrac{u'}{u}$.
\item إذا كانت $u$ قابلة للاشتقاق ولا تنعدم على $I$ فإن: $\big(\ln|u|\big)'=\dfrac{u'}{u}$.
\end{itemize}
\end{khassia}

\begin{molahada}
للدالتين $\ln u$ و $u$ نفس اتجاه التغير على $I$ (لأن $\frac{u'}{u}$ من إشارة $u'$ عندما $u>0$).
\end{molahada}

\begin{mithal}
\begin{itemize}
\item $f(x)=\ln\left(x^{2}+1\right)$ على $\R$: $f'(x)=\bx{\dfrac{2x}{x^{2}+1}}$.
\item $g(x)=x\ln x-x$ على $]0;+\infty[$: $g'(x)=\ln x+x\times\frac{1}{x}-1=\bx{\ln x}$.
\item $h(x)=\ln\dfrac{x-1}{x+1}$ على $]1;+\infty[$: $h'(x)=\dfrac{\frac{2}{(x+1)^{2}}}{\frac{x-1}{x+1}}=\bx{\dfrac{2}{x^{2}-1}}$.
\item $k(x)=(\ln x)^{2}$ على $]0;+\infty[$: $k'(x)=2\times\frac{1}{x}\times\ln x=\bx{\dfrac{2\ln x}{x}}$.
\end{itemize}
\end{mithal}

\section{دراسة الدالة $\ln$ وتمثيلها البياني}
\begin{khassia}[جدول التغيرات]
من أجل كل $x>0$: $\left(\ln x\right)'=\frac{1}{x}>0$، إذن الدالة $\ln$ متزايدة تمامًا على $]0;+\infty[$.
\end{khassia}

\begin{khassia}[المماس عند $1$ ووضعية المنحنى]
معادلة المماس للمنحنى عند النقطة ذات الفاصلة $1$: $y=x-1$.\par
نضع $\varphi(x)=\ln x-x+1$، فيكون $\varphi'(x)=\frac{1}{x}-1=\frac{1-x}{x}$: الدالة $\varphi$ متزايدة على $]0;1]$ ومتناقصة على $[1;+\infty[$، وقيمتها الحدية العظمى $\varphi(1)=0$. إذن:
\[ \text{من أجل كل } x>0:\qquad \bx{\ln x\leqslant x-1} \]
أي أن المنحنى يقع تحت مماسه عند $1$.
\end{khassia}

\section{اللوغاريتم العشري}
\begin{tarif}
نسمي \textbf{اللوغاريتم العشري} الدالة المعرفة على $]0;+\infty[$ بـ: $\log x=\dfrac{\ln x}{\ln 10}$.
\end{tarif}
\begin{khassia}
للدالة $\log$ نفس الخواص الجبرية لـ $\ln$، و $\log 10=1$، ومن أجل كل $n\in\Z$: $\log\left(10^{n}\right)=n$.\par
$\log x=a\iff x=10^{a}$.
\end{khassia}
\begin{mithal}
\begin{itemize}
\item $\log 1000=\bx{3}$ و $\log 0{,}01=\bx{-2}$.
\item $\log x=2\iff x=\bx{100}$.
\item في الكيمياء: $\text{pH}=-\log\left[\text{H}_{3}\text{O}^{+}\right]$؛ إذا كان $\left[\text{H}_{3}\text{O}^{+}\right]=10^{-3}$ فإن $\text{pH}=\bx{3}$.
\end{itemize}
\end{mithal}

\section{تمرين محلول: دراسة دالة كاملة}
\begin{tmr}
\textbf{I.} نعتبر الدالة $g$ المعرفة على $]0;+\infty[$ بـ: $g(x)=x^{2}-1+\ln x$.
\begin{sq}
\qa ادرس اتجاه تغير الدالة $g$.
\qb احسب $g(1)$ ثم استنتج إشارة $g(x)$.
\end{sq}
\textbf{II.} نعتبر الدالة $f$ المعرفة على $]0;+\infty[$ بـ: $f(x)=x-1-\dfrac{\ln x}{x}$، و $(C_f)$ تمثيلها البياني.
\begin{sq}
\qa احسب $\lim\limits_{x\to0^{+}}f(x)$ و $\lim\limits_{x\to+\infty}f(x)$.
\qb بيّن أنه من أجل كل $x>0$: $f'(x)=\dfrac{g(x)}{x^{2}}$، ثم شكّل جدول تغيرات $f$.
\qc بيّن أن المستقيم $(\Delta)$ ذو المعادلة $y=x-1$ مقارب مائل لـ $(C_f)$، وادرس الوضع النسبي.
\end{sq}
\end{tmr}

\begin{hal}
\textbf{I.}
\pa $g'(x)=2x+\frac{1}{x}$، وهو موجب تمامًا على $]0;+\infty[$، إذن \bxt{$g$ متزايدة تمامًا على $]0;+\infty[$}.
\pb $g(1)=1-1+\ln1=0$. وبما أن $g$ متزايدة تمامًا:

\textbf{II.}
\pa عند $0^{+}$: $\ln x\to-\infty$ و $\frac{1}{x}\to+\infty$ إذن $\frac{\ln x}{x}\to-\infty$، ومنه $\lim\limits_{x\to0^{+}}f(x)=\bx{+\infty}$.\par
عند $+\infty$: $\frac{\ln x}{x}\to0$ و $x-1\to+\infty$، إذن $\lim\limits_{x\to+\infty}f(x)=\bx{+\infty}$.
\pb $\left(\frac{\ln x}{x}\right)'=\frac{\frac{1}{x}\times x-\ln x}{x^{2}}=\frac{1-\ln x}{x^{2}}$، ومنه:
\[ f'(x)=1-\frac{1-\ln x}{x^{2}}=\frac{x^{2}-1+\ln x}{x^{2}}=\frac{g(x)}{x^{2}} \]
إشارة $f'(x)$ هي إشارة $g(x)$، و $f(1)=1-1-0=0$:

\pc $f(x)-(x-1)=-\dfrac{\ln x}{x}$ و $\lim\limits_{x\to+\infty}\left(-\dfrac{\ln x}{x}\right)=0$، إذن \bx{y=x-1} معادلة مقارب مائل لـ $(C_f)$ بجوار $+\infty$.\par
إشارة $f(x)-(x-1)$ هي إشارة $-\ln x$ (لأن $x>0$):
\begin{itemize}
\item على $]0;1[$: $-\ln x>0$، إذن $(C_f)$ فوق $(\Delta)$.
\item على $]1;+\infty[$: $-\ln x<0$، إذن $(C_f)$ تحت $(\Delta)$.
\item $(C_f)$ و $(\Delta)$ يتقاطعان في النقطة $A(1;0)$.
\end{itemize}
