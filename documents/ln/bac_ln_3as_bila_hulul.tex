\section{تمارين البكالوريا -- شعبة علوم تجريبية}

% ===================== التمرين 1 =====================
\begin{tamrin}{بكالوريا 2009 -- الموضوع الأول}
\textbf{I)} $h$ دالة عددية معرفة على $]-1;+\infty[$ كما يلي: $h(x)=x^2+2x+\ln(x+1)$.
\QQ{1} أحسب $\lim\limits_{x\to -1^{+}}h(x)$ و $\lim\limits_{x\to +\infty}h(x)$.
\QQ{2} بيّن أنه من أجل كل عدد حقيقي $x$ من المجال $]-1;+\infty[$: $h'(x)=\dfrac{1+2(x+1)^2}{x+1}$، واستنتج اتجاه تغيّر الدالة $h$ ثم أنجز جدول تغيّراتها.
\QQ{3} أحسب $h(0)$ واستنتج إشارة $h(x)$ حسب قيم $x$.

\textbf{II)} لتكن $f$ دالة عددية معرفة على المجال $]-1;+\infty[$ كما يلي: $f(x)=x-1-\dfrac{\ln(x+1)}{x+1}$.
نرمز بـ $(C_f)$ لتمثيلها البياني في المستوي المنسوب إلى المعلم المتعامد والمتجانس $(O;\vec i,\vec j)$.
\QQ{1}
\begin{sq}
\qa أحسب $\lim\limits_{x\to -1^{+}}f(x)$، ثم فسّر هذه النتيجة بيانيا.
\qb باستخدام النتيجة $\lim\limits_{t\to +\infty}\dfrac{e^{t}}{t}=+\infty$، برهن أن $\lim\limits_{u\to +\infty}\dfrac{\ln u}{u}=0$.
\qc استنتج $\lim\limits_{x\to +\infty}f(x)$.
\qd أحسب $\lim\limits_{x\to +\infty}\big[f(x)-(x-1)\big]$ واستنتج وجود مستقيم مقارب مائل للمنحنى $(C_f)$.
\qe أدرس وضعية المنحنى $(C_f)$ بالنسبة إلى المستقيم المقارب المائل.
\end{sq}
\QQ{2} بيّن أنه من أجل كل عدد حقيقي $x$ من المجال $]-1;+\infty[$: $f'(x)=\dfrac{h(x)}{(x+1)^2}$، ثم شكّل جدول تغيّرات الدالة $f$.
\QQ{3} بيّن أن المنحنى $(C_f)$ يقطع المستقيم ذا المعادلة $y=2$ عند النقطة التي فاصلتها محصورة بين $3{,}3$ و $3{,}4$.
\QQ{4} أرسم $(C_f)$.
\end{tamrin}

% ===================== التمرين 2 =====================
\begin{tamrin}{بكالوريا 2010 -- الموضوع الأول}
\textbf{I)} $f$ الدالة العددية المعرفة على المجال $I=\left]\frac12;+\infty\right[$ بـ: $f(x)=1+\ln(2x-1)$.
وليكن $(C_f)$ تمثيلها البياني في المستوي المنسوب إلى المعلم المتعامد والمتجانس $(O;\vec i,\vec j)$.
\QQ{1} أحسب $\lim\limits_{x\to +\infty}f(x)$ و $\lim\limits_{x\to \frac12^{+}}f(x)$.
\QQ{2} بيّن أن الدالة $f$ متزايدة تماما على المجال $I$، ثم شكّل جدول تغيّراتها.
\QQ{3} عيّن فاصلة النقطة من $(C_f)$ التي تكون فيها المماس موازيا للمستقيم $(d)$ ذي المعادلة $y=x$.
\QQ{4}
\begin{sq}
\qa أثبت أنه من أجل كل $x$ من $I$ يمكن كتابة $f(x)$ على الشكل: $f(x)=\ln(x+a)+b$ حيث $a$، $b$ عددان حقيقيان يطلب تعيينهما.
\qb استنتج أنه يمكن رسم $(C_f)$ انطلاقا من $(C)$ منحنى الدالة اللوغاريتمية النيبيرية $\ln$، ثم أرسم $(C)$ و $(C_f)$.
\end{sq}

\textbf{II)} نعتبر الدالة العددية $g$ المعرفة على المجال $I$ بـ: $g(x)=f(x)-x$.
\QQ{1} أحسب $\lim\limits_{x\to \frac12^{+}}g(x)$ ثم بيّن أن $\lim\limits_{x\to +\infty}g(x)=-\infty$.
\QQ{2} أدرس اتجاه تغيّر الدالة $g$ على $I$ ثم شكّل جدول تغيّراتها.
\QQ{3}
\begin{sq}
\qa أحسب $g(1)$ ثم بيّن أن المعادلة $g(x)=0$ تقبل في المجال $\left]\frac32;+\infty\right[$ حلا وحيدا $\alpha$. تحقق أن $2<\alpha<3$.
\qb أرسم $(C_g)$ منحنى الدالة $g$ على المجال $\left]\frac12;5\right]$ في المعلم السابق.
\qc استنتج إشارة $g(x)$ على المجال $I$ ثم حدّد وضعية المنحنى $(C_f)$ بالنسبة إلى $(d)$.
\end{sq}
\QQ{4} برهن أنه من أجل كل عدد حقيقي $x$ من المجال $]1;\alpha[$ فإن: $f(x)$ ينتمي إلى المجال $]1;\alpha[$.

\textbf{III)} نسمي $(u_n)$ المتتالية العددية المعرفة على $\N^{*}$ كما يأتي: $u_n=f\left(1+\dfrac{1}{2n}\right)$.
\QQ{1} عيّن قيمة العدد الطبيعي $n$ التي من أجلها يكون: $u_n=1+2\ln 3-3\ln 2$.
\QQ{2} أحسب بدلالة $n$ المجموع $S_n$ حيث: $S_n=u_1+u_2+\cdots+u_n$.
\end{tamrin}

% ===================== التمرين 3 =====================
\begin{tamrin}{بكالوريا 2011 -- الموضوع الأول}
\textbf{I)} نعتبر الدالة $g$ المعرفة على $\R-\{-1\}$ بـ: $g(x)=\dfrac{x-1}{x+1}$،
و $(C_g)$ تمثيلها البياني في المستوي المنسوب إلى المعلم المتعامد والمتجانس $(O;\vec i,\vec j)$ (الشكل المقابل).
\begin{GR}{-5}{5}{-3}{4}{0.6}
 \Ticks{1}{1}
 \draw[gray,dashed] (-5,1)--(5,1);
 \draw[gray,dashed] (-1,-3)--(-1,4);
 \draw[cDef,very thick,domain=-5:-1.45,samples=100,smooth] plot (\x,{(\x-1)/(\x+1)});
 \draw[cDef,very thick,domain=-0.5:5,samples=100,smooth] plot (\x,{(\x-1)/(\x+1)});
 \node[cDef] at (-3.6,3.2){$(C_g)$};
\end{GR}
\textbf{بقراءة بيانية:}
\begin{sq}
\qa شكّل جدول تغيّرات الدالة $g$.
\qb حلّ بيانيا المتراجحة $g(x)>0$.
\qc عيّن بيانيا قيم $x$ التي يكون من أجلها $0<g(x)<1$.
\end{sq}

\textbf{II)} لتكن الدالة $f$ المعرفة على المجال $]1;+\infty[$ بـ: $f(x)=\dfrac{x-1}{x+1}+\ln\left(\dfrac{x-1}{x+1}\right)$،
و $(C_f)$ تمثيلها البياني في المستوي المنسوب إلى المعلم المتعامد والمتجانس $(O;\vec i,\vec j)$.
\QQ{1} أحسب $\lim\limits_{x\to 1^{+}}f(x)$ و $\lim\limits_{x\to +\infty}f(x)$ ثم فسّر النتيجتين هندسيا.
\QQ{2}
\begin{sq}
\qa بيّن أنه من أجل كل عدد حقيقي $x$ من المجال $]1;+\infty[$: $g'(x)=\dfrac{2}{(x+1)^2}$.
\qb أحسب $f'(x)$ وأدرس إشارتها ثم شكّل جدول تغيّرات الدالة $f$.
\end{sq}
\QQ{3}
\begin{sq}
\qa باستعمال الجزء (I) السؤال (جـ)، عيّن إشارة العبارة $\ln\left(\dfrac{x-1}{x+1}\right)$ على المجال $]1;+\infty[$.
\qb $\alpha$ عدد حقيقي. بيّن أن الدالة $x\mapsto (x-\alpha)\ln(x-\alpha)-x$ هي دالة أصلية للدالة $x\mapsto \ln(x-\alpha)$ على المجال $]\alpha;+\infty[$.
\qc تحقق أنه من أجل كل عدد حقيقي $x$ من المجال $]1;+\infty[$: $g(x)=1-\dfrac{2}{x+1}$، ثم عيّن دالة أصلية للدالة $f$ على المجال $]1;+\infty[$.
\end{sq}
\end{tamrin}

\begin{tamrin}{بكالوريا 2012 -- الموضوع الأول}
نعتبر الدالة $f$ المعرفة على المجال $]-\infty;0[$ كما يلي: $f(x)=x+5+6\ln\left(\dfrac{x}{x-1}\right)$.
$(C_f)$ تمثيلها البياني في المستوي المنسوب إلى المعلم المتعامد والمتجانس $(O;\vec i,\vec j)$.
\QQ{1}
\begin{sq}
\qa أحسب $\lim\limits_{x\to 0^{-}}f(x)$، ثم فسّر النتيجة هندسيا.
\qb أحسب $\lim\limits_{x\to -\infty}f(x)$.
\end{sq}
\QQ{2} بيّن أنه من أجل كل $x$ من $]-\infty;0[$: $f'(x)=\dfrac{x^2-x-6}{x(x-1)}$.
استنتج اتجاه تغيّر الدالة $f$، ثم شكّل جدول تغيّراتها.
\QQ{3}
\begin{sq}
\qa بيّن أن المستقيم $(\Delta)$ الذي معادلة له: $y=x+5$ هو مستقيم مقارب مائل للمنحنى $(C_f)$ عند $-\infty$.
\qb أدرس وضع المنحنى $(C_f)$ بالنسبة للمستقيم $(\Delta)$.
\end{sq}
\QQ{4} بيّن أن المعادلة $f(x)=0$ تقبل حلين $\alpha$ و $\beta$ حيث $-3{,}5<\alpha<-3{,}4$ و $-1{,}1<\beta<-1$.
\QQ{5} أنشئ المنحنى $(C_f)$ والمستقيم $(\Delta)$.
\QQ{6}
\begin{sq}
\qa نعتبر النقطتين $A\left(-1;3+6\ln\frac34\right)$ و $B\left(-2;\frac52+6\ln\frac34\right)$.
بيّن أن $y=\dfrac12x+\dfrac72+6\ln\dfrac34$ معادلة ديكارتية للمستقيم $(AB)$.
\qb بيّن أن المستقيم $(AB)$ يمس المنحنى $(C_f)$ في نقطة $M_0$ يطلب تعيين إحداثيتيها.
\end{sq}
\end{tamrin}
