% ================= تمارين تطبيقية قصيرة: الدالة الأسية =================
\begin{tamrin}{خواص جبرية: التحقق من مساويات}
تحقق أنه من أجل كل عدد حقيقي $x$:
\begin{sq}
\qa $\dfrac{e^{x}}{x+e^{x}}=\dfrac{1}{1+xe^{-x}}$
\qb $\dfrac{4}{1+e^{-x}}-1=\dfrac{3e^{x}-1}{1+e^{x}}$
\qc $\dfrac{e^{-x}+2}{3x+e^{x}}\times\dfrac{6xe^{x}+2e^{2x}}{1+2e^{x}}=2$
\qd $\dfrac{1-e^{-2x}}{1+e^{-2x}}=\dfrac{e^{2x}-1}{e^{2x}+1}$
\qe $e^{-x}-e^{-2x}=\dfrac{e^{x}-1}{e^{2x}}$
\item[(و)] $\left(e^{x}+e^{-x}\right)^{2}=\dfrac{e^{4x}+1}{e^{2x}}+2$
\item[(ز)] $\dfrac{e^{x}-e^{-x}}{e^{x}+e^{-x}}=\dfrac{e^{2x}-1}{e^{2x}+1}$
\end{sq}
\end{tamrin}

\begin{tamrin}{تبسيط عبارات أسية}
اكتب على أبسط شكل ممكن ما يلي (حيث $x$ عدد حقيقي):
\begin{sq}
\qa $A=\dfrac{e^{4x}}{e^{x}\left(e^{-x}\right)^{5}}$
\qb $B=\dfrac{\left(e^{x-1}\right)^{2}\times e^{4x-1}}{\left(e^{2x}\right)^{3}\times e^{-x+3}}$
\qc $C=\left(e^{2x}-3\right)^{2}-e^{2x}\left(e^{2x}+9e^{-2x}\right)$
\qd $D=\dfrac{1+4e^{-x}}{1+e^{-x}}-\dfrac{3}{e^{x}+1}$
\end{sq}
\end{tamrin}

% ------------------------------------------------------------------
\begin{tamrin}{مركز تناظر منحنى}
$f$ دالة عددية معرفة على $\R$ كما يلي: $f(x)=\dfrac{e^{x}}{1+e^{x}}$، و $(C_f)$ تمثيلها البياني.
\QQ{1} تحقق أنه من أجل كل $x$ من $\R$: $f(-x)=\dfrac{1}{1+e^{x}}$.
\QQ{2} بيّن أنه من أجل كل $x$ من $\R$: $f(x)+f(-x)=1$، ثم فسّر النتيجة هندسيا.
\end{tamrin}

% ------------------------------------------------------------------
\begin{tamrin}{كتابة عبارة على شكل آخر}
\QQ{1} عيّن الأعداد الحقيقية $a$ و $b$ و $c$ بحيث من أجل كل $x$ من $\R$: \ $\dfrac{e^{2x}}{e^{x}+2}=ae^{x}+b+\dfrac{c}{e^{x}+2}$.
\QQ{2} عيّن العددين الحقيقيين $\alpha$ و $\beta$ بحيث من أجل كل $x$ من $\R$: \ $\dfrac{e^{x}-1}{3e^{x}+1}=\alpha+\dfrac{\beta}{3+e^{-x}}$.
\end{tamrin}

% ------------------------------------------------------------------
\begin{tamrin}{حل معادلات أسية}
حل في $\R$ المعادلات التالية:
\begin{sq}
\qa $e^{4x}+3=0$
\qb $e^{-2x+10}=1$
\qc $e^{5x-3}=\dfrac{1}{e^{11-3x}}$
\qd $e^{x^{2}}=\dfrac{\left(e^{x}\right)^{5}}{\left(e^{2}\right)^{3}}$
\qe $e^{2x}+6e^{x}-7=0$
\item[(و)] $7e^{x}-5-2e^{-x}=0$
\item[(ز)] $-2e^{x-4}+7e^{-x+4}-5=0$
\item[(ح)] $e^{\frac{x^{2}}{x-2}}-\left[e^{\frac{x+1}{3(x-2)}}\right]^{4}=0$
\end{sq}
\end{tamrin}

% ------------------------------------------------------------------
\begin{tamrin}{حل جمل معادلات}
حل في $\R\times\R$ الجمل التالية:
\begin{sq}
\qa $\left\{\begin{array}{l}\left(e^{x}\right)^{2}\cdot e^{y-1}=e^{9}\\[4pt] \dfrac{e^{x-2}}{e^{y}}=1\end{array}\right.$
\qb $\left\{\begin{array}{l}e^{4x}\times e^{y}=\dfrac{1}{e^{2}}\\[4pt] x\times y=-2\end{array}\right.$
\qc $\left\{\begin{array}{l}e^{x}\cdot e^{y}=e^{3}\\[2pt] e^{x-3}\times\left(e^{y}\right)^{2}=e^{2}\end{array}\right.$
\end{sq}
\end{tamrin}

% ------------------------------------------------------------------
\begin{tamrin}{حل متراجحات أسية}

\begin{sq}
\qa $3e^{x}+5>0$
\qb $e^{x+2}+4\leqslant0$
\qc $-3e^{-4x+8}+3\geqslant0$
\qd $e^{x^{2}}\geqslant\dfrac{\left(e^{x}\right)^{2}}{e^{-3}}$
\qe $e^{x^{2}}\leqslant\left[e^{2x-3}\right]^{3}$
\item[(و)] $2e^{2(x-3)}+5e^{x-3}-7\leqslant0$
\item[(ز)] $e^{2x}-e^{x}+4-4e^{-x}\leqslant0$
\item[(ح)] $(-x+3)\left(2e^{x}-2\right)\geqslant0$
\end{sq}
\end{tamrin}

\begin{tamrin}{دراسة دالة أسية ومناقشة بيانية -- شعبة رياضيات}
\QQ{1} $g$ دالة معرفة على $\R$ كما يلي: $g(x)=e^{x}-x+2$.
\begin{sq}
\qa ادرس تغيرات الدالة $g$.
\qb استنتج إشارة $g(x)$ على $\R$.
\end{sq}
\QQ{2} نعتبر الدالة $f$ المعرفة على $\R$ كما يلي: $f(x)=e^{-x}(x-1)+x+1$، وليكن $(C_f)$ المنحنى الممثل للدالة $f$ في المستوي المنسوب إلى المعلم المتعامد والمتجانس $\left(O;\vec i,\vec j\right)$.
\begin{sq}
\qa احسب نهايتي الدالة $f$ عند $+\infty$ و $-\infty$.
\qb \textbf{i.} بيّن أنه من أجل كل $x$ من $\R$: $f'(x)=\dfrac{g(x)}{e^{x}}$. \quad \textbf{ii.} استنتج اتجاه تغير الدالة $f$، ثم شكّل جدول تغيراتها.
\qc \textbf{i.} بيّن أن المستقيم $(\Delta)$ ذا المعادلة $y=x+1$ مقارب مائل لـ $(C_f)$ عند $+\infty$. \quad \textbf{ii.} ادرس الوضع النسبي لـ $(C_f)$ مع $(\Delta)$.
\qd بيّن أن المنحنى $(C_f)$ يقبل نقطة انعطاف $\omega$ يطلب تعيين إحداثييها.
\qe ليكن $(T_\alpha)$ مستقيما معادلته $y=x+\alpha$ حيث $\alpha\in\R$. عيّن قيمة $\alpha$ حتى يكون $(T_\alpha)$ مماسا لـ $(C_f)$.
\item[(و)] احسب $f(0)$، ثم أنشئ المنحنى $(C_f)$.
\item[(ز)] نعتبر المعادلة: $(1-m)e^{x}+x-1=0\ \ldots(I)$ حيث $m$ وسيط حقيقي. عيّن قيم العدد الحقيقي $m$ بحيث تقبل المعادلة $(I)$ حلين متمايزين.
\end{sq}
\end{tamrin}

\qa ادرس تغيرات الدالة $g$.
\qb بيّن أن المعادلة $g(x)=0$ تقبل حلا وحيدا $\alpha$ من المجال $]1{,}27;1{,}28[$.
\qc استنتج إشارة $g(x)$ على $\R$.
\end{sq}
\QQ{2} نعتبر الدالة $f$ المعرفة على $\R$ بـ: $f(x)=\dfrac{x}{e^{x}+1}+2$، وليكن $(C_f)$ التمثيل البياني الممثل للدالة $f$ في معلم متعامد ومتجانس $\left(O;\vec i,\vec j\right)$.
\begin{sq}
