\section{التعريف والعدد $e$}
\begin{tarif}[الدالة الأسية]
توجد دالة \textbf{وحيدة} $f$ قابلة للاشتقاق على $\R$ تحقق:
\[ \bx{f'=f}\qquad\qquad \bx{f(0)=1} \]
تسمى \textbf{الدالة الأسية النيبيرية} ويرمز لها بالرمز $\exp$. إذن: $\exp(0)=1$ ومن أجل كل $x\in\R$: $\exp'(x)=\exp(x)$.
\end{tarif}

\begin{khassia}[الدالة الأسية لا تنعدم]
من أجل كل عدد حقيقي $x$: \quad $\exp(x)\times\exp(-x)=1$ ، \ إذن \bx{\exp(x)\neq0}.
\par\smallskip
\textbf{البرهان:} نضع $\varphi(x)=\exp(x)\exp(-x)$، فيكون $\varphi'(x)=\exp(x)\exp(-x)-\exp(x)\exp(-x)=0$، إذن $\varphi$ ثابتة و $\varphi(x)=\varphi(0)=1$.
\end{khassia}

\begin{tarif}[العدد $e$ والترميز $e^{x}$]
\begin{itemize}
\item العدد $e$ هو صورة العدد $1$ بالدالة الأسية: \ \bx{e=\exp(1)\approx2{,}718}.
\item من أجل كل عدد صحيح نسبي $n$: $\exp(n)=\left[\exp(1)\right]^{n}=e^{n}$.
\item اصطلاحًا نرمز، من أجل كل عدد حقيقي $x$، إلى $\exp(x)$ بالرمز \bx{e^{x}} ونقرأ «$e$ أس $x$».
\end{itemize}
\end{tarif}

\section{الخواص الجبرية}
\begin{khassia}
من أجل كل عددين حقيقيين $x$ و $y$ ومن أجل كل عدد صحيح نسبي $n$:
\begin{align*}
e^{0}&=1 & e^{1}&=e & e^{x}&>0\\
e^{x+y}&=e^{x}\times e^{y} & e^{-x}&=\frac{1}{e^{x}} & e^{x-y}&=\frac{e^{x}}{e^{y}}\\
e^{nx}&=\left(e^{x}\right)^{n} & e^{\frac{x}{2}}&=\sqrt{e^{x}} & e^{x}\times e^{x}&=e^{2x}=\left(e^{x}\right)^{2}
\end{align*}
\end{khassia}

\begin{mithal}[تبسيط عبارات]
من أجل كل عدد حقيقي $x$:
\begin{itemize}
\item $\dfrac{e^{4x}}{e^{x}\left(e^{-x}\right)^{5}}=\dfrac{e^{4x}}{e^{x}\,e^{-5x}}=\dfrac{e^{4x}}{e^{-4x}}=e^{4x+4x}=\bx{e^{8x}}$.
\item $\dfrac{\left(e^{x-1}\right)^{2}\times e^{4x-1}}{\left(e^{2x}\right)^{3}\times e^{-x+3}}=\dfrac{e^{2x-2+4x-1}}{e^{6x-x+3}}=\dfrac{e^{6x-3}}{e^{5x+3}}=\bx{e^{x-6}}$.
\item $\left(e^{2x}-3\right)^{2}-e^{2x}\left(e^{2x}+9e^{-2x}\right)=e^{4x}-6e^{2x}+9-e^{4x}-9=\bx{-6e^{2x}}$.
\item $\dfrac{1-e^{-2x}}{1+e^{-2x}}=\dfrac{e^{2x}\left(1-e^{-2x}\right)}{e^{2x}\left(1+e^{-2x}\right)}=\bx{\dfrac{e^{2x}-1}{e^{2x}+1}}$ \quad (ضربنا البسط والمقام في $e^{2x}$).
\item $\left(e^{x}+e^{-x}\right)^{2}=e^{2x}+2e^{x}e^{-x}+e^{-2x}=\bx{\dfrac{e^{4x}+1}{e^{2x}}+2}$.
\end{itemize}
\end{mithal}

\begin{mithal}[التحقق من مساواة]
تحقق أنه من أجل كل $x\in\R$:
\begin{itemize}
\item $\dfrac{e^{x}}{x+e^{x}}=\dfrac{1}{1+xe^{-x}}$: \ نقسم البسط والمقام على $e^{x}\neq0$: \ $\dfrac{e^{x}}{x+e^{x}}=\dfrac{1}{\frac{x}{e^{x}}+1}=\bx{\dfrac{1}{1+xe^{-x}}}$.
\item $\dfrac{4}{1+e^{-x}}-1=\dfrac{3e^{x}-1}{e^{x}+1}$: \ نضرب البسط والمقام في $e^{x}$: \ $\dfrac{4e^{x}}{e^{x}+1}-1=\dfrac{4e^{x}-\left(e^{x}+1\right)}{e^{x}+1}=\bx{\dfrac{3e^{x}-1}{e^{x}+1}}$.
\item $\dfrac{e^{-x}+2}{3x+e^{x}}\times\dfrac{6xe^{x}+2e^{2x}}{1+2e^{x}}=2$: \ نحلل $6xe^{x}+2e^{2x}=2e^{x}\left(3x+e^{x}\right)$، ومنه:
\[ \frac{e^{-x}+2}{3x+e^{x}}\times\frac{2e^{x}\left(3x+e^{x}\right)}{1+2e^{x}}=\frac{\left(e^{-x}+2\right)e^{x}\times2}{1+2e^{x}}=\frac{\left(1+2e^{x}\right)\times2}{1+2e^{x}}=\bx{2} \]
\item $e^{-x}-e^{-2x}=\dfrac{1}{e^{x}}-\dfrac{1}{e^{2x}}=\dfrac{e^{x}-1}{e^{2x}}$ \quad (توحيد المقامات على $e^{2x}$).
\item $\dfrac{e^{x}-e^{-x}}{e^{x}+e^{-x}}=\dfrac{e^{x}\left(e^{x}-e^{-x}\right)}{e^{x}\left(e^{x}+e^{-x}\right)}=\bx{\dfrac{e^{2x}-1}{e^{2x}+1}}$.
\end{itemize}
\end{mithal}

\begin{tariqa}[إثبات مساواة أو تحويل كتابة]
\begin{itemize}
\item نضرب البسط والمقام في $e^{x}$ (أو $e^{-x}$) للتخلص من الأسس السالبة أو لإظهارها.
\item نخرج $e^{x}$ عاملًا مشتركًا، ونستعمل $e^{x}\times e^{-x}=1$.
\item \textbf{المطابقة:} نوحّد المقامات في الطرف المقترح ثم نطابق معاملات $e^{2x}$ ، $e^{x}$ والحد الثابت.
\end{itemize}
\end{tariqa}

\begin{mithal}[مطابقة]
عيّن الأعداد الحقيقية $a$ ، $b$ ، $c$ بحيث من أجل كل $x\in\R$: \ $\dfrac{e^{2x}}{e^{x}+2}=ae^{x}+b+\dfrac{c}{e^{x}+2}$.
\par\smallskip
\[ ae^{x}+b+\frac{c}{e^{x}+2}=\frac{(ae^{x}+b)(e^{x}+2)+c}{e^{x}+2}=\frac{ae^{2x}+(2a+b)e^{x}+2b+c}{e^{x}+2} \]
بالمطابقة: $a=1$ ، $2a+b=0$ ، $2b+c=0$؛ إذن $a=1$ ، $b=-2$ ، $c=4$ \ و\ \bx{\dfrac{e^{2x}}{e^{x}+2}=e^{x}-2+\dfrac{4}{e^{x}+2}}.
\par\medskip
عيّن العددين الحقيقيين $\alpha$ و $\beta$ بحيث من أجل كل $x\in\R$: \ $\dfrac{e^{x}-1}{3e^{x}+1}=\alpha+\dfrac{\beta}{3+e^{-x}}$.
\par\smallskip
نوحّد المقامات ثم نضرب البسط والمقام في $e^{x}$:
\[ \alpha+\frac{\beta}{3+e^{-x}}=\frac{3\alpha+\beta+\alpha e^{-x}}{3+e^{-x}}=\frac{(3\alpha+\beta)e^{x}+\alpha}{3e^{x}+1} \]
بالمطابقة مع $\dfrac{e^{x}-1}{3e^{x}+1}$: $3\alpha+\beta=1$ و $\alpha=-1$، إذن $\alpha=-1$ ، $\beta=4$ \ و\ \bx{\dfrac{e^{x}-1}{3e^{x}+1}=-1+\dfrac{4}{3+e^{-x}}}.
\end{mithal}

\begin{mithal}[مركز تناظر]
$f(x)=\dfrac{e^{x}}{1+e^{x}}$ على $\R$. \ $f(-x)=\dfrac{e^{-x}}{1+e^{-x}}=\dfrac{e^{-x}\times e^{x}}{(1+e^{-x})\times e^{x}}=\dfrac{1}{e^{x}+1}$، ومنه:
\[ f(x)+f(-x)=\frac{e^{x}}{1+e^{x}}+\frac{1}{1+e^{x}}=\frac{e^{x}+1}{e^{x}+1}=1=2\times\frac{1}{2} \]
إذن \bxt{النقطة $\Omega\left(0;\frac{1}{2}\right)$ مركز تناظر للمنحنى $(C_f)$}.
\end{mithal}

\section{دراسة الدالة الأسية}
\begin{khassia}[الإشارة واتجاه التغير]
\begin{itemize}
\item من أجل كل $x\in\R$: \bx{e^{x}>0} \quad (لأن $e^{x}=\left(e^{\frac{x}{2}}\right)^{2}\geqslant0$ و $e^{x}\neq0$).
\item $\left(e^{x}\right)'=e^{x}>0$، إذن \bxt{الدالة $x\mapsto e^{x}$ متزايدة تمامًا على $\R$}.
\end{itemize}
\end{khassia}

\begin{khassia}[نتائج]
من أجل كل عددين حقيقيين $a$ و $b$، ومن أجل كل $x\in\R$:
\begin{center}
$e^{a}=e^{b}\iff a=b$ \qquad $e^{a}<e^{b}\iff a<b$ \qquad $e^{a}>e^{b}\iff a>b$\\[4pt]
$e^{x}=1\iff x=0$ \qquad $e^{x}>1\iff x>0$ \qquad $0<e^{x}<1\iff x<0$
\end{center}
\end{khassia}

\begin{khassia}[النهايات عند حدود $\R$]
\[ \lim_{x\to-\infty}e^{x}=0 \qquad\qquad \lim_{x\to+\infty}e^{x}=+\infty \]
التفسير البياني: المستقيم ذو المعادلة $y=0$ (محور الفواصل) \textbf{مقارب أفقي} للمنحنى بجوار $-\infty$.
\end{khassia}

\section{المعادلات والمتراجحات}
\begin{tariqa}
\begin{itemize}
\item $e^{u(x)}=e^{v(x)}\iff u(x)=v(x)$ \quad و\quad $e^{u(x)}<e^{v(x)}\iff u(x)<v(x)$.
\item $e^{u(x)}=k$: \ إذا كان $k\leqslant0$ فلا حلول (لأن $e^{u(x)}>0$)؛ وإذا كان $k>0$ فإن $u(x)=\ln k$.
\item $ae^{2x}+be^{x}+c=0$: نضع $X=e^{x}$ \textbf{مع} $X>0$، نحل $aX^{2}+bX+c=0$، ونرفض الحلول $X\leqslant0$.
\item إذا ظهر $e^{-x}$ نضرب طرفي المعادلة في $e^{x}>0$.
\end{itemize}
\end{tariqa}

\begin{mithal}[معادلات]
حل في $\R$:
\begin{itemize}
\item $e^{4x}+3=0$ تكافئ $e^{4x}=-3$، وهذا مستحيل لأن $e^{4x}>0$؛ إذن \bx{S=\varnothing}.
\item $e^{-2x+10}=1$ تكافئ $e^{-2x+10}=e^{0}$ أي $-2x+10=0$، إذن \bx{S=\{5\}}.
\item $e^{5x-3}=\dfrac{1}{e^{11-3x}}=e^{3x-11}$ تكافئ $5x-3=3x-11$ أي $x=-4$، إذن \bx{S=\{-4\}}.
\item $e^{x^{2}}=\dfrac{\left(e^{x}\right)^{5}}{\left(e^{2}\right)^{3}}=e^{5x-6}$ تكافئ $x^{2}-5x+6=0$: \ $\Delta=1$، الحلان $2$ و $3$؛ إذن \bx{S=\{2;3\}}.
\item $e^{2x-1}=3$ تكافئ $2x-1=\ln3$، إذن \bx{S=\left\{\dfrac{1+\ln3}{2}\right\}}.
\item $e^{2x}+6e^{x}-7=0$: نضع $X=e^{x}>0$: $X^{2}+6X-7=0$، $\Delta=64$، $X=1$ أو $X=-7$ (مرفوض)؛ $e^{x}=1$ أي $x=0$؛ إذن \bx{S=\{0\}}.
\item $7e^{x}-5-2e^{-x}=0$: نضرب في $e^{x}>0$: $7e^{2x}-5e^{x}-2=0$؛ $X=e^{x}$: $7X^{2}-5X-2=0$، $\Delta=81$، $X=1$ أو $X=-\frac{2}{7}$ (مرفوض)؛ إذن \bx{S=\{0\}}.
\item $-2e^{x-4}+7e^{-x+4}-5=0$: نضرب في $e^{x-4}>0$: $-2e^{2(x-4)}-5e^{x-4}+7=0$. نضع $t=e^{x-4}>0$: $-2t^{2}-5t+7=0$، $\Delta=81$، $t=1$ أو $t=-\frac{7}{2}$ (مرفوض)؛ $e^{x-4}=1$ أي $x=4$؛ إذن \bx{S=\{4\}}.
\item $e^{\frac{x^{2}}{x-2}}-\left[e^{\frac{x+1}{3(x-2)}}\right]^{4}=0$ \ (الشرط $x\neq2$): تكافئ $\dfrac{x^{2}}{x-2}=\dfrac{4x+4}{3(x-2)}$ أي $3x^{2}-4x-4=0$، $\Delta=64$، $x=2$ (مرفوض) أو $x=-\frac{2}{3}$؛ إذن \bx{S=\left\{-\frac{2}{3}\right\}}.
\end{itemize}
\end{mithal}

\section{النهايات الشهيرة وحالات عدم التعيين}
\begin{khassia}[التزايد المقارن والنهايات الشهيرة]
من أجل كل عدد طبيعي غير معدوم $n$:
\begin{center}
\renewcommand{\arraystretch}{2.1}
\begin{tabular}{|c|c|}\hline
\rowcolor{cSec!12} \ar{\textbf{الحالة الخاصة ($n=1$ أو $a=1$)}} & \ar{\textbf{الحالة العامة}}\\\hline
$\displaystyle\lim_{x\to+\infty}\frac{e^{x}}{x}=+\infty$ & $\displaystyle\lim_{x\to+\infty}\frac{e^{x}}{x^{n}}=+\infty$\\\hline
$\displaystyle\lim_{x\to+\infty}\frac{x}{e^{x}}=\lim_{x\to+\infty}xe^{-x}=0$ & $\displaystyle\lim_{x\to+\infty}\frac{x^{n}}{e^{x}}=\lim_{x\to+\infty}x^{n}e^{-x}=0$\\\hline
$\displaystyle\lim_{x\to-\infty}xe^{x}=0$ & $\displaystyle\lim_{x\to-\infty}x^{n}e^{x}=0$\\\hline
$\displaystyle\lim_{x\to0}\frac{e^{x}-1}{x}=1$ & $\displaystyle\lim_{x\to0}\frac{e^{ax}-1}{x}=a$\\\hline
\end{tabular}
\end{center}
وعمومًا: إذا كان $\lim u=0$ فإن $\lim\dfrac{e^{u}-1}{u}=1$. \quad نقول: عند $+\infty$ «$e^{x}$ يتغلب على $x^{n}$».
\end{khassia}

\begin{molahada}[برهان $\lim\limits_{x\to+\infty}\frac{e^{x}}{x}=+\infty$ و $\lim\limits_{x\to0}\frac{e^{x}-1}{x}=1$]
\begin{itemize}
\item نضع $\psi(x)=e^{x}-\frac{x^{2}}{2}$ على $[0;+\infty[$: $\psi'(x)=e^{x}-x\geqslant x+1-x=1>0$، إذن $\psi$ متزايدة و $\psi(x)\geqslant\psi(0)=1>0$، ومنه $e^{x}>\frac{x^{2}}{2}$، أي $\frac{e^{x}}{x}>\frac{x}{2}$ من أجل $x>0$. وبما أن $\frac{x}{2}\to+\infty$ فإن $\frac{e^{x}}{x}\to+\infty$ (بالمقارنة).
\item $\dfrac{e^{x}-1}{x}=\dfrac{e^{x}-e^{0}}{x-0}$ هو معدل التغير، ونهايته العدد المشتق للدالة الأسية عند $0$: $e^{0}=1$.
\end{itemize}
\end{molahada}

\begin{tariqa}[إزالة حالات عدم التعيين]

\begin{mithal}[الحالة $\infty-\infty$]
\begin{itemize}
\item $\lim\limits_{x\to+\infty}\left(e^{x}-x+1\right)=\lim\limits_{x\to+\infty}x\left(\dfrac{e^{x}}{x}-1+\dfrac{1}{x}\right)=\bx{+\infty}$ \ لأن $\dfrac{e^{x}}{x}\to+\infty$.
\item $\lim\limits_{x\to-\infty}\left(e^{-x}+x\right)=\lim\limits_{x\to-\infty}e^{-x}\left(1+xe^{x}\right)=\bx{+\infty}$ \ لأن $xe^{x}\to0$ و $e^{-x}\to+\infty$.
\item $\lim\limits_{x\to+\infty}\left(x-e^{x}\right)=\lim\limits_{x\to+\infty}x\left(1-\dfrac{e^{x}}{x}\right)=\bx{-\infty}$.
\end{itemize}
\end{mithal}

\begin{mithal}[الحالة $\frac{\infty}{\infty}$]
\begin{itemize}
\item $\lim\limits_{x\to+\infty}\dfrac{e^{x}-2}{x+3}=\lim\limits_{x\to+\infty}\dfrac{\frac{e^{x}}{x}-\frac{2}{x}}{1+\frac{3}{x}}=\bx{+\infty}$.
\item $\lim\limits_{x\to+\infty}\dfrac{e^{x}-2x}{2-2x}=\lim\limits_{x\to+\infty}\dfrac{\frac{e^{x}}{x}-2}{\frac{2}{x}-2}=\bx{-\infty}$ \quad (البسط $\to+\infty$ والمقام $\to-2$).
\item $\lim\limits_{x\to-\infty}\dfrac{e^{x}-2x}{2-2x}=\lim\limits_{x\to-\infty}\dfrac{\frac{e^{x}}{x}-2}{\frac{2}{x}-2}=\dfrac{0-2}{0-2}=\bx{1}$ \quad (هنا $e^{x}\to0$: ليست حالة التزايد المقارن).
\item $\lim\limits_{x\to+\infty}\dfrac{e^{x}-x}{2e^{x}+x}=\lim\limits_{x\to+\infty}\dfrac{e^{x}\left(1-\frac{x}{e^{x}}\right)}{e^{x}\left(2+\frac{x}{e^{x}}\right)}=\bx{\dfrac{1}{2}}$.
\item $\lim\limits_{x\to+\infty}\dfrac{x^{2}}{e^{x}-1}=\lim\limits_{x\to+\infty}\dfrac{1}{\frac{e^{x}}{x^{2}}-\frac{1}{x^{2}}}=\bx{0}$.
\item $\lim\limits_{x\to+\infty}\dfrac{e^{x}-1}{x}=\lim\limits_{x\to+\infty}\left(\dfrac{e^{x}}{x}-\dfrac{1}{x}\right)=\bx{+\infty}$ \quad (تفريق الكسر).
\item $\lim\limits_{x\to+\infty}\dfrac{x-1}{e^{x}-1}=\lim\limits_{x\to+\infty}\dfrac{x}{e^{x}}\times\dfrac{1-\frac{1}{x}}{1-\frac{1}{e^{x}}}=0\times1=\bx{0}$.
\end{itemize}
\end{mithal}

\begin{mithal}[الحالتان $0\times\infty$ و $\frac{0}{0}$]
\begin{itemize}
\item $\lim\limits_{x\to-\infty}(x-3)e^{x}=\lim\limits_{x\to-\infty}\left(xe^{x}-3e^{x}\right)=0-0=\bx{0}$.
\item $\lim\limits_{x\to-\infty}(1+2x)e^{3x}=\lim\limits_{x\to-\infty}\left(e^{3x}+\frac{2}{3}\times3xe^{3x}\right)=\bx{0}$ \quad (نضع $t=3x\to-\infty$ و $te^{t}\to0$).
\item $\lim\limits_{x\to0}\dfrac{e^{2x}-1}{3x}=\lim\limits_{x\to0}\dfrac{2}{3}\times\dfrac{e^{2x}-1}{2x}=\dfrac{2}{3}\times1=\bx{\dfrac{2}{3}}$.
\item $\lim\limits_{x\to0}\dfrac{e^{2x}-e^{x}}{x}=\lim\limits_{x\to0}e^{x}\times\dfrac{e^{x}-1}{x}=1\times1=\bx{1}$.
\item $\lim\limits_{x\to0}\dfrac{e^{x}-1}{\sin x}=\lim\limits_{x\to0}\dfrac{\frac{e^{x}-1}{x}}{\frac{\sin x}{x}}=\dfrac{1}{1}=\bx{1}$.
\end{itemize}
\end{mithal}

\section{الدالة $x\mapsto e^{u(x)}$}
\begin{khassia}[الاشتقاق]
إذا كانت $u$ قابلة للاشتقاق على مجال $I$ فإن الدالة $e^{u}$ قابلة للاشتقاق على $I$ و:
\[ \bx{\left(e^{u}\right)'=u'\,e^{u}} \]
وبما أن $e^{u}>0$ فإن \bxt{للدالتين $e^{u}$ و $u$ نفس اتجاه التغير على $I$}.
\end{khassia}

\begin{khassia}[النهايات (نهاية دالة مركبة)]
إذا كان $\lim\limits_{x\to x_0}u(x)=\ell$ فإن $\lim\limits_{x\to x_0}e^{u(x)}=e^{\ell}$؛ \ وإذا كان $\lim u=+\infty$ فإن $\lim e^{u}=+\infty$؛ \ وإذا كان $\lim u=-\infty$ فإن $\lim e^{u}=0$.
\end{khassia}

\begin{mithal}[نهايات]
\begin{itemize}
\item $\lim\limits_{x\to0^{+}}e^{\frac{1}{x}}=\bx{+\infty}$ (لأن $\frac{1}{x}\to+\infty$) \quad و\quad $\lim\limits_{x\to0^{-}}e^{\frac{1}{x}}=\bx{0}$ (لأن $\frac{1}{x}\to-\infty$).
\item $\lim\limits_{x\to+\infty}e^{-x^{2}}=\bx{0}$ \quad و\quad $\lim\limits_{x\to+\infty}e^{\frac{x+1}{x-2}}=e^{1}=\bx{e}$ (لأن $\frac{x+1}{x-2}\to1$).
\end{itemize}
\end{mithal}

\begin{mithal}[مشتقات]

\section{المعادلات التفاضلية}
\begin{tarif}
$a$ و $b$ عددان حقيقيان مع $a\neq0$. حل المعادلة التفاضلية \bx{y'=ay+b} على $\R$ هو تعيين كل الدوال $f$ القابلة للاشتقاق على $\R$ والتي تحقق من أجل كل $x\in\R$: $f'(x)=af(x)+b$.
\end{tarif}

\begin{khassia}[المعادلة $y'=ay$]
حلول المعادلة التفاضلية $y'=ay$ هي الدوال: \quad \bx{x\mapsto Ce^{ax}} \quad حيث $C$ ثابت حقيقي.
\par\smallskip
\textbf{البرهان:} الدالة $f(x)=Ce^{ax}$ حل لأن $f'(x)=aCe^{ax}=af(x)$. \ عكسيًا إذا كانت $g$ حلًا نضع $h(x)=e^{-ax}g(x)$، فيكون $h'(x)=-ae^{-ax}g(x)+e^{-ax}ag(x)=0$، إذن $h$ ثابتة $C$ و $g(x)=Ce^{ax}$.
\end{khassia}

\begin{khassia}[المعادلة $y'=ay+b$]
حلول المعادلة التفاضلية $y'=ay+b$ هي الدوال: \quad \bx{x\mapsto Ce^{ax}-\dfrac{b}{a}} \quad حيث $C$ ثابت حقيقي.
\par\smallskip
\textbf{البرهان:} نضع $h=g+\frac{b}{a}$. لدينا $h'=g'$ و $ah=ag+b$، إذن: $g'=ag+b\iff h'=ah$، أي $g$ حل للمعادلة $y'=ay+b$ إذا وفقط إذا كانت $h$ حلًا للمعادلة $y'=ay$، أي $h(x)=Ce^{ax}$ ومنه $g(x)=Ce^{ax}-\frac{b}{a}$.
\par\smallskip
\textbf{الحل الخاص:} من أجل كل $x_0$ و $y_0$ حقيقيين يوجد حل \textbf{وحيد} $f$ يحقق الشرط الابتدائي $f(x_0)=y_0$.
\end{khassia}

\begin{molahada}
\begin{itemize}
\item الدالة $x\mapsto -\frac{b}{a}$ هي الحل الثابت للمعادلة $y'=ay+b$.
\item الدالة الأسية هي الحل الخاص للمعادلة $y'=y$ الذي يحقق $y(0)=1$.
\item قبل تطبيق القاعدة نكتب المعادلة على الشكل $y'=ay+b$ (نقسم على معامل $y'$).
\end{itemize}
\end{molahada}

\begin{mithal}
\begin{itemize}
\item $2y'+5y=0$ تكافئ $y'=-\frac{5}{2}y$؛ الحلول: $x\mapsto Ce^{-\frac{5}{2}x}$ ، $C\in\R$.
\item $y'+3y=0$ مع $f(7)=-2$: الحلول $f(x)=Ce^{-3x}$؛ $Ce^{-21}=-2$ أي $C=-2e^{21}$، إذن \bx{f(x)=-2e^{21-3x}}.
\item $2y'-16y=11$ تكافئ $y'=8y+\frac{11}{2}$؛ الحلول: $x\mapsto Ce^{8x}-\dfrac{11/2}{8}=Ce^{8x}-\dfrac{11}{16}$ ، $C\in\R$.
\end{itemize}
\end{mithal}

\begin{mithal}[حل خاص بشرط على المماس]
عيّن الحل الخاص $f$ للمعادلة $3y'+2y-15=0$ بحيث يقبل منحناه مماسًا موازيًا للمستقيم $y=4x-1$ عند النقطة ذات الفاصلة $3$.
\par\smallskip
المعادلة تكافئ $y'=-\frac{2}{3}y+5$، حلولها: $f(x)=Ce^{-\frac{2}{3}x}-\dfrac{5}{-\frac{2}{3}}=Ce^{-\frac{2}{3}x}+\dfrac{15}{2}$.\par
المماس موازٍ للمستقيم يعني $f'(3)=4$: $-\frac{2}{3}Ce^{-2}=4$ أي $C=-6e^{2}$. \ إذن \bx{f(x)=-6e^{2-\frac{2}{3}x}+\dfrac{15}{2}}.
\end{mithal}

\section{تمارين محلولة في المعادلات التفاضلية}
\origNeedspace{20\baselineskip}
\begin{tmr}
نعتبر المعادلة التفاضلية $(E):\ 2y'+3y=x^{2}+1$.
\begin{sq}
\qa عيّن الدالة كثيرة الحدود $p$ من الدرجة الثانية حلًا للمعادلة $(E)$.
\qb حل المعادلة التفاضلية $(E_0):\ 2y'+3y=0$.
\qc بيّن أن $g$ حل لـ $(E)$ إذا وفقط إذا كانت $g-p$ حلًا لـ $(E_0)$، ثم استنتج حلول $(E)$.
\qd عيّن الحل الخاص $g$ لـ $(E)$ الذي ينعدم من أجل $0$.
\end{sq}
\end{tmr}

\begin{hal}
\pa نضع $p(x)=ax^{2}+bx+c$، $p'(x)=2ax+b$:
\[ 2(2ax+b)+3\left(ax^{2}+bx+c\right)=3ax^{2}+(4a+3b)x+2b+3c=x^{2}+1 \]
بالمطابقة:
\[ 3a=1\ ;\quad 4a+3b=0\ ;\quad 2b+3c=1 \]
إذن $a=\frac{1}{3}$ ، $b=-\frac{4}{9}$ ، $c=\frac{17}{27}$، ومنه \bx{p(x)=\frac{1}{3}x^{2}-\frac{4}{9}x+\frac{17}{27}}.
\pb $(E_0)$ تكافئ $y'=-\frac{3}{2}y$، حلولها: $x\mapsto Ce^{-\frac{3}{2}x}$ ، $C\in\R$.
\pc بما أن $2p'+3p=x^{2}+1$:
\[ 2g'+3g=x^{2}+1\iff 2g'+3g=2p'+3p\iff 2(g-p)'+3(g-p)=0 \]
أي $g-p$ حل لـ $(E_0)$. إذن حلول $(E)$: \ \bx{g(x)=Ce^{-\frac{3}{2}x}+\frac{1}{3}x^{2}-\frac{4}{9}x+\frac{17}{27}} ، $C\in\R$.
\pd $g(0)=C+\frac{17}{27}=0$ أي $C=-\frac{17}{27}$، إذن \bx{g(x)=-\frac{17}{27}e^{-\frac{3}{2}x}+\frac{1}{3}x^{2}-\frac{4}{9}x+\frac{17}{27}}.
\end{hal}

\origNeedspace{20\baselineskip}
\begin{tmr}
نعتبر المعادلة التفاضلية $(E):\ y'=-2y+2e^{-2x}$.
\begin{sq}
\qa عيّن العدد الحقيقي $\alpha$ حتى تكون الدالة $g:x\mapsto\alpha xe^{-2x}$ حلًا للمعادلة $(E)$.
\qb بيّن أن الدالة $f$ حل لـ $(E)$ إذا وفقط إذا كانت $f-g$ حلًا للمعادلة $(E'):\ y'=-2y$.
\qc حل المعادلة $(E')$، ثم استنتج حلول $(E)$.
\qd عيّن حل $(E)$ الذي يشمل تمثيله البياني النقطة $A(0;1)$.
\end{sq}
\end{tmr}

\begin{hal}
\pa $g'(x)=\alpha e^{-2x}-2\alpha xe^{-2x}=-2g(x)+\alpha e^{-2x}$. \ $g$ حل لـ $(E)$ يعني $-2g(x)+\alpha e^{-2x}=-2g(x)+2e^{-2x}$ من أجل كل $x$، أي بالمطابقة \bx{\alpha=2} و $g(x)=2xe^{-2x}$.
\pb لدينا $g'=-2g+2e^{-2x}$. إذن:
\[ f'=-2f+2e^{-2x}\iff f'-g'=-2f+2g\iff(f-g)'=-2(f-g) \]
أي $f$ حل لـ $(E)$ إذا وفقط إذا كانت $f-g$ حلًا لـ $(E')$.
\pc حلول $(E')$: $x\mapsto Ce^{-2x}$ ، $C\in\R$. ومنه $f(x)-2xe^{-2x}=Ce^{-2x}$، أي حلول $(E)$: \ \bx{f_C(x)=(C+2x)e^{-2x}} ، $C\in\R$.
\pd $f_C(0)=C=1$، إذن \bx{f(x)=(1+2x)e^{-2x}}.
\end{hal}
