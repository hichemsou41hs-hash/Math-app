% ===================== 1 =====================
\begin{tamrin}{تمرين من بكالوريا 2008 -- الموضوع الأول}
الفضاء منسوب إلى المعلم المتعامد والمتجانس $\Rep$. نعتبر المستوي $(P)$ الذي معادلته: $x+2y-z+7=0$ ، والنقط: $A(2;0;1)$ ، $B(3;2;0)$ ، $C(-1;-2;2)$.
\QQ{1} تحقق أن النقط $A$ ، $B$ و $C$ ليست على استقامية ، ثم بيّن أن المعادلة الديكارتية للمستوي $(ABC)$ هي: $y+2z-2=0$.
\QQ{2}
\begin{sq}
\qa تحقق أن المستويين $(P)$ و $(ABC)$ متعامدان ، ثم عيّن تمثيلًا وسيطيًا للمستقيم $(\Delta)$ مستقيم تقاطع $(P)$ و $(ABC)$.
\qb احسب المسافة بين النقطة $A$ والمستقيم $(\Delta)$.
\end{sq}
\QQ{3} لتكن $G$ مرجح الجملة $\{(A;1),(B;\alpha),(C;\beta)\}$ حيث $\alpha$ و $\beta$ عددان حقيقيان يحققان $\alpha+\beta+1\neq0$.
\QQ{4} عيّن $\alpha$ حتى تنتمي النقطة $G$ إلى المستقيم $(\Delta)$.
\end{tamrin}
\begin{hal}
\QQ{1} $\V{AB}(1;2;-1)$ و $\V{AC}(-3;-2;1)$: \ $\dfrac{-3}1\neq\dfrac{-2}2$ ، إذن الشعاعان غير مرتبطين خطيًا و \bxt{$A$ ، $B$ ، $C$ ليست على استقامية}.
\par نعوض في $y+2z-2$: \ $A$: $0+2-2=0$ ، \ $B$: $2+0-2=0$ ، \ $C$: $-2+4-2=0$ ، إذن \bx{(ABC):\ y+2z-2=0}.
\QQ{2}
\pa $\vec n_P(1;2;-1)$ و $\vec n(0;1;2)$: \ $\vec n_P\cdot\vec n=0+2-2=0$ ، إذن \bxt{$(P)\perp(ABC)$}.
\par نحل الجملة: من $y+2z-2=0$: $y=2-2z$ ، ونعوض في $(P)$: $x+4-4z-z+7=0$ ، أي $x=-11+5z$. نضع $z=t$:
\[(\Delta):\ \bx{\Sys{x=-11+5t\\y=2-2t\\z=t}}\qquad(t\in\R)\]
\pb $A\in(ABC)$ و $(ABC)\perp(P)$ ، إذن المسقط العمودي لـ $A$ على $(P)$ يقع على $(\Delta)$ ، والمسافة بين $A$ و $(\Delta)$ هي المسافة بين $A$ و $(P)$:
\[d\big(A;(\Delta)\big)=\frac{|2+0-1+7|}{\sqrt{1+4+1}}=\frac8{\sqrt6}=\bx{\frac{4\sqrt6}3}\]
\QQ{3} $G$ موجودة لأن $1+\alpha+\beta\neq0$ ، و $G$ مرجح $A$ ، $B$ ، $C$ ، إذن $G\in(ABC)$.
\QQ{4} نضع $f(M)=x+2y-z+7$. بما أن $G\in(ABC)$ فإن $G\in(\Delta)\iff G\in(P)$. \ $f(A)=8$ ، $f(B)=14$ ، $f(C)=-1-4-2+7=0$ ، و:
\[f(G)=\frac{f(A)+\alpha f(B)+\beta f(C)}{1+\alpha+\beta}=\frac{8+14\alpha}{1+\alpha+\beta}=0\iff\bx{\alpha=-\frac47}\qquad\Big(\beta\neq-\frac37\Big)\]
\end{hal}

\begin{tamrin}{تمرين من بكالوريا 2008 -- الموضوع الثاني}
لكل سؤال من الأسئلة التالية جواب واحد فقط صحيح ، عيّن الجواب الصحيح معلّلًا اختيارك.
\par نعتبر في الفضاء المنسوب إلى معلم متعامد ومتجانس $\Rep$ النقط $A(1;3;-1)$ ، $B(4;1;0)$ ، $C(-2;0;-2)$ و $D(3;2;1)$ ، والمستوي $(P)$ الذي معادلته $x-3z-4=0$.
\QQ{1} المستوي $(P)$ هو: \ABC{$(BCD)$}{$(ABC)$}{$(ABD)$}
\QQ{2} شعاع ناظمي للمستوي $(P)$ هو: \ABC{$\vec n_1(1;2;1)$}{$\vec n_2(-2;0;6)$}{$\vec n_3(2;0;-1)$}
\QQ{3} المسافة بين النقطة $D$ والمستوي $(P)$ هي: \ABC{$\dfrac{\sqrt{10}}5$}{$\dfrac{\sqrt{10}}{10}$}{$\dfrac{2\sqrt{10}}5$}
\end{tamrin}
\begin{hal}
\QQ{1} $A$: $1+3-4=0$ ، $B$: $4-0-4=0$ ، $C$: $-2+6-4=0$ ، و $D$: $3-3-4=-4\neq0$ ، و $\V{AB}(3;-2;1)$ ، $\V{AC}(-3;-3;-1)$ غير مرتبطين خطيًا ، إذن \bxt{الجواب (ب): $(P)=(ABC)$}.
\QQ{2} $\vec n(1;0;-3)$ و $\vec n_2=-2\vec n$ ، إذن \bxt{الجواب (ب): $\vec n_2(-2;0;6)$}.
\QQ{3} $d\big(D;(P)\big)=\dfrac{|3-3-4|}{\sqrt{1+9}}=\dfrac4{\sqrt{10}}=\dfrac{4\sqrt{10}}{10}=\dfrac{2\sqrt{10}}5$ ، إذن \bxt{الجواب (جـ)}.
\end{hal}

\begin{tamrin}{تمرين من بكالوريا 2009 -- الموضوع الأول}
الفضاء مزود بمعلم متعامد ومتجانس $\Rep$. نعتبر النقط: $A(1;0;2)$ ، $B(0;2;1)$ ، $C(2;1;3)$.
\QQ{1} $(P)$ مستوٍ معادلة له من الشكل: $x-z+1=0$.
\begin{sq}
\qa بيّن أن المستوي $(P)$ هو المستوي $(ABC)$.
\qb ما طبيعة المثلث $ABC$ ؟
\end{sq}
\QQ{2}
\begin{sq}
\qa تحقق من أن النقطة $D(2;3;4)$ لا تنتمي إلى المستوي $(ABC)$.
\qb ما طبيعة $ABCD$ ؟
\end{sq}
\QQ{3}
\begin{sq}
\qa احسب المسافة بين النقطة $D$ والمستوي $(ABC)$.
\qb احسب حجم $ABCD$.
\end{sq}
\end{tamrin}
\begin{hal}
\QQ{1}
\pa $\V{AB}(-1;2;-1)$ و $\V{AC}(1;1;1)$ غير مرتبطين خطيًا. \ $A$: $1-2+1=0$ ، $B$: $0-1+1=0$ ، $C$: $2-3+1=0$ ، إذن \bx{(P)=(ABC)}.
\pb $AB=\sqrt6$ ، $AC=\sqrt3$ ، $\V{BC}(2;-1;2)$: $BC=3$. \ $\V{AB}\cdot\V{AC}=-1+2-1=0$ (و $AB^2+AC^2=9=BC^2$) ، إذن \bxt{$ABC$ قائم في $A$}.
\QQ{2}
\pa $2-4+1=-1\neq0$ ، إذن \bxt{$D\notin(ABC)$}.
\pb النقط $A$ ، $B$ ، $C$ ، $D$ ليست من نفس المستوي: \bxt{$ABCD$ رباعي وجوه}.
\QQ{3}
\pa $d\big(D;(ABC)\big)=\dfrac{|2-4+1|}{\sqrt{1+0+1}}=\dfrac1{\sqrt2}=\bx{\frac{\sqrt2}2}$
\pb مساحة القاعدة: $S=\dfrac12\,AB\times AC=\dfrac12\sqrt6\times\sqrt3=\dfrac{3\sqrt2}2$ ، ومنه:
\[V=\frac13\times\frac{3\sqrt2}2\times\frac{\sqrt2}2=\bx{\frac12}\ \text{\ (u.v)}\]
\end{hal}

\begin{tamrin}{تمرين من بكالوريا 2009 -- الموضوع الثاني}
في الفضاء المنسوب إلى معلم متعامد ومتجانس $\Rep$ نعتبر النقط: $A(2;3;-1)$ ، $B(1;-2;4)$ ، $C(3;0;-2)$ ، $D(1;-1;-2)$ ، وليكن $(\pi)$ المستوي المعرف بمعادلته الديكارتية: $2x-y+2z+1=0$.
\par\textbf{المطلوب:} أجب بصحيح أو خطأ مع تبرير الإجابة في كل حالة من الحالات التالية:
\QQ{1} النقط $A$ ، $B$ و $C$ في استقامية.
\QQ{2} $(ABD)$ مستوٍ معادلة ديكارتية له: $25x-6y-z-33=0$.
\QQ{3} المستقيم $(CD)$ عمودي على المستوي $(\pi)$.
\QQ{4} المسقط العمودي للنقطة $B$ على $(\pi)$ هو النقطة $H(1;1;-1)$.
\end{tamrin}
\begin{hal}
\QQ{1} $\V{AB}(-1;-5;5)$ و $\V{AC}(1;-3;-1)$: $\dfrac1{-1}\neq\dfrac{-3}{-5}$ ، غير مرتبطين خطيًا: \bxt{خطأ}.
