% ===================== 1 =====================
\begin{tamrin}{تمرين من بكالوريا 2008 -- الموضوع الأول}
الفضاء منسوب إلى المعلم المتعامد والمتجانس $\Rep$. نعتبر المستوي $(P)$ الذي معادلته: $x+2y-z+7=0$ ، والنقط: $A(2;0;1)$ ، $B(3;2;0)$ ، $C(-1;-2;2)$.
\QQ{1} تحقق أن النقط $A$ ، $B$ و $C$ ليست على استقامية ، ثم بيّن أن المعادلة الديكارتية للمستوي $(ABC)$ هي: $y+2z-2=0$.
\QQ{2}
\begin{sq}
\qa تحقق أن المستويين $(P)$ و $(ABC)$ متعامدان ، ثم عيّن تمثيلًا وسيطيًا للمستقيم $(\Delta)$ مستقيم تقاطع $(P)$ و $(ABC)$.
\qb احسب المسافة بين النقطة $A$ والمستقيم $(\Delta)$.
\end{sq}
\QQ{3} لتكن $G$ مرجح الجملة $\{(A;1),(B;\alpha),(C;\beta)\}$ حيث $\alpha$ و $\beta$ عددان حقيقيان يحققان $\alpha+\beta+1\neq0$.
\QQ{4} عيّن $\alpha$ حتى تنتمي النقطة $G$ إلى المستقيم $(\Delta)$.
\end{tamrin}

\begin{tamrin}{تمرين من بكالوريا 2008 -- الموضوع الثاني}
لكل سؤال من الأسئلة التالية جواب واحد فقط صحيح ، عيّن الجواب الصحيح معلّلًا اختيارك.
\par نعتبر في الفضاء المنسوب إلى معلم متعامد ومتجانس $\Rep$ النقط $A(1;3;-1)$ ، $B(4;1;0)$ ، $C(-2;0;-2)$ و $D(3;2;1)$ ، والمستوي $(P)$ الذي معادلته $x-3z-4=0$.
\QQ{1} المستوي $(P)$ هو: \ABC{$(BCD)$}{$(ABC)$}{$(ABD)$}
\QQ{2} شعاع ناظمي للمستوي $(P)$ هو: \ABC{$\vec n_1(1;2;1)$}{$\vec n_2(-2;0;6)$}{$\vec n_3(2;0;-1)$}
\QQ{3} المسافة بين النقطة $D$ والمستوي $(P)$ هي: \ABC{$\dfrac{\sqrt{10}}5$}{$\dfrac{\sqrt{10}}{10}$}{$\dfrac{2\sqrt{10}}5$}
\end{tamrin}

\begin{tamrin}{تمرين من بكالوريا 2009 -- الموضوع الأول}
الفضاء مزود بمعلم متعامد ومتجانس $\Rep$. نعتبر النقط: $A(1;0;2)$ ، $B(0;2;1)$ ، $C(2;1;3)$.
\QQ{1} $(P)$ مستوٍ معادلة له من الشكل: $x-z+1=0$.
\begin{sq}
\qa بيّن أن المستوي $(P)$ هو المستوي $(ABC)$.
\qb ما طبيعة المثلث $ABC$ ؟
\end{sq}
\QQ{2}
\begin{sq}
\qa تحقق من أن النقطة $D(2;3;4)$ لا تنتمي إلى المستوي $(ABC)$.
\qb ما طبيعة $ABCD$ ؟
\end{sq}
\QQ{3}
\begin{sq}
\qa احسب المسافة بين النقطة $D$ والمستوي $(ABC)$.
\qb احسب حجم $ABCD$.
\end{sq}
\end{tamrin}

\begin{tamrin}{تمرين من بكالوريا 2009 -- الموضوع الثاني}
في الفضاء المنسوب إلى معلم متعامد ومتجانس $\Rep$ نعتبر النقط: $A(2;3;-1)$ ، $B(1;-2;4)$ ، $C(3;0;-2)$ ، $D(1;-1;-2)$ ، وليكن $(\pi)$ المستوي المعرف بمعادلته الديكارتية: $2x-y+2z+1=0$.
\par\textbf{المطلوب:} أجب بصحيح أو خطأ مع تبرير الإجابة في كل حالة من الحالات التالية:
\QQ{1} النقط $A$ ، $B$ و $C$ في استقامية.
\QQ{2} $(ABD)$ مستوٍ معادلة ديكارتية له: $25x-6y-z-33=0$.
\QQ{3} المستقيم $(CD)$ عمودي على المستوي $(\pi)$.
\QQ{4} المسقط العمودي للنقطة $B$ على $(\pi)$ هو النقطة $H(1;1;-1)$.
\end{tamrin}

\begin{tamrin}{تمرين من بكالوريا 2010 -- الموضوع الأول}
نعتبر في الفضاء المنسوب إلى المعلم المتعامد والمتجانس $\Rep$ النقط: $A(1;1;0)$ ، $B(2;1;1)$ و $C(-1;2;-1)$.
\QQ{1}
\begin{sq}
\qa بيّن أن النقط $A$ ، $B$ و $C$ ليست في استقامية.
\qb بيّن أن المعادلة الديكارتية للمستوي $(ABC)$ هي: $x+y-z-2=0$.
\end{sq}
\QQ{2} نعتبر المستويين $(P)$ و $(Q)$ اللذين معادلتاهما على الترتيب: $(P):\ x+2y-3z+1=0$ ، $(Q):\ 2x+y-z-1=0$ ، والمستقيم $(D)$ الذي يشمل النقطة $F(0;4;3)$ و $\vec u(-1;5;3)$ شعاع توجيه له.
\begin{sq}
\qa اكتب تمثيلًا وسيطيًا للمستقيم $(D)$.
\qb تحقق أن تقاطع المستويين $(P)$ و $(Q)$ هو المستقيم $(D)$.
\end{sq}
\QQ{3} عيّن تقاطع المستويات الثلاثة $(ABC)$ ، $(P)$ و $(Q)$.
\end{tamrin}

\begin{tamrin}{تمرين من بكالوريا 2010 -- الموضوع الثاني}
في الفضاء المنسوب إلى المعلم المتعامد والمتجانس $\Rep$ نعتبر المستوي $(P)$ الذي معادلته: $x-2y+z+3=0$.
\QQ{1} نذكّر أن حامل محور الفواصل $(O;\vec i)$ يعرّف بالجملة $\Sys{y=0\\z=0}$. عيّن إحداثيات $A$ نقطة تقاطع حامل $(O;\vec i)$ مع المستوي $(P)$.
\QQ{2} $B$ و $C$ النقطتان من الفضاء حيث: $B(0;0;-3)$ و $C(-1;-4;2)$.
\begin{sq}
\qa تحقق أن النقطة $B$ تنتمي إلى المستوي $(P)$.
\qb احسب الطول $AB$.
\qc احسب المسافة بين النقطة $C$ والمستوي $(P)$.
\end{sq}
\QQ{3}
\begin{sq}
\qa اكتب تمثيلًا وسيطيًا للمستقيم $(\Delta)$ المار بالنقطة $C$ والعمودي على المستوي $(P)$.
\qb تحقق أن النقطة $A$ تنتمي إلى المستقيم $(\Delta)$.
\qc احسب مساحة المثلث $ABC$.
\end{sq}
\end{tamrin}

\begin{tamrin}{تمرين من بكالوريا 2011 -- الموضوع الأول}
نعتبر في الفضاء المنسوب إلى المعلم المتعامد والمتجانس $\Rep$ المستوي $(P)$ الذي يشمل النقطة $A(1;-2;1)$ ، و $\vec n(-2;1;5)$ شعاع ناظمي له ، وليكن $(Q)$ المستوي ذو المعادلة $x+2y-7=0$.
\QQ{1} اكتب معادلة ديكارتية للمستوي $(P)$.
\QQ{2}
\begin{sq}

\qb بيّن أن المستويين $(P)$ و $(Q)$ متقاطعان وفق مستقيم $(\Delta)$ يطلب تعيين تمثيل وسيطي له.
\end{sq}
\QQ{3} لتكن النقطة $C(5;-2;-1)$.
\begin{sq}
\qa احسب المسافة بين النقطة $C$ والمستوي $(P)$ ، ثم المسافة بين النقطة $C$ والمستوي $(Q)$.
\qb أثبت أن المستويين $(P)$ و $(Q)$ متعامدان.
\qc استنتج المسافة بين النقطة $C$ والمستقيم $(\Delta)$.
\end{sq}
\end{tamrin}

\begin{tamrin}{تمرين من بكالوريا 2011 -- الموضوع الثاني}
نعتبر في الفضاء المنسوب إلى المعلم المتعامد والمتجانس $\Rep$ النقط: $A(0;1;5)$ ، $B(2;1;7)$ و $C(3;-3;6)$.
\QQ{1}
\begin{sq}
\qa اكتب تمثيلًا وسيطيًا للمستقيم $(\Delta)$ الذي يشمل النقطة $B$ و $\vec u(1;-4;-1)$ شعاع توجيه له.
\qb تحقق أن النقطة $C$ تنتمي إلى المستقيم $(\Delta)$.

الفضاء منسوب إلى المعلم المتعامد والمتجانس $\Rep$. نعتبر النقط $A(-8;0;-2)$ ، $B(1;2;1)$ ، $C(2;3;-1)$ و $E(1;1;4)$ ، والمستوي $(P)$ ذا المعادلة $2x+y-3=0$.
\QQ{1}
\begin{sq}
\qa بيّن أن النقط $A$ ، $B$ و $C$ تعيّن مستويًا.
\qb عيّن قيمة العدد الحقيقي $\alpha$ حتى يكون $\vec n(1;\alpha;-1)$ شعاعًا ناظميًا للمستوي $(ABC)$ ، ثم عيّن معادلة ديكارتية له.
\end{sq}
\QQ{2} بيّن أن المستويين $(P)$ و $(ABC)$ يتقاطعان وفق مستقيم $(\Delta)$ ، ثم تحقق أن $E$ تنتمي إلى $(\Delta)$ و $\vec u(1;-2;7)$ شعاع توجيه له.
\QQ{3} لتكن $G$ مرجح الجملة $\{(A;1),(B;-2),(C;3)\}$ ، نرمز بـ $(\Gamma)$ إلى مجموعة النقط $M$ من الفضاء التي تحقق: $\big(\V{MA}-2\V{MB}+3\V{MC}\big)\cdot\big(\V{MB}-\V{MC}\big)=0$.
\par عيّن إحداثيات النقطة $G$ ، ثم حدد طبيعة المجموعة $(\Gamma)$ واكتب معادلة ديكارتية لها.
\QQ{4} عيّن إحداثيات نقطة تقاطع $(P)$ ، $(ABC)$ و $(\Gamma)$.
\end{tamrin}

\begin{tamrin}{تمرين من بكالوريا 2018 -- الموضوع الثاني}
الفضاء منسوب إلى المعلم المتعامد والمتجانس $\Rep$. نعتبر النقطة $A(1;-2;1)$ والمستويين $(P_1)$ و $(P_2)$ اللذين معادلتاهما على الترتيب: $-x+y+2z+1=0$ و $-3x+y+z+4=0$.
\QQ{1} اكتب تمثيلًا وسيطيًا للمستقيم $(\Delta)$ الذي يشمل النقطة $A$ و $\vec u(1;5;-2)$ شعاع توجيه له.
\QQ{2} بيّن أن المستويين $(P_1)$ و $(P_2)$ متقاطعان ، ثم تحقق أن تقاطعهما هو المستقيم $(\Delta)$.
\QQ{3} اكتب معادلة ديكارتية للمستوي $(Q)$ الذي يشمل $B(-1;4;0)$ ويعامد كلًا من $(P_1)$ و $(P_2)$ ، ثم استنتج تقاطع المستويات الثلاثة $(P_1)$ ، $(P_2)$ و $(Q)$.
\QQ{4} لتكن $E(2;3;-1)$ و $H(0;3;-2)$ نقطتين من الفضاء.
\begin{sq}
\qa تحقق أن $H$ هي المسقط العمودي للنقطة $B$ على المستوي $(P_1)$.
\qb حدد طبيعة المثلث $EBH$ ، ثم احسب $V$ حجم رباعي الوجوه $AEBH$.
\end{sq}
\end{tamrin}
