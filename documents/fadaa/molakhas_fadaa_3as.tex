\section{الأشعة في الفضاء}
\begin{tarif}[شعاعان مرتبطان خطيًّا -- استقامية]
\begin{itemize}
\item $\vu$ و $\vv$ \textbf{مرتبطان خطيًّا} (متوازيان) إذا وُجد عدد حقيقي $k$ حيث \bx{\vv=k\,\vu} (أو $\vu=\vec0$).
\item النقط $A$ ، $B$ ، $C$ \textbf{في استقامية} $\iff$ $\V{AB}$ و $\V{AC}$ مرتبطان خطيًّا.
\item $(AB)\parallel(CD)$ $\iff$ $\V{AB}$ و $\V{CD}$ مرتبطان خطيًّا.
\end{itemize}
\end{tarif}
\begin{tarif}[أشعة من نفس المستوي]
إذا كان $\vu$ و $\vv$ غير مرتبطين خطيًّا، فإن $\vu$ ، $\vv$ ، $\vw$ \textbf{من نفس المستوي} إذا وُجد عددان حقيقيان $a$ و $b$ حيث \bx{\vw=a\,\vu+b\,\vv}.\par
\textbf{تطبيق:} النقط $A$ ، $B$ ، $C$ ، $D$ \textbf{من نفس المستوي} $\iff$ الأشعة $\V{AB}$ ، $\V{AC}$ ، $\V{AD}$ من نفس المستوي.
\end{tarif}
\begin{mithal}
\textbf{(1)} $A(1;2;-1)$ ، $B(3;1;0)$ ، $C(7;-1;2)$: \ $\V{AB}(2;-1;1)$ و $\V{AC}(6;-3;3)$، إذن $\V{AC}=3\V{AB}$ و\bxt{$A$ ، $B$ ، $C$ في استقامية}.\par
\textbf{(2)} $\vu(1;0;2)$ ، $\vv(0;1;-1)$ ، $\vw(2;3;1)$: نبحث عن $a,b$ حيث $\vw=a\vu+b\vv$:
\[ \begin{cases} a=2\\ b=3\\ 2a-b=1 \end{cases}\quad (4-3=1\ \text{\ok}) \]
إذن $\vw=2\vu+3\vv$ و\bxt{الأشعة $\vu$ ، $\vv$ ، $\vw$ من نفس المستوي}.
\end{mithal}

\section{المرجح في الفضاء}
\begin{tarif}
$\alpha+\beta+\gamma\neq0$: مرجح الجملة $\{(A,\alpha),(B,\beta),(C,\gamma)\}$ هو النقطة الوحيدة $G$ حيث \bx{\alpha\V{GA}+\beta\V{GB}+\gamma\V{GC}=\vec0}، وإحداثياتها:
\[ x_G=\frac{\alpha x_A+\beta x_B+\gamma x_C}{\alpha+\beta+\gamma}\ ،\quad y_G=\frac{\alpha y_A+\beta y_B+\gamma y_C}{\alpha+\beta+\gamma}\ ،\quad z_G=\frac{\alpha z_A+\beta z_B+\gamma z_C}{\alpha+\beta+\gamma}. \]
\end{tarif}
\begin{khassia}[استعمال المرجح]
\begin{itemize}
\item مرجح نقطتين $A$ و $B$ ينتمي إلى المستقيم $(AB)$: يُستعمل لإثبات \textbf{الاستقامية}.
\item مرجح ثلاث نقط $A$ ، $B$ ، $C$ (ليست في استقامية) ينتمي إلى المستوي $(ABC)$: يُستعمل لإثبات أن \textbf{أربع نقط من نفس المستوي}.
\item المستقيم $(AB)$ هو مجموعة مرجحات $A$ و $B$؛ والمستوي $(ABC)$ هو مجموعة مرجحات $A$ ، $B$ ، $C$.
\end{itemize}
\end{khassia}
\begin{mithal}
$A(1;2;0)$ ، $B(3;0;2)$ ، $C(-1;2;2)$. مرجح $\{(A,2),(B,1),(C,1)\}$:
\[ x_G=\frac{2+3-1}{4}=1\ ،\quad y_G=\frac{4+0+2}{4}=\frac32\ ،\quad z_G=\frac{0+2+2}{4}=1, \]
إذن $G\big(1;\frac32;1\big)$، والنقطة $G$ \bxt{تنتمي إلى المستوي $(ABC)$}.
\end{mithal}

\begin{tariqa}[الانتقال بين التمثيلات]
\begin{itemize}
\item \textbf{تمثيل وسيطي $\leftarrow$ معادلة مستوٍ:} نضع مجهولين وسيطين. مثال: $x+2y-z-4=0$: نضع $x=t$ ، $y=s$ فنجد $z=t+2s-4$.
\item \textbf{معادلة مستوٍ $\leftarrow$ تمثيل وسيطي:} نحذف $t$ و $s$. مثال: $\begin{cases}x=1+t\\y=t+s\\z=2-s\end{cases}$: $t=x-1$ ، $s=2-z$، ومنه $y=x-1+2-z$، أي \bx{x-y-z+1=0}.
\item \textbf{تمثيل وسيطي $\leftarrow$ مستقيم معرّف بجملة معادلتين} (تقاطع مستويين): نضع أحد المجاهيل $t$ ونعبّر عن الآخرين. مثال:
\[ \begin{cases}x+y-z-1=0\\x-y+z-3=0\end{cases}\ \xrightarrow{\ \text{بالجمع}\ }\ 2x=4\ ؛\ \text{نضع } z=t:\quad \begin{cases}x=2\\y=-1+t\\z=t\end{cases}\ (t\in\R) \]
\item \textbf{جملة معادلتين $\leftarrow$ تمثيل وسيطي لمستقيم:} نستخرج $t$ من معادلة ونعوّض في الأخريين. مثال: $x=1+2t$ ، $y=-1+t$ ، $z=2-3t$: \ $t=y+1$ فنجد $\begin{cases}x-2y-3=0\\3y+z+1=0\end{cases}$.
\end{itemize}
\end{tariqa}
