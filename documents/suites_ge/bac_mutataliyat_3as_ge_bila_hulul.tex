\begin{tamrin}{تمرين من بكالوريا 2008 -- الموضوع الأول}
$(u_n)$ متتالية عددية معرفة بـ: $u_0=\alpha$ حيث $\alpha$ عدد حقيقي ، ومن أجل كل عدد طبيعي $n$: \ $u_{n+1}=\dfrac23u_n-\dfrac89$.
\begin{enumerate}
\item برهن بالتراجع أنه في حالة $\alpha=-\dfrac83$ تكون المتتالية $(u_n)$ ثابتة.
\item في كل ما يلي نضع $\alpha=2$ ، ونعرّف المتتالية $(v_n)$ كما يلي: من أجل كل عدد طبيعي $n$: \ $v_n=u_n+\dfrac83$.
\begin{sq}
\qa أحسب $u_1$ و $u_2$.
\qb أثبت أن $(v_n)$ متتالية هندسية يطلب تعيين أساسها $q$ وحدها الأول $v_0$.
\qc أكتب عبارة $v_n$ ثم $u_n$ بدلالة $n$.
\qd أحسب $\limn u_n$.
\end{sq}
\end{enumerate}
\end{tamrin}

\begin{tamrin}{تمرين من بكالوريا 2008 -- الموضوع الثاني}
$(u_n)$ متتالية عددية معرفة بـ: $u_0=1$ ، ومن أجل كل عدد طبيعي $n$: \ $u_{n+1}=\dfrac12u_n-1$.
\begin{enumerate}
\item أحسب $u_1$ ، $u_2$ و $u_3$.
\item \begin{sq}
\qa أثبت بالتراجع أنه من أجل كل عدد طبيعي $n$: $u_n\ge-2$.
\qb جد اتجاه تغير المتتالية $(u_n)$. ماذا تستنتج؟
\end{sq}
\item نعتبر المتتالية $(v_n)$ المعرفة من أجل كل عدد طبيعي $n$ بـ: $v_n=u_n+2$.
\begin{sq}
\qa بيّن أن $(v_n)$ متتالية هندسية يطلب تعيين أساسها وحدها الأول.
\qb عبّر بدلالة $n$ عن $v_n$ ثم عن $u_n$.
\qc أحسب $\limn u_n$.
\qd أحسب بدلالة $n$ المجموع: $S_n=u_0+u_1+\cdots+u_n$.
\end{sq}
\end{enumerate}
\end{tamrin}

\begin{tamrin}{تمرين من بكالوريا 2009 -- الموضوع الأول}
$(u_n)$ متتالية عددية معرفة بـ: $u_0=-1$ ، ومن أجل كل عدد طبيعي $n$: \ $3u_{n+1}=u_n+4$.
\begin{enumerate}
\item \begin{sq}
\qa برهن بالتراجع أنه من أجل كل عدد طبيعي $n$: $u_n\le2$.
\qb بيّن أن المتتالية $(u_n)$ متزايدة.
\qc استنتج مع التبرير أن المتتالية $(u_n)$ متقاربة.
\end{sq}
\item نضع من أجل كل عدد طبيعي $n$: $v_n=u_n-2$.
\begin{sq}
\qa بيّن أن $(v_n)$ متتالية هندسية يطلب تحديد حدها الأول وأساسها.
\qb أكتب $v_n$ ثم $u_n$ بدلالة $n$.
\qc أحسب $\limn u_n$.
\qd أحسب بدلالة $n$ المجموع: $S_n=u_0+u_1+\cdots+u_n$.
\end{sq}
\end{enumerate}
\end{tamrin}

\begin{tamrin}{تمرين من بكالوريا 2009 -- الموضوع الثاني}
$(u_n)$ متتالية عددية معرفة بـ: $u_0=-1$ ، ومن أجل كل عدد طبيعي $n$: \ $u_{n+1}=3u_n-2$.
\begin{enumerate}
\item أحسب $u_1$ و $u_2$.
\item نعتبر المتتالية $(v_n)$ المعرفة من أجل كل عدد طبيعي $n$ بـ: $v_n=u_n-1$.
\begin{sq}
\qa أثبت أن $(v_n)$ متتالية هندسية يطلب تعيين أساسها $q$ وحدها الأول $v_0$.
\qb أكتب $v_n$ بدلالة $n$ ، ثم استنتج $u_n$ بدلالة $n$.
\end{sq}
\item بيّن أنه من أجل كل عدد طبيعي $n$: $u_{n+1}-u_n=(-4)\times3^n$.
\item استنتج اتجاه تغير المتتالية $(u_n)$.
\item عيّن العدد الطبيعي $n$ بحيث: $u_0+u_1+\cdots+u_n=n-79$.
\end{enumerate}
\end{tamrin}

\begin{tamrin}{تمرين من بكالوريا 2010 -- الموضوع الأول ($\star\star$ بعد درس الأسية)}
\begin{enumerate}
\item $n$ عدد طبيعي. أحسب بدلالة $n$ المجموع: $S_n=1+e+e^2+\cdots+e^n$.
\item $(w_n)$ متتالية معرفة من أجل كل عدد طبيعي $n$ بـ: $w_n=2n+4+e^n$.\\
بيّن أن $w_n=u_n+v_n$ حيث $(u_n)$ متتالية حسابية و $(v_n)$ متتالية هندسية ، يطلب تعيين الحد الأول والأساس لكل منهما.
\item أثبت أنه من أجل كل عدد طبيعي $n$: $4+6+8+\cdots+(2n+4)=(n+1)(n+4)$.
\item استنتج بدلالة $n$ المجموع: $S'_n=w_0+w_1+\cdots+w_n$.
\end{enumerate}
\end{tamrin}

\begin{tamrin}{تمرين من بكالوريا 2010 -- الموضوع الثاني}
$(u_n)$ متتالية عددية معرفة بـ: $u_0=1$ ، ومن أجل كل عدد طبيعي $n$: \ $u_{n+1}=\dfrac{3u_n+2}{4}$.
\begin{enumerate}
\item أحسب $u_1$ ، $u_2$ و $u_3$.
\item \begin{sq}
\qa برهن بالتراجع أنه من أجل كل عدد طبيعي $n$: $u_n<2$.
\qb بيّن أن المتتالية $(u_n)$ متزايدة تماماً.
\qc استنتج أن المتتالية $(u_n)$ متقاربة.
\end{sq}
\item نضع من أجل كل عدد طبيعي $n$: $v_n=u_n-2$.
\begin{sq}
\qa بيّن أن $(v_n)$ متتالية هندسية يطلب تعيين أساسها وحدها الأول.
\qb أكتب $v_n$ بدلالة $n$ ، واستنتج أن: $u_n=2-\Big(\dfrac34\Big)^n$.
\qc ما هي نهاية المتتالية $(u_n)$؟
\end{sq}
\item \begin{sq}
\qa أحسب بدلالة $n$ المجموع: $S_n=v_0+v_1+\cdots+v_n$.
\qb استنتج أن: $u_0+u_1+\cdots+u_n=3\Big(\dfrac34\Big)^n+2n-2$.
\end{sq}
\end{enumerate}
\end{tamrin}

\begin{tamrin}{تمرين من بكالوريا 2011 -- الموضوع الثاني}
$(u_n)$ متتالية عددية معرفة بـ: $u_0=\dfrac12$ ، ومن أجل كل عدد طبيعي $n$: \ $u_{n+1}=\dfrac25u_n+\dfrac15$.
\begin{enumerate}

(جـ) \ $3n+5$ & (ب) \ $5n+3$ & (أ) \ $3\times5^n$\end{tabular}
\item $(u_n)$ متتالية حسابية حدها الأول $u_0$ وأساسها $r$ ، حيث $u_2=5$ و $u_4=9$:\\[2pt]
\begin{tabular}{p{0.28\textwidth}p{0.28\textwidth}p{0.28\textwidth}}
(جـ) \ $u_0=1$ ، $r=4$ & (ب) \ $u_0=1$ ، $r=2$ & (أ) \ $u_0=2$ ، $r=1$\end{tabular}
\end{enumerate}
