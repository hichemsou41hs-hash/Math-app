\end{tabular}
\end{center}

\begin{center}

\begin{tabular}{|M{0.44\textwidth}|M{0.44\textwidth}|}\hline

$\big(a^n\big)^m=a^{n\times m}$ & $a^n\times b^n=(a\times b)^n$\\\hline
$a^n\times a^m=a^{n+m}$ & $\dfrac{a^n}{a^m}=a^{n-m}$\\\hline
\end{tabular}

\begin{center}

\begin{tabular}{|p{0.44\textwidth}|p{0.44\textwidth}|}\hline

\section{تعريف متتالية عددية}
\begin{tarif}
نسمي \textbf{متتالية عددية} $\LW{u}$ كل دالة معرفة على مجموعة الأعداد الطبيعية $\LW{\N}$ (أو على جزء منها) نحو مجموعة الأعداد الحقيقية $\LW{\R}$.
\end{tarif}
\begin{molahada}[ملاحظات]
\begin{itemize}
\item نرمز للمتتالية بأحد الرموز: $(u_n)$ ، $(v_n)$ ، $u$ ، $W$ ، $\ldots$
\item $u_n$ هو \textbf{الحد العام} للمتتالية ، والعدد $n$ هو \textbf{دليله} (أو رتبته).
\item إذا كانت $(u_n)$ معرفة على $\N$ فإن حدها الأول هو $u_0$ ، أما إذا كانت معرفة على $\N^*$ فإن حدها الأول هو $u_1$.
\end{itemize}
\end{molahada}

\section{طرق توليد متتالية}
\begin{center}
\renewcommand{\arraystretch}{2}
\begin{tabular}{|M{0.44\textwidth}|M{0.44\textwidth}|}\hline
\textbf{متتالية معرفة بعلاقة تراجعية} & \textbf{متتالية معرفة بحدها العام}\\
\bx{u_{n+1}=f(u_n)} \ مع إعطاء الحد الأول & \bx{u_n=f(n)}\\\hline
مثال: $u_0=1$ و $u_{n+1}=2u_n+n+3$ & مثال: $u_n=\dfrac52n-\dfrac13$\\\hline
\end{tabular}
\end{center}

\begin{mithal}[مثال 1: متتالية معرفة بحدها العام]
لتكن $(u_n)$ المتتالية المعرفة على $\N$ كما يلي: من أجل كل عدد طبيعي $n$: \ $u_n=\dfrac23n+2$.
\begin{enumerate}
\item أحسب الحدود: $u_0$ ، $u_3$ و $u_{2n}$.
\item عيّن قيمة الحد الذي رتبته $10$.
\end{enumerate}
\textbf{الحل:}
\begin{enumerate}
\item نعوّض $n$ بالقيمة المطلوبة في عبارة الحد العام:
\begin{align*}
u_0&=\frac23\times0+2=2\\
u_3&=\frac23\times3+2=2+2=4\\
u_{2n}&=\frac23\times(2n)+2=\frac{4n}{3}+2=\frac{4n+6}{3}
\end{align*}
\item الحد الأول هو $u_0$ (رتبته $1$) ، والحد الثاني $u_1$ (رتبته $2$) ، $\ldots$ \ إذن الحد ذو الرتبة $p$ هو $u_{p-1}$.\\
الحد الذي رتبته $10$ هو $u_9$: \ $u_9=\dfrac23\times9+2=6+2=\bx{8}$.
\end{enumerate}
\end{mithal}

\begin{mithal}[مثال 2: متتالية معرفة بعلاقة تراجعية]
نعتبر المتتالية $(u_n)$ المعرفة على $\N$ بـ: $u_0=\dfrac12$ ، ومن أجل كل عدد طبيعي $n$: \ $u_{n+1}=\dfrac25u_n+\dfrac15$.
\begin{itemize}
\item أحسب الحدود $u_1$ ، $u_2$ و $u_3$.
\end{itemize}
\textbf{الحل:} نحسب كل حد انطلاقاً من الحد الذي قبله:
\begin{align*}
u_1&=\frac25u_0+\frac15=\frac25\times\frac12+\frac15=\frac15+\frac15=\frac25\\
u_2&=\frac25u_1+\frac15=\frac25\times\frac25+\frac15=\frac{4}{25}+\frac{5}{25}=\frac{9}{25}\\
u_3&=\frac25u_2+\frac15=\frac25\times\frac{9}{25}+\frac15=\frac{18}{125}+\frac{25}{125}=\frac{43}{125}
\end{align*}
\end{mithal}
\begin{molahada}
يمكن حساب الحدود باستعمال الآلة الحاسبة (Casio) ، ويُشرح ذلك في القسم.
\end{molahada}

\section{دراسة اتجاه تغير متتالية عددية}
لدراسة اتجاه تغير متتالية عددية $(u_n)$ نتبع إحدى الطريقتين التاليتين:
\begin{tariqa}[الطريقة الأولى: إشارة الفرق $u_{n+1}-u_n$]
ندرس إشارة الفرق $u_{n+1}-u_n$ من أجل كل عدد طبيعي $n$ ، ونميز ثلاث حالات:
\begin{itemize}
\item إذا كان $u_{n+1}-u_n>0$ فإن المتتالية $(u_n)$ \textbf{متزايدة تماماً}.
\item إذا كان $u_{n+1}-u_n<0$ فإن المتتالية $(u_n)$ \textbf{متناقصة تماماً}.
\item إذا كان $u_{n+1}-u_n=0$ فإن المتتالية $(u_n)$ \textbf{ثابتة}.
\end{itemize}
\end{tariqa}

\begin{mithal}[مثال 1]
$(u_n)$ متتالية عددية معرفة على $\N$ كما يلي: من أجل كل عدد طبيعي $n$: \ $u_n=-2n+3$.
\begin{itemize}
\item أدرس اتجاه تغير المتتالية $(u_n)$.
\end{itemize}
\textbf{الحل:} من أجل كل عدد طبيعي $n$ لدينا:
\begin{align*}
u_{n+1}-u_n&=\big[-2(n+1)+3\big]-\big(-2n+3\big)\\
&=-2n-2+3+2n-3\\
&=-2<0
\end{align*}
إذن المتتالية $(u_n)$ \textbf{متناقصة تماماً} على $\N$.
\end{mithal}

\begin{mithal}[مثال 2]
$(v_n)$ متتالية عددية معرفة على $\N$ كما يلي: من أجل كل عدد طبيعي $n$: \ $v_n=2\Big(\dfrac13\Big)^n-1$.
\begin{itemize}
\item أدرس اتجاه تغير المتتالية $(v_n)$.
\end{itemize}
\textbf{الحل:} من أجل كل عدد طبيعي $n$ لدينا:
\begin{align*}
v_{n+1}-v_n&=\Big[2\Big(\frac13\Big)^{n+1}-1\Big]-\Big[2\Big(\frac13\Big)^n-1\Big]
=2\Big(\frac13\Big)^n\times\frac13-1-2\Big(\frac13\Big)^n+1\\
&=2\Big(\frac13\Big)^n\Big[\frac13-1\Big]\qquad(\text{\ar{نخرج}}\ 2\big(\tfrac13\big)^n\ \text{\ar{عاملاً مشتركاً}})\\
&=2\Big(\frac13\Big)^n\times\Big(-\frac23\Big)=-\frac43\Big(\frac13\Big)^n
\end{align*}
وبما أن $\Big(\dfrac13\Big)^n>0$ فإن $v_{n+1}-v_n<0$ ، إذن المتتالية $(v_n)$ \textbf{متناقصة تماماً} على $\N$.
\end{mithal}

\begin{mithal}[مثال 3]
$(u_n)$ متتالية عددية معرفة على $\N$ بـ: $u_0=-1$ ، ومن أجل كل عدد طبيعي $n$: \ $u_{n+1}=\dfrac13u_n+\dfrac43$.\\
نقبل أنه من أجل كل عدد طبيعي $n$: \ $u_n\le2$.
\begin{itemize}
\item أدرس اتجاه تغير المتتالية $(u_n)$.
\end{itemize}
\textbf{الحل:} من أجل كل عدد طبيعي $n$ لدينا:
\[u_{n+1}-u_n=\frac13u_n+\frac43-u_n=\Big(\frac13-1\Big)u_n+\frac43=-\frac23u_n+\frac43.\]
ولدينا من أجل كل عدد طبيعي $n$: $u_n\le2$ ، نضرب الطرفين في العدد السالب $-\dfrac23$ فيتغير اتجاه المتباينة:
\[-\frac23u_n\ge-\frac23\times2\quad\text{\ar{أي}}\quad-\frac23u_n\ge-\frac43\quad\text{\ar{ومنه}}\quad-\frac23u_n+\frac43\ge-\frac43+\frac43=0.\]
إذن من أجل كل عدد طبيعي $n$: \ $u_{n+1}-u_n\ge0$ ، ومنه المتتالية $(u_n)$ \textbf{متزايدة} على $\N$.
\end{mithal}
\begin{molahada}[تنبيه]
عند ضرب طرفي متباينة في عدد \textbf{سالب} يجب قلب اتجاهها. في المثال 3 أعلاه: من $u_n\le2$ نحصل على $-\dfrac23u_n\ge-\dfrac43$ (وليس $\le$) ، لذلك المتتالية متزايدة وليست متناقصة.
\end{molahada}

\begin{tariqa}[الطريقة الثانية: مقارنة النسبة $\dfrac{u_{n+1}}{u_n}$ بالعدد $1$]
إذا كانت حدود المتتالية $(u_n)$ \textbf{موجبة تماماً} ، نحسب النسبة $\dfrac{u_{n+1}}{u_n}$ ونميز ثلاث حالات:
\begin{itemize}
\item إذا كان $\dfrac{u_{n+1}}{u_n}>1$ فإن المتتالية $(u_n)$ متزايدة تماماً.
\item إذا كان $\dfrac{u_{n+1}}{u_n}<1$ فإن المتتالية $(u_n)$ متناقصة تماماً.
\item إذا كان $\dfrac{u_{n+1}}{u_n}=1$ فإن المتتالية $(u_n)$ ثابتة.
\end{itemize}
\end{tariqa}

\begin{tarif}[المتتالية الثابتة]
نقول إن $\LW{(u_n)}$ متتالية \textbf{ثابتة} إذا كان من أجل كل عدد طبيعي $\LW{n}$: \ \bx{u_{n+1}=u_n=u_0} \ (كل الحدود متساوية وتساوي الحد الأول).
\end{tarif}
\begin{mithal}
نعتبر المتتالية العددية $(u_n)$ المعرفة على $\N$ بـ: $u_0=\alpha$ ، ومن أجل كل عدد طبيعي $n$: \ $u_{n+1}=2u_n-1$.
\begin{itemize}
\item عيّن قيمة العدد الحقيقي $\alpha$ حتى تكون المتتالية $(u_n)$ ثابتة.
\end{itemize}
\textbf{الحل:} $(u_n)$ ثابتة معناه $u_{n+1}=u_n=u_0=\alpha$. نعوّض $u_{n+1}$ و $u_n$ بـ $\alpha$ في العلاقة التراجعية:
\[\alpha=2\alpha-1\quad\Longleftrightarrow\quad\alpha-2\alpha=-1\quad\Longleftrightarrow\quad-\alpha=-1\quad\Longleftrightarrow\quad\bx{\alpha=1}.\]
\end{mithal}

\section{المتتالية الحسابية والمتتالية الهندسية}
\subsection{المتتالية الحسابية}
\begin{tarif}
نقول إن $\LW{(u_n)}$ متتالية \textbf{حسابية} إذا وُجد عدد حقيقي $\LW{r}$ (يسمى \textbf{الأساس}) بحيث من أجل كل عدد طبيعي $\LW{n}$: \ \bx{u_{n+1}=u_n+r}.\\
\textbf{للبرهان أن $\LW{(u_n)}$ حسابية:} نحسب الفرق $\LW{u_{n+1}-u_n}$ فنجد عدداً حقيقياً ثابتاً (لا يتعلق بـ $\LW{n}$): \ \bx{u_{n+1}-u_n=r}.
\end{tarif}
\begin{mithal}
نعتبر المتتالية العددية $(u_n)$ المعرفة على $\N$ بـ: $u_0=1$ ، ومن أجل كل عدد طبيعي $n$: \ $u_{n+1}=\dfrac{u_n-4}{u_n+5}$.\\
ونضع من أجل كل عدد طبيعي $n$: \ $v_n=\dfrac{1}{u_n+2}$ \ (نقبل أن $u_n+2\ne0$).
\begin{itemize}
\item برهن أن المتتالية $(v_n)$ حسابية.
\end{itemize}
\textbf{الحل:} نحسب أولاً $u_{n+1}+2$ بتوحيد المقامات:
\[u_{n+1}+2=\frac{u_n-4}{u_n+5}+\frac{2(u_n+5)}{u_n+5}=\frac{u_n-4+2u_n+10}{u_n+5}=\frac{3u_n+6}{u_n+5}=\frac{3(u_n+2)}{u_n+5}.\]
ومنه من أجل كل عدد طبيعي $n$:
\begin{align*}
v_{n+1}-v_n&=\frac{1}{u_{n+1}+2}-\frac{1}{u_n+2}=\frac{u_n+5}{3(u_n+2)}-\frac{3}{3(u_n+2)}\\
&=\frac{u_n+5-3}{3(u_n+2)}=\frac{u_n+2}{3(u_n+2)}=\frac13
\end{align*}
إذن $(v_n)$ متتالية حسابية أساسها \bx{r=\dfrac13} وحدها الأول $v_0=\dfrac{1}{u_0+2}=\dfrac13$.
\end{mithal}

\begin{khassia}[عبارة الحد العام لمتتالية حسابية]
إذا كانت $\LW{(u_n)}$ حسابية أساسها $\LW{r}$ فإنه من أجل كل عددين طبيعيين $\LW{n}$ و $\LW{p}$:
\[\bx{u_n=u_p+(n-p)\,r}\qquad\text{\ar{وبصفة خاصة (الحد الأول $\LW{u_0}$):}}\qquad\bx{u_n=u_0+n\,r}\]
تستعمل هذه العلاقة لحساب حد معين ، أو لحساب الأساس إذا عُلم حدان: \ \bx{r=\dfrac{u_p-u_q}{p-q}}.
\end{khassia}
\begin{mithal}
$(u_n)$ متتالية حسابية حدها الأول $u_0=2$ ، وتحقق: $u_2+u_5=25$.
\begin{enumerate}
\item عيّن أساس المتتالية $(u_n)$.
\item أكتب عبارة الحد العام $u_n$ بدلالة $n$.
\end{enumerate}
\textbf{الحل:}
\begin{enumerate}
\item لدينا $u_2=u_0+2r$ و $u_5=u_0+5r$ ، ومنه:
\[u_2+u_5=2u_0+7r=4+7r.\]
لكن $u_2+u_5=25$ ، ومنه $4+7r=25$ أي $7r=21$ ، إذن \bx{r=3}.
\item $u_n=u_0+nr$ ، أي من أجل كل عدد طبيعي $n$: \ \bx{u_n=2+3n}.
\end{enumerate}
\end{mithal}

\begin{khassia}[الوسط الحسابي]
إذا كانت $\LW{a}$ ، $\LW{b}$ و $\LW{c}$ ثلاثة حدود متتابعة من متتالية حسابية فإن: \ \bx{2b=a+c} \ أي \bx{3b=a+b+c}.
\end{khassia}
\begin{mithal}
$(u_n)$ متتالية حسابية معرفة على $\N$ أساسها $r$ ، حيث:
$\begin{cases}u_2+u_3+u_4=21 & (1)\\ u_4+u_5=20 & (2)\end{cases}$
\begin{enumerate}
\item[(أ)] بيّن أن $u_3=7$ و $r=2$ ، ثم استنتج قيمة $u_0$.
\item[(ب)] أكتب $u_n$ بدلالة $n$.
\end{enumerate}
\textbf{الحل:}
\begin{enumerate}
\item[(أ)] حسب خاصية الوسط الحسابي: $u_2+u_4=2u_3$ ، فتصبح المعادلة $(1)$: $2u_3+u_3=21$ أي $3u_3=21$ ، إذن \bx{u_3=7}.\\
من جهة أخرى: $u_4=u_3+r=7+r$ و $u_5=u_3+2r=7+2r$. بالتعويض في $(2)$:
\[(7+r)+(7+2r)=20\ \Longleftrightarrow\ 14+3r=20\ \Longleftrightarrow\ 3r=6\ \Longleftrightarrow\ \bx{r=2}.\]
ونعلم أن $u_3=u_0+3r$ ، ومنه: $u_0=u_3-3r=7-3\times2=\bx{1}$.
\item[(ب)] $u_n=u_0+nr$ ، إذن من أجل كل عدد طبيعي $n$: \ \bx{u_n=1+2n}.
\end{enumerate}
\end{mithal}

\begin{khassia}[مجموع حدود متتابعة من متتالية حسابية]
\[\bx{S=\frac{\text{\ar{عدد الحدود}}}{2}\times\big(\text{\ar{الحد الأول}}+\text{\ar{الحد الأخير}}\big)}\]
حيث: \ \textbf{عدد الحدود} $\LW{=}$ دليل الحد الأخير $\LW{-}$ دليل الحد الأول $\LW{+1}$.\\
مثلاً: $\LW{S_n=u_0+u_1+\cdots+u_n=\dfrac{n+1}{2}\big(u_0+u_n\big)}$.
\end{khassia}

\begin{khassia}[اتجاه تغير متتالية حسابية]
نستنتج اتجاه تغير متتالية حسابية $\LW{(u_n)}$ حسب إشارة أساسها $\LW{r}$:
\begin{center}
\renewcommand{\arraystretch}{1.8}
\begin{tabular}{|M{0.27\textwidth}|M{0.27\textwidth}|M{0.27\textwidth}|}\hline
\hl{r>0} & \hl{r<0} & \hl{r=0}\\\hline
$\LW{(u_n)}$ متزايدة تماماً & $\LW{(u_n)}$ متناقصة تماماً & $\LW{(u_n)}$ ثابتة\\\hline
\end{tabular}
\end{center}
\end{khassia}

\subsection{المتتالية الهندسية}
\begin{tarif}
نقول إن $\LW{(v_n)}$ متتالية \textbf{هندسية} إذا وُجد عدد حقيقي $\LW{q}$ (يسمى \textbf{الأساس}) بحيث من أجل كل عدد طبيعي $\LW{n}$: \ \bx{v_{n+1}=q\times v_n}.\\
\textbf{للبرهان أن $\LW{(v_n)}$ هندسية:} نحسب عبارة $\LW{v_{n+1}}$ فنجدها $\LW{v_n}$ مضروبة في عدد ثابت $\LW{q}$ ، أو (إذا كانت حدودها غير معدومة) نجد $\LW{\dfrac{v_{n+1}}{v_n}=q}$.
\end{tarif}
\begin{mithal}
نعتبر المتتالية العددية $(u_n)$ المعرفة على $\N$ بـ: $u_0=-3$ ، ومن أجل كل عدد طبيعي $n$: \ $u_{n+1}=2u_n+3$.\\
ونضع من أجل كل عدد طبيعي $n$: \ $v_n=u_n+3$.
\begin{itemize}
\item برهن أن المتتالية $(v_n)$ هندسية.
\end{itemize}
\textbf{الحل:} لدينا $v_n=u_n+3$ ومنه $u_n=v_n-3$. من أجل كل عدد طبيعي $n$:
\begin{align*}
v_{n+1}&=u_{n+1}+3=(2u_n+3)+3=2u_n+6\\
&=2(v_n-3)+6=2v_n-6+6=2v_n
\end{align*}
إذن $(v_n)$ متتالية هندسية أساسها \bx{q=2} وحدها الأول $v_0=u_0+3=-3+3=0$.
\end{mithal}

\begin{khassia}[عبارة الحد العام لمتتالية هندسية]
إذا كانت $\LW{(v_n)}$ هندسية أساسها $\LW{q}$ فإنه من أجل كل عددين طبيعيين $\LW{n}$ و $\LW{p}$:
\[\bx{v_n=v_p\times q^{\,n-p}}\qquad\text{\ar{وبصفة خاصة (الحد الأول $\LW{v_0}$):}}\qquad\bx{v_n=v_0\times q^{\,n}}\]
وتستعمل في حساب الأساس إذا عُلم حدان: \ \bx{q^{\,n-p}=\dfrac{v_n}{v_p}}.
\end{khassia}
\begin{mithal}
$(u_n)$ متتالية هندسية أساسها $3$ وحدها الأول $u_0$ بحيث: $u_0+u_3=28$.
\begin{itemize}
\item أحسب $u_0$ ، ثم أكتب $u_n$ بدلالة $n$.
\end{itemize}
\textbf{الحل:} لدينا $q=3$ و $u_3=u_0\times q^3=27u_0$. ومنه:
\[u_0+u_3=28\ \Longleftrightarrow\ u_0+27u_0=28\ \Longleftrightarrow\ 28u_0=28\ \Longleftrightarrow\ \bx{u_0=1}.\]
إذن من أجل كل عدد طبيعي $n$: \ $u_n=u_0\times q^n=\bx{3^n}$.
\end{mithal}

\begin{khassia}[الوسط الهندسي]
إذا كانت $\LW{a}$ ، $\LW{b}$ و $\LW{c}$ ثلاثة حدود متتابعة من متتالية هندسية فإن: \ \bx{b^2=a\times c} \ أي \bx{b^3=a\times b\times c}.
\end{khassia}
\begin{mithal}
لتكن $(v_n)$ متتالية هندسية حدودها موجبة ، معرفة على $\N$ بـ: $v_0=2$ و $v_2=18$.
\begin{itemize}
\item أحسب $v_1$ ، ثم استنتج أساس المتتالية $(v_n)$.
\end{itemize}
\textbf{الحل:} بتطبيق خاصية الوسط الهندسي: $v_1^2=v_0\times v_2=2\times18=36$ ، ومنه $v_1=6$ أو $v_1=-6$.\\
وبما أن حدود المتتالية موجبة فإن \bx{v_1=6} ($v_1=-6$ مرفوض).\\
الأساس: $q=\dfrac{v_1}{v_0}=\dfrac62=\bx{3}$.
\end{mithal}

\begin{khassia}[مجموع حدود متتابعة من متتالية هندسية أساسها $q\ne1$]
\[\bx{S=\text{\ar{الحد الأول}}\times\frac{1-q^{\,\text{\ar{عدد الحدود}}}}{1-q}}\]
حيث: \ \textbf{عدد الحدود} $\LW{=}$ دليل الحد الأخير $\LW{-}$ دليل الحد الأول $\LW{+1}$.\\
مثلاً: $\LW{S_n=v_0+v_1+\cdots+v_n=v_0\times\dfrac{1-q^{\,n+1}}{1-q}}$.
\end{khassia}

\begin{khassia}[اتجاه تغير متتالية هندسية $v_n=v_0\,q^n$]
نستنتج اتجاه تغير متتالية هندسية حسب إشارة حدها الأول $\LW{v_0}$ وقيمة أساسها $\LW{q}$:
\begin{center}
\renewcommand{\arraystretch}{1.8}
\begin{tabular}{|M{0.16\textwidth}|M{0.22\textwidth}|M{0.18\textwidth}|M{0.22\textwidth}|}\hline
 & \hl{0<q<1} & \hl{q=1} & \hl{q>1}\\\hline
\hl{v_0>0} & متناقصة تماماً & ثابتة & متزايدة تماماً\\\hline
\hl{v_0<0} & متزايدة تماماً & ثابتة & متناقصة تماماً\\\hline
\end{tabular}
\end{center}
\begin{itemize}
\item إذا كان $\LW{q<0}$ فإن المتتالية الهندسية $\LW{(v_n)}$ (بحد أول غير معدوم) \textbf{ليست رتيبة} (إشارات حدودها متناوبة).
\item إذا كان $\LW{q=0}$ فإن المتتالية $\LW{(v_n)}$ ثابتة (معدومة) ابتداءً من الحد $\LW{v_1}$.
\end{itemize}
\end{khassia}

\begin{khassia}[نهاية المتتالية $(q^n)$]
\begin{itemize}
\item إذا كان $\LW{-1<q<1}$ فإن \bx{\lim_{n\to+\infty}q^n=0}.
\item إذا كان $\LW{q=1}$ فإن \bx{\lim_{n\to+\infty}q^n=1}.
\item إذا كان $\LW{q>1}$ فإن \bx{\lim_{n\to+\infty}q^n=+\infty}.
\item إذا كان $\LW{q\le-1}$ فإن المتتالية $\LW{(q^n)}$ ليس لها نهاية.
\end{itemize}
\end{khassia}

\section{البرهان بالتراجع}
\begin{tariqa}[مبدأ الاستدلال بالتراجع]
لتكن $P(n)$ خاصية متعلقة بعدد طبيعي $n$ ، و $n_0$ عدداً طبيعياً.
للبرهان على صحة الخاصية $P(n)$ من أجل كل عدد طبيعي $n\ge n_0$ يكفي:
\begin{enumerate}
\item \textbf{التحقق:} نتأكد من صحة الخاصية من أجل $n_0$ أي صحة $P(n_0)$.
\item \textbf{الوراثة:} نفرض أن الخاصية صحيحة من أجل عدد طبيعي كيفي $n\ge n_0$ أي $P(n)$ صحيحة ، ونبرهن صحة الخاصية من أجل $n+1$ أي $P(n+1)$.
\item \textbf{الخلاصة:} من أجل كل عدد طبيعي $n$ أكبر من أو يساوي $n_0$ ، $P(n)$ صحيحة.
\end{enumerate}
\begin{center}
\begin{LTR}
\begin{tikzpicture}[font=\small]
\draw[-{Stealth[length=2.5mm]},thick] (0,0)--(11,0);
\foreach \x/\l in {1.5/{$P(n_0)$},5.2/{$P(n)$},8.8/{$P(n+1)$}}{
 \fill (\x,0) circle (2pt);
 \node[above=3pt] at (\x,0){\l};
 \node[below=3pt,font=\footnotesize] at (\x,0){\ar{صحيحة}};}
\draw[-{Stealth[length=2mm]},cThm,thick] (5.2,0.8) to[bend left=40] node[above,font=\footnotesize]{\ar{فإن}} (8.8,0.8);
\node[cThm,font=\footnotesize,left] at (5.0,0.95){\ar{إذا كانت}};
\node[cDef,font=\footnotesize\bfseries] at (1.5,-1){\ar{المرحلة 1: التحقق}};
\node[cDef,font=\footnotesize\bfseries] at (7,-1){\ar{المرحلة 2: الوراثة}};
\end{tikzpicture}
\end{LTR}
\end{center}
\end{tariqa}

\begin{mithal}
لتكن $(u_n)$ المتتالية المعرفة على $\N$ بـ: $u_0=1$ ، ومن أجل كل عدد طبيعي $n$: \ $u_{n+1}=\dfrac{3u_n+2}{4}$.
\begin{itemize}
\item برهن بالتراجع أنه من أجل كل عدد طبيعي $n$: \ $u_n<2$.
\end{itemize}
\textbf{الحل:} لتكن $P(n)$ الخاصية: «$u_n<2$».
\begin{itemize}
\item \textbf{التحقق:} من أجل $n=0$: $u_0=1<2$ ، ومنه $P(0)$ صحيحة.
\item \textbf{الوراثة:} نفرض أن $P(n)$ صحيحة من أجل عدد طبيعي كيفي $n$ (أي $u_n<2$) ونبرهن صحة $P(n+1)$ (أي $u_{n+1}<2$):\\
لدينا $u_n<2$ ، بضرب الطرفين في العدد الموجب $3$: $3u_n<6$ ، وبإضافة العدد $2$: $3u_n+2<8$ ، وبضرب الطرفين في العدد الموجب $\dfrac14$: $\dfrac{3u_n+2}{4}<2$ ، أي $u_{n+1}<2$ ، ومنه $P(n+1)$ صحيحة.
\item \textbf{الخلاصة:} حسب مبدأ الاستدلال بالتراجع فإن $P(n)$ صحيحة من أجل كل عدد طبيعي $n$ ، أي $u_n<2$.
\end{itemize}
\end{mithal}

\section{المتتالية المحدودة من الأعلى ، المحدودة من الأسفل والمحدودة}
\begin{tarif}
لتكن $\LW{(u_n)}$ متتالية عددية معرفة على $\LW{\N}$.
\begin{itemize}
\item نقول إن $\LW{(u_n)}$ \textbf{محدودة من الأعلى} إذا وُجد عدد حقيقي $\LW{A}$ بحيث من أجل كل عدد طبيعي $\LW{n}$: \ \bx{u_n\le A}.
\item نقول إن $\LW{(u_n)}$ \textbf{محدودة من الأسفل} إذا وُجد عدد حقيقي $\LW{B}$ بحيث من أجل كل عدد طبيعي $\LW{n}$: \ \bx{u_n\ge B}.
\item نقول إن $\LW{(u_n)}$ \textbf{محدودة} إذا كانت محدودة من الأعلى ومحدودة من الأسفل.
\end{itemize}
\end{tarif}

\section{المتتالية المتقاربة}
\begin{tarif}
نقول إن المتتالية $\LW{(u_n)}$ \textbf{متقاربة} إذا كانت لها نهاية منتهية: \ \bx{\lim_{n\to+\infty}u_n=\ell} \ حيث $\LW{\ell}$ عدد حقيقي.
وإذا لم تكن متقاربة نقول إنها \textbf{متباعدة}.
\end{tarif}
\begin{khassia}[مبرهنة التقارب]
\begin{itemize}
\item كل متتالية \textbf{متزايدة} ($\LW{u_{n+1}-u_n\ge0}$) و\textbf{محدودة من الأعلى} ($\LW{u_n\le a}$) هي متتالية متقاربة.
\item كل متتالية \textbf{متناقصة} ($\LW{u_{n+1}-u_n\le0}$) و\textbf{محدودة من الأسفل} ($\LW{u_n\ge a}$) هي متتالية متقاربة.
\end{itemize}
\end{khassia}

\section{ملاحظات مهمة: المتتاليات في المسائل الاقتصادية}
لتكن $(u_n)$ متتالية معرفة في مسألة اقتصادية ، حيث $u_n$ هو المقدار القديم و $u_{n+1}$ هو المقدار الجديد (بعد سنة ، شهر ، $\ldots$).
\begin{khassia}[الزيادة والتخفيض بنسبة مئوية]
\begin{itemize}
\item \textbf{الزيادة بـ $\LW{p\%}$}: المقدار الجديد هو المقدار القديم مضروباً في العدد $\LW{\Big(1+\dfrac{p}{100}\Big)}$ ، أي:
\[\bx{u_{n+1}=\Big(1+\frac{p}{100}\Big)\times u_n}\]
\item \textbf{التخفيض بـ $\LW{p\%}$}: المقدار الجديد هو المقدار القديم مضروباً في العدد $\LW{\Big(1-\dfrac{p}{100}\Big)}$ ، أي:
\[\bx{u_{n+1}=\Big(1-\frac{p}{100}\Big)\times u_n}\]
\item \textbf{الزيادة بـ $\LW{p\%}$ ثم إضافة أو طرح مقدار ثابت $\LW{a}$}: نضرب المقدار القديم في $\LW{\Big(1+\dfrac{p}{100}\Big)}$ ثم نضيف أو نطرح العدد $\LW{a}$ حسب سياق التمرين ، أي:
\[\bx{u_{n+1}=\Big(1+\frac{p}{100}\Big)\times u_n\pm a}\]
\item \textbf{التخفيض بـ $\LW{p\%}$ ثم إضافة أو طرح مقدار ثابت $\LW{a}$}: نضرب المقدار القديم في $\LW{\Big(1-\dfrac{p}{100}\Big)}$ ثم نضيف أو نطرح العدد $\LW{a}$ حسب سياق التمرين ، أي:
\[\bx{u_{n+1}=\Big(1-\frac{p}{100}\Big)\times u_n\pm a}\]
\end{itemize}
\end{khassia}
\begin{mithal}
\begin{itemize}
\item زيادة بـ $5\%$: \ $u_{n+1}=1{,}05\,u_n$.
\item تخفيض بـ $20\%$: \ $u_{n+1}=0{,}8\,u_n$.
\item ينخفض عدد السكان بـ $4\%$ سنوياً ويستقر $500$ ساكن جديد: \ $u_{n+1}=0{,}96\,u_n+500$.
\end{itemize}
\end{mithal}

\section{مخطط حل التمرين النموذجي في البكالوريا}
\begin{tariqa}[التمرين النموذجي: $u_{n+1}=a\,u_n+b$ \ مع $a\ne1$]
\begin{enumerate}
\item \textbf{النقطة الصامدة:} نحل المعادلة $\ell=a\ell+b$ فنجد $\ell=\dfrac{b}{1-a}$ ، وهي غالباً العدد الذي يُطلب البرهان بالتراجع على أن $u_n\le\ell$ أو $u_n\ge\ell$.
\item \textbf{الرتابة:} ندرس إشارة الفرق: $u_{n+1}-u_n=(a-1)u_n+b=(a-1)(u_n-\ell)$.
\item \textbf{التقارب:} متتالية متزايدة ومحدودة من الأعلى (أو متناقصة ومحدودة من الأسفل) هي متتالية متقاربة.
\item \textbf{المتتالية المساعدة:} $v_n=u_n-\ell$ متتالية هندسية أساسها $a$ ، لأن: $v_{n+1}=a u_n+b-\ell=a(u_n-\ell)=a\,v_n$.
\item \textbf{العبارة بدلالة $n$:} $v_n=v_0\times a^n$ ، ثم $u_n=v_n+\ell$.
\item \textbf{النهاية:} إذا كان $-1<a<1$ فإن $\displaystyle\lim_{n\to+\infty}a^n=0$ ومنه $\displaystyle\lim_{n\to+\infty}u_n=\ell$.
\item \textbf{المجموع:} $u_0+u_1+\cdots+u_n=(v_0+v_1+\cdots+v_n)+(n+1)\,\ell$.
\end{enumerate}
\end{tariqa}

\qc أكتب عبارة $v_n$ ثم $u_n$ بدلالة $n$.
\qd أحسب $\limn u_n$.
\end{sq}
\end{enumerate}
\end{tamrin}
\begin{hal}
\textbf{1.} نفرض $\alpha=-\dfrac83$ ، ونبرهن أنه من أجل كل عدد طبيعي $n$: $u_n=-\dfrac83$.
\Rec{$u_n=-\dfrac83$}{من أجل $n=0$: $u_0=\alpha=-\dfrac83$ ، إذن $P(0)$ صحيحة.}{لدينا $u_n=-\dfrac83$ ، ومنه:
\[u_{n+1}=\frac23u_n-\frac89=\frac23\times\Big(-\frac83\Big)-\frac89=-\frac{16}{9}-\frac89=-\frac{24}{9}=-\frac83.\]
إذن $P(n+1)$ صحيحة.}
ومنه من أجل كل عدد طبيعي $n$: $u_n=u_0=-\dfrac83$ ، أي المتتالية $(u_n)$ \textbf{ثابتة}.

\textbf{2.} $\alpha=2$ أي $u_0=2$.
\pa نعوّض في العلاقة التراجعية:
\begin{align*}
u_1&=\frac23u_0-\frac89=\frac23\times2-\frac89=\frac{12}{9}-\frac89=\bx{\frac49}\\
u_2&=\frac23u_1-\frac89=\frac23\times\frac49-\frac89=\frac{8}{27}-\frac{24}{27}=\bx{-\frac{16}{27}}
\end{align*}
\pb من أجل كل عدد طبيعي $n$:
\begin{align*}
v_{n+1}&=u_{n+1}+\frac83=\frac23u_n-\frac89+\frac{24}{9}=\frac23u_n+\frac{16}{9}\\
&=\frac23\Big(u_n+\frac{16}{9}\times\frac32\Big)=\frac23\Big(u_n+\frac83\Big)=\frac23\,v_n
\end{align*}
إذن $(v_n)$ متتالية هندسية أساسها \bx{q=\dfrac23} وحدها الأول $v_0=u_0+\dfrac83=2+\dfrac83=\bx{\dfrac{14}{3}}$.
\pc من أجل كل عدد طبيعي $n$: \ $v_n=v_0\times q^n=\bx{\dfrac{14}{3}\Big(\dfrac23\Big)^n}$.\\
ولدينا $v_n=u_n+\dfrac83$ ومنه $u_n=v_n-\dfrac83$ ، أي من أجل كل عدد طبيعي $n$: \ \bx{u_n=\dfrac{14}{3}\Big(\dfrac23\Big)^n-\dfrac83}.
\pd لدينا: $\limn u_n=\bx{-\dfrac83}$ لأنّ: \quad $\Car{\limn\Big(\dfrac23\Big)^n=0\qquad\left(-1<\dfrac23<1\right)}$
\end{hal}

$(u_n)$ متتالية عددية معرفة بـ: $u_0=1$ ، ومن أجل كل عدد طبيعي $n$: \ $u_{n+1}=\dfrac12u_n-1$.

\item \begin{sq}
\qa أثبت بالتراجع أنه من أجل كل عدد طبيعي $n$: $u_n\ge-2$.

\begin{hal}
\textbf{1.} حساب الحدود:
\begin{align*}
u_1&=\frac12u_0-1=\frac12-1=\bx{-\frac12}\\
u_2&=\frac12u_1-1=-\frac14-1=\bx{-\frac54}\\
u_3&=\frac12u_2-1=-\frac58-1=\bx{-\frac{13}{8}}
\end{align*}
\par\smallskip\textbf{2.} \textbf{(أ)} \Rec{$u_n\ge-2$}{من أجل $n=0$: $u_0=1\ge-2$ ، إذن $P(0)$ صحيحة.}{لدينا $u_n\ge-2$ ، بضرب الطرفين في العدد الموجب $\dfrac12$ نجد $\dfrac12u_n\ge-1$ ، وبإضافة $-1$ للطرفين نجد $\dfrac12u_n-1\ge-2$ أي $u_{n+1}\ge-2$ ، إذن $P(n+1)$ صحيحة.}
\pb من أجل كل عدد طبيعي $n$:
\[u_{n+1}-u_n=\frac12u_n-1-u_n=-\frac12u_n-1=-\frac12\big(u_n+2\big).\]
لدينا $u_n\ge-2$ أي $u_n+2\ge0$ ، ومنه $-\dfrac12(u_n+2)\le0$. إذن $u_{n+1}-u_n\le0$ والمتتالية $(u_n)$ \textbf{متناقصة}.\\
\textbf{الاستنتاج:} $(u_n)$ متناقصة ومحدودة من الأسفل بالعدد $-2$ ، إذن هي \textbf{متقاربة}.

\par\smallskip\textbf{3.} \textbf{(أ)} من أجل كل عدد طبيعي $n$:
\[v_{n+1}=u_{n+1}+2=\frac12u_n-1+2=\frac12u_n+1=\frac12\big(u_n+2\big)=\frac12\,v_n.\]
إذن $(v_n)$ هندسية أساسها \bx{q=\dfrac12} وحدها الأول $v_0=u_0+2=1+2=\bx{3}$.
\pb من أجل كل عدد طبيعي $n$: \ $v_n=v_0\,q^n=\bx{3\Big(\dfrac12\Big)^n}$ \ ، ومنه \ $u_n=v_n-2=\bx{3\Big(\dfrac12\Big)^n-2}$.
\pc لدينا: $\limn u_n=\bx{-2}$ لأنّ: \quad $\Car{\limn\Big(\dfrac12\Big)^n=0\qquad\left(-1<\dfrac12<1\right)}$
\pd لدينا $u_k=v_k-2$ ، إذن:
\[S_n=u_0+u_1+\cdots+u_n=(v_0+v_1+\cdots+v_n)-\underbrace{(2+2+\cdots+2)}_{n+1\ \text{\ar{حداً}}}=(v_0+\cdots+v_n)-2(n+1).\]
المجموع $v_0+\cdots+v_n$ فيه $n+1$ حداً من متتالية هندسية أساسها $\dfrac12$:
\[v_0+\cdots+v_n=v_0\times\frac{1-q^{n+1}}{1-q}=3\times\frac{1-\big(\frac12\big)^{n+1}}{\frac12}=6-6\Big(\frac12\Big)^{n+1}=6-3\Big(\frac12\Big)^{n}.\]
إذن: $S_n=6-3\Big(\dfrac12\Big)^n-2n-2$ ، أي \ \bx{S_n=4-2n-3\Big(\dfrac12\Big)^n}.
\end{hal}

\pc $(u_n)$ متزايدة ومحدودة من الأعلى بالعدد $2$ ، إذن هي \textbf{متقاربة}.

إذن $(v_n)$ هندسية أساسها \bx{q=\dfrac13} وحدها الأول $v_0=u_0-2=-1-2=\bx{-3}$.

\end{align*}
إذن من أجل كل عدد طبيعي $n$: \ \bx{S_n=2n-\dfrac52+\dfrac32\Big(\dfrac13\Big)^n}.
\end{hal}

\end{hal}
