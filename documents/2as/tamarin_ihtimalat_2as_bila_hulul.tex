\begin{tabular}{|C|c|c|c|c|c|c|c|}\hline

\hline

\end{tabular}

\end{enumerate}

\section{قانون الاحتمال ، الأمل والتباين}
% ------------------------------------------------- 2
\begin{tamrin}{التعرف على قانون احتمال}
\begin{enumerate}
\item أي الجداول التالية يمثل قانون احتمال على $\{x_1;x_2;x_3\}$؟ علّل.
\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{|C|c|c|c|}\hline
\rowcolor{cSec!12}\ar{القانون} & $x_1$ & $x_2$ & $x_3$\\
\hline
(1) & $\frac34$ & $\frac14$ & $\frac14$\\
\hline
(2) & $\frac26$ & $\frac16$ & $\frac36$\\
\hline
(3) & $\frac12$ & $\frac32$ & $\frac12$\\
\hline
(4) & $\frac34$ & $-\frac14$ & $\frac12$\\
\hline
\end{tabular}
\end{center}
\item نرمي حجر نرد مرقماً من $1$ إلى $6$ ، ونصف هذه التجربة بقانون الاحتمال التالي:
\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{|C|c|c|c|c|c|c|}\hline
\rowcolor{cSec!12}$x_i$ & $1$ & $2$ & $3$ & $4$ & $5$ & $6$\\
\hline
$p_i$ & $0{,}1$ & $0{,}2$ & $2\alpha$ & $0{,}15$ & $\alpha$ & $0{,}1$\\
\hline
\end{tabular}
\end{center}
\begin{sq}
\qa احسب العدد $\alpha$.
\qb احسب احتمال الحادثة $A$: «الحصول على رقم زوجي».
\qc احسب الأمل الرياضياتي $E$ لهذا القانون.
\end{sq}
\end{enumerate}
\end{tamrin}

% ------------------------------------------------- 3
\begin{tamrin}{إيجاد احتمالين مجهولين باستعمال الأمل}
$a$ و $b$ عددان حقيقيان. نعرف المتغير العشوائي $X$ الذي قانون احتماله هو:
\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{|C|c|c|c|c|c|}\hline
\rowcolor{cSec!12}$x_i$ & $-2$ & $-1$ & $0$ & $1$ & $3$\\
\hline
$P(X=x_i)$ & $0{,}25$ & $a$ & $b$ & $0{,}05$ & $0{,}25$\\
\hline
\end{tabular}
\end{center}
\begin{enumerate}
\item عيّن قيمتي $a$ و $b$ علماً أن: $E(X)=0$.
\item احسب التباين $V(X)$ والانحراف المعياري $\sigma(X)$.
\item احسب $P(X\ge0)$ و $P(X^2\le1)$.
\end{enumerate}
\end{tamrin}

% ------------------------------------------------- 4
\begin{tamrin}{اختيار من متعدد}
عيّن الاقتراح الصحيح الوحيد في كل حالة مع التعليل:
\begin{enumerate}
\item $A$ و $B$ حادثتان. إذا كان: $P(\overline A)=0{,}6$ ، $P(B)=0{,}5$ و $P(A\cup B)=0{,}65$ فإن $P(A\cap B)$ يساوي:
\ABC{$0{,}2$}{$0{,}25$}{$0{,}45$}
\item يحتوي صندوق على $3$ كريات بيضاء و $5$ كريات سوداء متشابهة. نسحب من الصندوق كريتين على التوالي مع إعادة الكرية المسحوبة إلى الصندوق في كل مرة. عدد الحالات الممكنة هو:
\ABC{$16$}{$56$}{$64$}
\item الجدول التالي يعرف قانون احتمال لتجربة عشوائية:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{|C|c|c|c|c|}\hline
\rowcolor{cSec!12}$x_i$ & $-2$ & $-1$ & $2$ & $3$\\
\hline
$p_i$ & $0{,}12$ & $0{,}50$ & $0{,}08$ & $0{,}30$\\
\hline
\end{tabular}
\end{center}
التباين $V$ يساوي (بالتقريب):
\ABC{$3{,}89$}{$3{,}68$}{$4$}
\item تجربة عشوائية تحتوي على $12$ حادثة بسيطة متساوية الاحتمال $\{x_1\}$ ، $\{x_2\}$ ، $\ldots$ ، $\{x_{12}\}$. \ $P\big(\{x_9\}\big)$ يساوي:
\ABC{$\frac{9}{12}$}{$\frac{1}{12}$}{$\frac{1}{9}$}
\item نرمي نرداً: $\Omega=\{1;2;3;4;5;6\}$. \ إذا كانت $A=\{2;3;5\}$ فإن $\overline A$ هي:
\ABC{$\{1;4;5;6\}$}{$\{1;4;6\}$}{$\{2;3;5\}$}
\item في كيس $5$ كريات ، نسحب كريتين في آن واحد. عدد الحالات الممكنة هو:
\ABC{$25$}{$20$}{$10$}
\end{enumerate}
\end{tamrin}
\begin{molahada}
في النص الأصلي للسؤال 1 كُتب $P(A\cup B)=0{,}45$ وهذا مستحيل لأن $P(A\cup B)\ge P(B)=0{,}5$ ، فصححناه إلى $0{,}65$.
\end{molahada}

\end{enumerate}
\end{tamrin}

\section{الحوادث وتساوي الاحتمال وخواص الاحتمال}

\begin{tamrin}{احتمالات من جدول إحصائي}
الجدول التالي يعطي توزيع $500$ تلميذ في إحدى الثانويات:
\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{|C|c|c|c|}\hline
\rowcolor{cSec!12}التلميذ & \ar{ذكور} & \ar{إناث} & \ar{المجموع}\\
\hline
يملك هاتفاً نقالاً & $240$ & $60$ & $300$\\
\hline
لا يملك هاتفاً نقالاً & $80$ & $120$ & $200$\\
\hline
\ar{المجموع} & $320$ & $180$ & $500$\\
\hline
\end{tabular}
\end{center}
نختار عشوائياً تلميذاً من الثانوية ونسمي الحوادث: \ $H$: «التلميذ المختار ذكر» ، \ $F$: «التلميذ المختار أنثى» ، \ $S$: «التلميذ يملك هاتفاً نقالاً».
\begin{enumerate}
\item احسب $P(H)$ ، $P(F)$ و $P(S)$.
\item احسب احتمال الحوادث التالية:
\begin{sq}
\qa «التلميذ المختار أنثى وتملك هاتفاً نقالاً».
\qb «التلميذ المختار لا يملك هاتفاً نقالاً».
\qc «التلميذ المختار ذكر أو يملك هاتفاً نقالاً».
\end{sq}
\item نفرض أن التلميذ المختار لا يملك هاتفاً نقالاً. ما احتمال أن يكون هذا التلميذ ذكراً؟
\end{enumerate}
\end{tamrin}

\begin{tamrin}{سحب كرية ثم كريتين مع الإرجاع}
كيس يحتوي على كريتين بيضاوين $B_1$ ، $B_2$ ، وثلاث كريات حمراء $R_0$ ، $R_1$ ، $R_3$ ، وكريتين سوداوين $N_2$ ، $N_3$ (الرقم يمثل الرقم المسجل على الكرية). الكريات لا نفرق بينها عند اللمس.\par
\textbf{الجزء الأول:} نسحب عشوائياً من الكيس كرية واحدة.
\begin{enumerate}
\item عيّن مجموعة إمكانيات الحادثتين $A$ و $B$ حيث: \ $A$: «الكرية المسحوبة تحمل رقماً من مضاعفات $3$» ، \ $B$: «الكرية المسحوبة تحمل رقماً أولياً».
\item احسب كلاً من: $P(A)$ ، $P(B)$ ، $P(A\cap B)$ ، ثم استنتج $P(A\cup B)$ و $P\big(\overline B\big)$.
\end{enumerate}
\textbf{الجزء الثاني:} نسحب عشوائياً كريتين من الكيس على التوالي مع إرجاع الكرية قبل السحب الموالي.
\begin{enumerate}[resume]
\item ما عدد النتائج الممكنة؟ \ احسب احتمال الحصول على كريتين تحملان نفس الرقم.
\item ليكن $X$ المتغير العشوائي الذي يرفق بكل عملية سحب جداء الرقمين الظاهرين.
\begin{sq}
\qa عيّن قيم المتغير العشوائي $X$ وقانون احتماله.
\qb احسب الأمل الرياضياتي $E(X)$.
\qc احسب: $P\big(-2023X^2+2022X+2006>0\big)$.
\end{sq}
\end{enumerate}
\end{tamrin}

\begin{tamrin}{رمي نرد مرتين}
نرمي زهرة نرد متوازنة ذات $6$ أوجه مرقمة من $1$ إلى $6$ للمرة الأولى ونسجل الرقم $\alpha$ ، ثم نرميها مرة ثانية ونسجل الرقم $\beta$.
\begin{enumerate}
\item احسب احتمال كل حادثة مما يلي:
\begin{sq}
\qa $A$: «$\alpha$ و $\beta$ يحققان المساواة $2\alpha+1=\beta$».
\qb $B$: «$\alpha$ و $\beta$ يحققان المتباينة $|\alpha-\beta|<1$».
\qc $C$: «$\alpha$ و $\beta$ يحققان المتباينة $|\alpha-\beta|>5$».
\end{sq}
\item $X$ المتغير العشوائي الذي يرفق بكل رميتين العدد $|\alpha-\beta|$.
\begin{sq}
\qa ما هي القيم الممكنة لـ $X$؟
\qb عيّن قانون احتمال $X$ ثم احسب $E(X)$.
\end{sq}
\item نعتبر المعادلة: $x^2+\alpha x+\beta=0$ \ ...$(E)$. \ ليكن $Y$ المتغير العشوائي الذي يرفق بكل رميتين عدد حلول المعادلة $(E)$ في $\R$. \ عرّف قانون احتمال $Y$ ثم احسب أمله الرياضياتي.
\end{enumerate}
\end{tamrin}

\begin{tamrin}{سحب مع الإرجاع ثم سحب في آن واحد}
في صندوق كريتان لونهما أحمر ، وثلاث كريات لونها أخضر ، وكرية سوداء (الكريات لا نفرق بينها عند اللمس).
\begin{enumerate}
\item نسحب عشوائياً كريتين على التوالي مع الإرجاع.
\begin{sq}
\qa ما هو عدد الإمكانيات؟
\qb احسب احتمال الحوادث: $A$: «كريتان لونهما أحمر» ، $B$: «كريتان من نفس اللون» ، $C$: «كريتان مختلفتان في اللون» ، $D$: «كرية خضراء على الأقل».
\qc ليكن $X$ المتغير العشوائي الذي يرفق بكل سحب عدد الكريات الخضراء المسحوبة. عرّف قانون احتماله ، واحسب أمله الرياضياتي وتباينه وانحرافه المعياري.
\end{sq}
\item نسحب الآن عشوائياً كريتين في آن واحد من هذا الصندوق. أعد الإجابة على الأسئلة السابقة.
\end{enumerate}
\end{tamrin}

\begin{tamrin}{سحب دون إرجاع ولعبة}
كيس به $7$ كريات متماثلة: منها $3$ سوداء و $4$ حمراء. نسحب عشوائياً كريتين على التوالي دون إرجاع.
\begin{enumerate}
\item ما هو عدد الإمكانيات؟
\item احسب احتمال كل حادثة: \ $A$: «سحب كريتين مختلفتين في اللون» ، \ $B$: «سحب كريتين من نفس اللون» ، \ $C$: «الحصول على كرية سوداء على الأقل».
\item نقترح اللعبة التالية: يدفع اللاعب $\alpha$ دج ($\alpha$ عدد طبيعي) ويسحب كريتين: إذا سحب كريتين سوداوين يحصل على $100$ دج ، وإذا سحب كريتين مختلفتين في اللون يحصل على $50$ دج ، وإذا سحب كريتين حمراوين يخسر ما دفعه. \ ليكن $X$ المتغير العشوائي الذي يمثل ربح أو خسارة اللاعب.
\begin{sq}
\qa برّر أن قيم $X$ هي: $\{100-\alpha;\,50-\alpha;\,-\alpha\}$ ، ثم عرّف قانون احتماله.
\qb أثبت أن: $E(X)=-\alpha+\dfrac{300}{7}$ ، ثم أوجد أكبر قيمة لـ $\alpha$ حتى تكون اللعبة في صالح اللاعب.
\end{sq}
\end{enumerate}
\end{tamrin}

\begin{tamrin}{مجموع رقمين}
كيس به $4$ كرات حمراء تحمل الأرقام $0$ ، $1$ ، $2$ ، $3$ ، وكريتان خضراوان تحملان الرقمين $0$ و $1$ (لا نميز بينها باللمس). نسحب من الكيس كريتين على التوالي دون إرجاع.
\begin{enumerate}
\item ما هو عدد الحالات الكلية للسحب؟
\item احسب احتمال كل من الأحداث الآتية: \ $A$: «الكريتان المسحوبتان من نفس اللون» ، \ $B$: «الكريتان تحملان نفس الرقم» ، \ $C$: «الكريتان مختلفتان في اللون ورقميهما مختلفين» ، \ $D$: «جداء الرقمين المسجلين على الكريتين معدوم».
\item نعتبر المتغير العشوائي $X$ الذي يربط كل عملية سحب بمجموع الرقمين المسجلين عليهما.
\begin{sq}
\qa عيّن قيم المتغير العشوائي $X$.
\qb عيّن قانون الاحتمال للمتغير العشوائي $X$ ، ثم احسب أمله الرياضياتي $E(X)$.
\qc احسب احتمال الحدث: «$X^2-X>0$».
\end{sq}
\end{enumerate}
\end{tamrin}

\begin{tamrin}{جداء الكريات المتبقية}
كيس غير شفاف يحوي $7$ كريات لا نفرق بينها عند اللمس: منها $3$ بيضاء مرقمة $1$ ، $1$ ، $5$ ، وكريتان حمراوان مرقمتان $0$ ، $5$ ، وكريتان خضراوان مرقمتان $1$ ، $5$. \ نسحب كريتين على التوالي دون إرجاع ونسمي الأحداث التالية:\par
$A$: «الكريتان من نفس اللون» ، $B$: «الكريتان من نفس الرقم» ، $C$: «الكريتان من نفس اللون ونفس الرقم» ، $E$: «جداء الرقمين معدوم» ، $F$: «الكريتان من نفس اللون أو نفس الرقم» ، $G$: «جداء الرقمين غير معدوم».
\begin{enumerate}
\item احسب احتمال الأحداث $A$ ، $B$ ، ثم بيّن أن: $P(C)=\dfrac{1}{21}$ و $P(E)=\dfrac27$.
\item استنتج $P(G)$ و $P(F)$.
\item نعتبر المتغير العشوائي $X$ الذي يرفق بكل عملية سحب جداء أرقام الكريات المتبقية في الكيس.
\begin{sq}
\qa بيّن أن قيم المتغير العشوائي $X$ هي: $\{0;25;125\}$.
\qb عرّف قانون احتمال المتغير العشوائي $X$ واحسب أمله الرياضياتي.
\qc احسب احتمال الحدث: $X^3-150X^2+5^5X=0$. \ كيف نسمي هذا الحدث؟
\end{sq}
\end{enumerate}
\end{tamrin}

\begin{tamrin}{متغير عشوائي معرّف بقاعدتين}
كيس غير شفاف به $5$ كريات متماثلة: كريتان بيضاوان مرقمتان بـ $1$ و $0$ ، وكرية خضراء مرقمة بـ $-1$ ، وكريتان حمراوان مرقمتان بـ $1$ ، $2$. نسحب عشوائياً على التوالي دون إرجاع كريتين ونسجل النتائج الظاهرة.
\begin{enumerate}
\item شكّل مخططاً توضح فيه جميع الحالات الممكنة.
\item احسب احتمال الحادثة $A$: «الحصول على كريتين مختلفتين في اللون وتحملان نفس الرقم».
\item ليكن $X$ المتغير العشوائي المعرف كما يلي: إذا كانت الكريتان من نفس اللون يحدد $X$ جداء رقميهما ، وإذا كانت مختلفتين في اللون يحدد $X$ مجموع رقميهما.
\begin{sq}
\qa عيّن القيم الممكنة لـ $X$ ، ثم أوجد قانون احتماله.
\qb احسب الأمل الرياضياتي $E(X)$ و $P(X\le1)$.
\end{sq}
\end{enumerate}
\end{tamrin}

\begin{tamrin}{سحب في آن واحد: عدد الكريات المتبقية}
يحتوي كيس على أربع كريات حمراء تحمل الأرقام $0$ ، $2$ ، $3$ ، $5$ ، وثلاث كريات خضراء تحمل الأرقام $1$ ، $2$ ، $7$ (لا نفرق بين الكريات باللمس). \ نسحب كريتين عشوائياً وفي آن واحد من هذا الكيس ونسجل رقميهما ولونيهما.
\begin{enumerate}
\item عيّن عدد عناصر مجموعة الإمكانيات $\Omega$.
\item احسب احتمالات الحوادث التالية: \ $A$: «الحصول على كريتين من نفس اللون» ، \ $B$: «الحصول على كرية حمراء على الأقل» ، \ $C$: «الحصول على كريتين تحملان نفس الرقم» ، \ $D$: «الحصول على كريتين مجموع رقميهما عدد أولي».
\item ليكن $X$ المتغير العشوائي الذي يرفق بكل عملية سحب عدد الكريات الحمراء المتبقية في الكيس.
\begin{sq}
\qa عيّن قيم $X$ وعرّف قانون احتماله.
\qb احسب الأمل الرياضياتي $E(X)$ والانحراف المعياري $\sigma(X)$.
\end{sq}
\end{enumerate}
\end{tamrin}
\begin{molahada}
في بعض الحلول المتداولة لهذا التمرين وُجد $P(D)=\dfrac{12}{21}$ و $\sigma(X)\approx0{,}62$ (بسبب تدوير $E(X)$ قبل حساب التباين). \ القيمتان الصحيحتان: $P(D)=\dfrac{11}{21}$ و $\sigma(X)=\dfrac{2\sqrt5}{7}\approx0{,}64$.
\end{molahada}

\begin{tamrin}{بطاقات مرقمة}
يحوي صندوق $5$ بطاقات متشابهة مرقمة من $5$ إلى $9$ ، لا نميز بينها باللمس. نسحب عشوائياً وفي آن واحد بطاقتين من الصندوق.
\begin{enumerate}
\item
\begin{sq}
\qa أنشئ مخططاً تبين فيه كل الحالات الممكنة.
\qb احسب احتمال الأحداث التالية: \ $A$: «إحدى البطاقتين المسحوبتين على الأقل تحمل رقماً يقبل القسمة على $3$» ، \ $B$: «سحب بطاقتين تحملان رقمين فرديين» ، \ $C$: «سحب بطاقتين مجموع رقميهما عدد أولي».
\end{sq}
\item نعرّف المتغير العشوائي $X$ الذي يرفق بكل عملية سحب مجموع الرقمين الظاهرين.
\begin{sq}
\qa عيّن مجموعة القيم الممكنة للمتغير العشوائي $X$.
\qb عرّف قانون احتمال $X$ ، ثم احسب أمله الرياضياتي $E(X)$.
\end{sq}
\end{enumerate}
\end{tamrin}

\begin{tamrin}{قريصات مع الإرجاع}
نضع في كيس $3$ قريصات تحمل الرقم $1$ ، وقريصتين تحملان الرقم $2$ ، وقريصة واحدة تحمل الرقم $3$. \ نسحب عشوائياً قريصتين على التوالي حيث نعيد القريصة المسحوبة إلى الكيس قبل السحب الموالي. ليكن $X$ المتغير العشوائي الذي يرفق بكل سحب مجموع الرقمين المسحوبين.
\begin{enumerate}
\item عيّن بالاستعانة بجدول مجموعة الإمكانيات (وعدد عناصرها).
\item عيّن القيم الممكنة للمتغير العشوائي $X$.
\item عيّن قانون الاحتمال لـ $X$ ، ثم احسب $P(X\le4)$.
\item احسب الأمل الرياضياتي $E(X)$.
\end{enumerate}
\end{tamrin}
\begin{molahada}
في النص الأصلي طُلب حساب $P(X\le1)$ ، وهي حادثة مستحيلة (أصغر مجموع $2$) فاحتمالها $0$ ؛ لذلك استبدلناها بـ $P(X\le4)$.
\end{molahada}

\section{مسائل: الألعاب والأمل الرياضياتي}

\begin{tamrin}{هل اللعبة عادلة؟}
يحتوي كيس على $4$ كريات مرقمة بـ $0$ ، و $3$ كريات مرقمة بـ $2$ ، وكريتين مرقمتين بـ $3$ (لا نميز بينها باللمس). يسحب لاعب عشوائياً كرية واحدة:
\begin{itemize}
\item إذا ظهر رقم زوجي يربح اللاعب ضعف الرقم الظاهر (بالدينار).
\item إذا ظهر رقم فردي يخسر ضعفه أيضاً.
\end{itemize}
نعتبر $X$ المتغير العشوائي الذي يرفق بكل عملية سحب قيمة الربح أو الخسارة الممكنة.
\begin{enumerate}
\item عيّن القيم الممكنة للمتغير $X$ ، ثم عرّف قانون احتماله.
\item احسب الأمل الرياضياتي $E(X)$. \ هل اللعبة في صالح اللاعب؟
\item احسب الانحراف المعياري $\sigma(X)$.
\end{enumerate}
\end{tamrin}
\begin{molahada}
في بعض المذكرات وُجد $E(X)\approx0{,}66$ وهو خطأ: الحساب الصحيح يعطي $E(X)=0$.
\end{molahada}

\begin{tamrin}{تعيين المبلغ العادل}
يرمي لاعب نرداً متوازناً مرة واحدة:
\begin{itemize}
\item إذا ظهر الرقم $6$ يربح $12$ دج.
\item إذا ظهر الرقم $4$ أو $5$ يربح $3$ دج.
\item في الحالات الأخرى يخسر $a$ دج ($a>0$).
\end{itemize}
ليكن $X$ ربح اللاعب الجبري.
\begin{enumerate}
\item عرّف قانون احتمال $X$ ، ثم احسب $E(X)$ بدلالة $a$.
\item عيّن قيمة $a$ حتى تكون اللعبة عادلة ، ثم احسب $\sigma(X)$ في هذه الحالة.
\item ما هي قيم $a$ التي تجعل اللعبة في صالح اللاعب؟
\end{enumerate}
\end{tamrin}

\begin{tamrin}{تمرين شامل: احتمال أعظمي (توظيف المشتقة)}
يحتوي كيس على $n$ قريصة متماثلة لا نفرق بينها عند اللمس ($n\in\N$ و $n\ge7$) ، منها $7$ بيضاء والباقي حمراء. نسحب قريصتين على التوالي وبدون إرجاع.
\begin{enumerate}
\item نفرض في هذا السؤال أن $n=10$. احسب احتمالات الحوادث التالية:
\begin{sq}
\qa $A$: «القريصة الأولى بيضاء والثانية حمراء».
\qb $B$: «قريصة بيضاء وقريصة حمراء».
\qc $C$: «القريصتان بيضاوان».
\qd $D$: «القريصتان من نفس اللون».
\end{sq}
\item نعتبر الآن $n\ge8$. نرمز بـ $P_n$ إلى احتمال أن تكون القريصتان المسحوبتان من لونين مختلفين.
\begin{sq}
\qa أثبت أن: $P_n=\dfrac{14(n-7)}{n^2-n}$.
\qb بدراسة الدالة $f:x\mapsto\dfrac{14(x-7)}{x^2-x}$ على $[8;+\infty[$ ، عيّن قيم العدد الطبيعي $n$ الذي يكون من أجله لـ $P_n$ قيمة حدية عظمى ، ثم عيّن هذه القيمة.
\end{sq}
\end{enumerate}
\end{tamrin}
