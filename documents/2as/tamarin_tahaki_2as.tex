\section{التعريف والخاصية المميزة}
% ------------------------------------------------- 1
\begin{tamrin}{تعيين تحاكيات}
\begin{enumerate}
\item $A$ و $B$ نقطتان متمايزتان ، و $I$ نقطة حيث $\V{IB}=-\dfrac12\V{IA}$. \ عيّن تحاكياً يحوّل $A$ إلى $B$.
\item $O$ ، $P$ و $Q$ نقط حيث $\V{OQ}=3\V{OP}$. عيّن تحاكياً يحوّل $P$ إلى $Q$.
\item $ABCD$ متوازي أضلاع. هل يوجد تحاكٍ يحوّل $A$ إلى $C$ و $B$ إلى $D$؟ عيّنه.
\item $A$ ، $B$ نقطتان و $f$ التحويل النقطي الذي يرفق بكل نقطة $M$ النقطة $M'$ حيث: $\V{MM'}=2\V{MA}+\V{MB}$.\par
أنشئ النقطة $G$ مرجح الجملة $\{(A;2);(B;1)\}$ ، وأثبت أن $\V{GM'}=-2\V{GM}$ ، ثم استنتج طبيعة $f$.
\end{enumerate}
\end{tamrin}
\begin{hal}
\textbf{1.} $\V{IB}=-\dfrac12\V{IA}$ معناه $B=h(A)$ حيث \bx{h=h\Big(I;-\dfrac12\Big)}.
\QQ{2} $\V{OQ}=3\V{OP}$ معناه $Q=h(P)$ حيث \bx{h=h(O;3)}.
\QQ{3} لدينا $\V{CD}=-\V{AB}$ (متوازي أضلاع) ، فالنسبة $k=-1$. \ المركز هو تقاطع القطرين $(AC)$ و $(BD)$ ، أي مركز متوازي الأضلاع $O$:\par
التحاكي هو \bx{h(O;-1)} ، أي التناظر المركزي الذي مركزه $O$.
\QQ{4} $3\ne0$ ، فـ $G$ موجودة و $\V{AG}=\dfrac13\V{AB}$. \ بالاختزال: $2\V{MA}+\V{MB}=3\V{MG}$ ، ومنه $\V{MM'}=3\V{MG}$.
\[\V{GM'}=\V{GM}+\V{MM'}=\V{GM}+3\V{MG}=\V{GM}-3\V{GM}=-2\V{GM}.\]
إذن $f$ هو \bx{\text{\ar{التحاكي}}\ h(G;-2)}.
\end{hal}

% ------------------------------------------------- 2
\begin{tamrin}{مثلث قائم ومساحات}
$ABC$ مثلث قائم في $A$ حيث $AB=2$ و $AC=3$. \ $D$ نقطة من المستوي تحقق: $2\V{AD}=\V{DB}$.
\begin{enumerate}
\item بيّن أن النقطة $D$ هي صورة النقطة $B$ بالتحاكي $h$ الذي مركزه $A$ يطلب تعيين نسبته.
\item $E$ صورة $C$ بالتحاكي $h$. حدّد طبيعة المثلث $ADE$ واحسب مساحته.
\item لتكن النقطة $F$ مرجح النقطتين $A$ و $C$ المرفقتين بالمعاملين $2$ و $-3$ على الترتيب. \ بيّن أن النقطة $F$ هي صورة $A$ بالتحاكي $h'$ الذي مركزه $C$ يطلب تعيين نسبته.
\item نعتبر النقطتين $I$ و $H$ صورتي النقطتين $D$ و $E$ بالتحاكي $h'$ على الترتيب. حدّد طبيعة المثلث $FHI$ ثم استنتج مساحته.
\end{enumerate}
\end{tamrin}
\begin{hal}
\textbf{1.} $2\V{AD}=\V{DB}=\V{DA}+\V{AB}=-\V{AD}+\V{AB}$ ، ومنه $3\V{AD}=\V{AB}$ ، أي $\V{AD}=\dfrac13\V{AB}$: \ $D=h(B)$ حيث \bx{h=h\Big(A;\dfrac13\Big)}.
\QQ{2} $h$ يحافظ على الزوايا ، و $h(A)=A$ ، $h(B)=D$ ، $h(C)=E$: المثلث $ADE$ \textbf{قائم في $A$} ، مع $AD=\dfrac23$ و $AE=1$.\par
$S_{ABC}=\dfrac12\times2\times3=3$ ، و $S_{ADE}=\Big(\dfrac13\Big)^2\times3=\bx{\dfrac13}$ \ (تحقق: $\dfrac12\times\dfrac23\times1=\dfrac13$ \ok).
\QQ{3} $2\V{FA}-3\V{FC}=\vec0$ ، أي $2\big(\V{FC}+\V{CA}\big)-3\V{FC}=\vec0$ ، ومنه $\V{FC}=2\V{CA}$ ، أي $\V{CF}=-2\V{CA}$:\par
$F=h'(A)$ حيث \bx{h'=h(C;-2)}.
\QQ{4} $h'(A)=F$ ، $h'(D)=I$ ، $h'(E)=H$: المثلث $FIH$ صورة المثلث $ADE$ ، فهو \textbf{قائم في $F$}.\par
$S_{FIH}=(-2)^2\times S_{ADE}=4\times\dfrac13=\bx{\dfrac43}$.
\end{hal}

% ------------------------------------------------- 3
\begin{tamrin}{صحيح أم خطأ}
أجب بصحيح أو خطأ مع التعليل:
\begin{enumerate}
\item النقطة $C$ مرجح الجملة $\{(A;2);(B;-3)\}$ معناه $C$ هي صورة $B$ بالتحاكي $h$ ذي المركز $A$ والنسبة $k=3$.
\item $ABC$ مثلث حيث $\widehat{BAC}=60^\circ$ ، $AB=5\ \text{cm}$ و $AC=7\ \text{cm}$:
\begin{sq}
\qa $BC=\sqrt{39}\ \text{cm}$.
\qb مساحة المثلث $A'B'C'$ صورة $ABC$ بالتحاكي الذي نسبته $k=-\dfrac13$ هي $S=\dfrac{35\sqrt3}{36}\ \text{cm}^2$.
\end{sq}
\item صورة مستقيم بتحاكٍ هي دائماً مستقيم يختلف عنه.
\item التحاكي الذي نسبته $-1$ هو تناظر مركزي.
\item إذا كان $h(A)=A'$ و $h(B)=B'$ حيث $AB=4$ و $A'B'=2$ فإن نسبة $h$ هي حتماً $\dfrac12$.
\end{enumerate}
\end{tamrin}
\begin{hal}
\textbf{1. صحيح:} $2\V{CA}-3\V{CB}=\vec0$ ، ومنه $\V{AC}=\dfrac{-3}{2-3}\V{AB}=3\V{AB}$ ، أي $C=h(A;3)(B)$.
\QQ{2} \pa \textbf{صحيح:} بقانون الكاشي (العلاقات المترية): $BC^2=AB^2+AC^2-2\,AB\cdot AC\cos60^\circ=25+49-35=39$.
\pb \textbf{صحيح:} $S_{ABC}=\dfrac12\times5\times7\times\dfrac{\sqrt3}{2}=\dfrac{35\sqrt3}{4}$ ، و $S'=\dfrac19\times\dfrac{35\sqrt3}{4}=\dfrac{35\sqrt3}{36}$.
\QQ{3} \textbf{خطأ:} صورة مستقيم يشمل المركز هي المستقيم نفسه.
\QQ{4} \textbf{صحيح:} $\V{\Omega M'}=-\V{\Omega M}$ ، أي $\Omega$ منتصف $[MM']$.
\QQ{5} \textbf{خطأ:} $A'B'=|k|AB$ يعطي $|k|=\dfrac12$ ، فالنسبة $\dfrac12$ أو $-\dfrac12$.
\end{hal}

\section{الصور ، الأطوال والمساحات}
% ------------------------------------------------- 4
\begin{tamrin}{صورة مستطيل}
$ABCD$ مستطيل مركزه $O$ ، و $I$ منتصف القطعة $[AB]$ ، و $G$ نقطة تقاطع المستقيمين $(IC)$ و $(BD)$. \ نرمز بـ $h$ إلى التحاكي الذي مركزه $G$ ونسبته $-\dfrac12$.
\begin{enumerate}
\item بيّن أن $G$ مركز ثقل المثلث $ABC$.
\item عيّن صور النقط $A$ ، $B$ ، $C$ و $D$ بالتحاكي $h$ ، ثم أرسم صورة المستطيل $ABCD$.
\item احسب محيط ومساحة صورة المستطيل $ABCD$ إذا علمت أن $AD=4\ \text{cm}$ و $AB=10\ \text{cm}$.
\end{enumerate}
\end{tamrin}
\begin{hal}
\textbf{1.} في المثلث $ABC$: $(CI)$ متوسط (لأن $I$ منتصف $[AB]$) ، و $(BO)$ متوسط (لأن $O$ منتصف $[AC]$) و $(BO)=(BD)$. \ $G$ تقاطع متوسطين: \bx{G\ \text{\ar{مركز ثقل}}\ ABC}.
