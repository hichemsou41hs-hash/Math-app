\section{التواترات والمحاكاة}
% ------------------------------------------------- 1
\begin{tamrin}{استقرار التواترات}
رمى تلميذ زهرة نرد $600$ مرة وسجل النتائج في الجدول التالي:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{|C|c|c|c|c|c|c|c|}\hline
\rowcolor{cSec!12}الرقم الظاهر & $1$ & $2$ & $3$ & $4$ & $5$ & $6$ & \ar{المجموع}\\
\hline
التكرار $n_i$ & $95$ & $104$ & $98$ & $107$ & $96$ & $100$ & $600$\\
\hline
\end{tabular}
\end{center}
\begin{enumerate}
\item احسب تواتر ظهور كل رقم (تُدوَّر النتائج إلى $10^{-3}$).
\item قارن هذه التواترات بالعدد $\dfrac16$. \ ماذا تستنتج حول النرد؟
\item كيف يمكن محاكاة التجربة العشوائية «نجاح أو رسوب $15$ تلميذاً» (باحتمالين متساويين) باستعمال قطعة نقدية؟
\end{enumerate}
\end{tamrin}
\begin{hal}
\textbf{1.} $f_i=\dfrac{n_i}{600}$:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{|C|c|c|c|c|c|c|c|}\hline
\rowcolor{cSec!12}الرقم & $1$ & $2$ & $3$ & $4$ & $5$ & $6$ & \ar{المجموع}\\
\hline
التواتر $f_i$ & $0{,}158$ & $0{,}173$ & $0{,}163$ & $0{,}178$ & $0{,}160$ & $0{,}167$ & $1$\\
\hline
\end{tabular}
\end{center}
(تحقق: مجموع التكرارات $95+104+98+107+96+100=600$ ، ومجموع التواترات يساوي $1$.)

\QQ{2} $\dfrac16\approx0{,}167$. \ كل التواترات قريبة من $0{,}167$ (الفروق أصغر من $0{,}012$) ، وهي تذبذبات عادية للعينة.\par
نستنتج أن النرد \textbf{متوازن} على الأرجح: احتمال ظهور كل رقم هو $\dfrac16$ ، وكلما زاد عدد الرميات اقتربت التواترات أكثر من $\dfrac16$.

\QQ{3} نرمي قطعة نقدية $15$ مرة ، ونرفق الوجه $F$ بالنجاح والظهر $P$ بالرسوب.\par
مثلاً العينة $\{P;P;F;F;F;P;F;P;F;F;P;F;P;F;F\}$ تعني نجاح $9$ تلاميذ ورسوب $6$.
\end{hal}

\section{قانون الاحتمال ، الأمل والتباين}
% ------------------------------------------------- 2
\begin{tamrin}{التعرف على قانون احتمال}
\begin{enumerate}
\item أي الجداول التالية يمثل قانون احتمال على $\{x_1;x_2;x_3\}$؟ علّل.
\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{|C|c|c|c|}\hline
\rowcolor{cSec!12}\ar{القانون} & $x_1$ & $x_2$ & $x_3$\\
\hline
(1) & $\frac34$ & $\frac14$ & $\frac14$\\
\hline
(2) & $\frac26$ & $\frac16$ & $\frac36$\\
\hline
(3) & $\frac12$ & $\frac32$ & $\frac12$\\
\hline
(4) & $\frac34$ & $-\frac14$ & $\frac12$\\
\hline
\end{tabular}
\end{center}
\item نرمي حجر نرد مرقماً من $1$ إلى $6$ ، ونصف هذه التجربة بقانون الاحتمال التالي:
\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{|C|c|c|c|c|c|c|}\hline
\rowcolor{cSec!12}$x_i$ & $1$ & $2$ & $3$ & $4$ & $5$ & $6$\\
\hline
$p_i$ & $0{,}1$ & $0{,}2$ & $2\alpha$ & $0{,}15$ & $\alpha$ & $0{,}1$\\
\hline
\end{tabular}
\end{center}
\begin{sq}
\qa احسب العدد $\alpha$.
\qb احسب احتمال الحادثة $A$: «الحصول على رقم زوجي».
\qc احسب الأمل الرياضياتي $E$ لهذا القانون.
\end{sq}
\end{enumerate}
\end{tamrin}
\begin{hal}
\textbf{1.} نتحقق من الشرطين: كل $p_i\in[0;1]$ ، و $\sum p_i=1$.
\begin{itemize}
\item (1): $\dfrac34+\dfrac14+\dfrac14=\dfrac54\ne1$: \ \textbf{لا يمثل} قانون احتمال.
\item (2): $\dfrac26+\dfrac16+\dfrac36=1$ وكلها بين $0$ و $1$: \ \textbf{يمثل} قانون احتمال.
\item (3): $\dfrac32>1$: \ \textbf{لا يمثل} قانون احتمال.
\item (4): $-\dfrac14<0$ (رغم أن المجموع يساوي $1$): \ \textbf{لا يمثل} قانون احتمال.
\end{itemize}

\QQ{2} \pa مجموع الاحتمالات يساوي $1$: $0{,}1+0{,}2+2\alpha+0{,}15+\alpha+0{,}1=1$ ، أي $3\alpha+0{,}55=1$ ، ومنه $3\alpha=0{,}45$: \ \bx{\alpha=0{,}15}.
\pb $A=\{2;4;6\}$: \ $P(A)=0{,}2+0{,}15+0{,}1=\bx{0{,}45}$.
\pc $E=1\times0{,}1+2\times0{,}2+3\times0{,}3+4\times0{,}15+5\times0{,}15+6\times0{,}1$\par
$E=0{,}1+0{,}4+0{,}9+0{,}6+0{,}75+0{,}6=\bx{3{,}35}$.
\end{hal}

\begin{tamrin}{إيجاد احتمالين مجهولين باستعمال الأمل}
$a$ و $b$ عددان حقيقيان. نعرف المتغير العشوائي $X$ الذي قانون احتماله هو:
\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{|C|c|c|c|c|c|}\hline
\rowcolor{cSec!12}$x_i$ & $-2$ & $-1$ & $0$ & $1$ & $3$\\
\hline
$P(X=x_i)$ & $0{,}25$ & $a$ & $b$ & $0{,}05$ & $0{,}25$\\
\hline
\end{tabular}
\end{center}
\begin{enumerate}
\item عيّن قيمتي $a$ و $b$ علماً أن: $E(X)=0$.
\item احسب التباين $V(X)$ والانحراف المعياري $\sigma(X)$.
\item احسب $P(X\ge0)$ و $P(X^2\le1)$.
\end{enumerate}
\end{tamrin}
\begin{hal}
\textbf{1.} \textbf{شرط الأمل:} $E(X)=-2\times0{,}25-1\times a+0\times b+1\times0{,}05+3\times0{,}25=0$\par
أي $-0{,}5-a+0{,}05+0{,}75=0$ ، ومنه $-a+0{,}3=0$: \ \bx{a=0{,}3}.\par
\textbf{مجموع الاحتمالات:} $0{,}25+0{,}3+b+0{,}05+0{,}25=1$ ، أي $b+0{,}85=1$: \ \bx{b=0{,}15}.

\QQ{2} $\displaystyle\sum x_i^2p_i=4\times0{,}25+1\times0{,}3+0+1\times0{,}05+9\times0{,}25=1+0{,}3+0{,}05+2{,}25=3{,}6$.\par
$V(X)=3{,}6-0^2=\bx{3{,}6}$ ، \quad $\sigma(X)=\sqrt{3{,}6}\approx\bx{1{,}90}$.

\QQ{3} $P(X\ge0)=P(X=0)+P(X=1)+P(X=3)=0{,}15+0{,}05+0{,}25=\bx{0{,}45}$.\par
$X^2\le1\iff-1\le X\le1$ ، أي $X\in\{-1;0;1\}$: \ $P(X^2\le1)=0{,}3+0{,}15+0{,}05=\bx{0{,}5}$.
\end{hal}

\begin{tamrin}{اختيار من متعدد}

\begin{tamrin}{تمرين شامل: احتمال أعظمي (توظيف المشتقة)}
يحتوي كيس على $n$ قريصة متماثلة لا نفرق بينها عند اللمس ($n\in\N$ و $n\ge7$) ، منها $7$ بيضاء والباقي حمراء. نسحب قريصتين على التوالي وبدون إرجاع.
\begin{enumerate}
\item نفرض في هذا السؤال أن $n=10$. احسب احتمالات الحوادث التالية:
\begin{sq}
\qa $A$: «القريصة الأولى بيضاء والثانية حمراء».
\qb $B$: «قريصة بيضاء وقريصة حمراء».
\qc $C$: «القريصتان بيضاوان».
\qd $D$: «القريصتان من نفس اللون».
\end{sq}
\item نعتبر الآن $n\ge8$. نرمز بـ $P_n$ إلى احتمال أن تكون القريصتان المسحوبتان من لونين مختلفين.
\begin{sq}
\qa أثبت أن: $P_n=\dfrac{14(n-7)}{n^2-n}$.
\qb بدراسة الدالة $f:x\mapsto\dfrac{14(x-7)}{x^2-x}$ على $[8;+\infty[$ ، عيّن قيم العدد الطبيعي $n$ الذي يكون من أجله لـ $P_n$ قيمة حدية عظمى ، ثم عيّن هذه القيمة.
\end{sq}
\end{enumerate}
\end{tamrin}
\begin{hal}
\textbf{1.} $n=10$: $7$ بيضاء و $3$ حمراء ، \ $\card\Omega=10\times9=90$.
\pa بيضاء ثم حمراء: $7\times3=21$ ، \ $P(A)=\dfrac{21}{90}=\bx{\dfrac{7}{30}}$.
\pb بيضاء ثم حمراء أو حمراء ثم بيضاء: $21+21=42$ ، \ $P(B)=\dfrac{42}{90}=\bx{\dfrac{7}{15}}$.
\pc $7\times6=42$ ، \ $P(C)=\dfrac{42}{90}=\bx{\dfrac{7}{15}}$.
\pd $D=\overline B$: \ $P(D)=1-\dfrac{7}{15}=\bx{\dfrac{8}{15}}$ \ (أو $\dfrac{42+3\times2}{90}=\dfrac{48}{90}$).

\QQ{2} \pa $\card\Omega=n(n-1)=n^2-n$. \ عدد الحمراء $n-7$. \ لونان مختلفان: $7(n-7)+(n-7)\times7=14(n-7)$ ، \ ومنه: \bx{P_n=\dfrac{14(n-7)}{n^2-n}}.
\pb \begin{align*}
f'(x)&=14\times\frac{1\times(x^2-x)-(x-7)(2x-1)}{(x^2-x)^2}=14\times\frac{x^2-x-2x^2+15x-7}{(x^2-x)^2}\\
&=\frac{14(-x^2+14x-7)}{(x^2-x)^2}.
\end{align*}
$-x^2+14x-7=0$: $\Delta=196-28=168$ ، والجذر الأكبر $x_0=\dfrac{14+\sqrt{168}}{2}=7+\sqrt{42}\approx13{,}48$.\par
على $[8;x_0]$: $f'(x)\ge0$ ($f$ متزايدة) ، وعلى $[x_0;+\infty[$: $f'(x)\le0$ ($f$ متناقصة).\par
إذن القيمة العظمى لـ $P_n$ (مع $n$ طبيعي) تكون عند $n=13$ أو $n=14$:
\[P_{13}=\frac{14\times6}{156}=\frac{84}{156}=\frac{7}{13}\qquad;\qquad P_{14}=\frac{14\times7}{182}=\frac{98}{182}=\frac{7}{13}.\]
القيمة العظمى هي \bx{\dfrac{7}{13}\approx0{,}54} ، وتُبلغ من أجل \bx{n=13} و \bx{n=14}.
\end{hal}
