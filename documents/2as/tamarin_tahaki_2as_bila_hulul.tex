\section{التعريف والخاصية المميزة}
% ------------------------------------------------- 1
\begin{tamrin}{تعيين تحاكيات}
\begin{enumerate}
\item $A$ و $B$ نقطتان متمايزتان ، و $I$ نقطة حيث $\V{IB}=-\dfrac12\V{IA}$. \ عيّن تحاكياً يحوّل $A$ إلى $B$.
\item $O$ ، $P$ و $Q$ نقط حيث $\V{OQ}=3\V{OP}$. عيّن تحاكياً يحوّل $P$ إلى $Q$.
\item $ABCD$ متوازي أضلاع. هل يوجد تحاكٍ يحوّل $A$ إلى $C$ و $B$ إلى $D$؟ عيّنه.
\item $A$ ، $B$ نقطتان و $f$ التحويل النقطي الذي يرفق بكل نقطة $M$ النقطة $M'$ حيث: $\V{MM'}=2\V{MA}+\V{MB}$.\par
أنشئ النقطة $G$ مرجح الجملة $\{(A;2);(B;1)\}$ ، وأثبت أن $\V{GM'}=-2\V{GM}$ ، ثم استنتج طبيعة $f$.
\end{enumerate}
\end{tamrin}

% ------------------------------------------------- 2
\begin{tamrin}{مثلث قائم ومساحات}
$ABC$ مثلث قائم في $A$ حيث $AB=2$ و $AC=3$. \ $D$ نقطة من المستوي تحقق: $2\V{AD}=\V{DB}$.
\begin{enumerate}
\item بيّن أن النقطة $D$ هي صورة النقطة $B$ بالتحاكي $h$ الذي مركزه $A$ يطلب تعيين نسبته.
\item $E$ صورة $C$ بالتحاكي $h$. حدّد طبيعة المثلث $ADE$ واحسب مساحته.
\item لتكن النقطة $F$ مرجح النقطتين $A$ و $C$ المرفقتين بالمعاملين $2$ و $-3$ على الترتيب. \ بيّن أن النقطة $F$ هي صورة $A$ بالتحاكي $h'$ الذي مركزه $C$ يطلب تعيين نسبته.
\item نعتبر النقطتين $I$ و $H$ صورتي النقطتين $D$ و $E$ بالتحاكي $h'$ على الترتيب. حدّد طبيعة المثلث $FHI$ ثم استنتج مساحته.
\end{enumerate}
\end{tamrin}

% ------------------------------------------------- 3
\begin{tamrin}{صحيح أم خطأ}
أجب بصحيح أو خطأ مع التعليل:
\begin{enumerate}
\item النقطة $C$ مرجح الجملة $\{(A;2);(B;-3)\}$ معناه $C$ هي صورة $B$ بالتحاكي $h$ ذي المركز $A$ والنسبة $k=3$.
\item $ABC$ مثلث حيث $\widehat{BAC}=60^\circ$ ، $AB=5\ \text{cm}$ و $AC=7\ \text{cm}$:
\begin{sq}
\qa $BC=\sqrt{39}\ \text{cm}$.
\qb مساحة المثلث $A'B'C'$ صورة $ABC$ بالتحاكي الذي نسبته $k=-\dfrac13$ هي $S=\dfrac{35\sqrt3}{36}\ \text{cm}^2$.
\end{sq}
\item صورة مستقيم بتحاكٍ هي دائماً مستقيم يختلف عنه.
\item التحاكي الذي نسبته $-1$ هو تناظر مركزي.
\item إذا كان $h(A)=A'$ و $h(B)=B'$ حيث $AB=4$ و $A'B'=2$ فإن نسبة $h$ هي حتماً $\dfrac12$.
\end{enumerate}
\end{tamrin}

\section{الصور ، الأطوال والمساحات}
% ------------------------------------------------- 4
\begin{tamrin}{صورة مستطيل}
$ABCD$ مستطيل مركزه $O$ ، و $I$ منتصف القطعة $[AB]$ ، و $G$ نقطة تقاطع المستقيمين $(IC)$ و $(BD)$. \ نرمز بـ $h$ إلى التحاكي الذي مركزه $G$ ونسبته $-\dfrac12$.
\begin{enumerate}
\item بيّن أن $G$ مركز ثقل المثلث $ABC$.
\item عيّن صور النقط $A$ ، $B$ ، $C$ و $D$ بالتحاكي $h$ ، ثم أرسم صورة المستطيل $ABCD$.
\item احسب محيط ومساحة صورة المستطيل $ABCD$ إذا علمت أن $AD=4\ \text{cm}$ و $AB=10\ \text{cm}$.
\end{enumerate}
\end{tamrin}

% ------------------------------------------------- 5
\begin{tamrin}{تحاكيان متتابعان في مثلث}
$ABC$ مثلث مركز ثقله $G$ ، و $A'$ ، $B'$ ، $C'$ منتصفات $[BC]$ ، $[CA]$ و $[AB]$ على الترتيب. \ ليكن $h$ التحاكي الذي مركزه $G$ ويحوّل $A$ إلى $A'$ ، ولنضع $M=h(A')$ ، $N=h(B')$ و $P=h(C')$.
\begin{enumerate}
\item عيّن نسبة التحاكي $h$.
\item بيّن أن $h(B)=B'$ و $h(C)=C'$ ، ثم عيّن صورة المثلث $ABC$.
\item قارن بين مساحتي المثلثين $MNP$ و $ABC$.
\end{enumerate}
\end{tamrin}

% ------------------------------------------------- 6
\begin{tamrin}{دوائر في معلم}
المستوي منسوب إلى معلم متعامد ومتجانس $(O;\vec i,\vec j)$.
\begin{enumerate}
\item $(C)$ الدائرة التي مركزها $A(2;3)$ ونصف قطرها $2$ ، و $h$ التحاكي الذي مركزه $O$ ونسبته $-\dfrac12$. \ عيّن ثم أرسم $(C')=h(C)$.
\item $(\Gamma)$ مجموعة النقط $M(x;y)$ حيث: $x^2+y^2-2x-6y+5=0$.
\begin{sq}
\qa بيّن أن $(\Gamma)$ دائرة يطلب تعيين مركزها ونصف قطرها.
\qb عيّن معادلة الدائرة $(C_1)$ صورة $(\Gamma)$ بالتحاكي الذي مركزه $B(0;2)$ ونسبته $3$.
\end{sq}
\item $(E)$ مجموعة النقط $M(x;y)$ حيث: $x^2+y^2+4x-2y-31=0$. \ عيّن صورة $(E)$ بالتحاكي الذي مركزه $O$ ونسبته $3$.
\end{enumerate}
\end{tamrin}

\section{مسائل شاملة: الاستقامية والتوازي}
% ------------------------------------------------- 7
\begin{tamrin}{تحويل نقطي وتحاكيان}
$ABCD$ متوازي أضلاع ، و $G$ مرجح الجملة المثقلة $\{(A;3);(B;1)\}$.\par
\textbf{الجزء الأول:} نسمي $f$ التحويل النقطي الذي يرفق بكل نقطة $M$ من المستوي النقطة $M'$ حيث: $\V{MM'}-3\V{MA}-\V{MB}=\vec0$.
\begin{enumerate}
\item حدّد صورة النقطة $A$ بواسطة $f$.
\item
\begin{sq}
\qa أثبت أن التحويل $f$ يقبل نقطة صامدة وحيدة يطلب تعيينها.
\qb أثبت أن $f$ هو تحاكٍ يطلب تعيين مركزه ونسبته.
\end{sq}
\end{enumerate}
\textbf{الجزء الثاني:} ليكن $h$ التحاكي الذي مركزه $G$ ونسبته $-3$.
\begin{enumerate}[resume]
\item عيّن صورة المستقيم $(AD)$ بواسطة $h$ معللاً إجابتك.
\item المستقيم $(GD)$ يقطع المستقيم $(BC)$ في $E$. \ أثبت أن $E=h(D)$.
\item نعتبر $(C)$ الدائرة التي مركزها $G$ وتشمل $A$.
\begin{sq}
\qa حدّد ثم أنشئ $(C')$ صورة $(C)$ بواسطة التحاكي $h$.
\qb نصف المستقيم $[GD)$ يقطع $(C)$ في $M$ ، ونصف المستقيم $[GE)$ يقطع $(C')$ في $N$. \ أثبت أن المستقيمين $(AM)$ و $(BN)$ متوازيان.
\end{sq}
\item ليكن $h'$ التحاكي حيث: $h'(A)=N$ و $h'(M)=B$.
\begin{sq}
\qa حدّد نسبة التحاكي $h'$.
\qb أنشئ $J$ مركز التحاكي $h'$.
\qc عيّن $G'$ صورة $G$ وفق التحاكي $h'$.
\end{sq}
\end{enumerate}
\end{tamrin}

% ------------------------------------------------- 8
\begin{tamrin}{دائرة وتحاكٍ نسبته سالبة}
في المستوي ، $(\mathcal C)$ الدائرة التي مركزها $O$ وقطرها $[AB]$. \ نرمز بـ $h$ للتحاكي الذي مركزه $A$ ونسبته $k=-\dfrac32$.
\begin{enumerate}
\item أنشئ النقطة $B'$ صورة $B$ بالتحاكي $h$.
\item $M$ نقطة من $(\mathcal C)$ مختلفة عن $A$ و $B$. \ المستقيم $(\Delta)$ يشمل $B'$ ويوازي $(BM)$ ويقطع $(AM)$ في $M'$.
\begin{sq}
\qa عيّن صورة كل من المستقيمين $(BM)$ و $(AM)$ بالتحاكي $h$.
\qb استنتج أن $h(M)=M'$.
\end{sq}
\item نفرض أن $M$ تتغير على $(\mathcal C)$. بيّن أن $M'$ تتغير على دائرة $(\mathcal C')$ يطلب تعيينها.
\item $E$ منتصف $[BM]$ و $F$ منتصف $[B'M']$. بيّن أن $A$ ، $E$ و $F$ على استقامة واحدة.
\item المستقيم $(EF)$ يقطع $(\mathcal C)$ في $C$ و $(\mathcal C')$ في $D$ (غير $A$). بيّن أن $(CM)\parallel(DM')$.
\end{enumerate}
\end{tamrin}

\section{المحلات الهندسية والإنشاءات}
% ------------------------------------------------- 9
\begin{tamrin}{محل هندسي لنقطة تقاطع}
$(C)$ دائرة مركزها $O$ و $[AB]$ أحد أقطارها. \ $I$ النقطة التي تحقق $\V{AI}=-\V{AO}$ ، و $M$ نقطة متغيرة من $(C)$ تختلف عن $A$ و $B$ ، و $M'$ نظيرة $M$ بالنسبة لـ $O$. \ $N$ نقطة تقاطع المستقيمين $(IM)$ و $(BM')$.
\begin{enumerate}
\item بيّن أن $(AM)\parallel(BM')$.
\item بيّن أن $N$ صورة $M$ بتحاكٍ $h$ مركزه $I$ يطلب تعيين نسبته.
\item عيّن المحل الهندسي للنقطة $N$ عندما تمسح $M$ الدائرة $(C)$ ماعدا النقطتين $A$ و $B$.
\end{enumerate}
\end{tamrin}

% ------------------------------------------------- 10
\begin{tamrin}{محل منتصف قطعة}
$(\Delta)$ مستقيم ثابت و $A$ نقطة ثابتة لا تنتمي إلى $(\Delta)$. \ من أجل كل نقطة $M$ من $(\Delta)$ نرسم النقطة $N$ منتصف القطعة $[AM]$.\par
ما هو المحل الهندسي للنقطة $N$ لما تتغير النقطة $M$ على المستقيم $(\Delta)$؟
\end{tamrin}

\begin{tamrin}{إنشاء مربع داخل مثلث (تمرين نموذجي)}
$ABC$ مثلث زواياه حادة. \ نريد إنشاء مربع $PQRS$ حيث $P$ و $Q$ على $[BC]$ ، $R$ على $[AC]$ و $S$ على $[AB]$.
\begin{enumerate}
\item نختار نقطة $S_1$ من $[AB]$ ، ونرسم المربع $P_1Q_1R_1S_1$ حيث $P_1$ و $Q_1$ على $(BC)$ ($P_1$ المسقط العمودي لـ $S_1$). \ بيّن أن المربع المطلوب صورة هذا المربع بتحاكٍ مركزه $B$.
\item استنتج طريقة لإنشاء المربع $PQRS$.
\end{enumerate}
\end{tamrin}

\section{الاختيار من متعدد}
% ------------------------------------------------- 12
\begin{tamrin}{اختيار من متعدد}
عيّن الاقتراح الصحيح الوحيد في كل حالة مع التعليل:
\begin{enumerate}
\item إذا كان $\V{\Omega M'}=-2\V{\Omega M}$ فإن $M'$ صورة $M$ بالتحاكي:
\ABC{$h(\Omega;2)$}{$h(\Omega;-2)$}{$h\Big(\Omega;-\dfrac12\Big)$}
\item صورة دائرة نصف قطرها $6$ بتحاكٍ نسبته $-\dfrac23$ دائرة نصف قطرها:
\ABC{$-4$}{$4$}{$9$}
\item مثلث مساحته $18\ \text{cm}^2$. مساحة صورته بتحاكٍ نسبته $\dfrac13$ هي:
\ABC{$6\ \text{cm}^2$}{$2\ \text{cm}^2$}{$54\ \text{cm}^2$}
\item في معلم ، صورة $M(4;-2)$ بالتحاكي الذي مركزه $\Omega(1;1)$ ونسبته $2$ هي:
\ABC{$(8;-4)$}{$(7;-5)$}{$(9;-3)$}
\item التحاكي الذي يحوّل $A$ إلى $B$ ومركزه $I$ منتصف $[AB]$ نسبته:
\ABC{$-1$}{$1$}{$2$}
\end{enumerate}
\end{tamrin}
