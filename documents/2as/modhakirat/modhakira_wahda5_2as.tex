\textbf{(2)} $f(10)=3{,}1$ ، $f(100)=3{,}01$ ، $f(1000)=3{,}001$: القيم تقترب من \RR{$3$}.\par
\textbf{(3)} المنحنى يقترب من المستقيم الأفقي \RR{$y=3$}.}{}{}

\Lab{cDef}{تعريف}\par

\textbf{(1)} تحقق أن $g(x)=2+\dfrac3{x-1}$.\par
\textbf{(2)} احسب نهايات $g$ واستنتج المستقيمات المقاربة.\par

\Row{}{%
\Lab{cEx}{حل التمرين التطبيقي}\par
\textbf{(1)} $2+\dfrac3{x-1}=\dfrac{2(x-1)+3}{x-1}=\dfrac{2x+1}{x-1}=g(x)$ \ck\par
\textbf{(2)}\ \LDx{g(x)}{\pm\infty}{2} $\Car{\Lm{\pm\infty}2=2\\ \Lm{\pm\infty}\dfrac3{x-1}=0}$
\LD{g(x)}{1^-}{-\infty} $\Car{\Lm{1}(2x+1)=3\\ \Lm{1^-}(x-1)=0^-}$
\LD{g(x)}{1^+}{+\infty} $\Car{\Lm{1}(2x+1)=3\\ \Lm{1^+}(x-1)=0^+}$
\par إذن \RR{$y=2$} مقارب أفقي و \RR{$x=1$} مقارب عمودي.\par
\textbf{(3)} $g^{\prime}(x)=-\dfrac3{(x-1)^2}<0$: $g$ \RR{متناقصة تمامًا} على كل من $]-\infty;1[$ و $]1;+\infty[$.\par
