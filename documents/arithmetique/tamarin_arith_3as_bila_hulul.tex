\section{القسمة في $\Z$ والقسمة الإقليدية}

% ------------------------------------------------------------------
\begin{tamrin}{القسمة الإقليدية: تطبيق مباشر}
\QQ{1} أجرِ القسمة الإقليدية في كل حالة:
\begin{sq}
\qa العدد $2026$ على $17$.
\qb العدد $-2026$ على $17$.
\end{sq}
\QQ{2} عيّن الأعداد الطبيعية غير المعدومة $a$ التي يكون حاصل قسمتها الإقليدية على $13$ مساويا لباقيها.
\QQ{3} $a$ و $b$ عددان طبيعيان، حاصل القسمة الإقليدية لـ $a$ على $b$ هو $7$ والباقي $15$، و $a+b=255$. عيّن $a$ و $b$.
\end{tamrin}

% ------------------------------------------------------------------
\begin{tamrin}{قابلية القسمة في $\Z$}
\QQ{1} عيّن الأعداد الصحيحة $n$ (مع $n\neq -3$) بحيث يقسم $n+3$ العدد $2n+13$.
\QQ{2} عيّن الأعداد الطبيعية $n$ (مع $n\neq 2$) بحيث يقسم $n-2$ العدد $n^2+5$.
\QQ{3} عيّن الثنائيات $(x;y)$ من الأعداد الطبيعية التي تحقق: $x^2-y^2=15$.
\end{tamrin}

% ------------------------------------------------------------------
\begin{tamrin}{باقي قسمة عبارة بدلالة $n$}
$n$ عدد طبيعي.
\QQ{1} تحقق أن: $4n+15=4(n+2)+7$.
\QQ{2} استنتج حاصل وباقي القسمة الإقليدية للعدد $4n+15$ على $n+2$ عندما $n\ge 6$.
\QQ{3} عيّن الحاصل والباقي من أجل كل قيمة لـ $n$ من $\{0;1;2;3;4;5\}$.
\QQ{4} استنتج قيم $n$ التي من أجلها يقسم $n+2$ العدد $4n+15$.
\end{tamrin}

\section{الموافقات}

% ------------------------------------------------------------------
\begin{tamrin}{خواص الموافقات}
\QQ{1}
\begin{sq}
\qa عيّن باقي القسمة الإقليدية للعدد $2026$ على $7$.
\qb استنتج باقي قسمة العدد $2026^2+2026$ على $7$.
\end{sq}
\QQ{2} عيّن باقي قسمة $2026^{2027}$ على $5$، ثم باقي قسمة $39^{100}$ على $8$.
\QQ{3} بيّن أنه من أجل كل عدد طبيعي $n$: العدد $3^{2n+1}+2^{n+2}$ يقبل القسمة على $7$.
\QQ{4} باستعمال بواقي القسمة على $3$، بيّن أنه من أجل كل عدد طبيعي $n$: $n^3-n\equiv 0\,[3]$.
\end{tamrin}

% ------------------------------------------------------------------
\begin{tamrin}{بواقي قسمة $2^n$ على $7$}
\QQ{1} ادرس، حسب قيم العدد الطبيعي $n$، بواقي القسمة الإقليدية للعدد $2^n$ على $7$.
\QQ{2} استنتج باقي قسمة كل من العددين $2^{2026}$ و $1437^{2026}$ على $7$.
\QQ{3} عيّن الأعداد الطبيعية $n$ التي من أجلها يكون $2^n+3\equiv 0\,[7]$.
\QQ{4} بيّن أنه من أجل كل عدد طبيعي $n$: $2^{3n}+2^{3n+1}+2^{3n+2}\equiv 0\,[7]$.
\end{tamrin}

% ------------------------------------------------------------------
\begin{tamrin}{بواقي قسمة $3^n$ على $11$ ومجموع قوى}
\QQ{1} ادرس، حسب قيم العدد الطبيعي $n$، بواقي القسمة الإقليدية للعدد $3^n$ على $11$.
\QQ{2} عيّن باقي قسمة $3^{2027}$ على $11$، ثم باقي قسمة $245^{2026}$ على $11$.
\QQ{3} عيّن الأعداد الطبيعية $n$ التي من أجلها يقبل $3^n+7$ القسمة على $11$.
\QQ{4} نضع: $S=1+3+3^2+\cdots+3^{2026}$.
\begin{sq}
\qa بيّن أنه من أجل كل عدد طبيعي $k$: $3^{5k}+3^{5k+1}+3^{5k+2}+3^{5k+3}+3^{5k+4}\equiv 0\,[11]$.
\qb استنتج باقي قسمة $S$ على $11$.
\end{sq}
\end{tamrin}

% ------------------------------------------------------------------
\begin{tamrin}{بواقي قسمة $2^n$ على $9$ -- نحو البكالوريا}
\QQ{1} ادرس، حسب قيم العدد الطبيعي $n$، بواقي القسمة الإقليدية للعدد $2^n$ على $9$.
\QQ{2} عيّن باقي قسمة العدد $1442^{2026}$ على $9$.
\QQ{3} بيّن أنه من أجل كل عدد طبيعي $n$: $2^{6n+3}+1\equiv 0\,[9]$.
\QQ{4}
\begin{sq}
\qa عيّن بواقي قسمة $4^n$ على $9$ حسب قيم $n$ (لاحظ أن $4^n=2^{2n}$).
\qb استنتج الأعداد الطبيعية $n$ التي من أجلها يكون: $4^n+2^n\equiv 0\,[9]$.
\end{sq}
\end{tamrin}

\section{أنظمة التعداد}

% ------------------------------------------------------------------
\begin{tamrin}{التحويل بين الأنظمة}
\QQ{1} اكتب العدد $2026$ في نظام التعداد ذي الأساس $2$، ثم في النظام ذي الأساس $5$، ثم في النظام ذي الأساس $7$.
\QQ{2} اكتب في النظام العشري العددين: $\overline{3102}^{\,5}$ و $\overline{101101}^{\,2}$.
\QQ{3} عيّن العدد الطبيعي $b$ (حيث $b\ge 6$) بحيث: $\overline{235}^{\,b}=124$.
\end{tamrin}

% ------------------------------------------------------------------
\begin{tamrin}{عدد مكتوب في نظامين -- نحو البكالوريا}
\QQ{1} $n$ عدد طبيعي حيث $n\ge 3$. بيّن أن $\overline{121}^{\,n}=(n+1)^2$، واستنتج أن العدد $\overline{121}^{\,n}$ مربع تام في كل أساس $n\ge 3$.
\QQ{2} $N$ عدد طبيعي يُكتب $\overline{1\alpha\beta}^{\,6}$ في النظام ذي الأساس $6$، ويُكتب $\overline{1\beta\alpha}^{\,7}$ في النظام ذي الأساس $7$.
\begin{sq}
\qa بيّن أن $\alpha$ و $\beta$ يحققان: $5\alpha-6\beta=13$.
\qb عيّن $\alpha$ و $\beta$، ثم اكتب $N$ في النظام العشري.
\qc اكتب $N$ في النظام ذي الأساس $3$.
\end{sq}
\end{tamrin}

\section{القاسم المشترك الأكبر والمضاعف المشترك الأصغر}

% ------------------------------------------------------------------
\begin{tamrin}{خوارزمية إقليدس}
\QQ{1} باستعمال خوارزمية إقليدس، احسب $\pgcd(3150;1386)$.
\QQ{2} استنتج $\ppcm(3150;1386)$، ثم مجموعة القواسم المشتركة للعددين.
\QQ{3} عيّن الثنائيات $(a;b)$ من الأعداد الطبيعية حيث $a<b$ و $a+b=96$ و $\pgcd(a;b)=12$.
\end{tamrin}

% ------------------------------------------------------------------
\begin{tamrin}{جداء و PPCM}
عيّن الثنائيات $(a;b)$ من الأعداد الطبيعية حيث $a\le b$ و:
\[\begin{cases} a\times b=1512\\ \ppcm(a;b)=252\end{cases}\]
\end{tamrin}

% ------------------------------------------------------------------
\begin{tamrin}{القاسم المشترك الأكبر لعبارتين بدلالة $n$}
$n$ عدد طبيعي. نضع: $a=2n+3$ و $b=5n+2$، و $d=\pgcd(a;b)$.
\QQ{1} بيّن أن $d$ يقسم $11$، واستنتج القيم الممكنة لـ $d$.
\QQ{2} عيّن الأعداد الطبيعية $n$ التي من أجلها يكون $d=11$.
\QQ{3} في حالة $n=4$: احسب $a$ و $b$ ثم $\ppcm(a;b)$.
\end{tamrin}

\section{مبرهنتا بيزو وغوص والمعادلات الديوفانتية}

% ------------------------------------------------------------------
\begin{tamrin}{تطبيق مباشر لمبرهنتي بيزو وغوص}
\QQ{1} بيّن أنه من أجل كل عدد طبيعي $n$، العددان $2n+1$ و $3n+2$ أوليان فيما بينهما.
\QQ{2} باستعمال خوارزمية إقليدس، عيّن ثنائية $(u;v)$ من الأعداد الصحيحة تحقق: $23u+17v=1$.
\QQ{3} $x$ عدد صحيح حيث $7$ يقسم $15x$. بيّن أن $7$ يقسم $x$.
\end{tamrin}

% ------------------------------------------------------------------
\begin{tamrin}{حل معادلة ديوفانتية}
نعتبر في $\Z^2$ المعادلة $(E)$: $11x-7y=5$.
\QQ{1} تحقق أن $(3;4)$ حل للمعادلة $(E)$، ثم حل المعادلة $(E)$.
\QQ{2} عيّن الحلول $(x;y)$ للمعادلة $(E)$ التي تحقق: $0<x+y<60$.
\QQ{3} نضع $d=\pgcd(x;y)$ حيث $(x;y)$ حل للمعادلة $(E)$.
\begin{sq}
\qa ما هي القيم الممكنة لـ $d$؟
\qb عيّن الحلول $(x;y)$ التي من أجلها $d=5$.
\end{sq}
\end{tamrin}

% ------------------------------------------------------------------
\begin{tamrin}{مسألة: صندوق البيض}
عند بائع عدد $N$ من البيض محصور بين $300$ و $400$. إذا رتّبه في علب بها $7$ بيضات بقيت $3$ بيضات، وإذا رتّبه في علب بها $11$ بيضة بقيت $5$ بيضات.
\QQ{1} بيّن أنه توجد ثنائية $(x;y)$ من الأعداد الطبيعية تحقق: $7x-11y=2$.
\QQ{2} حل في $\Z^2$ المعادلة $7x-11y=2$.
\QQ{3} استنتج أن $N\equiv 38\,[77]$، ثم عيّن $N$.
\end{tamrin}

\begin{tamrin}{معادلة ديوفانتية و PGCD -- نحو البكالوريا}
نعتبر في $\Z^2$ المعادلة $(E)$: $156x-102y=18$.
\QQ{1}
\begin{sq}
\qa احسب $\pgcd(156;102)$ بخوارزمية إقليدس.
\qb استنتج أن المعادلة $(E)$ تقبل حلولا، وأنها تكافئ المعادلة $26x-17y=3$.
\end{sq}
\QQ{2} بيّن أنه إذا كانت $(x;y)$ حلا للمعادلة $(E)$ فإن $x\equiv 6\,[17]$، ثم حل المعادلة $(E)$.
\QQ{3} ليكن $(x;y)$ حلا للمعادلة $(E)$ و $d=\pgcd(x;y)$.
\begin{sq}
\qa بيّن أن $d\in\{1;3\}$.
\qb عيّن الحلول $(x;y)$ من الأعداد الطبيعية التي تحقق $d=3$ و $x<100$.
\end{sq}
\end{tamrin}

\QQ{3} عيّن أصغر عدد طبيعي غير معدوم $k$ بحيث يكون $2520k$ مربعا تاما.
\QQ{4} حلّل $1764$ إلى جداء عوامل أولية، واستنتج $\sqrt{1764}$.

\begin{tamrin}{مسائل حول الأعداد الأولية -- نحو البكالوريا}
\QQ{1} علمًا أن $2026=2\times 1013$ حيث $1013$ أولي، بيّن أن المعادلة $x^2-y^2=2026$ لا تقبل حلولا في $\N^2$.
\QQ{2} $p$ عدد أولي حيث $p\ge 5$.
\begin{sq}
\qa بيّن أن $p^2\equiv 1\,[3]$ و $p^2\equiv 1\,[8]$.
\qb استنتج أن $p^2\equiv 1\,[24]$.
\end{sq}
\QQ{3} $n$ عدد طبيعي.
\begin{sq}
\qa تحقق أن: $n^4+4=(n^2+2n+2)(n^2-2n+2)$.
\qb استنتج أن $n^4+4$ ليس أوليا من أجل كل $n\ge 2$.
\end{sq}
\end{tamrin}

\section{اختيار من متعدد وصحيح أم خطأ}

% ------------------------------------------------------------------
\begin{tamrin}{اختيار من متعدد}
اختر الإجابة الصحيحة الوحيدة في كل حالة مع التعليل:
\QQ{1} باقي القسمة الإقليدية للعدد $2^{2026}$ على $5$ هو:
\ABC{$1$}{$4$}{$3$}
\QQ{2} من أجل كل عدد طبيعي غير معدوم $n$، $\pgcd(2n+1;n)$ يساوي:
\ABC{$n$}{$1$}{$2$}
\QQ{3} العدد $\overline{1011}^{\,2}$ يساوي في النظام العشري:
\ABC{$13$}{$9$}{$11$}
\QQ{4} حلول المعادلة $3x\equiv 2\,[7]$ ذات المجهول الصحيح $x$ هي:
\ABC{$x\equiv 3\,[7]$}{$x\equiv 5\,[7]$}{$x\equiv 2\,[7]$}
\QQ{5} عدد القواسم الطبيعية للعدد $72$ هو:
\ABC{$8$}{$12$}{$6$}
\QQ{6} المعادلة $6x+9y=10$ ذات المجهولين الصحيحين $x$ و $y$:
\ABC{لها حل وحيد}{عدد غير منته من الحلول}{ليس لها حلول}
\end{tamrin}

% ------------------------------------------------------------------
\begin{tamrin}{صحيح أم خطأ}
أجب بصحيح أو خطأ مع التعليل ($a$، $b$، $c$ أعداد صحيحة و $n$ عدد طبيعي):
\QQ{1} إذا كان $a\equiv b\,[n]$ فإن $a^2\equiv b^2\,[n]$ (مع $n\ge 2$).
\QQ{2} إذا كان $6$ يقسم $ab$ فإن $6$ يقسم $a$ أو $6$ يقسم $b$.
\QQ{3} من أجل كل عدد طبيعي $n$: العدد $3^{4n}-1$ يقبل القسمة على $5$.
\QQ{4} إذا كان $\pgcd(a;b)=1$ و $a$ يقسم $bc$ فإن $a$ يقسم $c$.
\QQ{5} العدد $2^{11}-1$ أولي.
\QQ{6} من أجل كل عدد طبيعي $n$: العدد $n^2+n$ زوجي.
\end{tamrin}

\begin{tamrin}{الموافقات والمتتاليات}
$(u_n)$ المتتالية المعرفة بـ: $u_0=1$ ومن أجل كل عدد طبيعي $n$: $u_{n+1}=5u_n+4$.
\QQ{1} نضع $v_n=u_n+1$. بيّن أن $(v_n)$ متتالية هندسية، ثم اكتب $u_n$ بدلالة $n$.
\QQ{2} احسب بدلالة $n$ المجموع $S_n=u_0+u_1+\cdots+u_n$.
\QQ{3} ادرس، حسب قيم $n$، بواقي القسمة الإقليدية للعدد $5^n$ على $7$.
\QQ{4} عيّن الأعداد الطبيعية $n$ التي من أجلها يكون $u_n\equiv 0\,[7]$.
\QQ{5}
\begin{sq}
\qa بيّن أنه من أجل كل $n\in\N$: $u_n$ عدد فردي.
\qb استنتج $\pgcd(u_n;u_{n+1})$.
\end{sq}
\end{tamrin}

\begin{tamrin}{القاسم المشترك الأكبر لعبارتين من الدرجة الثانية}
$n$ عدد طبيعي. نضع: $A=n^2+5n+4$ و $B=n^2+3n+2$.
\QQ{1} حلّل كلا من $A$ و $B$ إلى جداء عاملين من الدرجة الأولى.
\QQ{2} بيّن أن $\pgcd(n+4;n+2)=\pgcd(n+2;2)$، واستنتج قيمته حسب زوجية $n$.
\QQ{3} استنتج $\pgcd(A;B)$ حسب زوجية $n$.
\QQ{4} عيّن الأعداد الطبيعية $n$ التي من أجلها يكون $\pgcd(A;B)=20$.
\QQ{5} عبّر عن $\ppcm(A;B)$ بدلالة $n$.
\end{tamrin}
