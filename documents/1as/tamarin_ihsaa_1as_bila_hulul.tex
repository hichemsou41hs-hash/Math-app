\section{التوزيعات التكرارية والتمثيلات البيانية}
% ------------------------------------------------- 1
\begin{tamrin}{جدول تكراري وتواترات}
لتكن السلسلة التالية لعلامات $10$ تلاميذ:\quad $12\ ;\ 14\ ;\ 12\ ;\ 9\ ;\ 9\ ;\ 9\ ;\ 11\ ;\ 10\ ;\ 10\ ;\ 12$.
\begin{enumerate}
\item ما هي الميزة المدروسة؟ ما طبيعتها؟
\item مثّل هذه السلسلة في جدول يتضمن التكرارات ، التواترات والتكرارات المجمعة الصاعدة.
\item عيّن المنوال والوسيط والوسط الحسابي.
\item مثّل السلسلة بمخطط بالأعمدة.
\end{enumerate}
\end{tamrin}

% ------------------------------------------------- 2
\begin{tamrin}{تعيين تكرارين مجهولين}
يمثل الجدول التالي التوزيع التكراري لعلامات $30$ تلميذاً من قسم ، حيث $a$ و $b$ عددان طبيعيان:
\begin{center}
\renewcommand{\arraystretch}{1.45}
\begin{tabular}{|c|c|c|c|c|c|}\hline
\cellcolor{cSec!12}العلامة & $7$ & $9$ & $11$ & $13$ & $15$\\\hline
\cellcolor{cSec!12}التكرار & $a$ & $b$ & $2a$ & $b+1$ & $a+1$\\\hline
\cellcolor{cSec!12}ت. مجمع صاعد & & & & $25$ & \\\hline
\end{tabular}
\end{center}
\begin{enumerate}
\item عيّن قيمتي $a$ و $b$.
\item من أجل $a=4$ و $b=6$ أعد رسم الجدول مبيّناً التواترات والتكرارات المجمعة الصاعدة والنازلة.
\item احسب الوسط الحسابي ، الوسيط ، $Q_1$ و $Q_3$.
\end{enumerate}
\end{tamrin}

% ------------------------------------------------- 3
\begin{tamrin}{مخطط دائري}
يعطى توزيع تلاميذ ثانوية حسب المستويات: $160$ تلميذاً في السنة الأولى ، $120$ في السنة الثانية و $140$ في السنة الثالثة.
\begin{enumerate}
\item احسب التكرار الكلي وتواتر كل مستوى (بنسبة مئوية).
\item مثّل هذه السلسلة بمخطط دائري.
\end{enumerate}
\end{tamrin}

% ------------------------------------------------- 4
\begin{tamrin}{سلسلة ذات فئات: معدلات قسم}
يبيّن الجدول التالي نتائج $25$ تلميذاً لقسم من أقسام السنة الأولى ثانوي:
\begin{center}
\renewcommand{\arraystretch}{1.45}
\begin{tabular}{|c|c|c|c|c|c|}\hline
\cellcolor{cSec!12}المعدلات & $[8;10[$ & $[10;12[$ & $[12;14[$ & $[14;16[$ & $[16;18[$\\\hline
\cellcolor{cSec!12}عدد التلاميذ & $8$ & $7$ & $5$ & $3$ & $2$\\\hline
\end{tabular}
\end{center}
\begin{enumerate}
\item
\begin{sq}
\qa هل ميزة هذه السلسلة كمية أم نوعية؟
\qb هل هي مستمرة أم متقطعة؟
\end{sq}
\item أعد كتابة الجدول مبيّناً فيه: مراكز الفئات ، التواترات ، التكرار المجمع الصاعد والتكرار المجمع النازل.
\item عيّن الفئة المنوالية.
\item احسب الوسط الحسابي لهذه السلسلة ، ثم انحرافها المعياري.
\item أنشئ المدرج التكراري لهذه السلسلة ، ثم قدّر الوسيط.
\end{enumerate}
\end{tamrin}

\section{مؤشرات الموقع والتشتت}
% ------------------------------------------------- 5
\begin{tamrin}{دراسة كاملة لسلسلة متقطعة}
علامات $20$ تلميذاً في فرض الرياضيات:
\begin{center}
\renewcommand{\arraystretch}{1.45}
\begin{tabular}{|c|c|c|c|c|c|c|}\hline
\cellcolor{cSec!12}العلامة $x_i$ & $6$ & $8$ & $10$ & $12$ & $14$ & $16$\\\hline
\cellcolor{cSec!12}التكرار $n_i$ & $2$ & $3$ & $5$ & $6$ & $3$ & $1$\\\hline
\end{tabular}
\end{center}
\begin{enumerate}
\item عيّن المنوال والمدى.
\item احسب التكرارات المجمعة الصاعدة ، ثم عيّن الوسيط ، الربعي الأول $Q_1$ والربعي الثالث $Q_3$.
\item احسب الوسط الحسابي $\overline x$ ، التباين $V$ والانحراف المعياري $\sigma$.
\item مثّل المخطط بالعلبة لهذه السلسلة.
\item قرّر الأستاذ إضافة نقطة واحدة لكل تلميذ. ما هو المعدل الجديد؟ وهل يتغير الانحراف المعياري؟
\end{enumerate}
\end{tamrin}

% ------------------------------------------------- 6
\begin{tamrin}{أوزان تلاميذ ومضلع التكرارات المجمعة}
يمثل الجدول التالي أوزان تلاميذ قسم موزعة في فئات:
\begin{center}
\renewcommand{\arraystretch}{1.45}
\begin{tabular}{|c|c|c|c|c|}\hline
\cellcolor{cSec!12}الوزن ($kg$) & $[40;43[$ & $[43;46[$ & $[46;49[$ & $[49;52[$\\\hline
\cellcolor{cSec!12}التكرار & $5$ & $7$ & $12$ & $8$\\\hline
\end{tabular}
\end{center}
\begin{enumerate}
\item أنشئ المدرج التكراري لهذه السلسلة.
\item احسب التكرارات المجمعة الصاعدة ، وارسم مضلعها.
\item عيّن بيانياً ثم حسابياً قيمة تقريبية للوسيط.
\item احسب قيمة تقريبية للوسط الحسابي.
\end{enumerate}
\end{tamrin}

% ------------------------------------------------- 7
\begin{tamrin}{مقارنة قسمين}
علامات قسمين في نفس الفرض ($20$ تلميذاً في كل قسم):
\begin{center}
\renewcommand{\arraystretch}{1.45}
\begin{tabular}{|c|c|c|c|}\hline
\cellcolor{cSec!12}علامات القسم الأول & $8$ & $10$ & $12$\\\hline
\cellcolor{cSec!12}علامات القسم الثاني & $4$ & $10$ & $16$\\\hline
\cellcolor{cSec!12}التكرار (في القسمين) & $5$ & $10$ & $5$\\\hline
\end{tabular}
\end{center}
\begin{enumerate}
\item احسب الوسط الحسابي لكل قسم.
\item احسب الانحراف المعياري لكل قسم.
\item أي القسمين نتائجه أكثر تجانساً؟ علّل.
\end{enumerate}
\end{tamrin}

% ------------------------------------------------- 8
\begin{tamrin}{خواص الوسط الحسابي}
\begin{enumerate}
\item معدل علامات قسم من $30$ تلميذاً هو $9{,}5$. \ قرر الأستاذ ضرب كل علامة في $1{,}2$ ثم إضافة $0{,}5$. ما هو المعدل الجديد؟
\item قسم فيه $18$ بنتاً معدلهن $12$ و $12$ ولداً معدلهم $10{,}5$. احسب معدل القسم.
\item علامات تلميذ في أربعة فروض: $11$ ، $13$ ، $9$ و $x$. عيّن $x$ حتى يكون معدله $12$.
\end{enumerate}
\end{tamrin}

\section{صحيح أم خطأ ، اختيار من متعدد}
% ------------------------------------------------- 9
\begin{tamrin}{صحيح أم خطأ}
أجب بصحيح أو خطأ مع التعليل:
\begin{enumerate}
\item لون العينين ميزة كمية.
\item مجموع تواترات سلسلة إحصائية يساوي $1$.
\item الوسيط يتأثر بالقيم الشاذة أكثر من الوسط الحسابي.
\item للسلسلتين $3,3,7,8,9$ و $4,5,6,8,18,20$ نفس الوسيط.
\item إذا ضربنا كل قيم سلسلة في $2$ فإن وسطها الحسابي يُضرب في $2$.
\item الانحراف المعياري يُقاس بنفس وحدة قيم السلسلة.
\item في مخطط دائري لسلسلة تكرارها الكلي $60$ ، القيمة ذات التكرار $15$ يمثلها قطاع زاويته $90^\circ$.
\end{enumerate}
\end{tamrin}
