\section{العبارات الجبرية وقيمتها العددية}
\begin{tamrin}{تسمية عبارة جبرية}
سمِّ كل عبارة من العبارات التالية (مجموع ، فرق ، جداء ، حاصل قسمة) حسب آخر عملية نجريها:
\[A=3(2x-1)^2\quad;\quad B=2-x(x+1)\quad;\quad C=\frac{x+1}{x-3}+2\]
\[D=(x-2)(x+5)-(x-2)^2\quad;\quad E=\sqrt x\,(1-x)\]
\end{tamrin}

\begin{tamrin}{القيمة العددية والقيم الممنوعة}
\begin{enumerate}
\item $E(x)=2x^2-3x+5$. احسب $E(-1)$ ، $E(0)$ ، $E\Big(\dfrac12\Big)$ و $E\big(\sqrt2\big)$.
\item $F(x)=\dfrac{x^2-1}{x+3}$. ما هي القيمة الممنوعة؟ احسب $F(0)$ ، $F(1)$ و $F(-2)$.
\item $G(x)=\dfrac1{x-1}-\dfrac2x$. عيّن القيم الممنوعة ، ثم احسب $G(2)$ و $G(-1)$.
\item احسب $-x^2$ و $(-x)^2$ من أجل $x=-3$.
\end{enumerate}
\end{tamrin}

\section{النشر ، التحليل والمتطابقات الشهيرة}
\begin{tamrin}{النشر}
انشر ثم بسّط العبارات التالية:
\[A=(2x-3)^2\quad;\quad B=(x+2)(x-5)\quad;\quad C=(3x+1)(3x-1)\]
\[D=(x-1)^2-(x+2)(x-2)\quad;\quad E=2(x+1)^2-(2x-1)^2\]
\end{tamrin}

\begin{tamrin}{التحليل}
حلّل إلى جداء عوامل:
\[A=9x^2-16\ ;\ B=x^2-6x+9\ ;\ C=4x^2+20x+25\ ;\ D=(x-1)^2-9\]
\[E=(x+1)(2x-3)-(x+1)(x+4)\ ;\ F=(2x+1)^2-(x-3)^2\ ;\ G=x^3-4x\ ;\ H=3x^2-12x\]
\end{tamrin}

\begin{tamrin}{حساب ذهني بالمتطابقات}
احسب دون آلة حاسبة: \quad $99^2$ \ ؛ \ $101\times99$ \ ؛ \ $1002^2$ \ ؛ \ $1999^2-1998^2$.
\end{tamrin}

\begin{tamrin}{اختزال عبارة كسرية}
بسّط كل عبارة بعد تعيين قيمها الممنوعة:
\[A=\frac{x^2-1}{x+1}\qquad;\qquad B=\frac{x^2-4x+4}{x^2-4}\qquad;\qquad C=\frac{2x^2-8}{x^2+4x+4}\]
\end{tamrin}

\begin{tamrin}{البرهان على صحة مساواة}
\begin{enumerate}
\item بيّن أنه من أجل كل عددين حقيقيين $a$ و $b$: $(a+b)^2-(a-b)^2=4ab$.
\item بيّن أنه من أجل كل عدد طبيعي $n$: $(n+1)^2-n^2=2n+1$ ، ثم استنتج $2026^2-2025^2$.
\item بيّن أنه من أجل كل عدد حقيقي $x\neq2$: $\dfrac{x^2-x-2}{x-2}=x+1$.
\end{enumerate}
\end{tamrin}

\section{الشكل النموذجي لثلاثي حدود}
\begin{tamrin}{الشكل النموذجي والتحليل}
اكتب كل ثلاثي حدود على الشكل النموذجي ، ثم حلّله إن أمكن:
\[A=x^2+6x+5\ ;\ B=x^2-4x+7\ ;\ C=2x^2+4x-6\ ;\ D=-x^2+2x+3\ ;\ E=3x^2-6x+3\]
\end{tamrin}

\begin{tamrin}{القيمة الحدية}
\begin{enumerate}
\item بيّن أن $x^2-4x+7\geqslant3$ من أجل كل عدد حقيقي $x$. متى تتحقق المساواة؟
\item عيّن القيمة الحدية العظمى للعبارة $-x^2+2x+3$.
\item مستطيل محيطه $20\ \text{cm}$ ، وأحد بعديه $x$. عبّر عن مساحته $S(x)$ بدلالة $x$ ، ثم عيّن $x$ لتكون المساحة أكبر ما يمكن.
\end{enumerate}
\end{tamrin}

\begin{tamrin}{لمحة تاريخية: الخوارزمي والعدد الذهبي}
\begin{enumerate}
\item حل الخوارزمي المعادلة $x^2+21=10x$. اكتب $x^2-10x+21$ على الشكل النموذجي ، ثم حل المعادلة.
\item العدد الذهبي $\varphi$ هو الحل الموجب للمعادلة $\varphi^2-\varphi-1=0$. باستعمال الشكل النموذجي عيّن $\varphi$.
\end{enumerate}
\end{tamrin}

\section{حل المعادلة $ax^2+bx+c=0$ باستعمال المميز}
\begin{tamrin}{المميز}
\begin{enumerate}
\item حل في $\R$ المعادلات التالية:\par
(أ) $2x^2-5x+2=0$ \qquad (ب) $x^2-6x+9=0$ \qquad (جـ) $x^2+x+1=0$\par
(د) $3x^2+2x-1=0$ \qquad (هـ) $x^2-2x-4=0$ \qquad (و) $-2x^2+3x+2=0$
\item استنتج تحليل العبارات $2x^2-5x+2$ ، $3x^2+2x-1$ و $-2x^2+3x+2$.
\end{enumerate}
\end{tamrin}

\begin{tamrin}{معادلة بوسيط}
$m$ عدد حقيقي. نعتبر المعادلة $x^2+mx+4=0$.
\begin{enumerate}
\item احسب المميز $\Delta$ بدلالة $m$.
\item عيّن قيم $m$ التي تقبل من أجلها المعادلة حلًا مضاعفًا ، وأعط هذا الحل.
\item حل المعادلة من أجل $m=5$.
\end{enumerate}
\end{tamrin}

\section{حل المعادلات والمتراجحات جبريًا}
\begin{tamrin}{معادلات تؤول إلى جداء أو حاصل قسمة}
حل في $\R$ المعادلات التالية:\par
(أ) $x^2=5x$ \qquad (ب) $(2x-1)(x+3)=0$ \qquad (جـ) $\dfrac{x+1}{x-2}=3$ \qquad (د) $\dfrac{x^2-9}{x+3}=0$ \qquad (هـ) $(x-1)^2=4(x+2)^2$
\end{tamrin}

\begin{tamrin}{متراجحات وجدول الإشارة}
حل في $\R$ المتراجحات التالية:\par
(أ) $(x-1)(x+2)\leqslant0$ \qquad (ب) $x^2-4x\geqslant0$ \qquad (جـ) $x^2-x-6<0$\par
(د) $\dfrac{2x-4}{x+1}>0$ \qquad (هـ) $\dfrac{3-x}{x+2}\geqslant0$ \qquad (و) $\dfrac1x\leqslant2$
\end{tamrin}

\begin{tamrin}{إشارة ثلاثي حدود}
\begin{enumerate}
\item حلّل $2x^2-5x+2$ ، ثم ادرس إشارته.
\item استنتج حلول المتراجحة $2x^2-5x+2>0$ ، ثم المتراجحة $2x^2+2\leqslant5x$.
\end{enumerate}
\end{tamrin}

\section{الحل البياني}
\begin{tamrin}{تقاطع منحنى ومستقيم}
$f(x)=x^2-2x-3$ و $g(x)=x+1$.
\begin{enumerate}
\item ارسم في نفس المعلم المنحنى $(C_f)$ والمستقيم $(D)$ الممثل لـ $g$.
\item حل بيانيًا: $f(x)=0$ ، $f(x)=g(x)$ و $f(x)<g(x)$.
\item تحقق جبريًا من النتائج.
\end{enumerate}
\end{tamrin}

\begin{tamrin}{قراءة بيانية}
$f(x)=-x^2+4x$ و $(C_f)$ تمثيلها البياني.
\begin{enumerate}
\item ارسم $(C_f)$ والمستقيم ذا المعادلة $y=3$.
\item حل بيانيًا: $f(x)=3$ ، $f(x)\geqslant0$ و $f(x)>3$.
\item تحقق جبريًا (حلّل $f(x)-3$).
\end{enumerate}
\end{tamrin}

\section{ترييض المشكلات}
\begin{tamrin}{أبعاد مستطيل}
مستطيل محيطه $34\ \text{m}$ ومساحته $60\ \text{m}^2$. ما بعداه؟
\end{tamrin}

\begin{tamrin}{أعداد}
\begin{enumerate}
\item عيّن عددين طبيعيين متتابعين جداؤهما $156$.
\item عيّن عددين مجموعهما $10$ وجداؤهما $21$.
\item «عدد مربعه يساوي ضعفه»: ما هو؟
\end{enumerate}
\end{tamrin}

\begin{tamrin}{حديقة وممر}
\begin{enumerate}
\item حديقة مربعة ضلعها $x$ (بالمتر) يحيط بها ممر عرضه $1\ \text{m}$. المساحة الكلية (الحديقة والممر) $64\ \text{m}^2$. احسب $x$.
\item ما قيم $x$ التي تجعل مساحة مربع ضلعه $x$ أكبر من محيطه؟
\end{enumerate}
\end{tamrin}

\begin{tamrin}{مقارنة عرضين}
يقترح متعامل للهاتف عرضين: \textbf{العرض A}: اشتراك $1000\ DA$ زائد $5\ DA$ للدقيقة ؛ \textbf{العرض B}: $15\ DA$ للدقيقة دون اشتراك.
\begin{enumerate}
\item عبّر عن كلفة كل عرض بدلالة عدد الدقائق $x$.
\item ابتداءً من كم دقيقة يصبح العرض A أفضل؟
\end{enumerate}
\end{tamrin}

\begin{tamrin}{قذف كرة}
يُقذف جسم إلى الأعلى ، فيكون ارتفاعه (بالمتر) بعد $t$ ثانية: $h(t)=-5t^2+20t$ ، حيث $0\leqslant t\leqslant4$.
\begin{enumerate}
\item متى يكون الارتفاع $15\ \text{m}$؟
\item اكتب $h(t)$ على الشكل النموذجي ، ثم عيّن أقصى ارتفاع يبلغه الجسم.
\item خلال أي مدة يكون الارتفاع أكبر من أو يساوي $15\ \text{m}$؟
\end{enumerate}
\end{tamrin}

\section{تمارين شاملة}
\begin{tamrin}{المميز والتحليل والشكل النموذجي}
\begin{enumerate}
\item حل في $\R$ المعادلات التالية: \quad $x^2-4x+3=0$ \ ؛ \ $2x^2-x-10=0$ \ ؛ \ $x^2+x-2=-3$.
\item استنتج تحليلًا للعبارات السابقة.
\item اكتب العبارات السابقة على الشكل النموذجي.
\end{enumerate}
\end{tamrin}

\begin{tamrin}{عبارة من الدرجة الثانية ودالة}
نعتبر في $\R$ العبارة: $E(x)=x^2-x-6$.
\begin{enumerate}
\item حل في $\R$ المعادلة $E(x)=0$.
\item حلّل العبارة $E(x)$ إلى جداء عاملين من الدرجة الأولى.
\item ادرس إشارة $E(x)$ ، ثم استنتج حلول المتراجحة $E(x)<0$.
\item لتكن الدالة $f$ المعرفة على $\R$ بـ: $f(x)=x^2-x-6$.
\begin{sq}
\qa باستعمال الشكل النموذجي ، بيّن أنه من أجل كل $x$ من $\R$: $f(x)=\Big(x-\dfrac12\Big)^2-\dfrac{25}4$.
\qb عيّن السوابق الممكنة للعدد $-6$.
\qc ادرس اتجاه تغير الدالة $f$ على كل من المجالين $\left]-\infty;\dfrac12\right]$ و $\left[\dfrac12;+\infty\right[$ ، ثم شكّل جدول تغيراتها.
\end{sq}
\end{enumerate}
\end{tamrin}

\begin{tamrin}{عبارة كسرية وإشارتها}
نعتبر العبارة $A(x)$ المعرفة على $\R$ بـ: $A(x)=4x^2-8x+3$.
\begin{enumerate}
\item حل في $\R$ المعادلة $A(x)=0$ ، ثم استنتج تحليلًا للعبارة $A(x)$.
\item ادرس إشارة العبارة $A(x)$.
\item نضع: $g(x)=\dfrac{A(x)}{2-x}$.
\begin{sq}
\qa عيّن القيمة الممنوعة للعبارة $g(x)$.
\qb ادرس إشارة العبارة $2-x$ ، ثم استنتج إشارة العبارة $g(x)$.
\qc حل في $\R$ المتراجحة $g(x)\leqslant0$.
\end{sq}
\end{enumerate}
\end{tamrin}

\begin{tamrin}{تحليل وإشارة حاصل قسمة}
لتكن العبارتان الجبريتان: $P(x)=(x+1)(2x+3)-\left(x^2-1\right)$ و $Q(x)=x^2-4x+3$.
\begin{enumerate}
\item حلّل كلًا من العبارتين $P(x)$ و $Q(x)$.
\item نضع $K(x)=\dfrac{P(x)}{Q(x)}$.
\begin{sq}
\qa عيّن القيم الممنوعة للعبارة $K(x)$.
\qb ادرس إشارة العبارة $K(x)$.
\qc استنتج حلول المتراجحة $K(x)\geqslant0$.
\end{sq}
\end{enumerate}
\end{tamrin}

\section{صحيح أم خطأ ، اختيار من متعدد}
\begin{tamrin}{صحيح أم خطأ}
أجب بصحيح أو خطأ مع التعليل:
\begin{enumerate}
\item من أجل كل عددين حقيقيين $a$ و $b$: $(a+b)^2=a^2+b^2$.
\item $x^2=4x$ يكافئ $x=4$.
\item من أجل كل عدد حقيقي $x\neq1$: $\dfrac{x^2-1}{x-1}=x+1$.
\item إذا كان $\Delta<0$ فإن المعادلة $ax^2+bx+c=0$ لا تقبل حلولًا في $\R$.
\item $x^2+x+1>0$ من أجل كل عدد حقيقي $x$.
\item القيمة الممنوعة للعبارة $\dfrac{x+2}{x^2+1}$ هي $-2$.
\end{enumerate}
\end{tamrin}

\begin{tamrin}{اختيار من متعدد}
اختر الإجابة الصحيحة الوحيدة مع التعليل:
\begin{enumerate}
\item الشكل النموذجي للعبارة $x^2-8x+10$ هو:
\ABC{$(x-4)^2+10$}{$(x-4)^2-6$}{$(x+4)^2-6$}
\item مميز العبارة $2x^2-3x-2$ هو:
\ABC{$-7$}{$25$}{$1$}
\item مجموعة حلول المعادلة $x^2=9$ هي:
\ABC{$\{3\}$}{$\{-3;3\}$}{$\{9\}$}
\item مجموعة حلول المتراجحة $x^2<4$ هي:
\ABC{$]-\infty;2[$}{$]-2;2[$}{$]-\infty;-2[\cup]2;+\infty[$}
\item القيم الممنوعة للعبارة $\dfrac1{x^2-4}$ هي:
\ABC{$4$}{$2$ فقط}{$-2$ و $2$}
\end{enumerate}
\end{tamrin}
