\section{العبارات الجبرية وقيمتها العددية}
\begin{tamrin}{القيمة العددية والقيم الممنوعة}
\begin{enumerate}
\item $E(x)=2x^2-3x+5$. احسب $E(-1)$ ، $E(0)$ ، $E\Big(\dfrac12\Big)$ و $E\big(\sqrt2\big)$.
\item $F(x)=\dfrac{x^2-1}{x+3}$. ما هي القيمة الممنوعة؟ احسب $F(0)$ ، $F(1)$ و $F(-2)$.
\item $G(x)=\dfrac1{x-1}-\dfrac2x$. عيّن القيم الممنوعة ، ثم احسب $G(2)$ و $G(-1)$.
\item احسب $-x^2$ و $(-x)^2$ من أجل $x=-3$.
\end{enumerate}
\end{tamrin}
\begin{hal}
\textbf{1.} $E(-1)=2+3+5=\bx{10}$ \quad ؛ \quad $E(0)=\bx{5}$ \quad ؛ \quad $E\Big(\dfrac12\Big)=\dfrac12-\dfrac32+5=\bx{4}$ \quad ؛ \quad $E\big(\sqrt2\big)=4-3\sqrt2+5=\bx{9-3\sqrt2}$
\QQ{2} $x+3=0\iff x=-3$: القيمة الممنوعة \bx{-3}.\par
$F(0)=\dfrac{-1}3=\bx{-\frac13}$ \quad ؛ \quad $F(1)=\dfrac03=\bx{0}$ \quad ؛ \quad $F(-2)=\dfrac{4-1}{1}=\bx{3}$
\QQ{3} المقامان $x-1$ و $x$ ينعدمان عند $1$ و $0$: القيمتان الممنوعتان \bx{0} و \bx{1}.\par
$G(2)=\dfrac11-\dfrac22=\bx{0}$ \quad ؛ \quad $G(-1)=\dfrac1{-2}-\dfrac2{-1}=-\dfrac12+2=\bx{\frac32}$
\QQ{4} $-x^2=-(-3)^2=\bx{-9}$ \quad و \quad $(-x)^2=3^2=\bx{9}$ \quad (الأس يُطبَّق قبل الإشارة $-$).
\end{hal}

\section{النشر ، التحليل والمتطابقات الشهيرة}
\begin{tamrin}{النشر}
انشر ثم بسّط العبارات التالية:
\[A=(2x-3)^2\quad;\quad B=(x+2)(x-5)\quad;\quad C=(3x+1)(3x-1)\]
\[D=(x-1)^2-(x+2)(x-2)\quad;\quad E=2(x+1)^2-(2x-1)^2\]
\end{tamrin}
\begin{hal}
\LW{$(a+b)^2=a^2+2ab+b^2$ \ ؛ \ $(a-b)^2=a^2-2ab+b^2$ \ ؛ \ $(a-b)(a+b)=a^2-b^2$}\par
$A=\bx{4x^2-12x+9}$ \qquad ؛ \qquad $B=x^2-5x+2x-10=\bx{x^2-3x-10}$ \qquad ؛ \qquad $C=\bx{9x^2-1}$\par
$D=x^2-2x+1-\left(x^2-4\right)=\bx{-2x+5}$\par
$E=2\left(x^2+2x+1\right)-\left(4x^2-4x+1\right)=2x^2+4x+2-4x^2+4x-1=\bx{-2x^2+8x+1}$
\end{hal}

\begin{tamrin}{التحليل}
حلّل إلى جداء عوامل:
\[A=9x^2-16\ ;\ B=x^2-6x+9\ ;\ C=4x^2+20x+25\ ;\ D=(x-1)^2-9\]
\[E=(x+1)(2x-3)-(x+1)(x+4)\ ;\ F=(2x+1)^2-(x-3)^2\ ;\ G=x^3-4x\ ;\ H=3x^2-12x\]
\end{tamrin}
\begin{hal}
$A=(3x)^2-4^2=\bx{(3x-4)(3x+4)}$ \qquad ؛ \qquad $B=x^2-2\times3x+3^2=\bx{(x-3)^2}$ \qquad ؛ \qquad $C=(2x)^2+2\times2x\times5+5^2=\bx{(2x+5)^2}$\par
$D=(x-1)^2-3^2=(x-1-3)(x-1+3)=\bx{(x-4)(x+2)}$\par
$E=(x+1)\big[(2x-3)-(x+4)\big]=\bx{(x+1)(x-7)}$ \quad (عامل مشترك $x+1$)\par
$F=\big[(2x+1)-(x-3)\big]\big[(2x+1)+(x-3)\big]=\bx{(x+4)(3x-2)}$\par
$G=x\left(x^2-4\right)=\bx{x(x-2)(x+2)}$ \qquad ؛ \qquad $H=\bx{3x(x-4)}$
\end{hal}

\begin{tamrin}{حساب ذهني بالمتطابقات}
احسب دون آلة حاسبة: \quad $99^2$ \ ؛ \ $101\times99$ \ ؛ \ $1002^2$ \ ؛ \ $1999^2-1998^2$.
\end{tamrin}
\begin{hal}
$99^2=(100-1)^2=10000-200+1=\bx{9801}$ \qquad ؛ \qquad $101\times99=(100+1)(100-1)=10000-1=\bx{9999}$\par
$1002^2=(1000+2)^2=1000000+4000+4=\bx{1004004}$\par
$1999^2-1998^2=(1999-1998)(1999+1998)=1\times3997=\bx{3997}$
\end{hal}

\begin{tamrin}{اختزال عبارة كسرية}
بسّط كل عبارة بعد تعيين قيمها الممنوعة:
\[A=\frac{x^2-1}{x+1}\qquad;\qquad B=\frac{x^2-4x+4}{x^2-4}\qquad;\qquad C=\frac{2x^2-8}{x^2+4x+4}\]
\end{tamrin}
\begin{hal}
\textbf{القاعدة:} نحلل البسط والمقام ، ثم نختزل العامل المشترك بشرط ألا ينعدم.\par
$A$: القيمة الممنوعة $-1$. \quad $A=\dfrac{(x-1)(x+1)}{x+1}=\bx{x-1}$ \quad من أجل $x\neq-1$.\par
$B$: $x^2-4=(x-2)(x+2)$ ، القيمتان الممنوعتان $2$ و $-2$. \quad $B=\dfrac{(x-2)^2}{(x-2)(x+2)}=\bx{\frac{x-2}{x+2}}$ \quad من أجل $x\neq\pm2$.\par
$C$: $x^2+4x+4=(x+2)^2$ ، القيمة الممنوعة $-2$. \quad $C=\dfrac{2(x-2)(x+2)}{(x+2)^2}=\bx{\frac{2(x-2)}{x+2}}$ \quad من أجل $x\neq-2$.
\end{hal}

\begin{tamrin}{البرهان على صحة مساواة}
\begin{enumerate}
\item بيّن أنه من أجل كل عددين حقيقيين $a$ و $b$: $(a+b)^2-(a-b)^2=4ab$.
\item بيّن أنه من أجل كل عدد طبيعي $n$: $(n+1)^2-n^2=2n+1$ ، ثم استنتج $2026^2-2025^2$.
\item بيّن أنه من أجل كل عدد حقيقي $x\neq2$: $\dfrac{x^2-x-2}{x-2}=x+1$.
\end{enumerate}
\end{tamrin}
\begin{hal}
\textbf{1.} $(a+b)^2-(a-b)^2=\left(a^2+2ab+b^2\right)-\left(a^2-2ab+b^2\right)=\bx{4ab}$
\QQ{2} $(n+1)^2-n^2=n^2+2n+1-n^2=\bx{2n+1}$. \quad من أجل $n=2025$: \quad $2026^2-2025^2=2\times2025+1=\bx{4051}$
\QQ{3} $x^2-x-2=(x-2)(x+1)$ (تحقق بالنشر: $x^2+x-2x-2$) \quad إذن من أجل $x\neq2$: \quad $\dfrac{(x-2)(x+1)}{x-2}=\bx{x+1}$
\end{hal}

\section{الشكل النموذجي لثلاثي حدود}
\begin{tamrin}{الشكل النموذجي والتحليل}
اكتب كل ثلاثي حدود على الشكل النموذجي ، ثم حلّله إن أمكن:
\[A=x^2+6x+5\ ;\ B=x^2-4x+7\ ;\ C=2x^2+4x-6\ ;\ D=-x^2+2x+3\ ;\ E=3x^2-6x+3\]
\end{tamrin}
\begin{hal}
\textbf{الطريقة:} \LW{نخرج $a$ عاملًا ، ثم نكمل المربع: $x^2+px=\Big(x+\dfrac p2\Big)^2-\dfrac{p^2}4$.}\par

\section{صحيح أم خطأ ، اختيار من متعدد}
\begin{tamrin}{صحيح أم خطأ}
أجب بصحيح أو خطأ مع التعليل:
\begin{enumerate}
\item من أجل كل عددين حقيقيين $a$ و $b$: $(a+b)^2=a^2+b^2$.
\item $x^2=4x$ يكافئ $x=4$.
\item من أجل كل عدد حقيقي $x\neq1$: $\dfrac{x^2-1}{x-1}=x+1$.
\item إذا كان $\Delta<0$ فإن المعادلة $ax^2+bx+c=0$ لا تقبل حلولًا في $\R$.
\item $x^2+x+1>0$ من أجل كل عدد حقيقي $x$.
\item القيمة الممنوعة للعبارة $\dfrac{x+2}{x^2+1}$ هي $-2$.
\end{enumerate}
\end{tamrin}
\begin{hal}
\textbf{1.} \bxt{خطأ}: $(a+b)^2=a^2+2ab+b^2$. مثال مضاد: $a=b=1$: \ $4\neq2$.
\QQ{2} \bxt{خطأ}: $x^2=4x\iff x(x-4)=0\iff x=0$ أو $x=4$.
\QQ{3} \bxt{صحيح}: $\dfrac{(x-1)(x+1)}{x-1}=x+1$ من أجل $x\neq1$.
\QQ{4} \bxt{صحيح}: من الشكل النموذجي $\Big(x+\dfrac b{2a}\Big)^2=\dfrac\Delta{4a^2}<0$ مستحيلة.
\QQ{5} \bxt{صحيح}: $x^2+x+1=\Big(x+\dfrac12\Big)^2+\dfrac34\geqslant\dfrac34>0$.
\QQ{6} \bxt{خطأ}: المقام $x^2+1\geqslant1$ لا ينعدم ، فلا توجد قيمة ممنوعة ($-2$ تعدم البسط فقط).
\end{hal}

\begin{tamrin}{اختيار من متعدد}
اختر الإجابة الصحيحة الوحيدة مع التعليل:
\begin{enumerate}
\item الشكل النموذجي للعبارة $x^2-8x+10$ هو:
\ABC{$(x-4)^2+10$}{$(x-4)^2-6$}{$(x+4)^2-6$}
\item مميز العبارة $2x^2-3x-2$ هو:
\ABC{$-7$}{$25$}{$1$}
\item مجموعة حلول المعادلة $x^2=9$ هي:
\ABC{$\{3\}$}{$\{-3;3\}$}{$\{9\}$}
\item مجموعة حلول المتراجحة $x^2<4$ هي:
\ABC{$]-\infty;2[$}{$]-2;2[$}{$]-\infty;-2[\cup]2;+\infty[$}
\item القيم الممنوعة للعبارة $\dfrac1{x^2-4}$ هي:
\ABC{$4$}{$2$ فقط}{$-2$ و $2$}
\end{enumerate}
\end{tamrin}
\begin{hal}
\textbf{1.} (ب): $x^2-8x+10=(x-4)^2-16+10=\bx{(x-4)^2-6}$
\QQ{2} (ب): $\Delta=9+16=\bx{25}$
\QQ{3} (ب): $x^2-9=(x-3)(x+3)=0$ \quad ؛ \quad \bx{S=\{-3;3\}}
\QQ{4} (ب): $x^2-4=(x-2)(x+2)<0$ بين الجذرين \quad ؛ \quad \bx{S=\,]-2;2[}
\QQ{5} (جـ): $x^2-4=0\iff x=-2$ أو $x=2$.
\end{hal}
