\section{الحساب الشعاعي دون معلم}
% ------------------------------------------------- 1
\begin{tamrin}{متوازي أضلاع ونقط معرّفة شعاعياً}
\begin{enumerate}
\item
\begin{sq}
\qa أنشئ متوازي الأضلاع $ABCD$ الذي مركزه $O$ حيث $AB=6\,cm$ و $AD=4\,cm$.
\qb أنشئ النقط $E$ ، $F$ و $G$ حيث: \ $\V{AE}=\dfrac14\V{AC}$ ، \ $\V{AF}=\dfrac13\V{DC}$ \ و \ $\V{OG}=\V{OC}-\V{OD}$.
\end{sq}
\item بيّن أن الرباعي $ABGO$ متوازي أضلاع.
\item
\begin{sq}
\qa بيّن أن: $\V{EB}+\V{ED}=2\V{EO}$ \ وأن \ $\V{EB}+\V{ED}=2\V{AE}$.
\qb استنتج أن النقطة $E$ هي منتصف $[AO]$.
\end{sq}
\item
\begin{sq}
\qa بيّن أن: $4\V{DE}=3\V{DF}$.
\qb هل النقط $D$ ، $E$ و $F$ في استقامية؟ علّل.
\end{sq}
\item
\begin{sq}
\qa برّر لماذا يمكن اعتبار $\big(A;\V{AO},\V{AB}\big)$ معلماً للمستوي.
\qb عيّن إحداثيات النقط $A$ ، $O$ ، $B$ ، $C$ ، $D$ ، $E$ ، $F$ و $G$ في هذا المعلم.
\qc باستعمال الإحداثيات تحقق من صحة النتائج 3(ب) و 4(ب).
\end{sq}
\end{enumerate}
\end{tamrin}
\begin{hal}
\RL{0.6}{%
\textbf{1.} الشكل مقابل: $E$ على $[AC]$ و $AE=\dfrac14AC$ ، و $F$ على $[AB]$ حيث $AF=\dfrac13AB=2\,cm$ (لأن $\V{DC}=\V{AB}$).
\QQ{2} $\V{OG}=\V{OC}-\V{OD}=\V{OC}+\V{DO}=\V{DO}+\V{OC}=\V{DC}=\V{AB}$.\par
$\V{AB}=\V{OG}$ ، إذن \textbf{$ABGO$ متوازي أضلاع}.}{0.37}{%
\begin{LTR}\centering
\begin{tikzpicture}[scale=0.55,font=\scriptsize]
\coordinate (A) at (0,0); \coordinate (B) at (6,0); \coordinate (D) at (1.5,3.7); \coordinate (C) at (7.5,3.7);
\coordinate (O) at (3.75,1.85); \coordinate (E) at (1.875,0.925); \coordinate (F) at (2,0); \coordinate (G) at (9.75,1.85);
\draw[thick] (A)--(B)--(C)--(D)--cycle; \draw[gray] (A)--(C) (B)--(D);
\draw[cRem,dashed] (B)--(G)--(O); \draw[cThm,thick] (D)--(F);
\foreach \p/\n/\a in {A/A/below left,B/B/below,C/C/above,D/D/above left,O/O/above,E/E/above left,F/F/below,G/G/right} \fill (\p) circle (2pt) node[\a]{$\n$};
\end{tikzpicture}
\end{LTR}}
\QQ{3}
\begin{sq}
\qa $O$ منتصف $[BD]$ ، إذن $\V{EB}+\V{ED}=2\V{EO}$ (خاصية المنتصف).\par
$\V{AE}=\dfrac14\V{AC}=\dfrac14\times2\V{AO}=\dfrac12\V{AO}$ ، و $\V{EO}=\V{EA}+\V{AO}=-\dfrac12\V{AO}+\V{AO}=\dfrac12\V{AO}=\V{AE}$ ، ومنه $2\V{EO}=2\V{AE}$.
\qb $\V{EO}=\V{AE}$ ، أي $\V{AE}=\V{EO}$: \ \textbf{$E$ منتصف $[AO]$}.
\end{sq}
\QQ{4}
\begin{sq}
\qa $\V{DE}=\V{DA}+\V{AE}=\V{DA}+\dfrac14\big(\V{AB}+\V{AD}\big)=-\dfrac34\V{AD}+\dfrac14\V{AB}$ ، إذن $4\V{DE}=-3\V{AD}+\V{AB}$.\par
$\V{DF}=\V{DA}+\V{AF}=-\V{AD}+\dfrac13\V{AB}$ ، إذن $3\V{DF}=-3\V{AD}+\V{AB}$. \quad إذن \bx{4\V{DE}=3\V{DF}}.
\qb $\V{DE}=\dfrac34\V{DF}$: الشعاعان مرتبطان خطياً ولهما المبدأ $D$ ، إذن \textbf{$D$ ، $E$ ، $F$ في استقامية}.
\end{sq}
\QQ{5}
\begin{sq}
\qa $A$ ، $O$ ، $B$ ليست في استقامية ، فالشعاعان $\V{AO}$ و $\V{AB}$ غير مرتبطين خطياً: $\big(A;\V{AO},\V{AB}\big)$ معلم.
\qb $A(0;0)$ ، $O(1;0)$ ، $B(0;1)$ ، $C(2;0)$ (لأن $\V{AC}=2\V{AO}$) ، $D(2;-1)$ (لأن $O$ منتصف $[BD]$) ،\par
$E\Big(\dfrac12;0\Big)$ ، $F\Big(0;\dfrac13\Big)$ (لأن $\V{DC}\cp{0}{1}$) ، $G(1;1)$ (لأن $\V{OG}=\V{DC}$).
\qc منتصف $[AO]$: $\Big(\dfrac12;0\Big)=E$ \ok. \quad $\V{DE}\cp{-\frac32}{1}$ و $\V{DF}\cp{-2}{\frac43}$: \ $-\dfrac32\times\dfrac43-(-2)\times1=-2+2=0$ \ok.
\end{sq}
\end{hal}

\begin{tamrin}{استقامية ومعلم كيفي}
$ABCD$ متوازي أضلاع ، نعتبر النقطتين $E$ و $F$ حيث: $\V{DE}=\dfrac52\V{DA}$ و $\V{CF}=\dfrac23\V{DC}$.
\begin{enumerate}
\item بيّن أن: $\V{BF}=\dfrac23\V{DC}+\V{BC}$ \ و \ $\V{BE}=\dfrac32\V{DA}-\V{AB}$.
\item اكتب الشعاعين $\V{BE}$ و $\V{BF}$ بدلالة الشعاعين $\V{AB}$ و $\V{BC}$.
\item بيّن أن $2\V{BE}+3\V{BF}=\vec0$. ماذا تستنتج؟
\item لتكن $A'$ نظيرة $A$ بالنسبة إلى $D$ ، و $M$ منتصف $[CD]$.
\begin{sq}
\qa بيّن لماذا يمكن اعتبار $\big(B;\V{BC},\V{BA}\big)$ معلماً للمستوي.
\qb عيّن إحداثيات النقط $A$ ، $B$ ، $C$ ، $D$ ، $M$ و $A'$ في هذا المعلم.
\qc بيّن أن النقطة $M$ هي منتصف القطعة $[BA']$.
\end{sq}
\end{enumerate}
\end{tamrin}
\begin{hal}
\textbf{1.} $\V{BF}=\V{BC}+\V{CF}=\V{BC}+\dfrac23\V{DC}$ \ok. \quad $\V{BE}=\V{BA}+\V{AD}+\V{DE}=-\V{AB}-\V{DA}+\dfrac52\V{DA}=\dfrac32\V{DA}-\V{AB}$ \ok.
\QQ{2} في متوازي الأضلاع: $\V{DC}=\V{AB}$ و $\V{DA}=\V{CB}=-\V{BC}$:
\[\bx{\V{BF}=\frac23\V{AB}+\V{BC}}\qquad;\qquad\bx{\V{BE}=-\V{AB}-\frac32\V{BC}}.\]
\QQ{3} $2\V{BE}+3\V{BF}=-2\V{AB}-3\V{BC}+2\V{AB}+3\V{BC}=\vec0$.\par
إذن $\V{BE}=-\dfrac32\V{BF}$: الشعاعان مرتبطان خطياً ، و\textbf{النقط $B$ ، $E$ ، $F$ في استقامية}.
\QQ{4}
\begin{sq}
\qa $B$ ، $C$ ، $A$ ليست في استقامية ($ABC$ مثلث) ، فالشعاعان $\V{BC}$ و $\V{BA}$ غير مرتبطين خطياً.

\end{sq}
\end{hal}

\begin{tamrin}{إنشاء نقط وعلاقات شعاعية}
$A$ ، $B$ و $C$ ثلاث نقط مختلفة مثنى مثنى وليست في استقامية.
\begin{enumerate}
\item علّم النقطة $D$ بحيث: $\V{DA}-3\V{DC}=\vec0$.
\item علّم النقطة $E$ بحيث: $\V{BE}=\V{BA}+\V{BD}$.
\item برهن أن $AEDB$ متوازي أضلاع.
\item برهن أنه من أجل كل نقطة $M$ من المستوي: \ $\V{MA}-3\V{MC}=-2\V{MD}$.
\item ليكن $O$ مركز متوازي الأضلاع $AEDB$. \ برهن أن: $\V{AB}+\V{AE}=2\V{AO}$.
\end{enumerate}
\end{tamrin}
\begin{hal}
\textbf{1.} $\V{DA}-3\big(\V{DA}+\V{AC}\big)=\vec0\iff-2\V{DA}=3\V{AC}\iff\bx{\V{AD}=\dfrac32\V{AC}}$: \ $D$ على $(AC)$ وراء $C$ حيث $AD=1{,}5\,AC$.
\QQ{2} $E$ هي الرأس الرابع لمتوازي الأضلاع المنشأ على $\V{BA}$ و $\V{BD}$ (قاعدة متوازي الأضلاع).
\QQ{3} $\V{BE}=\V{BA}+\V{BD}\iff\V{BE}-\V{BA}=\V{BD}\iff\V{AE}=\V{BD}$: \ \textbf{$AEDB$ متوازي أضلاع}.
\QQ{4} $\V{MA}-3\V{MC}=\big(\V{MD}+\V{DA}\big)-3\big(\V{MD}+\V{DC}\big)=-2\V{MD}+\underbrace{\V{DA}-3\V{DC}}_{=\vec0}=-2\V{MD}$ \ok.
\QQ{5} في متوازي الأضلاع $AEDB$: $\V{AB}+\V{AE}=\V{AD}$ ، و $O$ منتصف القطر $[AD]$: $\V{AD}=2\V{AO}$ ، إذن $\V{AB}+\V{AE}=2\V{AO}$ \ok.
\end{hal}

\begin{tamrin}{منتصفات في متوازي أضلاع}

\begin{enumerate}
\item برهن أن: $\V{AC}+\V{BD}=2\V{BC}$.
\item عبّر عن $\V{AE}+\V{AF}$ بدلالة $\V{AC}$.
\end{enumerate}
\end{tamrin}
\begin{hal}
\textbf{1.} $\V{AC}=\V{AB}+\V{BC}$ و $\V{BD}=\V{BA}+\V{AD}=-\V{AB}+\V{BC}$ (لأن $\V{AD}=\V{BC}$). \ بالجمع: $\V{AC}+\V{BD}=2\V{BC}$ \ok.
\QQ{2} $\V{AE}=\V{AD}+\V{DE}=\V{AD}+\dfrac12\V{AB}$ \ و \ $\V{AF}=\V{AB}+\V{BF}=\V{AB}+\dfrac12\V{AD}$.
\[\V{AE}+\V{AF}=\frac32\V{AB}+\frac32\V{AD}=\frac32\big(\V{AB}+\V{AD}\big)=\bx{\frac32\V{AC}}.\]
\begin{molahada}[تصحيح]
نص السؤال الأصلي يطلب إثبات $\V{AE}+\V{AF}=2\V{AC}$ ، وهذا خطأ: الصحيح $\V{AE}+\V{AF}=\dfrac32\V{AC}$. \ (تحقق في معلم $\big(A;\V{AB},\V{AD}\big)$: $E\big(\frac12;1\big)$ ، $F\big(1;\frac12\big)$ ، $C(1;1)$ ، والمجموع $\big(\frac32;\frac32\big)$.)
\end{molahada}
\end{hal}

\begin{tamrin}{وضعية: الطاولات والتلاميذ}

\end{tamrin}
\begin{hal}

\end{itemize}

\end{hal}
