\begin{tamrin}{الشكل النموذجي ودالة تآلفية}
$f$ الدالة المعرّفة على $\R$ بـ: $f(x)=x^2+2x-1$ ، و $(C)$ تمثيلها البياني في معلم متعامد ومتجانس $(O;\vec i,\vec j)$.
\begin{enumerate}
\item بيّن أنه يمكن كتابة $f(x)$ على الشكل $f(x)=(x+a)^2+b$ ، مع تعيين $a$ و $b$.
\item ادرس اتجاه تغير الدالة $f$ على المجالين $]-\infty;-1]$ و $[-1;+\infty[$.
\item شكّل جدول تغيرات الدالة $f$ ، ثم استنتج القيمة الحدية للدالة $f$ وبيّن نوعها.
\item ارسم المنحنى $(C)$ اعتماداً على التمثيل البياني للدالة مربع.
\item لتكن $g$ الدالة التآلفية المعرّفة على $\R$ حيث $g(-2)=-1$ و $g(1)=2$.
\begin{sq}
\qa أعط عبارة $g(x)$.
\qb ارسم التمثيل البياني لـ $g$ في نفس المعلم السابق.
\qc حل بيانياً ثم جبرياً في $\R$: \ $f(x)=g(x)$ و $f(x)>g(x)$.
\end{sq}
\end{enumerate}
\end{tamrin}
\begin{hal}
\textbf{1.} $x^2+2x-1=(x+1)^2-1-1=(x+1)^2-2$: \ \bx{a=1} و \bx{b=-2}.
\QQ{2} ليكن $u<v$.
\begin{itemize}
\item على $[-1;+\infty[$: $0\le u+1<v+1\Rightarrow(u+1)^2<(v+1)^2\Rightarrow f(u)<f(v)$: \textbf{متزايدة تماماً}.
\item على $]-\infty;-1]$: $u+1<v+1\le0\Rightarrow(u+1)^2>(v+1)^2\Rightarrow f(u)>f(v)$: \textbf{متناقصة تماماً}.
\end{itemize}

\begin{tamrin}{دراسة شاملة لدالة من الدرجة الثانية}
$h$ الدالة المعرّفة بـ: $h(x)=x^2-6x+3$ ، و $(C_h)$ تمثيلها البياني في معلم متعامد ومتجانس.
\begin{enumerate}
\item جد مجموعة تعريف الدالة $h$ ، ثم احسب صور الأعداد $-2$ ، $0$ و $1$.
\item تحقق أنه من أجل كل عدد حقيقي $x$: \ $h(x)=(x-3)^2-6$.
\item احسب السوابق الممكنة للأعداد $3$ ، $-2$ و $-5$.
\item ادرس اتجاه تغير $h$ على $]-\infty;3]$ و $[3;+\infty[$ ، ثم شكّل جدول تغيراتها.
\item تحقق أن $h(x)+6\ge0$ من أجل كل $x$. ماذا يمثل العدد $-6$ بالنسبة للدالة $h$؟
\item عيّن نقطة تقاطع $(C_h)$ مع محور التراتيب.
\item هل يمكن رسم $(C_h)$ انطلاقاً من منحنى الدالة المرجعية $x\mapsto x^2$؟ علّل ، ثم ارسمه.
\item انطلاقاً من $(C_h)$ ، ما هي شفعية الدالة $h$؟
\end{enumerate}
\end{tamrin}
\begin{hal}
\textbf{1.} كثير حدود: \bx{D_h=\R}. \quad $h(-2)=4+12+3=\bx{19}$ ، $h(0)=\bx{3}$ ، $h(1)=1-6+3=\bx{-2}$.
\QQ{2} $(x-3)^2-6=x^2-6x+9-6=x^2-6x+3=h(x)$ \ok.
\QQ{3}
\begin{itemize}
\item $h(x)=3\iff(x-3)^2=9\iff x-3=\pm3\iff x=0$ أو $x=6$.
\item $h(x)=-2\iff(x-3)^2=4\iff x=1$ أو $x=5$.
\item $h(x)=-5\iff(x-3)^2=1\iff x=2$ أو $x=4$.
\end{itemize}
\QQ{4} ليكن $u<v$: على $[3;+\infty[$: $0\le u-3<v-3\Rightarrow h(u)<h(v)$ (متزايدة) ؛ على $]-\infty;3]$: $u-3<v-3\le0\Rightarrow h(u)>h(v)$ (متناقصة).

\QQ{5} $h(x)+6=(x-3)^2\ge0$ ، أي $h(x)\ge-6=h(3)$: \ $-6$ هي \textbf{القيمة الحدية الصغرى} لـ $h$ (عند $x=3$).
\RL{0.55}{%

\begin{tamrin}{حساب الجيب من جيب التمام والعكس}
\begin{enumerate}
\item احسب $\sin x$ إذا علمت أن $\cos x=-\dfrac{\sqrt3}2$ و $x\in\,]-\pi;0[$.
\item احسب $\cos x$ إذا علمت أن $\sin x=\dfrac12$ و $x\in\Big[0;\dfrac\pi2\Big[$.
\item احسب $\cos x$ إذا علمت أن $\sin x=\dfrac35$ و $x\in\Big[\dfrac\pi2;\pi\Big]$.
\item نضع $\cos x=-\dfrac35$ حيث $x\in\Big[\dfrac\pi2;\pi\Big]$. \ احسب $\sin x$ ، ثم $\cos(\pi-x)+\sin(2\pi-x)$.
\end{enumerate}
\end{tamrin}
\begin{hal}
\textbf{1.} $\sin^2x=1-\cos^2x=1-\dfrac34=\dfrac14$ ، و $x\in\,]-\pi;0[$ معناه $\sin x<0$: \ \bx{\sin x=-\dfrac12} \ (أي $x=-\dfrac{5\pi}6$).
\QQ{2} $\cos^2x=1-\dfrac14=\dfrac34$ ، و $\cos x>0$ على $\Big[0;\dfrac\pi2\Big[$: \ \bx{\cos x=\dfrac{\sqrt3}2}.
\QQ{3} $\cos^2x=1-\dfrac9{25}=\dfrac{16}{25}$ ، و $\cos x\le0$ على $\Big[\dfrac\pi2;\pi\Big]$: \ \bx{\cos x=-\dfrac45}.
\QQ{4} $\sin^2x=1-\dfrac9{25}=\dfrac{16}{25}$ ، و $\sin x\ge0$ على $\Big[\dfrac\pi2;\pi\Big]$: \ \bx{\sin x=\dfrac45}.\par
$\cos(\pi-x)+\sin(2\pi-x)=-\cos x+\sin(-x)=-\cos x-\sin x=\dfrac35-\dfrac45=\bx{-\dfrac15}$.
\end{hal}

\begin{tamrin}{صحيح أم خطأ}

\item من أجل كل عدد حقيقي $x$: $\sqrt{x^2}=x$.
\item الدالة مكعب متزايدة تماماً على $\R$.
\item منحنى الدالة $x\mapsto(x-3)^2+1$ رأسه النقطة $S(-3;1)$.
\end{enumerate}
\end{tamrin}
\begin{hal}

\QQ{2} \textbf{صحيح:} $\vec v\cp13$ يعطي معادلة $3x-y+c=0$ ، و $A$: $\dfrac32-\dfrac32+c=0$ ، $c=0$.

\QQ{4} \textbf{صحيح:} $x^2-x=x(x-1)<0$.
\QQ{5} \textbf{خطأ:} $\sqrt{x^2}=|x|$ (مثلاً $\sqrt{(-2)^2}=2\ne-2$).
\QQ{6} \textbf{صحيح.}
\QQ{7} \textbf{خطأ:} الرأس هو $S(3;1)$ (الانسحاب بالشعاع $\cp31$).
\end{hal}
