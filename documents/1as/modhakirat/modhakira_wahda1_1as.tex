\Row{بناء التعلّمات: تعاريف}{%
\begin{itemize}
\item $\N=\{0;1;2;\ldots\}$: مجموعة \textbf{الأعداد الطبيعية}.
\item $\Z=\{\ldots;-2;-1;0;1;2;\ldots\}$: مجموعة \textbf{الأعداد الصحيحة النسبية}.
\item $\D$: مجموعة \textbf{الأعداد العشرية} ، أي الأعداد التي تُكتب $\dfrac{a}{10^{n}}$ حيث $a\in\Z$ و $n\in\N$.
\item $\Q$: مجموعة \textbf{الأعداد الناطقة} ، أي الأعداد التي تُكتب $\dfrac{a}{b}$ حيث $a\in\Z$ و $b\in\Z^{*}$.
\item $\R$: مجموعة \textbf{الأعداد الحقيقية} (فواصل نقط المستقيم العددي). العدد الحقيقي غير الناطق يسمى \textbf{أصمّ}.
\end{itemize}
\textbf{خاصية:} \bx{\N\subset\Z\subset\D\subset\Q\subset\R}. \quad \textbf{ترميز:} $3\in\N$ ، $\frac23\notin\D$ ، $\R^{*}=\R-\{0\}$ ، $\R_{+}$ ، $\R_{-}$.}{12 د}{تُدوَّن في الكراريس. نرسم مخطط المجموعات المتداخلة على السبورة.}
\Row{أمثلة}{%
\begin{itemize}
\item $\frac{7}{4}=1{,}75=\frac{175}{100}$: إذن $\frac74\in\D$.
\item $\frac{1}{3}\in\Q$ و $\frac13\notin\D$.
\item $-\sqrt{16}=-4\in\Z$ و $-4\notin\N$.
\item $1+\sqrt2\notin\Q$ ، و $1+\sqrt2\in\R$.
\end{itemize}}{8 د}{نُشرك التلاميذ في تعليل كل انتماء.}

\textbf{تمرين:} أكمل بأحد الرمزين $\in$ أو $\notin$:\quad

\ModhHead{10}{1 ج م ع ت}{الأعداد والحساب}{الأعداد والحساب}{1 سا}{القيمة المطلقة والمسافة}%
{\begin{itemize}\item المستقيم العددي والمجالات.\end{itemize}}%
{\begin{itemize}\item كتابة عبارة بقيمة مطلقة دون الرمز.\item التعبير عن مجال بأربع صيغ: مجال ، حصر ، مسافة ، قيمة مطلقة.\end{itemize}}%
{الكتاب المدرسي (باب «الترتيب في $\R$»: نشاط 4 المسافة).}
\begin{Sayr}
\Row{وضعية انطلاقية}{%
على المستقيم العددي: ما المسافة بين النقطتين اللتين فاصلتاهما $-2$ و $5$؟ وبين $5$ و $-2$؟}{5 د}{المسافة موجبة ولا تتعلق بالترتيب.}
\Row{النشاط}{%
(نشاط 4 من الكتاب) عيّن الأعداد $x$ التي تبعد عن $3$ بأقل من $2$ أو تساوي $2$. مثّلها.}{12 د}{عمل ثنائي.}
\Row{مناقشة النشاط}{%
$1\leqslant x\leqslant5$ ، أي $x\in[1;5]$: مركز المجال $3$ ونصف قطره $2$.}{8 د}{نربط المسافة بمركز المجال ونصف قطره.}
\Row{بناء التعلّمات}{%
\begin{itemize}\item $|x|=x$ إذا كان $x\geqslant0$ ، و $|x|=-x$ إذا كان $x\leqslant0$.\item المسافة بين $a$ و $b$: $d(a;b)=|a-b|=|b-a|$.\item $|xy|=|x|\,|y|$ ، $\sqrt{x^2}=|x|$ ، $|x+y|\leqslant|x|+|y|$.\item من أجل $r>0$: $|x-a|\leqslant r\iff a-r\leqslant x\leqslant a+r\iff x\in[a-r;a+r]$.\end{itemize}}{15 د}{نلخص الصيغ الأربع في جدول.}
\Row{أمثلة}{%
$|\sqrt2-2|=2-\sqrt2$ (لأن $\sqrt2<2$) ؛ \ $[-1;7]$: المركز $3$ ونصف القطر $4$ ، إذن $x\in[-1;7]\iff|x-3|\leqslant4$.}{8 د}{نطلب تبرير إشارة ما داخل القيمة المطلقة.}
\Row{تقويم}{%
اكتب بمجال: $|x+1|<3$. اكتب $[2;8]$ بقيمة مطلقة.\par\textbf{الحل:} $]-4;2[$ ؛ $|x-5|\leqslant3$.}{12 د}{تقويم فردي.}

\Row{مناقشة النشاط}{%
\begin{itemize}\item $|x-2|=3\iff x=5$ أو $x=-1$.\item $|x+1|=|x-3|$: $x$ على نفس البعد من $-1$ و $3$ ، إذن $x=1$.\item $|2x-1|\leqslant5\iff-5\leqslant2x-1\leqslant5\iff-2\leqslant x\leqslant3$.\end{itemize}}{10 د}{نعرض الطريقتين: المسافة ، والحساب.}
