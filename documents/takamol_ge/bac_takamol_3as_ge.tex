\RL{0.48}{\textbf{دالة أصلية:} $F$ دالة أصلية لـ $f$ على $I$ يعني $\LW{F'(x)=f(x)}$ من أجل كل $x$ من $I$.\par
\textbf{التكامل:} $\LW{\displaystyle\int_a^bf(x)\,dx=F(b)-F(a)}$.\par
\textbf{القيمة المتوسطة:} $\LW{\mu=\dfrac1{b-a}\displaystyle\int_a^bf(x)\,dx}$.}{0.48}{\textbf{المساحة:} إذا كان $(C_f)$ فوق $(C_g)$ على $[a;b]$ فإن $\LW{\mathcal A=\displaystyle\int_a^b\big[f(x)-g(x)\big]dx}$ (بوحدة المساحة).\par
\textbf{وحدة المساحة:} إذا كانت الوحدة على المحورين $u$ و $v$ (بالسنتيمتر) فإن $\LW{1\,ua=u\times v\ \text{cm}^2}$.\par
\textbf{قبل كل مساحة:} ندرس الوضع النسبي ، ونرسم الحيز.}

\begin{tamrin}{تمرين من بكالوريا 2022 -- الموضوع الأول}
المستوي منسوب إلى معلم $(O;\vec i,\vec j)$. \ $(C_f)$ و $(C_g)$ التمثيلان البيانيان للدالتين $f$ و $g$ المعرفتين على $\R$ بـ:
\[g(x)=(x+1)^2(x-1)\qquad;\qquad f(x)=ax^2+bx-1\quad\text{\ar{حيث $a$ و $b$ عددان حقيقيان.}}\]
\begin{enumerate}
\item \pa بيّن أنه من أجل كل عدد حقيقي $x$: $g(x)=x^3+x^2-x-1$.
\pb عيّن العددين $a$ و $b$ حتى تكون $g$ دالة أصلية لـ $f$ على $\R$.
\item تحقق أنه من أجل كل عدد حقيقي $x$: $f(x)=(x+1)(3x-1)$.
\item \pa حلّل العبارة $g(x)-f(x)$.
\pb استنتج أن $(C_f)$ و $(C_g)$ يتقاطعان في ثلاث نقط يطلب تعيينها.
\item \textbf{(سؤال إضافي)} ادرس الوضع النسبي لـ $(C_f)$ و $(C_g)$ ، ثم احسب مساحة الحيز المحدد بالمنحنيين والمستقيمين $x=0$ و $x=3$.
\end{enumerate}
\end{tamrin}

\pb $g$ دالة أصلية لـ $f$ يعني $g'(x)=f(x)$: \quad $g'(x)=3x^2+2x-1=ax^2+bx-1$ \quad إذن \quad \bx{a=3} \ و \ \bx{b=2}
\QQ{2} $(x+1)(3x-1)=3x^2-x+3x-1=3x^2+2x-1=f(x)$ \ok

\section{الكلفة الإجمالية دالة أصلية للكلفة الهامشية (\Ss)}

\begin{tamrin}{تمرين من بكالوريا 2014 -- الموضوع الثاني (\Ss\ بعد درس الأسية)}
$f$ الدالة المعرفة على $[0;7]$ بـ: $f(x)=6(1-2x)e^{-x}+5$.
\begin{enumerate}
\item \pa عيّن العددين الحقيقيين $a$ و $b$ بحيث تكون الدالة $g$ المعرفة على $[0;7]$ بـ: $g(x)=(ax+b)e^{-x}$ دالة أصلية للدالة $h:x\mapsto6(1-2x)e^{-x}$.
\pb استنتج دالة أصلية للدالة $f$ على $[0;7]$.
\item الكلفة الهامشية $C_m$ لصناعة كمية $x$ (مقدرة بالطن) من منتوج ، حيث $x\in[0;7]$ ، تُنمذج بالدالة $f$: $C_m(x)=f(x)$ (الكلفة مقدرة بملايين الدنانير). نذكّر أن دالة الكلفة الإجمالية دالة أصلية لدالة الكلفة الهامشية.
\pa بيّن أن الكلفة الإجمالية $C_T$ معرفة بـ: $C_T(x)=(12x+6)e^{-x}+5x+k$ حيث $k$ عدد حقيقي.
\pb حدّد قيمة $k$ إذا علمت أن الكلفة الثابتة $2$ مليون دينار (أي $C_T(0)=2$).
\end{enumerate}
\end{tamrin}
\begin{hal}
\textbf{1.} \pa $g'(x)=ae^{-x}-(ax+b)e^{-x}=(-ax+a-b)e^{-x}$ ، ونريد $g'(x)=(-12x+6)e^{-x}$:\par
بالمطابقة: $-a=-12$ و $a-b=6$ \quad إذن \quad \bx{a=12} \ و \ \bx{b=6}
\pb دالة أصلية لـ $f$: \quad \bx{F(x)=(12x+6)e^{-x}+5x}
\QQ{2} \pa $C_T$ دالة أصلية لـ $C_m=f$ ، والدوال الأصلية لـ $f$ تختلف بثابت: \quad $C_T(x)=(12x+6)e^{-x}+5x+k$ \ok
\pb $C_T(0)=6+k=2$ \quad ومنه \quad \bx{k=-4} \quad و \quad $C_T(x)=(12x+6)e^{-x}+5x-4$.
\end{hal}
