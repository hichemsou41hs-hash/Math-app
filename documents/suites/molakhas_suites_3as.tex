\section{توليد متتالية وتمثيلها البياني}
\begin{tarif}
\textbf{المتتالية العددية} $\LW{(u_n)}$ هي دالة ترفق بكل عدد طبيعي $\LW{n}$ (ابتداءً من رتبة $\LW{n_0}$) عددًا حقيقيًا $\LW{u_n}$ يسمى \textbf{الحد العام} أو الحد ذا الدليل $\LW{n}$.
\begin{itemize}
\item \textbf{التوليد الصريح:} $\LW{u_n=f(n)}$؛ نحسب أي حد مباشرة بدلالة $\LW{n}$.
\item \textbf{التوليد بعلاقة تراجعية:} نعطي الحد الأول $\LW{u_0}$ وعلاقة $\LW{u_{n+1}=f(u_n)}$؛ نحسب الحدود واحدًا بعد الآخر.
\end{itemize}
\end{tarif}

\begin{mithal}
\begin{itemize}
\item $u_n=\dfrac{2n+1}{n+1}$ (توليد صريح): $u_0=1$ ، $u_1=\dfrac{3}{2}$ ، $u_2=\dfrac{5}{3}$.
\item $u_0=1$ و $u_{n+1}=2u_n+3$ (توليد تراجعي): $u_1=5$ ، $u_2=13$ ، $u_3=29$.
\end{itemize}
\end{mithal}

\begin{molahada}
لا نخلط بين $u_{n+1}$ (الحد الذي يلي $u_n$) و $u_n+1$ (الحد $u_n$ مضافًا إليه $1$). مثلًا إذا كان $u_n=n^{2}$ فإن $u_{n+1}=(n+1)^{2}=n^{2}+2n+1$ بينما $u_n+1=n^{2}+1$.
\begin{itemize}
\item \textbf{الدليل والرتبة:} في $u_{16}$ العدد $16$ هو \textbf{الدليل}، أما \textbf{الرتبة} فهي موضع الحد في القائمة: $u_{16}$ هو الحد السابع عشر إذا بدأت المتتالية من $u_0$، والسادس عشر إذا بدأت من $u_1$.
\end{itemize}
\end{molahada}

\begin{tariqa}[تمثيل الحدود الأولى لمتتالية $u_{n+1}=f(u_n)$]
\begin{enumerate}
\item نرسم المنحنى $\LW{(C_f)}$ والمستقيم $\LW{(\Delta): y=x}$.
\item نعلّم $\LW{u_0}$ على محور الفواصل، ونصعد شاقوليًا إلى $\LW{(C_f)}$: ترتيب النقطة هو $\LW{u_1=f(u_0)}$.
\item ننتقل أفقيًا إلى $\LW{(\Delta)}$، ثم ننزل شاقوليًا إلى محور الفواصل فنحصل على $\LW{u_1}$ عليه.
\item نكرر العملية للحصول على $\LW{u_2}$ ، $\LW{u_3}$ ، $\LW{\ldots}$
\end{enumerate}
\end{tariqa}

\begin{mithal}[سلّم وحلزون]
على اليمين: $u_0=0$ و $u_{n+1}=\frac{1}{2}u_n+2$ ؛ نجد $u_1=2$ ، $u_2=3$ ، $u_3=\frac{7}{2}$ : الحدود تتزايد وتقترب من $4$ (شكل \textbf{سلّم}).\par

\section{اتجاه تغير متتالية}
\begin{tarif}
نقول إن المتتالية $\LW{(u_n)}$:
\begin{itemize}
\item \textbf{متزايدة تمامًا} إذا كان من أجل كل عدد طبيعي $\LW{n}$: $\LW{u_{n+1}>u_n}$.
\item \textbf{متناقصة تمامًا} إذا كان من أجل كل عدد طبيعي $\LW{n}$: $\LW{u_{n+1}<u_n}$.
\item \textbf{ثابتة} إذا كان من أجل كل عدد طبيعي $\LW{n}$: $\LW{u_{n+1}=u_n}$.
\end{itemize}
المتتالية \textbf{رتيبة} إذا كانت متزايدة أو متناقصة.
\end{tarif}

\begin{tariqa}[دراسة اتجاه التغير]
\begin{enumerate}
\item \textbf{إشارة الفرق:} ندرس إشارة $\LW{u_{n+1}-u_n}$.
\item \textbf{النسبة} (إذا كانت جميع الحدود موجبة تمامًا): نقارن $\LW{\dfrac{u_{n+1}}{u_n}}$ بالعدد $\LW{1}$.
\item \textbf{الدالة المرفقة} (توليد صريح $\LW{u_n=f(n)}$): إذا كانت $\LW{f}$ متزايدة (متناقصة) على $\LW{[0;+\infty[}$ فإن $\LW{(u_n)}$ متزايدة (متناقصة).
\item \textbf{التراجع} (توليد تراجعي): نبرهن مثلًا أن $\LW{u_n\leqslant u_{n+1}}$ من أجل كل $\LW{n}$ (انظر الفقرة 4).
\end{enumerate}
\end{tariqa}

\begin{mithal}[الفرق]
$u_n=n^{2}-4n$:
\[u_{n+1}-u_n=(n+1)^{2}-4(n+1)-n^{2}+4n=\bx{2n-3}\]
من أجل $n\geqslant2$: $2n-3>0$، إذن \bxt{المتتالية $(u_n)$ متزايدة تمامًا ابتداءً من الرتبة $2$} (لاحظ: $u_1=-3$ ، $u_2=-4$ ، $u_3=-3$).
\end{mithal}

\begin{mithal}[النسبة]
$v_n=\dfrac{3^{n}}{n+1}>0$:
\[\dfrac{v_{n+1}}{v_n}=\dfrac{3^{n+1}}{n+2}\times\dfrac{n+1}{3^{n}}=\bx{\dfrac{3n+3}{n+2}}\]
بما أن $3n+3>n+2$ (لأن $2n+1>0$) فإن $\dfrac{v_{n+1}}{v_n}>1$، إذن \bxt{$(v_n)$ متزايدة تمامًا}.
\end{mithal}

\begin{mithal}[الدالة المرفقة]
$w_n=\dfrac{2n+1}{n+1}=f(n)$ حيث $f(x)=\dfrac{2x+1}{x+1}$ ؛ $f'(x)=\dfrac{1}{(x+1)^{2}}>0$ على $[0;+\infty[$، إذن \bxt{$(w_n)$ متزايدة تمامًا}.
\end{mithal}

\begin{mithal}[توليد تراجعي]
$u_0=2$ و $u_{n+1}=u_n-u_n^{2}$: \quad $u_{n+1}-u_n=-u_n^{2}\leqslant0$، إذن $(u_n)$ متناقصة.
\end{mithal}

\begin{khassia}[حالة $u_{n+1}=f(u_n)$ مع $f$ متزايدة]
إذا كانت $\LW{f}$ \textbf{متزايدة} على مجال $\LW{I}$ بحيث $\LW{f(I)\subset I}$ وكل الحدود من $\LW{I}$، فإن:
\begin{itemize}
\item إذا كان $\LW{u_0\leqslant u_1}$ فإن $\LW{(u_n)}$ متزايدة؛
\item إذا كان $\LW{u_0\geqslant u_1}$ فإن $\LW{(u_n)}$ متناقصة.
\end{itemize}
(البرهان بالتراجع: من $\LW{u_n\leqslant u_{n+1}}$ وتزايد $\LW{f}$ نجد $\LW{f(u_n)\leqslant f(u_{n+1})}$ أي $\LW{u_{n+1}\leqslant u_{n+2}}$.) يكفي إذن مقارنة الحدين الأولين.
\end{khassia}

\begin{molahada}
إذا كانت $u_{n+1}=f(u_n)$ فإن اتجاه تغير $f$ \textbf{لا يعطي} مباشرة اتجاه تغير $(u_n)$: مثال الحلزون السابق ($f$ متناقصة) يعطي متتالية غير رتيبة، ومثال السلّم ($f$ متزايدة) يعطي متتالية متزايدة لأن $u_1>u_0$.
\end{molahada}

\section{المتتاليات المحدودة}
\begin{tarif}
\begin{itemize}
\item $\LW{(u_n)}$ \textbf{محدودة من الأعلى} إذا وجد عدد حقيقي $\LW{M}$ بحيث من أجل كل عدد طبيعي $\LW{n}$: $\LW{u_n\leqslant M}$.
\item $\LW{(u_n)}$ \textbf{محدودة من الأسفل} إذا وجد عدد حقيقي $\LW{m}$ بحيث من أجل كل عدد طبيعي $\LW{n}$: $\LW{u_n\geqslant m}$.
\item $\LW{(u_n)}$ \textbf{محدودة} إذا كانت محدودة من الأعلى ومن الأسفل: $\LW{m\leqslant u_n\leqslant M}$.
\end{itemize}
\end{tarif}

\begin{molahada}
\begin{itemize}
\item كل متتالية متزايدة محدودة من الأسفل بحدها الأول، وكل متتالية متناقصة محدودة من الأعلى بحدها الأول.
\item $(u_n)$ محدودة إذا وفقط إذا وجد $A>0$ بحيث $|u_n|\leqslant A$ من أجل كل $n$.
\end{itemize}
\end{molahada}

\begin{mithal}
\begin{itemize}
\item $w_n=\dfrac{2n+1}{n+1}=2-\dfrac{1}{n+1}$: بما أن $0<\dfrac{1}{n+1}\leqslant1$ فإن \bx{1\leqslant w_n<2}.
\item $u_n=\dfrac{(-1)^{n}}{n+1}$: لدينا $|u_n|=\dfrac{1}{n+1}\leqslant1$ إذن \bx{-1\leqslant u_n\leqslant1}.
\item $v_n=n^{2}+1$ محدودة من الأسفل بـ $1$ وغير محدودة من الأعلى.
\end{itemize}
\end{mithal}

\begin{tariqa}[إثبات أن $(u_n)$ محدودة من الأعلى بـ $M$ (أو من الأسفل بـ $m$)]
\begin{enumerate}
\item \textbf{الفرق:} ندرس إشارة $\LW{u_n-M}$ (أو $\LW{u_n-m}$) ونبين أنها سالبة (أو موجبة).
\item \textbf{الدالة المرفقة:} إذا كان $\LW{u_n=f(n)}$ ندرس تغيرات $\LW{f}$ على $\LW{[0;+\infty[}$ ونستعمل قيمها الحدية.
\item \textbf{التراجع:} خاصة إذا كانت $\LW{u_{n+1}=f(u_n)}$ (انظر الفقرة 4).
\end{enumerate}
\end{tariqa}

\section{الاستدلال بالتراجع}
\begin{khassia}[مبدأ الاستدلال بالتراجع]
لتكن $\LW{P(n)}$ خاصية متعلقة بعدد طبيعي $\LW{n}$، و $\LW{n_0}$ عددًا طبيعيًا. إذا كان:
\begin{enumerate}
\item \textbf{التحقق:} $\LW{P(n_0)}$ صحيحة؛
\item \textbf{الوراثة:} من أجل كل عدد طبيعي $\LW{n\geqslant n_0}$، إذا كانت $\LW{P(n)}$ صحيحة فإن $\LW{P(n+1)}$ صحيحة؛
\end{enumerate}
فإن $\LW{P(n)}$ صحيحة من أجل كل عدد طبيعي $\LW{n\geqslant n_0}$.
\end{khassia}

\begin{tariqa}[تحرير برهان بالتراجع]
نكتب الخاصية $\LW{P(n)}$ بوضوح، نتحقق من $\LW{P(n_0)}$، ثم \textbf{نفرض} أن $\LW{P(n)}$ صحيحة (فرضية التراجع) و\textbf{نبرهن} صحة $\LW{P(n+1)}$ باستعمالها، ثم نستنتج.
\end{tariqa}

\begin{mithal}[مساواة]
نبرهن أن من أجل كل عدد طبيعي $n\geqslant1$: $1+2+\cdots+n=\dfrac{n(n+1)}{2}$.
\begin{itemize}
\item \textbf{التحقق:} من أجل $n=1$: الطرف الأيسر $1$، والأيمن $\frac{1\times2}{2}=1$. \ok
\item \textbf{الوراثة:} نفرض $1+\cdots+n=\frac{n(n+1)}{2}$، إذن
\[1+\cdots+n+(n+1)=\frac{n(n+1)}{2}+(n+1)=\frac{(n+1)(n+2)}{2}\]
وهي $P(n+1)$. إذن الخاصية صحيحة من أجل كل $n\geqslant1$.
\end{itemize}
\end{mithal}

\begin{mithal}[حصر حدود متتالية تراجعية]
$u_0=1$ و $u_{n+1}=\frac{1}{2}u_n+2$. نبرهن أن من أجل كل عدد طبيعي $n$: $u_n<4$.
\begin{itemize}
\item \textbf{التحقق:} $u_0=1<4$. \ok
\item \textbf{الوراثة:} نفرض $u_n<4$؛ إذن $\frac{1}{2}u_n<2$ ومنه $\frac{1}{2}u_n+2<4$ أي $u_{n+1}<4$.
\end{itemize}
إذن \bx{u_n<4} من أجل كل $n\in\N$. ثم:
\[u_{n+1}-u_n=-\dfrac{1}{2}u_n+2=\dfrac{1}{2}(4-u_n)>0\]
إذن \bxt{المتتالية $(u_n)$ متزايدة تمامًا}.
\end{mithal}

\begin{mithal}[متباينة]
نبرهن أن من أجل كل عدد طبيعي $n$: $3^{n}\geqslant1+2n$.
\begin{itemize}
\item \textbf{التحقق:} $3^{0}=1\geqslant1$. \ok
\item \textbf{الوراثة:} نفرض $3^{n}\geqslant1+2n$؛ إذن
\[3^{n+1}=3\times3^{n}\geqslant3+6n=\left(1+2(n+1)\right)+4n\geqslant\bx{1+2(n+1)}\]
\end{itemize}
\end{mithal}

\begin{molahada}
عندما تكون $u_{n+1}=f(u_n)$ و $f$ \textbf{متزايدة} على مجال $I$ يشمل الحدود، فإن الوراثة تكون سريعة: من $a\leqslant u_n\leqslant b$ نستنتج $f(a)\leqslant u_{n+1}\leqslant f(b)$.
\end{molahada}

\section{المتتاليات الحسابية}
\begin{tarif}
$\LW{(u_n)}$ \textbf{حسابية} إذا وجد عدد حقيقي $\LW{r}$ (يسمى \textbf{الأساس}) بحيث من أجل كل عدد طبيعي $\LW{n}$: $\LW{u_{n+1}=u_n+r}$.\par
لإثبات ذلك نبين أن الفرق $\LW{u_{n+1}-u_n}$ ثابت (لا يتعلق بـ $\LW{n}$).
\end{tarif}

\begin{khassia}
إذا كانت $\LW{(u_n)}$ حسابية أساسها $\LW{r}$:
\begin{itemize}
\item \textbf{الحد العام:} $\LW{u_n=u_0+nr}$، وعمومًا $\LW{u_n=u_p+(n-p)r}$.
\item \textbf{الوسط الحسابي:} $\LW{a}$ ، $\LW{b}$ ، $\LW{c}$ حدود متتابعة من متتالية حسابية $\LW{\iff 2b=a+c}$.
\item \textbf{المجموع:} \bx{S=N\times\dfrac{P+L}{2}} حيث $\LW{N}$ عدد الحدود، $\LW{P}$ الحد الأول، $\LW{L}$ الحد الأخير.\par
مثلًا: $\LW{u_0+u_1+\cdots+u_n=(n+1)\dfrac{u_0+u_n}{2}}$.
\item \textbf{اتجاه التغير:} متزايدة تمامًا إذا كان $\LW{r>0}$، متناقصة تمامًا إذا كان $\LW{r<0}$، ثابتة إذا كان $\LW{r=0}$.
\end{itemize}
\end{khassia}

\begin{molahada}
عدد الحدود من $u_p$ إلى $u_n$ هو $n-p+1$. وبالخصوص $1+2+\cdots+n=\dfrac{n(n+1)}{2}$.
\end{molahada}

\begin{mithal}
\begin{itemize}
\item $u_0=5$ و $r=3$: \quad $u_n=5+3n$ \quad و \quad $u_{20}=5+3\times20=65$.
\[u_0+u_1+\cdots+u_{20}=21\times\dfrac{5+65}{2}=\bx{735}\]
\item $(u_n)$ حسابية حيث $u_3=10$ و $u_8=25$:
\[u_8-u_3=5r=15\ \Rightarrow\ r=3\quad;\quad u_0=u_3-3r=10-9=1\quad;\quad \bx{u_n=3n+1}\]
\item $u_n=\dfrac{n}{2}-1$: $u_{n+1}-u_n=\dfrac{1}{2}$ ثابت، إذن حسابية أساسها $\frac{1}{2}$ وحدها الأول $u_0=-1$.
\item \textbf{الوسط الحسابي:} $a$ ، $b$ ، $c$ حدود متتابعة من متتالية حسابية مجموعها $36$:
\[a+b+c=36\ \text{\ar{و}}\ a+c=2b\ \Rightarrow\ 3b=36\ \Rightarrow\ \bx{b=12}\]
\end{itemize}
\end{mithal}

\begin{tariqa}[متتالية الفروق $v_n=u_{n+1}-u_n$]
إذا كانت $\LW{u_{n+1}=u_n+v_n}$ حيث $\LW{(v_n)}$ حسابية (أو هندسية)، فإن الحدود تختزل:
\[\LW{u_n=u_0+\left(v_0+v_1+\cdots+v_{n-1}\right)}\]
والمجموع بين القوسين يضم $\LW{n}$ حدًّا.
\end{tariqa}

\begin{mithal}
$u_0=2$ و $u_{n+1}=u_n+2n+3$: \quad $v_n=u_{n+1}-u_n=2n+3$ حسابية أساسها $2$ وحدها الأول $v_0=3$.\par
\[v_0+v_1+\cdots+v_{n-1}=n\times\dfrac{3+(2n+1)}{2}=n(n+2)\]
\[u_n=u_0+n(n+2)=\bx{n^{2}+2n+2}\]
تحقق: $u_1=5$ ، $u_2=10$ \ok
\end{mithal}

\section{المتتاليات الهندسية}
\begin{tarif}
$\LW{(u_n)}$ \textbf{هندسية} إذا وجد عدد حقيقي $\LW{q}$ (يسمى \textbf{الأساس}) بحيث من أجل كل عدد طبيعي $\LW{n}$: $\LW{u_{n+1}=q\,u_n}$.\par
لإثبات ذلك نكتب $\LW{u_{n+1}}$ على الشكل $\LW{q\times u_n}$ (أو نبين أن $\LW{\frac{u_{n+1}}{u_n}}$ ثابت إذا كانت الحدود غير معدومة).
\end{tarif}

\begin{khassia}
إذا كانت $\LW{(u_n)}$ هندسية أساسها $\LW{q}$:
\begin{itemize}
\item \textbf{الحد العام:} $\LW{u_n=u_0\,q^{n}}$، وعمومًا $\LW{u_n=u_p\,q^{\,n-p}}$.
\item \textbf{الوسط الهندسي:} $\LW{a}$ ، $\LW{b}$ ، $\LW{c}$ حدود متتابعة من متتالية هندسية $\LW{\iff b^{2}=ac}$.
\item \textbf{المجموع} ($\LW{q\neq1}$): \bx{S=P\times\dfrac{1-q^{N}}{1-q}} حيث $\LW{P}$ الحد الأول و $\LW{N}$ عدد الحدود.\par
مثلًا: $\LW{u_0+u_1+\cdots+u_n=u_0\dfrac{1-q^{n+1}}{1-q}}$.
\item إذا كان $\LW{q=1}$ فإن المتتالية ثابتة والمجموع يساوي $\LW{N\times u_0}$.
\end{itemize}
\end{khassia}

\begin{khassia}[اتجاه التغير والنهاية لـ $u_n=u_0q^{n}$ ($u_0\neq0$)]
\begin{itemize}
\item $\LW{q>1}$: إذا كان $\LW{u_0>0}$ فهي متزايدة تمامًا ونهايتها $\LW{+\infty}$؛ وإذا كان $\LW{u_0<0}$ فهي متناقصة تمامًا ونهايتها $\LW{-\infty}$.
\item $\LW{0<q<1}$: إذا كان $\LW{u_0>0}$ فهي متناقصة تمامًا؛ وإذا كان $\LW{u_0<0}$ فهي متزايدة تمامًا؛ وفي الحالتين نهايتها $\LW{0}$.
\item $\LW{q=1}$: المتتالية ثابتة.
\item $\LW{q<0}$: غير رتيبة (إشارات متناوبة)؛ نهايتها $\LW{0}$ إذا كان $\LW{-1<q<0}$، وليس لها نهاية إذا كان $\LW{q\leqslant-1}$.
\end{itemize}
\end{khassia}

\begin{mithal}
\begin{itemize}
\item $v_0=3$ و $q=2$: \quad $v_n=3\times2^{n}$.
\[v_0+v_1+\cdots+v_9=3\times\dfrac{1-2^{10}}{1-2}=3\times1023=\bx{3069}\]
\item $(v_n)$ هندسية حيث $v_1=6$ و $v_4=48$:
\[q^{3}=\dfrac{v_4}{v_1}=\dfrac{48}{6}=8\ \Rightarrow\ \bx{q=2}\quad;\quad v_0=\dfrac{v_1}{q}=\bx{3}\]
\item \textbf{تطبيق:} سعر سلعة $2000$ دج يرتفع بـ $5\%$ كل سنة: $P_{n+1}=1{,}05\,P_n$.
\[P_n=2000\times1{,}05^{n}\quad;\quad P_{10}=2000\times1{,}05^{10}\approx\bx{3257{,}79}\ \text{\ar{دج}}\]
\item مجموع هندسي أساسه $\frac{1}{2}$ وحده الأول $1$:
\[1+\dfrac{1}{2}+\dfrac{1}{4}+\cdots+\dfrac{1}{2^{n}}=\dfrac{1-\left(\frac{1}{2}\right)^{n+1}}{1-\frac{1}{2}}=\bx{2-\dfrac{1}{2^{n}}}\]
\item \textbf{الوسط الهندسي:} $a$ ، $b$ ، $c$ حدود متتابعة من متتالية هندسية جداؤها $216$:
\[abc=216\ \text{\ar{و}}\ ac=b^{2}\ \Rightarrow\ b^{3}=216\ \Rightarrow\ \bx{b=6}\]
\item $(u_n)$ هندسية حدودها موجبة تمامًا حيث $u_0\times u_2=4$ و $\ln u_1+\ln u_4=5\ln2$:\par
\[u_1^{2}=u_0u_2=4\ \text{\ar{و}}\ u_1>0\ \Rightarrow\ u_1=2\]
\[\ln(u_1u_4)=\ln32\ \Rightarrow\ u_1^{2}q^{3}=32\ \Rightarrow\ q^{3}=8\ \Rightarrow\ q=2\quad;\quad \bx{u_n=u_1q^{n-1}=2^{n}}\]
\end{itemize}
\end{mithal}

\begin{molahada}[الربط مع $\ln$ و $\exp$]
\begin{itemize}
\item إذا كانت $(u_n)$ هندسية حدودها موجبة تمامًا وأساسها $q$ فإن $v_n=\ln u_n$ \textbf{حسابية} أساسها $\ln q$، لأن:
\[v_{n+1}=\ln u_{n+1}=\ln(q\,u_n)=\ln u_n+\ln q=v_n+\ln q\]
\item إذا كانت $(v_n)$ حسابية أساسها $r$ فإن $w_n=e^{v_n}$ \textbf{هندسية} أساسها $e^{r}$.
\end{itemize}
\end{molahada}

\section{المجاميع والجداءات}
\begin{khassia}[مجاميع وجداءات مرجعية]
\begin{itemize}
\item $\LW{0+1+2+\cdots+n=\dfrac{n(n+1)}{2}}$ \quad و \quad $\LW{1+a+a^{2}+\cdots+a^{n}=\dfrac{a^{n+1}-1}{a-1}}$ \ ($\LW{a\neq1}$).
\item مجموع $\LW{N}$ حدًّا كلها تساوي $\LW{a}$ هو $\LW{N\times a}$.
\item $\LW{a^{0}\times a^{1}\times a^{2}\times\cdots\times a^{n}=a^{0+1+\cdots+n}=a^{\frac{n(n+1)}{2}}}$.
\item إذا كانت $\LW{(v_n)}$ هندسية: $\LW{v_0\times v_1\times\cdots\times v_n=v_0^{\,n+1}\times q^{\frac{n(n+1)}{2}}}$.
\item إذا كانت الحدود موجبة تمامًا: $\LW{\ln v_0+\ln v_1+\cdots+\ln v_n=\ln\left(v_0\times v_1\times\cdots\times v_n\right)}$.
\end{itemize}
\end{khassia}

\Needspace{14\baselineskip}
\begin{tariqa}[مجموع مشتق من متتالية هندسية $(v_n)$ أساسها $q$]
نضع $\LW{w_n}$ مساويًا للحد العام للمجموع، ونبين أن $\LW{\dfrac{w_{n+1}}{w_n}}$ ثابت، ثم نطبق دستور المجموع الهندسي:
\begin{itemize}
\item $\LW{w_n=v_n^{2}}$: هندسية أساسها $\LW{q^{2}}$. \quad $\LW{w_n=\dfrac{1}{v_n}}$: هندسية أساسها $\LW{\dfrac{1}{q}}$.
\item $\LW{w_n=k^{n}v_n}$: هندسية أساسها $\LW{kq}$. \quad $\LW{w_n=\dfrac{v_n}{k^{n}}}$: هندسية أساسها $\LW{\dfrac{q}{k}}$.
\item إذا كان الأساس الجديد يساوي $\LW{1}$ فإن $\LW{(w_n)}$ ثابتة والمجموع يساوي عدد الحدود مضروبًا في $\LW{w_0}$.
\end{itemize}
\end{tariqa}

\begin{mithal}[$v_n=2\times3^{n}$ ($v_0=2$ ، $q=3$)]
\begin{itemize}
\item \textbf{الجداء:}
\[v_0\times v_1\times\cdots\times v_n=2^{n+1}\times3^{0+1+\cdots+n}=\bx{2^{n+1}\times3^{\frac{n(n+1)}{2}}}\]
\item \textbf{مجموع المربعات:} $(v_n^{2})$ هندسية أساسها $9$ وحدها الأول $4$:
\[v_0^{2}+v_1^{2}+\cdots+v_n^{2}=4\times\dfrac{9^{n+1}-1}{9-1}=\bx{\dfrac{9^{n+1}-1}{2}}\]
\item \textbf{مجموع المقلوبات:} $\left(\frac{1}{v_n}\right)$ هندسية أساسها $\frac{1}{3}$ وحدها الأول $\frac{1}{2}$:
\[\dfrac{1}{v_0}+\dfrac{1}{v_1}+\cdots+\dfrac{1}{v_n}=\dfrac{1}{2}\times\dfrac{1-\left(\frac{1}{3}\right)^{n+1}}{1-\frac{1}{3}}=\bx{\dfrac{3}{4}\left(1-\left(\tfrac{1}{3}\right)^{n+1}\right)}\]
\item \textbf{المجموع $\sum 2^{k}v_k$:} $(2^{n}v_n)$ هندسية أساسها $6$ وحدها الأول $2$:
\[v_0+2v_1+4v_2+\cdots+2^{n}v_n=2\times\dfrac{6^{n+1}-1}{6-1}=\bx{\dfrac{2}{5}\left(6^{n+1}-1\right)}\]
\item \textbf{المجموع $\sum \frac{v_k}{3^{k}}$:} الأساس $\frac{3}{3}=1$، فكل الحدود تساوي $2$ وعددها $n+1$:
\[v_0+\dfrac{v_1}{3}+\cdots+\dfrac{v_n}{3^{n}}=(n+1)\times2=\bx{2(n+1)}\]
\end{itemize}
\end{mithal}

\section{تقارب متتالية والنهايات}
\begin{tarif}
\begin{itemize}
\item نقول إن $\LW{(u_n)}$ \textbf{متقاربة} نحو العدد الحقيقي $\LW{\ell}$ إذا كان كل مجال مفتوح يشمل $\LW{\ell}$ يشمل جميع الحدود ابتداءً من رتبة معينة، ونكتب $\LW{\limn u_n=\ell}$.
\item إذا لم تكن متقاربة (نهايتها $\LW{\pm\infty}$ أو ليس لها نهاية) نقول إنها \textbf{متباعدة}.
\item إذا كان $\LW{u_n=f(n)}$ و $\LW{\displaystyle\lim_{x\to+\infty}f(x)=\ell}$ فإن $\LW{\limn u_n=\ell}$ (نفس قواعد نهايات الدوال).
\end{itemize}
\end{tarif}

\begin{khassia}[نهاية $q^{n}$]
\begin{itemize}
\item إذا كان $\LW{q>1}$: $\LW{\limn q^{n}=+\infty}$.\qquad إذا كان $\LW{q=1}$: $\LW{\limn q^{n}=1}$.
\item إذا كان $\LW{-1<q<1}$: $\LW{\limn q^{n}=0}$.\qquad إذا كان $\LW{q\leqslant-1}$: ليس لـ $\LW{(q^{n})}$ نهاية.
\end{itemize}
\end{khassia}

\begin{mithal}
\begin{itemize}
\item لأن $\left(\frac{2}{3}\right)^{n}\to0$ و $\left(\frac{1}{3}\right)^{n}\to0$:
\[\limn\dfrac{2^{n}-3^{n}}{3^{n}+1}=\limn\dfrac{\left(\frac{2}{3}\right)^{n}-1}{1+\left(\frac{1}{3}\right)^{n}}=\bx{-1}\]
\item $\limn\left(5-3\times0{,}8^{n}\right)=\bx{5}$ \quad و \quad $\limn\left(2^{n}-n\right)=+\infty$.
\end{itemize}
\end{mithal}

\begin{khassia}[مبرهنات المقارنة]
\begin{enumerate}
\item إذا كان $\LW{u_n\geqslant v_n}$ ابتداءً من رتبة معينة و $\LW{\limn v_n=+\infty}$ فإن $\LW{\limn u_n=+\infty}$.
\item إذا كان $\LW{u_n\leqslant v_n}$ ابتداءً من رتبة معينة و $\LW{\limn v_n=-\infty}$ فإن $\LW{\limn u_n=-\infty}$.
\item \textbf{(الحصر)} إذا كان $\LW{v_n\leqslant u_n\leqslant w_n}$ ابتداءً من رتبة معينة و $\LW{\limn v_n=\limn w_n=\ell}$ فإن $\LW{\limn u_n=\ell}$.
\item إذا كان $\LW{|u_n-\ell|\leqslant v_n}$ و $\LW{\limn v_n=0}$ فإن $\LW{\limn u_n=\ell}$.
\end{enumerate}
\end{khassia}

\begin{mithal}
\begin{itemize}
\item $u_n=n+(-1)^{n}$: لدينا $u_n\geqslant n-1$ و $\limn(n-1)=+\infty$، إذن \bx{\limn u_n=+\infty}.
\item $u_n=\dfrac{\cos n}{n}$ ($n\geqslant1$): $-\dfrac{1}{n}\leqslant u_n\leqslant\dfrac{1}{n}$ والطرفان يؤولان إلى $0$، إذن \bx{\limn u_n=0}.
\item $u_n=\dfrac{2n+(-1)^{n}}{n+1}$: $\dfrac{2n-1}{n+1}\leqslant u_n\leqslant\dfrac{2n+1}{n+1}$ والطرفان يؤولان إلى $2$، إذن \bx{\limn u_n=2}.
\end{itemize}
\end{mithal}

\begin{khassia}[المتتاليات الرتيبة]
\begin{itemize}
\item كل متتالية \textbf{متزايدة ومحدودة من الأعلى} متقاربة، وكل متتالية \textbf{متناقصة ومحدودة من الأسفل} متقاربة.
\item كل متتالية متزايدة وغير محدودة من الأعلى نهايتها $\LW{+\infty}$.
\end{itemize}
\end{khassia}

\begin{molahada}
هذه المبرهنة تثبت \textbf{وجود} النهاية ولا تعطي قيمتها: إذا كانت $(u_n)$ متزايدة و $u_n\leqslant M$ فإن نهايتها $\ell\leqslant M$ وليس بالضرورة $\ell=M$. مثلًا $w_n=2-\dfrac{1}{n+1}\leqslant5$ لكن $\limn w_n=2$.
\end{molahada}

\begin{khassia}[نهاية متتالية $u_{n+1}=f(u_n)$]
إذا كانت $\LW{(u_n)}$ متقاربة نحو $\LW{\ell}$، وكانت $\LW{f}$ مستمرة عند $\LW{\ell}$ (و $\LW{\ell}$ من مجال يحوي الحدود)، فإن $\LW{\ell}$ حل للمعادلة \bx{f(\ell)=\ell}؛ أي فاصلة نقطة تقاطع $\LW{(C_f)}$ مع المستقيم $\LW{y=x}$.
\end{khassia}

\begin{molahada}
هذه المبرهنة تعطي القيم الممكنة للنهاية فقط؛ يجب أولًا إثبات التقارب (مثلًا: رتيبة ومحدودة)، ثم نختار الحل المناسب بالحصر.
\end{molahada}

\begin{mithal}[$u_0=0$ و $u_{n+1}=\sqrt{2u_n+3}$]
\begin{itemize}
\item \textbf{بالتراجع:} $0\leqslant u_n<3$ صحيح من أجل $n=0$؛ ونفرض $0\leqslant u_n<3$:
\[3\leqslant2u_n+3<9\ \Rightarrow\ \sqrt3\leqslant u_{n+1}<3\ \Rightarrow\ \bx{0\leqslant u_{n+1}<3}\]
\item \textbf{اتجاه التغير:} (نضرب في المرافق)
\[u_{n+1}-u_n=\dfrac{2u_n+3-u_n^{2}}{\sqrt{2u_n+3}+u_n}=\dfrac{(3-u_n)(u_n+1)}{\sqrt{2u_n+3}+u_n}>0\]
إذن \bxt{$(u_n)$ متزايدة تمامًا}.
\item \textbf{التقارب:} متزايدة ومحدودة من الأعلى بـ $3$ فهي متقاربة نحو $\ell\in[0;3]$، و $f:x\mapsto\sqrt{2x+3}$ مستمرة:
\[\ell=\sqrt{2\ell+3}\ \Rightarrow\ \ell^{2}-2\ell-3=0\ \Rightarrow\ (\ell-3)(\ell+1)=0\]
$\ell=-1$ مرفوض، إذن \bx{\limn u_n=3}.
\end{itemize}

\section{المتتاليتان المتجاورتان}
\begin{tarif}
نقول إن المتتاليتين $\LW{(u_n)}$ و $\LW{(v_n)}$ \textbf{متجاورتان} إذا كانت:
\begin{itemize}
\item إحداهما متزايدة والأخرى متناقصة؛
\item و $\LW{\limn\left(v_n-u_n\right)=0}$.
\end{itemize}
\end{tarif}

\begin{khassia}
إذا كانت $\LW{(u_n)}$ متزايدة و $\LW{(v_n)}$ متناقصة ومتجاورتين، فإنهما \textbf{متقاربتان نحو نفس النهاية} $\LW{\ell}$، ومن أجل كل عدد طبيعي $\LW{n}$:
\[\LW{\bx{u_n\leqslant\ell\leqslant v_n}}\]
\end{khassia}

\begin{mithal}
من أجل $n\geqslant1$: $u_n=2-\dfrac{1}{n}$ و $v_n=2+\dfrac{1}{n^{2}}$.
\begin{itemize}
\item $u_{n+1}-u_n=\dfrac{1}{n}-\dfrac{1}{n+1}=\dfrac{1}{n(n+1)}>0$ \quad إذن $(u_n)$ متزايدة.
\item $v_{n+1}-v_n=\dfrac{1}{(n+1)^{2}}-\dfrac{1}{n^{2}}<0$: $(v_n)$ متناقصة.
\item $v_n-u_n=\dfrac{1}{n^{2}}+\dfrac{1}{n}\to0$.
\end{itemize}
إذن \bxt{$(u_n)$ و $(v_n)$ متجاورتان} ونهايتهما المشتركة $2$.
\end{mithal}

\begin{molahada}
إذا كانت $w_n=v_n-u_n$ متتالية هندسية أساسها $q\in]0;1[$ وحدها الأول موجب، فإن $w_n>0$ و $\limn w_n=0$؛ وهذا شائع في تمارين البكالوريا (انظر التمرين المحلول 2).
\end{molahada}

\section{المتتاليات من الشكل $u_{n+1}=au_n+b$}
\begin{tariqa}[المتتالية المساعدة]
لتكن $\LW{u_{n+1}=au_n+b}$ حيث $\LW{a\neq1}$ و $\LW{a\neq0}$.
\begin{enumerate}
\item نحل $\LW{\alpha=a\alpha+b}$ فنجد $\LW{\alpha=\dfrac{b}{1-a}}$ (النقطة الصامدة).
\item نضع $\LW{v_n=u_n-\alpha}$؛ عندئذ $\LW{v_{n+1}=au_n+b-\alpha=a(u_n-\alpha)=a\,v_n}$: إذن $\LW{(v_n)}$ \textbf{هندسية أساسها $\LW{a}$}.
\item $\LW{v_n=v_0\,a^{n}}$ ومنه $\LW{u_n=\alpha+(u_0-\alpha)a^{n}}$، ثم النهاية والمجاميع.
\end{enumerate}
\end{tariqa}

\begin{mithal}
$u_0=1$ و $u_{n+1}=\frac{1}{2}u_n+2$.
\begin{itemize}
\item $\alpha=\frac{1}{2}\alpha+2$ ومنه $\alpha=4$. نضع $v_n=u_n-4$:
\[v_{n+1}=u_{n+1}-4=\dfrac{1}{2}u_n+2-4=\dfrac{1}{2}(u_n-4)=\bx{\dfrac{1}{2}v_n}\]
\item $(v_n)$ هندسية أساسها $\frac{1}{2}$ وحدها الأول $v_0=1-4=-3$:
\[v_n=-3\left(\tfrac{1}{2}\right)^{n}\quad;\quad u_n=v_n+4=\bx{4-3\left(\tfrac{1}{2}\right)^{n}}\]
\item $\limn\left(\frac{1}{2}\right)^{n}=0$ إذن \bx{\limn u_n=4}.
\item المجموع $S_n=u_0+\cdots+u_n$: لأن $u_k=4+v_k$
\[S_n=4(n+1)+\left(v_0+\cdots+v_n\right)=4(n+1)-3\times\dfrac{1-\left(\frac{1}{2}\right)^{n+1}}{1-\frac{1}{2}}=\bx{4n-2+3\left(\tfrac{1}{2}\right)^{n}}\]
\end{itemize}
\end{mithal}

\begin{molahada}
حالة متتالية من الشكل $u_{n+1}=\dfrac{au_n+b}{cu_n+d}$: يُعطى غالبًا في التمرين متتالية مساعدة مثل $v_n=\dfrac{u_n-\beta}{u_n-\gamma}$ (حيث $\beta,\gamma$ حلّا $f(x)=x$) ونبرهن أنها هندسية (انظر التمرين المحلول 1).
\end{molahada}

\begin{tariqa}[أشكال أخرى شائعة للمتتالية المساعدة]
\begin{itemize}
\item $\LW{u_{n+1}=\dfrac{au_n+b}{cu_n+d}}$ وللمعادلة $\LW{f(x)=x}$ \textbf{حل مضاعف} $\LW{\alpha}$: نضع $\LW{v_n=\dfrac{1}{u_n-\alpha}}$ فتكون \textbf{حسابية}.
\item $\LW{u_{n+1}-a=\sqrt{u_n-a}}$ (أو قوة أخرى): نضع $\LW{v_n=\ln(u_n-a)}$ فتكون \textbf{هندسية} أساسها $\LW{\frac{1}{2}}$.
\end{itemize}
\end{tariqa}

\begin{mithal}[حل مضاعف]
$u_0=2$ و $u_{n+1}=\dfrac{3u_n-1}{u_n+1}$. المعادلة $\frac{3x-1}{x+1}=x$ تكافئ $(x-1)^{2}=0$ إذن $\alpha=1$.
\begin{itemize}
\item $u_{n+1}-1=\dfrac{2(u_n-1)}{u_n+1}$، ومنه
\[v_{n+1}=\dfrac{1}{u_{n+1}-1}=\dfrac{u_n+1}{2(u_n-1)}=\dfrac{(u_n-1)+2}{2(u_n-1)}=\bx{\dfrac{1}{2}+v_n}\]
\item $(v_n)$ حسابية أساسها $\frac{1}{2}$ و $v_0=1$: $v_n=1+\dfrac{n}{2}$، إذن \bx{u_n=1+\dfrac{2}{n+2}} و \bx{\limn u_n=1}.
\end{itemize}
\end{mithal}

\begin{mithal}[مع اللوغاريتم]
$u_0=3+e^{2}$ و $u_{n+1}=3+\sqrt{u_n-3}$. نضع $v_n=\ln(u_n-3)$:
\[v_{n+1}=\ln(u_{n+1}-3)=\ln\sqrt{u_n-3}=\dfrac{1}{2}\ln(u_n-3)=\bx{\dfrac{1}{2}v_n}\quad;\quad v_0=\ln e^{2}=2\]
\[v_n=2\left(\tfrac{1}{2}\right)^{n}\quad;\quad u_n=3+e^{v_n}=\bx{3+e^{\,2\left(\frac{1}{2}\right)^{n}}}\]
و $v_n\to0$ ومنه:
\[\limn u_n=3+e^{0}=\bx{4}\]
\end{mithal}

\section{تمارين محلولة}
\begin{tmr}
لتكن $f$ الدالة المعرفة على $[0;+\infty[$ بـ $f(x)=\dfrac{3x+2}{x+2}$، و $(u_n)$ المتتالية المعرفة بـ $u_0=0$ ومن أجل كل عدد طبيعي $n$: $u_{n+1}=f(u_n)$.
\QQ{1} ادرس اتجاه تغير $f$ على $[0;+\infty[$.
\QQ{2} مثّل على محور الفواصل الحدود $u_0$ ، $u_1$ ، $u_2$ ، $u_3$ باستعمال $(C_f)$ والمستقيم $y=x$. ضع تخمينًا حول اتجاه تغير $(u_n)$ وتقاربها.
\QQ{3} برهن بالتراجع أن من أجل كل عدد طبيعي $n$: $0\leqslant u_n<2$.
\QQ{4} بيّن أن $(u_n)$ متزايدة تمامًا، واستنتج أنها متقاربة.
\QQ{5} نضع $v_n=\dfrac{u_n-2}{u_n+1}$.
\begin{sq}
\qa بيّن أن $(v_n)$ متتالية هندسية يطلب تعيين أساسها وحدها الأول.
\qb اكتب $v_n$ ثم $u_n$ بدلالة $n$، واحسب $\limn u_n$.
\end{sq}
\end{tmr}

\begin{hal}
\QQ{1} $f'(x)=\dfrac{3(x+2)-(3x+2)}{(x+2)^{2}}=\dfrac{4}{(x+2)^{2}}>0$، إذن \bxt{$f$ متزايدة تمامًا على $[0;+\infty[$}.

\QQ{2} $u_1=f(0)=1$ ، $u_2=f(1)=\dfrac{5}{3}\approx1{,}67$ ، $u_3=f\!\left(\dfrac{5}{3}\right)=\dfrac{21}{11}\approx1{,}91$.

\QQ{3} نسمي $P(n)$: «$0\leqslant u_n<2$».
\begin{itemize}
\item \textbf{التحقق:} $u_0=0$ إذن $0\leqslant u_0<2$: $P(0)$ صحيحة.
\item \textbf{الوراثة:} نفرض $0\leqslant u_n<2$. بما أن $f$ متزايدة تمامًا على $[0;+\infty[$:
\[f(0)\leqslant f(u_n)<f(2)\ \Rightarrow\ 1\leqslant u_{n+1}<2\ \Rightarrow\ 0\leqslant u_{n+1}<2\]
إذن $P(n+1)$ صحيحة.
\end{itemize}
إذن من أجل كل عدد طبيعي $n$: \bx{0\leqslant u_n<2}.

\QQ{4}
\[u_{n+1}-u_n=\dfrac{3u_n+2-u_n(u_n+2)}{u_n+2}=\dfrac{-u_n^{2}+u_n+2}{u_n+2}=\dfrac{(2-u_n)(u_n+1)}{u_n+2}\]
لأن $0\leqslant u_n<2$ فإن $2-u_n>0$ ، $u_n+1>0$ ، $u_n+2>0$؛ إذن $u_{n+1}-u_n>0$ و \bxt{$(u_n)$ متزايدة تمامًا}.\par
$(u_n)$ متزايدة ومحدودة من الأعلى بالعدد $2$، إذن \bxt{$(u_n)$ متقاربة}.

\QQ{5}
\pa
\[u_{n+1}-2=\dfrac{3u_n+2-2u_n-4}{u_n+2}=\dfrac{u_n-2}{u_n+2}\]
\[u_{n+1}+1=\dfrac{3u_n+2+u_n+2}{u_n+2}=\dfrac{4(u_n+1)}{u_n+2}\]
\[v_{n+1}=\dfrac{u_{n+1}-2}{u_{n+1}+1}=\dfrac{u_n-2}{4(u_n+1)}=\bx{\dfrac{1}{4}v_n}\]
$(v_n)$ \bxt{هندسية أساسها $q=\frac{1}{4}$ وحدها الأول $v_0=\frac{0-2}{0+1}=-2$}.
\pb $v_n=-2\left(\dfrac{1}{4}\right)^{n}$. لدينا $v_n\neq1$ و:
\[v_n(u_n+1)=u_n-2\ \Rightarrow\ u_n(1-v_n)=2+v_n\]
\[ \bx{u_n=\dfrac{2+v_n}{1-v_n}=\dfrac{2-2\left(\frac{1}{4}\right)^{n}}{1+2\left(\frac{1}{4}\right)^{n}}} \]
تحقق: $u_1=\frac{2-\frac{1}{2}}{1+\frac{1}{2}}=1$ \ok\par
بما أن $0<\frac{1}{4}<1$ فإن $\limn\left(\frac{1}{4}\right)^{n}=0$ ومنه \bx{\limn u_n=2}، وهذا يؤكد التخمين.
\end{hal}

\begin{tmr}
لتكن $(u_n)$ و $(v_n)$ المتتاليتان المعرفتان بـ $u_0=1$ ، $v_0=8$، ومن أجل كل عدد طبيعي $n$:
\[ u_{n+1}=\frac{2u_n+v_n}{3}\qquad\text{و}\qquad v_{n+1}=\frac{u_n+3v_n}{4} \]
\QQ{1} نضع $w_n=v_n-u_n$. بيّن أن $(w_n)$ هندسية، ثم اكتب $w_n$ بدلالة $n$ وعيّن نهايتها.
\QQ{2} بيّن أن $(u_n)$ متزايدة و $(v_n)$ متناقصة، واستنتج أنهما متجاورتان.
\QQ{3} نضع $t_n=3u_n+4v_n$. بيّن أن $(t_n)$ ثابتة، ثم عيّن النهاية المشتركة لـ $(u_n)$ و $(v_n)$.
\end{tmr}

\begin{hal}
\QQ{1}
\begin{align*}
w_{n+1}=v_{n+1}-u_{n+1}&=\dfrac{u_n+3v_n}{4}-\dfrac{2u_n+v_n}{3}=\dfrac{3u_n+9v_n-8u_n-4v_n}{12}\\
&=\dfrac{5v_n-5u_n}{12}=\dfrac{5}{12}\left(v_n-u_n\right)=\dfrac{5}{12}w_n
\end{align*}
إذن $(w_n)$ هندسية أساسها $\frac{5}{12}$ وحدها الأول $w_0=8-1=7$، ومنه \bx{w_n=7\left(\tfrac{5}{12}\right)^{n}}.\par
بما أن $0<\frac{5}{12}<1$: \bx{\limn w_n=0}. ولاحظ أن $w_n>0$ أي $u_n<v_n$ من أجل كل $n$.

\QQ{2}
\[u_{n+1}-u_n=\dfrac{2u_n+v_n-3u_n}{3}=\dfrac{v_n-u_n}{3}=\dfrac{w_n}{3}>0\]
إذن \bxt{$(u_n)$ متزايدة تمامًا}.
\[v_{n+1}-v_n=\dfrac{u_n+3v_n-4v_n}{4}=\dfrac{u_n-v_n}{4}=-\dfrac{w_n}{4}<0\]
إذن \bxt{$(v_n)$ متناقصة تمامًا}.\par
وبما أن $\limn(v_n-u_n)=\limn w_n=0$ فإن \bxt{$(u_n)$ و $(v_n)$ متجاورتان}، إذن متقاربتان نحو نفس النهاية $\ell$.

\QQ{3}
\[t_{n+1}=3\times\dfrac{2u_n+v_n}{3}+4\times\dfrac{u_n+3v_n}{4}=2u_n+v_n+u_n+3v_n=3u_n+4v_n=t_n\]
إذن $(t_n)$ ثابتة:
\[t_n=t_0=3\times1+4\times8=\bx{35}\]
بالمرور إلى النهاية:
\[3\ell+4\ell=35\ \Rightarrow\ 7\ell=35\ \Rightarrow\ \bx{\ell=5}\]
\end{hal}
