\begin{molahada}[اصطلاحات]
في كل التمارين، المستوي المركب منسوب إلى المعلم المتعامد والمتجانس $(O;\vec u,\vec v)$، و$i$ هو العدد المركب الذي يحقق $i^2=-1$. نقول عن النقطة $M$ إن لاحقتها $z$ ونكتب $M(z)$.
\end{molahada}

\section{الشكل الجبري والمعادلات}

\begin{tamrin}{العمليات على الشكل الجبري}
نعتبر العددين المركبين $z_1=2-3i$ و$z_2=1+2i$.
\QQ{1} اكتب على الشكل الجبري كلًّا من: $z_1+z_2$، $z_1z_2$، $z_1^2$ و$\dfrac{z_1}{z_2}$.
\QQ{2} احسب $|z_1|$ و$|z_2|$، ثم تحقق أن $|z_1z_2|=|z_1|\times|z_2|$.
\QQ{3} حل في $\mathbb{C}$ المعادلة: $(1+i)z=3-i$.
\QQ{4} عيّن العدد المركب $z$ الذي يحقق: $2z+i\,\overline{z}=3+3i$.
\QQ{5} $x$ عدد حقيقي، و$Z=(x+i)(2-i)$.
\begin{sq}
\qa اكتب $Z$ على الشكل الجبري.
\qb عيّن $x$ حتى يكون $Z$ حقيقيًا، ثم حتى يكون $Z$ تخيليًا صرفًا.
\end{sq}
\end{tamrin}
\begin{hal}
\QQ{1} نحسب كما في $\R$ مع مراعاة $i^2=-1$:
\[z_1+z_2=(2+1)+(-3+2)i=\bx{3-i}.\]
\[z_1z_2=(2-3i)(1+2i)=2+4i-3i-6i^2=2+i+6=\bx{8+i}.\]
\[z_1^2=(2-3i)^2=4-12i+9i^2=4-12i-9=\bx{-5-12i}.\]
للقسمة نضرب البسط والمقام في مرافق المقام $\overline{z_2}=1-2i$، مع العلم أن $(1+2i)(1-2i)=1^2+2^2=5$:
\[\frac{z_1}{z_2}=\frac{(2-3i)(1-2i)}{(1+2i)(1-2i)}=\frac{2-4i-3i+6i^2}{5}=\frac{-4-7i}{5}=\bx{-\frac45-\frac75\,i}.\]
\QQ{2} طويلة $a+ib$ هي $\sqrt{a^2+b^2}$:
\[|z_1|=\sqrt{2^2+(-3)^2}=\bx{\sqrt{13}},\qquad |z_2|=\sqrt{1^2+2^2}=\bx{\sqrt5}.\]
من جهة: $|z_1z_2|=|8+i|=\sqrt{64+1}=\sqrt{65}$، ومن جهة أخرى: $|z_1|\times|z_2|=\sqrt{13}\times\sqrt5=\sqrt{65}$. إذن $|z_1z_2|=|z_1|\times|z_2|$ \ok.
\QQ{3} بما أن $1+i\neq0$:
\[z=\frac{3-i}{1+i}=\frac{(3-i)(1-i)}{(1+i)(1-i)}=\frac{3-3i-i+i^2}{2}=\frac{2-4i}{2}=\bx{1-2i}.\]
تحقق: $(1+i)(1-2i)=1-2i+i-2i^2=3-i$ \ok.
\QQ{4} نضع $z=x+iy$ مع $x,y$ حقيقيين، فيكون $\overline{z}=x-iy$ و:
\[2z+i\,\overline{z}=2x+2iy+ix-i^2y=(2x+y)+i(x+2y).\]
عددان مركبان متساويان معناه تساوي الجزأين الحقيقيين وتساوي الجزأين التخيليين:
\[\begin{cases}2x+y=3\\ x+2y=3\end{cases}\]
بطرح المعادلتين: $x-y=0$ أي $x=y$، ثم $3x=3$ أي $x=1$. إذن \bx{z=1+i}.
تحقق: $2(1+i)+i(1-i)=2+2i+i+1=3+3i$ \ok.
\QQ{5}
\pa $Z=(x+i)(2-i)=2x-ix+2i-i^2=\bx{(2x+1)+i(2-x)}$.
\pb $Z$ حقيقي معناه جزؤه التخيلي معدوم: $2-x=0$ أي \bx{x=2} (ونجد $Z=5$).

$Z$ تخيلي صرف معناه جزؤه الحقيقي معدوم: $2x+1=0$ أي \bx{x=-\tfrac12} (ونجد $Z=\tfrac52\,i$).
\end{hal}

\begin{tamrin}{معادلات من الدرجة الثانية والثالثة}
\QQ{1} حل في $\mathbb{C}$ المعادلات التالية:
\begin{sq}
\qa $z^2-4z+13=0$.
\qb $z^2+z+1=0$.
\qc $(z^2+4)(z^2-2z+2)=0$.
\end{sq}
\QQ{2} نعتبر كثير الحدود $P(z)=z^3-3z^2+7z-5$.
\begin{sq}
\qa احسب $P(1)$.
\qb عيّن العددين الحقيقيين $a$ و$b$ بحيث من أجل كل عدد مركب $z$: $P(z)=(z-1)(z^2+az+b)$.
\qc حل في $\mathbb{C}$ المعادلة $P(z)=0$.
\end{sq}
\QQ{3} حل في $\mathbb{C}$ المعادلة: $\dfrac{z-2i}{z+1}=i$ (مع $z\neq-1$).
\end{tamrin}
\begin{hal}
\begin{molahada}[تذكير]
المعادلة $az^2+bz+c=0$ (حيث $a,b,c$ أعداد حقيقية و$a\neq0$) مميزها $\Delta=b^2-4ac$. إذا كان $\Delta<0$ فلها حلان مركبان مترافقان: $z_1=\dfrac{-b-i\sqrt{-\Delta}}{2a}$ و$z_2=\overline{z_1}=\dfrac{-b+i\sqrt{-\Delta}}{2a}$.
\end{molahada}
\QQ{1}
\pa $\Delta=(-4)^2-4\times1\times13=16-52=-36=(6i)^2<0$، إذن:
\[z_1=\frac{4-6i}{2}=2-3i,\qquad z_2=\frac{4+6i}{2}=2+3i.\]
مجموعة الحلول: \bx{S=\{2-3i\,;\,2+3i\}}.
\pb $\Delta=1-4=-3<0$ و$\sqrt{-\Delta}=\sqrt3$، إذن:
\[\bx{S=\left\{-\frac12-\frac{\sqrt3}{2}i\,;\,-\frac12+\frac{\sqrt3}{2}i\right\}}.\]
\pc جداء معدوم معناه أحد عامليه معدوم:
\begin{itemize}
\item $z^2+4=0$ معناه $z^2=-4=(2i)^2$، أي $z=2i$ أو $z=-2i$.
\item $z^2-2z+2=0$: $\Delta=4-8=-4=(2i)^2$، إذن $z=\dfrac{2-2i}{2}=1-i$ أو $z=1+i$.
\end{itemize}
\[\bx{S=\{-2i\,;\,2i\,;\,1-i\,;\,1+i\}}.\]
\QQ{2}
\pa $P(1)=1-3+7-5=\bx{0}$، إذن $1$ جذر لـ $P$ و$P(z)$ يقبل القسمة على $(z-1)$.
\pb ننشر الطرف الثاني:
\[(z-1)(z^2+az+b)=z^3+az^2+bz-z^2-az-b=z^3+(a-1)z^2+(b-a)z-b.\]
بالمطابقة مع $P(z)=z^3-3z^2+7z-5$:
\[\begin{cases}a-1=-3\\ b-a=7\\ -b=-5\end{cases}\iff\begin{cases}a=-2\\ b=5\end{cases}\]
(المعادلة الثانية محققة: $b-a=5+2=7$).
إذن \bx{P(z)=(z-1)(z^2-2z+5)}.
\pc $P(z)=0$ معناه $z-1=0$ أو $z^2-2z+5=0$.

للمعادلة الثانية: $\Delta=4-20=-16=(4i)^2$، إذن $z=\dfrac{2-4i}{2}=1-2i$ أو $z=1+2i$.
\[\bx{S=\{1\,;\,1-2i\,;\,1+2i\}}.\]
\QQ{3} من أجل $z\neq-1$:
\[\frac{z-2i}{z+1}=i\iff z-2i=i(z+1)\iff z-iz=3i\iff z(1-i)=3i.\]
\[z=\frac{3i}{1-i}=\frac{3i(1+i)}{(1-i)(1+i)}=\frac{3i+3i^2}{2}=\bx{-\frac32+\frac32\,i}.\]
وهو يختلف عن $-1$، فهو مقبول.
\end{hal}

\section{الشكل المثلثي والأسي -- دستور موافر}

\begin{tamrin}{الشكل الأسي واستنتاج قيم مضبوطة}
نعتبر العددين المركبين $z_1=1+i\sqrt3$ و$z_2=1-i$.
\QQ{1} عيّن الطويلة وعمدة لكل من $z_1$ و$z_2$، ثم اكتبهما على الشكل الأسي.
\QQ{2} اكتب على الشكل الأسي كلًّا من $z_1z_2$ و$\dfrac{z_1}{z_2}$.
\QQ{3} اكتب $\dfrac{z_1}{z_2}$ على الشكل الجبري.
\QQ{4} استنتج القيمتين المضبوطتين لـ $\cos\dfrac{7\pi}{12}$ و$\sin\dfrac{7\pi}{12}$.
\QQ{5} احسب $\left(\dfrac{z_1}{z_2}\right)^{12}$.
\end{tamrin}
\begin{hal}
\QQ{1} $|z_1|=\sqrt{1+3}=2$، ونبحث عن $\theta_1$ بحيث:

\QQ{1} حسب موافر: $(\cos\theta+i\sin\theta)^3=e^{3i\theta}=\cos3\theta+i\sin3\theta$. وبالنشر (دستور $(a+b)^3$) مع $i^2=-1$ و$i^3=-i$، وبوضع $c=\cos\theta$ و$s=\sin\theta$:
\[(c+is)^3=c^3+3c^2(is)+3c(is)^2+(is)^3=\left(c^3-3cs^2\right)+i\left(3c^2s-s^3\right).\]
بمطابقة الجزأين الحقيقيين ثم التخيليين، واستعمال $s^2=1-c^2$ و$c^2=1-s^2$:
\begin{align*}
\cos3\theta&=c^3-3c\left(1-c^2\right)=\bx{4\cos^3\theta-3\cos\theta},\\
\sin3\theta&=3\left(1-s^2\right)s-s^3=\bx{3\sin\theta-4\sin^3\theta}.
\end{align*}
\QQ{2}
\pa نبيّن أولًا أن كل حلول $(E)$ تنتمي إلى $[-1;1]$: إذا كان $|x|>1$ فإن $4x^2-3>1$، ومنه:
\[|8x^3-6x|=2|x|\times\left|4x^2-3\right|>2\times1\times1=2>1,\]
فلا يمكن أن يكون $8x^3-6x=1$. والدالة $\theta\mapsto\cos\theta$ تقابل من $[0;\pi]$ نحو $[-1;1]$، فيمكن وضع $x=\cos\theta$ مع $\theta\in[0;\pi]$. عندئذ:
\[8x^3-6x-1=0\iff2\left(4\cos^3\theta-3\cos\theta\right)=1\iff\bx{\cos3\theta=\frac12}.\]
\pb $\cos3\theta=\cos\frac\pi3\iff3\theta=\pm\frac\pi3+2k\pi\iff\theta=\pm\frac\pi9+\frac{2k\pi}3$. مع $\theta\in[0;\pi]$ أي $3\theta\in[0;3\pi]$:
\[3\theta\in\left\{\frac\pi3;\frac{5\pi}3;\frac{7\pi}3\right\}\iff\theta\in\left\{\frac\pi9;\frac{5\pi}9;\frac{7\pi}9\right\}.\]
الدالة $\cos$ متناقصة تمامًا على $[0;\pi]$ فهذه القيم تعطي ثلاثة أعداد مختلفة، و$(E)$ من الدرجة الثالثة فلها ثلاثة حلول على الأكثر:
\[\bx{S=\left\{\cos\frac\pi9\,;\,\cos\frac{5\pi}9\,;\,\cos\frac{7\pi}9\right\}}\quad(\approx0{,}940\ ;\ -0{,}174\ ;\ -0{,}766).\]
\pc الحلول الثلاثة $x_1,x_2,x_3$ فـ $8x^3-6x-1=8(x-x_1)(x-x_2)(x-x_3)$. بنشر الطرف الثاني ومطابقة معامل $x^2$ والحد الثابت:
\[-8(x_1+x_2+x_3)=0,\qquad -8x_1x_2x_3=-1.\]
\[\bx{\cos\frac\pi9+\cos\frac{5\pi}9+\cos\frac{7\pi}9=0},\qquad\bx{\cos\frac\pi9\cos\frac{5\pi}9\cos\frac{7\pi}9=\frac18}.\]
\QQ{3} حسب أولر $\cos\theta=\dfrac{e^{i\theta}+e^{-i\theta}}{2}$، وبدستور ثنائي الحد ($1,4,6,4,1$):
\[\cos^4\theta=\frac{1}{16}\left(e^{4i\theta}+4e^{2i\theta}+6+4e^{-2i\theta}+e^{-4i\theta}\right)=\frac1{16}\left(2\cos4\theta+8\cos2\theta+6\right).\]
\[\bx{\cos^4\theta=\frac18\cos4\theta+\frac12\cos2\theta+\frac38}.\]
\QQ{4}
\pa $S$ مجموع $n+1$ حدًّا متتابعًا لمتتالية هندسية أساسها $e^{i\theta}\neq1$ (لأن $\theta\in]0;2\pi[$):
\[S=\frac{e^{i(n+1)\theta}-1}{e^{i\theta}-1}=\frac{e^{i\frac{(n+1)\theta}2}\left(e^{i\frac{(n+1)\theta}2}-e^{-i\frac{(n+1)\theta}2}\right)}{e^{i\frac\theta2}\left(e^{i\frac\theta2}-e^{-i\frac\theta2}\right)}=\frac{e^{i\frac{(n+1)\theta}2}\times2i\sin\frac{(n+1)\theta}2}{e^{i\frac\theta2}\times2i\sin\frac\theta2}.\]
و$\frac{(n+1)\theta}{2}-\frac\theta2=\frac{n\theta}2$، إذن \bx{S=e^{i\frac{n\theta}2}\dfrac{\sin\frac{(n+1)\theta}{2}}{\sin\frac\theta2}} ($\sin\frac\theta2\neq0$ لأن $\frac\theta2\in]0;\pi[$).
\pb $\operatorname{Re}S=\displaystyle\sum_{k=0}^n\cos k\theta$، وبأخذ الجزء الحقيقي للعبارة السابقة نحصل على المطلوب.

من أجل $\theta=\frac{2\pi}5$ و$n=4$: $\sin\frac{5\theta}{2}=\sin\pi=0$، إذن:
\[1+\cos\frac{2\pi}5+\cos\frac{4\pi}5+\cos\frac{6\pi}5+\cos\frac{8\pi}5=0.\]
لكن $\cos\frac{6\pi}5=\cos\left(2\pi-\frac{4\pi}5\right)=\cos\frac{4\pi}5$ و$\cos\frac{8\pi}5=\cos\frac{2\pi}5$، فـ $1+2\left(\cos\frac{2\pi}5+\cos\frac{4\pi}5\right)=0$:
\[\bx{\cos\frac{2\pi}5+\cos\frac{4\pi}5=-\frac12}.\]
\end{hal}

\begin{tamrin}{معادلة من الدرجة الثالثة وتشابه -- شعبة رياضيات}
نعتبر كثير الحدود $P(z)=z^3-(2+2i)z^2+(2+4i)z-4i$.
\QQ{1} بيّن أن المعادلة $P(z)=0$ تقبل حلًّا تخيليًا صرفًا $z_0=iy$ ($y$ حقيقي) يطلب تعيينه.
\QQ{2} عيّن العددين الحقيقيين $\alpha$ و$\beta$ بحيث: $P(z)=(z-2i)(z^2+\alpha z+\beta)$، ثم حل في $\mathbb{C}$ المعادلة $P(z)=0$.
\QQ{3} $A$، $B$ و$C$ النقط ذات اللواحق $z_A=1+i$، $z_B=1-i$ و$z_C=2i$. بيّن أن المثلث $OAB$ قائم ومتساوي الساقين.
\QQ{4} $S$ التشابه المباشر الذي مركزه $O$ ويحول $B$ إلى $C$.
\begin{sq}
\qa عيّن الكتابة المركبة لـ $S$ وعناصره المميزة.
\qb عيّن لاحقة $D=S(A)$، واستنتج طبيعة المثلث $OCD$.
\qc احسب النسبة بين مساحتي المثلثين $OCD$ و$OAB$. أنشئ الشكل.
\end{sq}
\end{tamrin}
\begin{hal}
\QQ{1} نعوض $z=iy$، مع $i^2=-1$ و$i^3=-i$:
\[P(iy)=-iy^3-(2+2i)(-y^2)+(2+4i)(iy)-4i=\left(2y^2-4y\right)+i\left(-y^3+2y^2+2y-4\right).\]
$P(iy)=0$ معناه انعدام الجزأين معًا:
\[\begin{cases}2y^2-4y=0\\ -y^3+2y^2+2y-4=0\end{cases}\]
الأولى: $2y(y-2)=0$ أي $y=0$ أو $y=2$. نعوض في الثانية: $y=0$ يعطي $-4\neq0$ (مرفوض)، و$y=2$ يعطي $-8+8+4-4=0$ \ok.
\bxt{الحل التخيلي الصرف هو $z_0=2i$}.
\QQ{2} ننشر: $(z-2i)(z^2+\alpha z+\beta)=z^3+(\alpha-2i)z^2+(\beta-2i\alpha)z-2i\beta$. بالمطابقة مع $P(z)$:
\[\begin{cases}\alpha-2i=-2-2i\\ \beta-2i\alpha=2+4i\\ -2i\beta=-4i\end{cases}\iff\begin{cases}\alpha=-2\\ \beta=2\end{cases}\]
(والمعادلة الثانية محققة: $2-2i(-2)=2+4i$). إذن \bx{P(z)=(z-2i)(z^2-2z+2)}.

$z^2-2z+2=0$: $\Delta=4-8=-4=(2i)^2$ فـ $z=1-i$ أو $z=1+i$. إذن:
\[\bx{S=\{2i\,;\,1-i\,;\,1+i\}}.\]
\QQ{3} نحسب:
\[\frac{z_A-z_O}{z_B-z_O}=\frac{1+i}{1-i}=\frac{(1+i)^2}{2}=\frac{2i}2=i.\]
بما أن $|i|=1$ و$\arg i=\frac\pi2$:
\[OA=OB,\qquad \left(\overrightarrow{OB},\overrightarrow{OA}\right)=\frac\pi2\ [2\pi].\]
\bxt{المثلث $OAB$ قائم في $O$ ومتساوي الساقين}.
\QQ{4}
\pa $S$ مركزه $O$: كتابته $z'=az$. و$S(B)=C$ يعطي:
\[a=\frac{z_C}{z_B}=\frac{2i}{1-i}=\frac{2i(1+i)}{2}=i+i^2=-1+i.\]
إذن \bx{S:\ z'=(-1+i)z}. ولدينا $|a|=\sqrt2$ و:
\[a=\sqrt2\left(-\frac{\sqrt2}2+\frac{\sqrt2}2i\right)=\sqrt2e^{i\frac{3\pi}4}.\]
\bxt{$S$ تشابه مباشر مركزه $O$، نسبته $\sqrt2$ وزاويته $\frac{3\pi}4$}.
\pb $z_D=(-1+i)(1+i)=-1-i+i+i^2=\bx{-2}$.
التشابه المباشر يحافظ على الزوايا الموجهة ويضرب الأطوال في $\sqrt2$، وصورة المثلث $OAB$ هي $OCD$ (لأن $S(O)=O$، $S(A)=D$، $S(B)=C$). تحقق مباشر: $\dfrac{z_D}{z_C}=\dfrac{-2}{2i}=i$:
\bxt{المثلث $OCD$ قائم في $O$ ومتساوي الساقين}.
\pc التشابه ذو النسبة $k$ يضرب المساحات في $k^2$:
\[\mathcal A(OAB)=\frac{OA\times OB}{2}=\frac{\sqrt2\times\sqrt2}{2}=1,\qquad \mathcal A(OCD)=\frac{OC\times OD}{2}=\frac{2\times2}{2}=2.\]
\[\frac{\mathcal A(OCD)}{\mathcal A(OAB)}=\bx{2=\left(\sqrt2\right)^2}.\]

\RA{O}{A}{B}\RA{O}{C}{D}
\end{CP}
\end{hal}
