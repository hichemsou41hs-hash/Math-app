\begin{molahada}[اصطلاحات]
في كل التمارين، المستوي المركب منسوب إلى المعلم المتعامد والمتجانس $(O;\vec u,\vec v)$، و$i$ هو العدد المركب الذي يحقق $i^2=-1$. نقول عن النقطة $M$ إن لاحقتها $z$ ونكتب $M(z)$.
\end{molahada}

\section{الشكل الجبري والمعادلات}

\begin{tamrin}{العمليات على الشكل الجبري}
نعتبر العددين المركبين $z_1=2-3i$ و$z_2=1+2i$.
\QQ{1} اكتب على الشكل الجبري كلًّا من: $z_1+z_2$، $z_1z_2$، $z_1^2$ و$\dfrac{z_1}{z_2}$.
\QQ{2} احسب $|z_1|$ و$|z_2|$، ثم تحقق أن $|z_1z_2|=|z_1|\times|z_2|$.
\QQ{3} حل في $\mathbb{C}$ المعادلة: $(1+i)z=3-i$.
\QQ{4} عيّن العدد المركب $z$ الذي يحقق: $2z+i\,\overline{z}=3+3i$.
\QQ{5} $x$ عدد حقيقي، و$Z=(x+i)(2-i)$.
\begin{sq}
\qa اكتب $Z$ على الشكل الجبري.
\qb عيّن $x$ حتى يكون $Z$ حقيقيًا، ثم حتى يكون $Z$ تخيليًا صرفًا.
\end{sq}
\end{tamrin}

\begin{tamrin}{معادلات من الدرجة الثانية والثالثة}
\QQ{1} حل في $\mathbb{C}$ المعادلات التالية:
\begin{sq}
\qa $z^2-4z+13=0$.
\qb $z^2+z+1=0$.
\qc $(z^2+4)(z^2-2z+2)=0$.
\end{sq}
\QQ{2} نعتبر كثير الحدود $P(z)=z^3-3z^2+7z-5$.
\begin{sq}
\qa احسب $P(1)$.
\qb عيّن العددين الحقيقيين $a$ و$b$ بحيث من أجل كل عدد مركب $z$: $P(z)=(z-1)(z^2+az+b)$.
\qc حل في $\mathbb{C}$ المعادلة $P(z)=0$.
\end{sq}
\QQ{3} حل في $\mathbb{C}$ المعادلة: $\dfrac{z-2i}{z+1}=i$ (مع $z\neq-1$).
\end{tamrin}

\section{الشكل المثلثي والأسي -- دستور موافر}

\begin{tamrin}{الشكل الأسي واستنتاج قيم مضبوطة}
نعتبر العددين المركبين $z_1=1+i\sqrt3$ و$z_2=1-i$.
\QQ{1} عيّن الطويلة وعمدة لكل من $z_1$ و$z_2$، ثم اكتبهما على الشكل الأسي.
\QQ{2} اكتب على الشكل الأسي كلًّا من $z_1z_2$ و$\dfrac{z_1}{z_2}$.
\QQ{3} اكتب $\dfrac{z_1}{z_2}$ على الشكل الجبري.
\QQ{4} استنتج القيمتين المضبوطتين لـ $\cos\dfrac{7\pi}{12}$ و$\sin\dfrac{7\pi}{12}$.
\QQ{5} احسب $\left(\dfrac{z_1}{z_2}\right)^{12}$.
\end{tamrin}

\begin{tamrin}{قوى عدد مركب}
نعتبر العددين المركبين $z=\sqrt3+i$ و$w=1+i$.
\QQ{1} اكتب $z$ و$w$ على الشكل الأسي.
\QQ{2} عيّن قيم العدد الطبيعي $n$ التي يكون من أجلها $z^n$:
\begin{sq}
\qa عددًا حقيقيًا.
\qb عددًا تخيليًا صرفًا.
\end{sq}
\QQ{3} احسب $z^6$، ثم اكتب $z^{2026}$ على الشكل الجبري.
\QQ{4} عيّن أصغر عدد طبيعي غير معدوم $n$ يكون من أجله $(zw)^n$ عددًا حقيقيًا، ثم احسب $(zw)^n$ من أجل هذه القيمة.
\end{tamrin}

\begin{tamrin}{الشكل الأسي لـ $1+e^{i\theta}$ و$1-e^{i\theta}$}
$\theta$ عدد حقيقي من المجال $]-\pi;\pi[$.
\QQ{1} بيّن أن: $1+e^{i\theta}=2\cos\dfrac\theta2\,e^{i\frac\theta2}$.
\QQ{2} استنتج طويلة وعمدة للعدد $1+e^{i\theta}$.
\QQ{3} تحقق من النتيجة من أجل $\theta=\dfrac{2\pi}{3}$.
\QQ{4} بيّن أن $1-e^{i\theta}=-2i\sin\dfrac\theta2\,e^{i\frac\theta2}$، ثم استنتج، من أجل $\theta\in]0;\pi[$، طويلة وعمدة للعدد $1-e^{i\theta}$.
\QQ{5} استنتج أن العدد $\dfrac{1+e^{i\theta}}{1-e^{i\theta}}$ تخيلي صرف من أجل $\theta\in]0;\pi[$.
\end{tamrin}

\begin{tamrin}{معادلة وحلولها على الشكل الأسي}
\QQ{1} حل في $\mathbb{C}$ المعادلة: $z^2-2\sqrt3\,z+4=0$. نرمز $z_1$ للحل الذي جزؤه التخيلي موجب و$z_2$ للحل الآخر.
\QQ{2} اكتب $z_1$ و$z_2$ على الشكل الأسي، ثم $\dfrac{z_1}{z_2}$.
\QQ{3} عيّن الأعداد الطبيعية $n$ التي تحقق $\left(\dfrac{z_1}{z_2}\right)^n=1$.
\QQ{4} بيّن أنه من أجل كل عدد طبيعي $n$: $z_1^n+z_2^n=2^{n+1}\cos\dfrac{n\pi}{6}$، ثم احسب $z_1^6+z_2^6$.
\QQ{5} $A$ و$B$ النقطتان اللتان لاحقتاهما $z_1$ و$z_2$. بيّن أن المثلث $OAB$ متقايس الأضلاع، وارسمه.
\end{tamrin}
\section{التفسير الهندسي ومجموعات النقط}

\begin{tamrin}{طبيعة مثلث ومربع ودائرة}
نعتبر النقط $A$، $B$ و$C$ التي لواحقها على الترتيب: $z_A=1+i$، $z_B=4+2i$ و$z_C=4i$.
\QQ{1} اكتب العدد $\dfrac{z_C-z_A}{z_B-z_A}$ على الشكل الجبري، ثم استنتج طبيعة المثلث $ABC$.
\QQ{2} عيّن لاحقة النقطة $D$ بحيث يكون الرباعي $ABDC$ مربعًا.
\QQ{3} عيّن لاحقة المركز $\Omega$ ونصف قطر الدائرة $(\Gamma)$ المحيطة بالمثلث $ABC$.
\QQ{4} تحقق أن $D$ تنتمي إلى $(\Gamma)$، ثم أنشئ الشكل.
\end{tamrin}

\begin{tamrin}{نقط من نفس الدائرة ومثلث متقايس الأضلاع}
نعتبر النقط $A$، $B$، $C$ و$D$ ذات اللواحق: $z_A=2$، $z_B=1+i\sqrt3$، $z_C=1-i\sqrt3$ و$z_D=-2$.
\QQ{1} اكتب $z_B$ و$z_C$ على الشكل الأسي، ثم استنتج أن النقط $A$، $B$، $C$ و$D$ تنتمي إلى دائرة يطلب تعيين مركزها ونصف قطرها.
\QQ{2} احسب $\dfrac{z_B-z_D}{z_C-z_D}$ على الشكل الأسي، واستنتج طبيعة المثلث $BCD$.
\QQ{3} بيّن أن العدد $\dfrac{z_A-z_B}{z_D-z_B}$ تخيلي صرف. ماذا تستنتج؟
\QQ{4} $E$ النقطة ذات اللاحقة $z_E=-1-i\sqrt3$. بيّن أن النقط $O$، $B$ و$E$ في استقامية. أنشئ الشكل.
\end{tamrin}

\begin{tamrin}{مجموعات نقط بسيطة}
عيّن في كل حالة مجموعة النقط $M$ ذات اللاحقة $z$ التي تحقق:
\begin{sq}
\qa $(E_1)$: $|z-1+2i|=2$.
\qb $(E_2)$: $|z-3|=|z+i|$.
\qc $(E_3)$: $|iz+2|=4$.
\qd $(E_4)$: $\arg(z-1-i)=\dfrac\pi4\ [2\pi]$.
\end{sq}
ثم مثّلها في المعلم.
\end{tamrin}

\begin{tamrin}{مجموعات نقط بواسطة عدد مركب}
نعتبر النقطتين $A(-1)$ و$B(2i)$. من أجل كل عدد مركب $z\neq-1$ نضع: $Z=\dfrac{z-2i}{z+1}$.
\QQ{1} نضع $z=x+iy$ ($x,y$ حقيقيان). اكتب الجزء الحقيقي والجزء التخيلي لـ $Z$ بدلالة $x$ و$y$.
\QQ{2} عيّن المجموعة $(\Delta)$ للنقط $M(z)$ بحيث يكون $Z$ حقيقيًا.
\QQ{3} عيّن المجموعة $(\Gamma)$ للنقط $M(z)$ بحيث يكون $Z$ تخيليًا صرفًا.
\QQ{4} عيّن المجموعة $(\mathcal D)$ للنقط $M(z)$ بحيث $|Z|=1$.
\QQ{5} أعط تفسيرًا هندسيًا لعمدة $Z$، وتحقق به من نتيجة السؤال 3. أنشئ المجموعات الثلاث.
\end{tamrin}

\section{التحويلات النقطية}

\begin{tamrin}{الانسحاب والتحاكي}
نعتبر التحويلين النقطيين $t$ و$h$ اللذين يرفقان بكل نقطة $M(z)$ النقطة $M'(z')$ حيث:
\[t:\ z'=z+2-i,\qquad\qquad h:\ z'=-2z+3i.\]
\QQ{1} عيّن طبيعة كل من $t$ و$h$ وعناصرهما المميزة.
\QQ{2} $A$ النقطة ذات اللاحقة $1+i$. عيّن لاحقتي $A_1=t(A)$ و$A_2=h(A)$.
\QQ{3} $(\mathcal C)$ الدائرة التي مركزها $A$ ونصف قطرها $1$. عيّن صورتيها بـ $t$ ثم بـ $h$.
\QQ{4} عيّن الكتابة المركبة للتحويل $h\circ t$، ثم طبيعته.
\end{tamrin}

\begin{tamrin}{الدوران}
\QQ{1} $r$ التحويل النقطي الذي يرفق بكل نقطة $M(z)$ النقطة $M'(z')$ حيث $z'=iz+1+i$.
\begin{sq}
\qa بيّن أن $r$ دوران يطلب تعيين مركزه $\Omega$ وزاويته.
\qb عيّن لاحقة $A'$ صورة النقطة $A(2+i)$ بـ $r$، ثم استنتج طبيعة المثلث $\Omega AA'$.
\end{sq}
\QQ{2} $r'$ الدوران الذي مركزه $B(1+i)$ وزاويته $-\dfrac\pi2$.
\begin{sq}
\qa عيّن الكتابة المركبة لـ $r'$.
\qb عيّن لاحقة $C'$ صورة النقطة $C(3+i)$ بـ $r'$.
\qc عيّن لاحقة النقطة $E$ التي صورتها بـ $r'$ هي $O$.
\end{sq}
\end{tamrin}

\begin{tamrin}{تعيين تحويل بمعرفة صورتي نقطتين}
نعتبر النقط $A(0)$، $B(4)$، $A'\left(1-i\sqrt3\right)$ و$B'\left(3+i\sqrt3\right)$.
$f$ التحويل النقطي ذو الكتابة المركبة $z'=az+b$ ($a,b$ عددان مركبان) الذي يحول $A$ إلى $A'$ و$B$ إلى $B'$.
\QQ{1} عيّن $a$ و$b$.
\QQ{2} اكتب $a$ على الشكل الأسي، ثم استنتج طبيعة $f$ وعناصره المميزة.
\QQ{3} بيّن أن المثلثين $\Omega AA'$ و$\Omega BB'$ متقايسا الأضلاع، حيث $\Omega$ مركز $f$. أنشئ الشكل.
\end{tamrin}

\begin{tamrin}{المرجح ومجموعة نقط وصورتها}
نعتبر النقط $A(4+2i)$، $B(2-2i)$ و$C(-1+2i)$.
\QQ{1} عيّن لاحقة النقطة $G$ مرجح الجملة $\{(A;1),(B;1),(C;2)\}$.
\QQ{2} عيّن المجموعة $(\Gamma)$ للنقط $M$ من المستوي التي تحقق: $\left\|\overrightarrow{MA}+\overrightarrow{MB}+2\overrightarrow{MC}\right\|=8$.
\QQ{3} اكتب هذا الشرط بدلالة لاحقة $M$.
\QQ{4} عيّن صورة $(\Gamma)$ بالانسحاب $t$ الذي شعاعه $\overrightarrow{AB}$، ثم صورتها بالتحاكي $h$ الذي مركزه $A$ ونسبته $\frac12$. أنشئ الشكل.
\end{tamrin}
\section{تمارين للتعمق: التشابهات المباشرة ومسائل}
\textbf{(للشعب الثلاث: رياضيات، تقني رياضي، علوم تجريبية)}

\begin{tamrin}{التشابه المباشر وعناصره المميزة}
$S$ التحويل النقطي الذي يرفق بكل نقطة $M(z)$ النقطة $M'(z')$ حيث: $z'=(1+i)z+2-i$.
\QQ{1} بيّن أن $S$ تشابه مباشر يطلب تعيين مركزه $\Omega$ ونسبته وزاويته.
\QQ{2} بيّن أنه من أجل كل نقطة $M\neq\Omega$: $\dfrac{z'-z}{\omega-z}=-i$، واستنتج طبيعة المثلث $\Omega MM'$.
\QQ{3} عيّن لاحقة $A'$ صورة النقطة $A(1)$ بـ $S$، وتحقق من نتيجة السؤال 2 على المثلث $\Omega AA'$.
\QQ{4} $(\mathcal C)$ الدائرة ذات المركز $A$ ونصف القطر $1$. عيّن صورتها $(\mathcal C')$ بـ $S$.
\QQ{5} عيّن الكتابة المركبة للتشابه المباشر $T$ الذي يحول $O$ إلى $E(2)$ و$E$ إلى $F(4+2i)$، ثم عناصره المميزة.
\end{tamrin}

\begin{tamrin}{تركيب دوران وتحاكٍ}
$r$ الدوران الذي مركزه $A(1)$ وزاويته $\dfrac\pi2$، و$h$ التحاكي الذي مركزه $B(-i)$ ونسبته $2$.
\QQ{1} عيّن الكتابة المركبة لكل من $r$ و$h$.
\QQ{2} عيّن الكتابة المركبة لكل من $f=h\circ r$ و$g=r\circ h$.
\QQ{3} بيّن أن $f$ و$g$ تشابهان مباشران لهما نفس النسبة ونفس الزاوية، وعيّن مركز كل منهما. هل $h\circ r=r\circ h$؟
\QQ{4} عيّن طبيعة التحويل $f\circ f$ وعناصره المميزة.
\QQ{5} عيّن لاحقتي $f(O)$ و$g(O)$، ومثّل النقط في المعلم.
\end{tamrin}

\begin{tamrin}{التحليل القانوني لتشابه مباشر}
$S$ التحويل النقطي الذي يرفق بكل نقطة $M(z)$ النقطة $M'(z')$ حيث: $z'=(-1+i)z+3+i$.
\QQ{1} بيّن أن $S$ تشابه مباشر يطلب تعيين مركزه $\Omega$ ونسبته $k$ وزاويته $\theta$.
\QQ{2} $r$ الدوران الذي مركزه $\Omega$ وزاويته $\theta$، و$h$ التحاكي الذي مركزه $\Omega$ ونسبته $k$.
\begin{sq}
\qa اكتب الكتابة المركبة لكل من $r$ و$h$.
\qb بيّن أن $h\circ r=S$ وأن $r\circ h=S$ (التحليل القانوني لـ $S$).
\end{sq}
\QQ{3} $A$ النقطة ذات اللاحقة $2+2i$. عيّن لاحقة $A_1=r(A)$ ثم لاحقة $A'=h(A_1)$، وتحقق أن $A'=S(A)$. أنشئ النقط.
\QQ{4} $(\mathcal C)$ الدائرة ذات المركز $A$ ونصف القطر $1$. عيّن صورتها $(\mathcal C')$ بـ $S$.
\QQ{5} عيّن الكتابة المركبة لـ $S\circ S$، ثم عناصره المميزة وتحليله القانوني.
\end{tamrin}

\begin{tamrin}{مجموعات نقط معرفة وسيطيًا}
$\theta$ و$t$ و$k$ وسائط حقيقية.
\QQ{1} عيّن في كل حالة مجموعة النقط $M(z)$:
\begin{sq}
\qa $(\Gamma_1)$: $z=-1+i+\sqrt2\,e^{i\theta}$ مع $\theta\in\R$.
\qb $(\Delta)$: $z=2i+t(1+i)$ مع $t\in\R$.
\qc $(\Delta')$: $z=k\,e^{i\frac\pi3}$ مع $k\in]0;+\infty[$.
\qd $(\Gamma_2)$: $z=2+e^{i\theta}$ مع $\theta\in\left[-\frac\pi2;\frac\pi2\right]$.
\end{sq}
\QQ{2} عيّن نقطتي تقاطع $(\Gamma_1)$ و$(\Delta)$.
\QQ{3} نرفق بكل نقطة $M(z)$ النقطة $M'(z')$ حيث $z'=(1+i)z$. عيّن المجموعة $(\Gamma_2')$ التي ترسمها $M'$ لما ترسم $M$ المجموعة $(\Gamma_2)$. ارسم كل المجموعات.
\end{tamrin}

\begin{tamrin}{دستور موافر ودستورا أولر}
$\theta$ عدد حقيقي.
\QQ{1} باستعمال دستور موافر ونشر $(\cos\theta+i\sin\theta)^3$، بيّن أن:
\[\cos3\theta=4\cos^3\theta-3\cos\theta,\qquad \sin3\theta=3\sin\theta-4\sin^3\theta.\]
\QQ{2} نعتبر المعادلة $(E)$: $8x^3-6x-1=0$ حيث $x$ عدد حقيقي.
\begin{sq}
\qa بوضع $x=\cos\theta$ مع $\theta\in[0;\pi]$، بيّن أن $(E)$ تكافئ $\cos3\theta=\frac12$.
\qb استنتج حلول $(E)$.
\qc استنتج قيمتي $\cos\frac\pi9+\cos\frac{5\pi}9+\cos\frac{7\pi}9$ و$\cos\frac\pi9\times\cos\frac{5\pi}9\times\cos\frac{7\pi}9$.
\end{sq}
\QQ{3} باستعمال دستوري أولر، اكتب $\cos^4\theta$ خطيًا (بدلالة $\cos4\theta$ و$\cos2\theta$).
\QQ{4} $n$ عدد طبيعي غير معدوم و$\theta\in]0;2\pi[$. نضع $S=1+e^{i\theta}+e^{2i\theta}+\dots+e^{in\theta}$.
\begin{sq}
\qa بيّن أن $S=e^{i\frac{n\theta}2}\,\dfrac{\sin\frac{(n+1)\theta}{2}}{\sin\frac\theta2}$.
\qb استنتج أن $\displaystyle\sum_{k=0}^n\cos k\theta=\frac{\cos\frac{n\theta}2\,\sin\frac{(n+1)\theta}2}{\sin\frac\theta2}$، ثم أن $\cos\frac{2\pi}5+\cos\frac{4\pi}5=-\frac12$.
\end{sq}
\end{tamrin}

\begin{tamrin}{معادلة من الدرجة الثالثة وتشابه}
نعتبر كثير الحدود $P(z)=z^3-(2+2i)z^2+(2+4i)z-4i$.
\QQ{1} بيّن أن المعادلة $P(z)=0$ تقبل حلًّا تخيليًا صرفًا $z_0=iy$ ($y$ حقيقي) يطلب تعيينه.
\QQ{2} عيّن العددين الحقيقيين $\alpha$ و$\beta$ بحيث: $P(z)=(z-2i)(z^2+\alpha z+\beta)$، ثم حل في $\mathbb{C}$ المعادلة $P(z)=0$.
\QQ{3} $A$، $B$ و$C$ النقط ذات اللواحق $z_A=1+i$، $z_B=1-i$ و$z_C=2i$. بيّن أن المثلث $OAB$ قائم ومتساوي الساقين.
\QQ{4} $S$ التشابه المباشر الذي مركزه $O$ ويحول $B$ إلى $C$.
\begin{sq}
\qa عيّن الكتابة المركبة لـ $S$ وعناصره المميزة.
\qb عيّن لاحقة $D=S(A)$، واستنتج طبيعة المثلث $OCD$.
\qc احسب النسبة بين مساحتي المثلثين $OCD$ و$OAB$. أنشئ الشكل.
\end{sq}
\end{tamrin}
