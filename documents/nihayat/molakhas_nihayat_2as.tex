\section{نهاية غير منتهية عند عدد حقيقي}
\begin{mithal}[نشاط تمهيدي]
الدالة $f(x)=\dfrac1{(x-3)^2}$ معرفة على $\R-\{3\}$. نحسب بعض القيم:
\begin{center}
\begin{LTR}\renewcommand{\arraystretch}{1.5}
\begin{tabular}{|c|c|c|c|c|c|c|}\hline
\rowcolor{cSec!12}$x$ & $2{,}9$ & $2{,}99$ & $2{,}999$ & $3{,}001$ & $3{,}01$ & $3{,}1$\\\hline
$f(x)$ & $10^2$ & $10^4$ & $10^6$ & $10^6$ & $10^4$ & $10^2$\\\hline
\end{tabular}\end{LTR}
\end{center}
كلما اقترب $x$ من $3$ أخذت $f(x)$ قيمًا كبيرة جدًا: نقول إن نهاية $f$ عند $3$ هي $+\infty$.
\end{mithal}

\begin{tarif}
$\LW{f}$ دالة معرفة على مجال من الشكل $\LW{]a-r;a[\cup]a;a+r[}$.\par
القول إن نهاية $\LW{f}$ عند $\LW{a}$ هي $\LW{+\infty}$ يعني أن $\LW{f(x)}$ يصبح أكبر من أي عدد موجب $\LW{A}$ (مهما كان كبيرًا) بشرط أن يكون $\LW{x}$ قريبًا بالقدر الكافي من $\LW{a}$. نكتب:
\[\LW{\Lm{a}f(x)=+\infty}\]
وبالمثل نعرّف $\LW{\Lm{a}f(x)=-\infty}$.
\end{tarif}

\begin{khassia}[نتائج أساسية (تقبل دون برهان)]
من أجل كل عدد حقيقي $\LW{a}$ ومن أجل كل عدد طبيعي $\LW{n\geq1}$:
\[\LW{\Lm{0}\frac1{x^2}=+\infty \qquad;\qquad \Lm{a}\frac1{(x-a)^2}=+\infty \qquad;\qquad \Lm{a}\frac1{(x-a)^{2n}}=+\infty}\]
\end{khassia}

\section{النهاية من اليمين ومن اليسار}
\begin{tarif}
\begin{itemize}
\item \textbf{النهاية من اليمين} عند $\LW{a}$: نقرّب $\LW{x}$ من $\LW{a}$ بقيم أكبر من $\LW{a}$ ، ونكتب $\LW{\Lm{a^+}f(x)}$ أو $\LW{\displaystyle\lim_{\substack{x\to a\\x>a}}f(x)}$.
\item \textbf{النهاية من اليسار} عند $\LW{a}$: نقرّب $\LW{x}$ من $\LW{a}$ بقيم أصغر من $\LW{a}$ ، ونكتب $\LW{\Lm{a^-}f(x)}$ أو $\LW{\displaystyle\lim_{\substack{x\to a\\x<a}}f(x)}$.
\end{itemize}
\end{tarif}

\begin{khassia}[مبرهنة]
\[\LW{\Lm{a^+}\frac1{x-a}=+\infty \qquad;\qquad \Lm{a^-}\frac1{x-a}=-\infty}\]

\[\LW{\Lm{0^+}\frac1x=+\infty \qquad;\qquad \Lm{0^-}\frac1x=-\infty}\]
\end{khassia}

\begin{tariqa}[حساب نهاية دالة ناطقة عند قيمة يعدم فيها المقام]
\begin{enumerate}
\item نحسب نهاية البسط (عدد غير معدوم $l$).
\item ندرس \textbf{إشارة المقام} بجوار $a$ (جدول إشارة) لنعرف هل يؤول إلى $0^+$ أم إلى $0^-$.
\item نطبق قاعدة الإشارات: $\dfrac{l}{0^+}$ و $\dfrac{l}{0^-}$ تساوي $+\infty$ أو $-\infty$ حسب إشارة $l$.
\end{enumerate}
\end{tariqa}

\begin{mithal}
$f(x)=\dfrac1{x^2+x-2}$ على $\R-\{-2;1\}$ ، ولدينا $x^2+x-2=(x+2)(x-1)$.

\LD{f(x)}{1^+}{+\infty} $\Car{\Lm{1}1=1\\ \Lm{1^+}\left(x^2+x-2\right)=0^+}$
\end{mithal}

\section{النهاية عند $+\infty$ وعند $-\infty$}
\subsection{نهاية غير منتهية}
\begin{tarif}
القول إن نهاية $\LW{f}$ عند $\LW{+\infty}$ هي $\LW{+\infty}$ يعني أن $\LW{f(x)}$ يصبح أكبر من أي عدد موجب $\LW{A}$ بشرط أن يكون $\LW{x}$ كبيرًا بالقدر الكافي. نكتب $\LW{\Lm{+\infty}f(x)=+\infty}$ (وبالمثل لباقي الحالات).
\end{tarif}

\begin{khassia}[نهايات الدوال المرجعية]
\begin{center}
\begin{LTR}\renewcommand{\arraystretch}{1.6}
\begin{tabular}{|c|c|c|c|c|c|}\hline
\rowcolor{cSec!12}\ar{الدالة} & $\LW{x}$ & $\LW{x^2}$ & $\LW{x^3}$ & $\LW{\sqrt x}$ & $\LW{x^n\ (n\geq1)}$\\\hline
\ar{النهاية عند $\LW{+\infty}$} & $\LW{+\infty}$ & $\LW{+\infty}$ & $\LW{+\infty}$ & $\LW{+\infty}$ & $\LW{+\infty}$\\\hline
\ar{النهاية عند $\LW{-\infty}$} & $\LW{-\infty}$ & $\LW{+\infty}$ & $\LW{-\infty}$ & \ar{غير معرفة} & \ar{$\LW{+\infty}$ إذا كان $\LW{n}$ زوجيًا ، $\LW{-\infty}$ إذا كان $\LW{n}$ فرديًا}\\\hline
\end{tabular}\end{LTR}
\end{center}
\end{khassia}

\subsection{نهاية منتهية}
\begin{tarif}
القول إن نهاية $\LW{f}$ عند $\LW{+\infty}$ هي العدد $\LW{b}$ يعني أن $\LW{f(x)}$ يقترب من $\LW{b}$ بالقدر الذي نريد بشرط أن يكون $\LW{x}$ كبيرًا بالقدر الكافي. نكتب: $\LW{\Lm{+\infty}f(x)=b}$.
\end{tarif}

\begin{khassia}
\[\LW{\Lm{\pm\infty}\frac1x=0 \qquad;\qquad \Lm{\pm\infty}\frac1{x^n}=0 \qquad;\qquad \Lm{+\infty}\frac1{\sqrt x}=0 \qquad;\qquad \Lm{\pm\infty}\frac1{x+a}=0}\]
\end{khassia}

\section{المستقيم المقارب الموازي لمحور الفواصل}
\begin{tarif}
إذا كانت $\LW{\Lm{+\infty}f(x)=b}$ (أو $\LW{\Lm{-\infty}f(x)=b}$) حيث $\LW{b}$ عدد حقيقي ، نقول إن المستقيم ذا المعادلة $\LW{y=b}$ \textbf{مستقيم مقارب} للمنحنى $\LW{(C_f)}$ \textbf{يوازي محور الفواصل} بجوار $\LW{+\infty}$ (أو $\LW{-\infty}$).
\end{tarif}

\begin{mithal}
$f(x)=\dfrac{2x-1}{x+1}$ على $\R-\{-1\}$.\E
لدينا: $\Lm{\pm\infty}f(x)=\Lm{\pm\infty}\dfrac{2x}{x}=\bx{2}$\E
إذن المستقيم ذو المعادلة \bx{y=2} مقارب أفقي.
\LD{f(x)}{-1^-}{+\infty} $\Car{\Lm{-1}(2x-1)=-3\\ \Lm{-1^-}(x+1)=0^-}$
\LD{f(x)}{-1^+}{-\infty} $\Car{\Lm{-1}(2x-1)=-3\\ \Lm{-1^+}(x+1)=0^+}$
\E إذن المستقيم ذو المعادلة \bx{x=-1} مقارب عمودي.

\section{نهاية كثير حدود ودالة ناطقة عند $+\infty$ و $-\infty$}
\begin{khassia}[قاعدتان]
\begin{enumerate}
\item نهاية \textbf{كثير حدود} عند $\LW{+\infty}$ أو $\LW{-\infty}$ هي نهاية \textbf{حدّه الأعلى درجة}.
\item نهاية \textbf{دالة ناطقة} عند $\LW{+\infty}$ أو $\LW{-\infty}$ هي نهاية \textbf{حاصل قسمة الحدين الأعلى درجة} في البسط والمقام.
\end{enumerate}
\textbf{تنبيه:} هاتان القاعدتان صالحتان \textbf{فقط} عند $\LW{\pm\infty}$ ، وليس عند عدد حقيقي.
\end{khassia}

\begin{mithal}
\begin{enumerate}
\item $\Lm{+\infty}\left(-x^3+2x-2\right)=\Lm{+\infty}\left(-x^3\right)=\bx{-\infty}$
\item $\Lm{-\infty}\left(3x^2+x+2\right)=\Lm{-\infty}3x^2=\bx{+\infty}$
\item $\Lm{+\infty}\dfrac{2x^2+3x+1}{x^2-1}=\Lm{+\infty}\dfrac{2x^2}{x^2}=\bx{2}$
\item $\Lm{-\infty}\dfrac{1+x-x^2}{x^3-1}=\Lm{-\infty}\dfrac{-x^2}{x^3}=\Lm{-\infty}\dfrac{-1}{x}=\bx{0}$
\item $\Lm{-\infty}\dfrac{x^3+x-2}{x-1}=\Lm{-\infty}\dfrac{x^3}{x}=\Lm{-\infty}x^2=\bx{+\infty}$
\end{enumerate}
\end{mithal}

\section{طرق إزالة حالات عدم التعيين}
\subsection{الحالة $\frac\infty\infty$}
\begin{tariqa}
\begin{itemize}
\item \textbf{دالة ناطقة:} نطبق قاعدة الحدين الأعلى درجة.
\item \textbf{دالة صمّاء (بها جذر):} نخرج الحد الأعلى درجة عاملًا مشتركًا ، مع الانتباه إلى أن $\sqrt{x^2}=|x|$:\E
$\sqrt{x^2}=x$ عند $+\infty$ ، و $\sqrt{x^2}=-x$ عند $-\infty$.
\end{itemize}
\end{tariqa}

\begin{mithal}
نحسب $\Lm{-\infty}\dfrac{\sqrt{2x^2-3}}{x+2}$ (حالة $\frac\infty\infty$). من أجل $x<0$ نكتب:
\[\frac{\sqrt{2x^2-3}}{x+2}=\frac{|x|\sqrt{2-\frac3{x^2}}}{x\left(1+\frac2x\right)}=\frac{-x\sqrt{2-\frac3{x^2}}}{x\left(1+\frac2x\right)}=\frac{-\sqrt{2-\frac3{x^2}}}{1+\frac2x}\]
\LD{\dfrac{\sqrt{2x^2-3}}{x+2}}{-\infty}{-\sqrt2} $\Car{\Lm{-\infty}\left(-\sqrt{2-\frac3{x^2}}\right)=-\sqrt2\\ \Lm{-\infty}\left(1+\frac2x\right)=1}$
\end{mithal}

\subsection{الحالة $+\infty-\infty$}
\begin{tariqa}
\begin{itemize}
\item \textbf{كثير حدود:} نهاية الحد الأعلى درجة.
\item \textbf{عبارة بها جذر:} نستعمل العامل المشترك أو الضرب في المرافق: \quad $\sqrt A-B=\dfrac{A-B^2}{\sqrt A+B}$.
\end{itemize}
\end{tariqa}

\begin{mithal}
\textbf{1. بالمرافق:} نحسب $\Lm{+\infty}\left(\sqrt{x^2+1}-x\right)$ (حالة $+\infty-\infty$). نكتب:
\[\sqrt{x^2+1}-x=\frac{\left(x^2+1\right)-x^2}{\sqrt{x^2+1}+x}=\frac1{\sqrt{x^2+1}+x}\]
\LD{\left(\sqrt{x^2+1}-x\right)}{+\infty}{0} $\Car{\Lm{+\infty}1=1\\ \Lm{+\infty}\left(\sqrt{x^2+1}+x\right)=+\infty}$
\E
\textbf{2. بالعامل المشترك:} نحسب $\Lm{+\infty}\left(\sqrt{2x-3}-3x+2\right)$. من أجل $x>0$ نكتب:
\[\sqrt{2x-3}-3x+2=x\left(\sqrt{\frac2x-\frac3{x^2}}-3+\frac2x\right)\]
\LD{\left(\sqrt{2x-3}-3x+2\right)}{+\infty}{-\infty} $\Car{\Lm{+\infty}x=+\infty\\ \Lm{+\infty}\left(\sqrt{\frac2x-\frac3{x^2}}-3+\frac2x\right)=-3}$
\end{mithal}

\subsection{الحالة $\frac00$ (عند عدد حقيقي $x_0$)}
\begin{tariqa}
البسط والمقام ينعدمان عند $x_0$ ، إذن كلاهما يقبل القسمة على $(x-x_0)$. نستعمل:
\begin{itemize}
\item \textbf{التحليل} (الجداءات الشهيرة ، القسمة الإقليدية) ثم الاختزال على $(x-x_0)$.
\item \textbf{الضرب في المرافق} إذا وُجد جذر.
\item \textbf{العدد المشتق:} $\Lm{x_0}\dfrac{f(x)-f(x_0)}{x-x_0}=f'(x_0)$ \quad (مثلًا: $\Lm{0}\dfrac{\sin x}{x}=\sin'(0)=\cos0=1$).
\end{itemize}
\end{tariqa}

\begin{mithal}
\textbf{1.} $\Lm{2}\dfrac{x^3-8}{x^2-4}$ (حالة $\frac00$). نكتب:
\[\frac{x^3-8}{x^2-4}=\frac{(x-2)\left(x^2+2x+4\right)}{(x-2)(x+2)}=\frac{x^2+2x+4}{x+2}\]
\LD{\dfrac{x^3-8}{x^2-4}}{2}{3} $\Car{\Lm{2}\left(x^2+2x+4\right)=12\\ \Lm{2}(x+2)=4}$
\E
\textbf{2.} $\Lm{1}\dfrac{\sqrt{x+3}-2}{x-1}$ (حالة $\frac00$). نكتب:
\[\frac{\sqrt{x+3}-2}{x-1}=\frac{(x+3)-4}{(x-1)\left(\sqrt{x+3}+2\right)}=\frac1{\sqrt{x+3}+2}\]
\LD{\dfrac{\sqrt{x+3}-2}{x-1}}{1}{\frac14} $\Car{\Lm{1}1=1\\ \Lm{1}\left(\sqrt{x+3}+2\right)=4}$
\end{mithal}

\section{المستقيم المقارب المائل والوضع النسبي}
\begin{tarif}
$\LW{(\Delta)}$ المستقيم ذو المعادلة $\LW{y=ax+b}$ ($\LW{a\neq0}$). نقول إن $\LW{(\Delta)}$ \textbf{مستقيم مقارب مائل} للمنحنى $\LW{(C_f)}$ بجوار $\LW{+\infty}$ (على الترتيب $\LW{-\infty}$) إذا كان:
\[\LW{\Lm{+\infty}\big[f(x)-(ax+b)\big]=0}\]
(على الترتيب: \ $\LW{\Lm{-\infty}\big[f(x)-(ax+b)\big]=0}$).
\end{tarif}

\begin{khassia}[حالة شائعة]
إذا أمكن كتابة $\LW{f(x)=ax+b+g(x)}$ حيث $\LW{\Lm{\pm\infty}g(x)=0}$ ، فإن المستقيم $\LW{y=ax+b}$ مقارب مائل لـ $\LW{(C_f)}$ بجوار $\LW{\pm\infty}$.\E
\textbf{مثال:} $\LW{f(x)=ax+b+\dfrac{c}{x-d}}$ \quad و \quad $\LW{\Lm{\pm\infty}\dfrac{c}{x-d}=0}$ ، إذن $\LW{y=ax+b}$ مقارب مائل.
\end{khassia}

\begin{tariqa}[الوضع النسبي لـ $(C_f)$ والمستقيم المقارب المائل]
ندرس إشارة الفرق $d(x)=f(x)-(ax+b)$ ونلخصها في \textbf{جدول الوضع النسبي}:
\begin{itemize}
\item $d(x)>0$: $(C_f)$ \textbf{فوق} $(\Delta)$ \quad؛\quad $d(x)<0$: $(C_f)$ \textbf{تحت} $(\Delta)$ \quad؛\quad $d(x)=0$: $(C_f)$ \textbf{يقطع} $(\Delta)$.
\end{itemize}
\end{tariqa}

\begin{tariqa}[كتابة $f(x)=ax+b+\frac{c}{x-d}$]
\textbf{بالمطابقة:} نوحد المقامات في $ax+b+\dfrac{c}{x-d}$ ، ثم نطابق معاملات البسط مع بسط $f$. \textbf{أو بالقسمة الإقليدية} لبسط $f$ على $(x-d)$.
\end{tariqa}

\begin{tmr}
$f$ الدالة المعرفة على $\R-\{1\}$ بـ: $f(x)=x+1+\dfrac5{1-x}$.
\begin{enumerate}
\item احسب نهايات $f$ عند حدود مجموعة التعريف وفسّر النتائج بيانيًا.
\item أثبت أن المستقيم $(\Delta):\ y=x+1$ مقارب مائل لـ $(C_f)$ بجوار $+\infty$ و $-\infty$.
\item ادرس وضعية $(C_f)$ بالنسبة إلى $(\Delta)$.
\end{enumerate}
\end{tmr}

\begin{hal}
\textbf{1.} \LDx{f(x)}{-\infty}{-\infty} $\Car{\Lm{-\infty}(x+1)=-\infty\\ \Lm{-\infty}\dfrac5{1-x}=0}$
\LD{f(x)}{+\infty}{+\infty} $\Car{\Lm{+\infty}(x+1)=+\infty\\ \Lm{+\infty}\dfrac5{1-x}=0}$
\E عند $1$: إذا كان $x<1$ فإن $1-x\to0^+$ ، وإذا كان $x>1$ فإن $1-x\to0^-$:
\LD{f(x)}{1^-}{+\infty} $\Car{\Lm{1}(x+1)=2\\ \Lm{1^-}\dfrac5{1-x}=+\infty}$
\LD{f(x)}{1^+}{-\infty} $\Car{\Lm{1}(x+1)=2\\ \Lm{1^+}\dfrac5{1-x}=-\infty}$
\E\textbf{التفسير:} المستقيم ذو المعادلة \bx{x=1} مقارب للمنحنى $(C_f)$ يوازي محور التراتيب.

\E إذن \bxt{المستقيم $(\Delta):\ y=x+1$ مقارب مائل بجوار $\pm\infty$}.

\section{مخطط دراسة دالة وخلاصة الطرائق}
\begin{tariqa}[مخطط دراسة دالة]
\begin{enumerate}
\item \textbf{مجموعة التعريف} ، ودراسة الشفعية أو مراكز ومحاور التناظر إن أمكن (لتقليص مجال الدراسة).
\item \textbf{النهايات} عند حدود مجموعة التعريف وتفسيرها (المستقيمات المقاربة).
\item \textbf{اتجاه التغير:} حساب $f'(x)$ ودراسة إشارتها ، والقيم الحدية.
\item \textbf{جدول التغيرات} (نضع فيه النهايات والقيم الحدية).
\item \textbf{المستقيمات المقاربة} المائلة والوضع النسبي (جدول الوضع النسبي).
\item \textbf{التمثيل البياني:} المستقيمات المقاربة ، نقط التقاطع مع المحورين ، المماسات المطلوبة ، ثم المنحنى.
\end{enumerate}
\end{tariqa}
