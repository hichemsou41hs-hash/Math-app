\section{النهاية عند عدد حقيقي والمستقيمات المقاربة العمودية}
\begin{tamrin}{النهاية من اليمين ومن اليسار}
احسب النهايات التالية ، ثم فسّر كل نتيجة بيانيًا:
\begin{enumerate}
\item $f(x)=\dfrac3{x-1}$: \quad $\Lm{1^-}f(x)$ و $\Lm{1^+}f(x)$.
\item $g(x)=\dfrac{2x+1}{x+2}$: \quad $\Lm{-2^-}g(x)$ و $\Lm{-2^+}g(x)$.
\item $h(x)=\dfrac{-x+3}{(x-2)^2}$: \quad $\Lm{2}h(x)$.
\item $k(x)=\dfrac{x^2+1}{x^2-4}$: \quad النهايات عند $-2$ وعند $2$ من اليمين ومن اليسار.
\end{enumerate}
\end{tamrin}

\begin{hal}
\textbf{1.} إذا كان $x<1$ فإن $x-1\to0^-$ ، وإذا كان $x>1$ فإن $x-1\to0^+$:
\LD{f(x)}{1^-}{-\infty} $\Car{\Lm{1}3=3\\ \Lm{1^-}(x-1)=0^-}$
\LD{f(x)}{1^+}{+\infty} $\Car{\Lm{1}3=3\\ \Lm{1^+}(x-1)=0^+}$
\E\textbf{التفسير:} المستقيم ذو المعادلة \bx{x=1} مقارب لـ $(C_f)$ يوازي محور التراتيب.
\E\textbf{2.} إذا كان $x<-2$ فإن $x+2\to0^-$ ، وإذا كان $x>-2$ فإن $x+2\to0^+$:
\LD{g(x)}{-2^-}{+\infty} $\Car{\Lm{-2}(2x+1)=-3\\ \Lm{-2^-}(x+2)=0^-}$
\LD{g(x)}{-2^+}{-\infty} $\Car{\Lm{-2}(2x+1)=-3\\ \Lm{-2^+}(x+2)=0^+}$
\E\textbf{التفسير:} المستقيم ذو المعادلة \bx{x=-2} مقارب عمودي لـ $(C_g)$.
\E\textbf{3.} المقام $(x-2)^2$ مربع فهو موجب دائمًا:
\LD{h(x)}{2}{+\infty} $\Car{\Lm{2}(-x+3)=1\\ \Lm{2}(x-2)^2=0^+}$
\E\textbf{التفسير:} المستقيم ذو المعادلة \bx{x=2} مقارب عمودي لـ $(C_h)$.
\E\textbf{4.} $x^2-4=(x-2)(x+2)$ ، وإشارة المقام:
\begin{TAB}
\tkzTabInit[lgt=2,espcl=2.4]{$x$/1,$x^2-4$/1}{$-\infty$,$-2$,$2$,$+\infty$}
\tkzTabLine{,+,z,-,z,+,}
\end{TAB}
\LD{k(x)}{-2^-}{+\infty} $\Car{\Lm{-2}\left(x^2+1\right)=5\\ \Lm{-2^-}\left(x^2-4\right)=0^+}$
\LD{k(x)}{-2^+}{-\infty} $\Car{\Lm{-2}\left(x^2+1\right)=5\\ \Lm{-2^+}\left(x^2-4\right)=0^-}$
\LD{k(x)}{2^-}{-\infty} $\Car{\Lm{2}\left(x^2+1\right)=5\\ \Lm{2^-}\left(x^2-4\right)=0^-}$
\LD{k(x)}{2^+}{+\infty} $\Car{\Lm{2}\left(x^2+1\right)=5\\ \Lm{2^+}\left(x^2-4\right)=0^+}$
\E\textbf{التفسير:} المستقيمان ذوا المعادلتين \bx{x=-2} و \bx{x=2} مقاربان لـ $(C_k)$ يوازيان محور التراتيب.
\end{hal}

\section{حساب النهايات وإزالة حالات عدم التعيين}
\begin{tamrin}{النهايات عند $\pm\infty$}
احسب النهايات التالية:\par
\RL{0.48}{\textbf{1.} $\Lm{-\infty}\left(3x^2+x+2\right)$\par \textbf{2.} $\Lm{+\infty}\left(-x^3+2x-2\right)$\par \textbf{3.} $\Lm{+\infty}\dfrac{1+x-x^2}{x^3-1}$\par \textbf{4.} $\Lm{-\infty}\dfrac{x^3+x-2}{x-1}$}{0.48}{\textbf{5.} $\Lm{+\infty}\dfrac{2x+1}{x+1}$\par \textbf{6.} $\Lm{+\infty}\dfrac{x^2+3}{x^3-2x^2+3x}$\par \textbf{7.} $\Lm{+\infty}\dfrac{-4x^2+5}{2x^2+x}$\par \textbf{8.} $\Lm{-\infty}\left(x+\dfrac1{x+1}\right)$}
\end{tamrin}

\begin{hal}
\textbf{1.} كثير حدود: نهاية الحد الأعلى درجة: \quad $\Lm{-\infty}\left(3x^2+x+2\right)=\Lm{-\infty}3x^2=\bx{+\infty}$
\E\textbf{2.} حالة عدم تعيين من الشكل $-\infty+\infty$ ، نأخذ الحد الأعلى درجة:\E $\Lm{+\infty}\left(-x^3+2x-2\right)=\Lm{+\infty}\left(-x^3\right)=\bx{-\infty}$
\E\textbf{3.} دالة ناطقة: \quad $\Lm{+\infty}\dfrac{1+x-x^2}{x^3-1}=\Lm{+\infty}\dfrac{-x^2}{x^3}=\Lm{+\infty}\dfrac{-1}{x}=\bx{0}$
\E\textbf{4.} $\Lm{-\infty}\dfrac{x^3+x-2}{x-1}=\Lm{-\infty}\dfrac{x^3}{x}=\Lm{-\infty}x^2=\bx{+\infty}$
\E\textbf{5.} $\Lm{+\infty}\dfrac{2x+1}{x+1}=\Lm{+\infty}\dfrac{2x}{x}=\bx{2}$
\E\textbf{6.} $\Lm{+\infty}\dfrac{x^2+3}{x^3-2x^2+3x}=\Lm{+\infty}\dfrac{x^2}{x^3}=\Lm{+\infty}\dfrac1x=\bx{0}$
\E\textbf{7.} $\Lm{+\infty}\dfrac{-4x^2+5}{2x^2+x}=\Lm{+\infty}\dfrac{-4x^2}{2x^2}=\bx{-2}$
\E\textbf{8.} \LDx{\left(x+\dfrac1{x+1}\right)}{-\infty}{-\infty} $\Car{\Lm{-\infty}x=-\infty\\ \Lm{-\infty}\dfrac1{x+1}=0}$
\end{hal}

\begin{tamrin}{حالات عدم التعيين مع الجذور}
احسب النهايات التالية:\par
\RL{0.48}{\textbf{1.} $\Lm{-\infty}\dfrac{\sqrt{2x^2-3}}{x+2}$\par \textbf{2.} $\Lm{-\infty}\dfrac{\sqrt{3x^2+7}}{2-x}$\par \textbf{3.} $\Lm{-\infty}\dfrac{x}{\sqrt{x^2-1}}$\par \textbf{4.} $\Lm{+\infty}\left(\sqrt{x^2+1}-x\right)$}{0.48}{\textbf{5.} $\Lm{+\infty}\left(\sqrt{x^2+x+1}-x\right)$\par \textbf{6.} $\Lm{-\infty}\left(\sqrt{4x^2-3}+2x-2\right)$\par \textbf{7.} $\Lm{+\infty}\left(\sqrt{2x-3}-3x+2\right)$\par \textbf{8.} $\Lm{+\infty}\left(\sqrt{9x^2+x}-3x\right)$}
\end{tamrin}

\begin{hal}
\textbf{تذكير:} $\LW{\sqrt{x^2}=|x|}$ ، أي $\LW{\sqrt{x^2}=-x}$ عندما $x\to-\infty$.
\E\textbf{1.} حالة $\frac\infty\infty$. من أجل $x<0$ نكتب:
\[\frac{\sqrt{2x^2-3}}{x+2}=\frac{-x\sqrt{2-\frac3{x^2}}}{x\left(1+\frac2x\right)}=\frac{-\sqrt{2-\frac3{x^2}}}{1+\frac2x}\]
\LD{\dfrac{\sqrt{2x^2-3}}{x+2}}{-\infty}{-\sqrt2} $\Car{\Lm{-\infty}\left(-\sqrt{2-\frac3{x^2}}\right)=-\sqrt2\\ \Lm{-\infty}\left(1+\frac2x\right)=1}$
\E\textbf{2.} حالة $\frac\infty\infty$. من أجل $x<0$ نكتب:
\[\frac{\sqrt{3x^2+7}}{2-x}=\frac{-x\sqrt{3+\frac7{x^2}}}{-x\left(1-\frac2x\right)}=\frac{\sqrt{3+\frac7{x^2}}}{1-\frac2x}\]
\LD{\dfrac{\sqrt{3x^2+7}}{2-x}}{-\infty}{\sqrt3} $\Car{\Lm{-\infty}\sqrt{3+\frac7{x^2}}=\sqrt3\\ \Lm{-\infty}\left(1-\frac2x\right)=1}$
\E\textbf{3.} حالة $\frac\infty\infty$. من أجل $x<-1$ نكتب:
\[\frac{x}{\sqrt{x^2-1}}=\frac{x}{-x\sqrt{1-\frac1{x^2}}}=\frac{-1}{\sqrt{1-\frac1{x^2}}}\]
\LD{\dfrac{x}{\sqrt{x^2-1}}}{-\infty}{-1} $\Car{\Lm{-\infty}(-1)=-1\\ \Lm{-\infty}\sqrt{1-\frac1{x^2}}=1}$
\E\textbf{4.} حالة $+\infty-\infty$. نضرب في المرافق ونكتب:
\[\sqrt{x^2+1}-x=\frac{\left(x^2+1\right)-x^2}{\sqrt{x^2+1}+x}=\frac1{\sqrt{x^2+1}+x}\]
\LD{\left(\sqrt{x^2+1}-x\right)}{+\infty}{0} $\Car{\Lm{+\infty}1=1\\ \Lm{+\infty}\left(\sqrt{x^2+1}+x\right)=+\infty}$
\E\textbf{5.} حالة $+\infty-\infty$. بالمرافق ، ومن أجل $x>0$ نكتب:
\[\sqrt{x^2+x+1}-x=\frac{x^2+x+1-x^2}{\sqrt{x^2+x+1}+x}=\frac{x\left(1+\frac1x\right)}{x\left(\sqrt{1+\frac1x+\frac1{x^2}}+1\right)}=\frac{1+\frac1x}{\sqrt{1+\frac1x+\frac1{x^2}}+1}\]
\LD{\left(\sqrt{x^2+x+1}-x\right)}{+\infty}{\frac12} $\Car{\Lm{+\infty}\left(1+\frac1x\right)=1\\ \Lm{+\infty}\left(\sqrt{1+\frac1x+\frac1{x^2}}+1\right)=2}$
\E\textbf{6.} حالة $+\infty-\infty$ (لأن $\Lm{-\infty}(2x-2)=-\infty$). بالمرافق ، ومن أجل $x<0$ نكتب:
\[\sqrt{4x^2-3}+(2x-2)=\frac{\left(4x^2-3\right)-(2x-2)^2}{\sqrt{4x^2-3}-(2x-2)}=\frac{8x-7}{\sqrt{4x^2-3}-2x+2}\]
\[\sqrt{4x^2-3}+(2x-2)=\frac{x\left(8-\frac7x\right)}{-x\left(\sqrt{4-\frac3{x^2}}+2-\frac2x\right)}=\frac{8-\frac7x}{-\left(\sqrt{4-\frac3{x^2}}+2-\frac2x\right)}\]
\LD{\left(\sqrt{4x^2-3}+2x-2\right)}{-\infty}{-2} $\Car{\Lm{-\infty}\left(8-\frac7x\right)=8\\ \Lm{-\infty}\left(-\sqrt{4-\frac3{x^2}}-2+\frac2x\right)=-4}$
\E\textbf{7.} بالعامل المشترك ، ومن أجل $x>0$ نكتب:
\[\sqrt{2x-3}-3x+2=x\left(\sqrt{\frac2x-\frac3{x^2}}-3+\frac2x\right)\]
\LD{\left(\sqrt{2x-3}-3x+2\right)}{+\infty}{-\infty} $\Car{\Lm{+\infty}x=+\infty\\ \Lm{+\infty}\left(\sqrt{\frac2x-\frac3{x^2}}-3+\frac2x\right)=-3}$
\E\textbf{8.} حالة $+\infty-\infty$. بالمرافق ، ومن أجل $x>0$ نكتب:

\section{الاختيار من متعدد وصحيح أم خطأ}
\begin{tamrin}{اختيار من متعدد}
عيّن الاقتراح الصحيح الوحيد في كل حالة مع التعليل:
\begin{enumerate}
\item $\Lm{+\infty}\dfrac{3x^2-x}{1-2x^2}$ تساوي: \ABC{$3$}{$-\dfrac32$}{$+\infty$}
\item $\Lm{2^-}\dfrac{x}{x-2}$ تساوي: \ABC{$+\infty$}{$-\infty$}{$0$}
\item $\Lm{+\infty}\left(\sqrt{x+1}-\sqrt x\right)$ تساوي: \ABC{$0$}{$1$}{$+\infty$}
\item المنحنى الممثل للدالة $x\mapsto\dfrac{3x-2}{x+4}$ يقبل مستقيمًا مقاربًا أفقيًا معادلته: \ABC{$y=-\dfrac12$}{$y=3$}{$x=-4$}
\item المستقيم المقارب المائل للمنحنى الممثل للدالة $x\mapsto\dfrac{x^2+3x+6}{x+2}$ هو: \ABC{$y=x+1$}{$y=x+3$}{$y=x$}
\item $\Lm{3}\dfrac{x^2-9}{x-3}$ تساوي: \ABC{$0$}{$6$}{\ar{غير موجودة}}
\end{enumerate}
\end{tamrin}

\begin{hal}
\textbf{1. (ب):} \quad $\Lm{+\infty}\dfrac{3x^2-x}{1-2x^2}=\Lm{+\infty}\dfrac{3x^2}{-2x^2}=\bx{-\frac32}$
\E\textbf{2. (ب):} \LDx{\dfrac{x}{x-2}}{2^-}{-\infty} $\Car{\Lm{2}x=2\\ \Lm{2^-}(x-2)=0^-}$
\E\textbf{3. (أ):} بالمرافق نكتب: \quad $\sqrt{x+1}-\sqrt x=\dfrac{(x+1)-x}{\sqrt{x+1}+\sqrt x}=\dfrac1{\sqrt{x+1}+\sqrt x}$
\LD{\left(\sqrt{x+1}-\sqrt x\right)}{+\infty}{0} $\Car{\Lm{+\infty}1=1\\ \Lm{+\infty}\left(\sqrt{x+1}+\sqrt x\right)=+\infty}$
\E\textbf{4. (ب):} \quad $\Lm{\pm\infty}\dfrac{3x-2}{x+4}=\Lm{\pm\infty}\dfrac{3x}{x}=3$ \quad إذن \quad \bx{y=3} \quad ($x=-4$ مقارب عمودي وليس أفقيًا).
\E\textbf{5. (أ):} \quad $x^2+3x+6=(x+2)(x+1)+4$ \quad إذن \quad $f(x)=x+1+\dfrac4{x+2}$ \quad ومنه \quad \bx{y=x+1}
\E\textbf{6. (ب):} \quad $\dfrac{x^2-9}{x-3}=\dfrac{(x-3)(x+3)}{x-3}=x+3$ \quad إذن \quad $\Lm{3}\dfrac{x^2-9}{x-3}=\Lm{3}(x+3)=\bx{6}$
\end{hal}

\begin{tamrin}{صحيح أم خطأ}
أجب بصحيح أو خطأ مع التبرير:
\begin{enumerate}
\item إذا كان $\Lm{+\infty}f(x)=+\infty$ و $\Lm{+\infty}g(x)=-\infty$ فإن $\Lm{+\infty}\big(f(x)+g(x)\big)=0$.
\item $\Lm{+\infty}\dfrac{2x^3+1}{x^2-5}=+\infty$.
\item $\Lm{-\infty}\sqrt{x^2+1}=-\infty$.
\item إذا كانت $f(x)=-x+1-\dfrac1{x-1}$ فإن المستقيم $y=-x+1$ مقارب مائل لـ $(C_f)$.
\item $\Lm{0}\dfrac1x=+\infty$.
\item الدالة $x\mapsto\dfrac{x^2}{x^2+1}$ يقبل منحناها مستقيمًا مقاربًا عموديًا.
\end{enumerate}
\end{tamrin}

\begin{hal}
\textbf{1. \bxt{خطأ}:} $+\infty-\infty$ حالة عدم تعيين. مثال مضاد: $f(x)=2x$ و $g(x)=-x$:\E $\Lm{+\infty}\big(f(x)+g(x)\big)=\Lm{+\infty}x=+\infty$.
\E\textbf{2. \bxt{صحيح}:} \quad $\Lm{+\infty}\dfrac{2x^3+1}{x^2-5}=\Lm{+\infty}\dfrac{2x^3}{x^2}=\Lm{+\infty}2x=+\infty$
\E\textbf{3. \bxt{خطأ}:} الجذر التربيعي موجب دائمًا: \quad $\Lm{-\infty}\sqrt{x^2+1}=+\infty$.
\E\textbf{4. \bxt{صحيح}:} \quad $f(x)-(-x+1)=-\dfrac1{x-1}$ \quad و \quad $\Lm{\pm\infty}\left(-\dfrac1{x-1}\right)=0$.
\E\textbf{5. \bxt{خطأ}:} \quad $\Lm{0^+}\dfrac1x=+\infty$ \quad لكن \quad $\Lm{0^-}\dfrac1x=-\infty$ ، فالنهاية عند $0$ غير موجودة (النهايتان من الجهتين مختلفتان).
\E\textbf{6. \bxt{خطأ}:} المقام $x^2+1\geq1$ لا ينعدم أبدًا ، فالدالة معرفة على $\R$ ولا يوجد مقارب عمودي (لها مقارب أفقي $y=1$).
\end{hal}
