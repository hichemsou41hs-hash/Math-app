\begin{tamrin}{النهاية من اليمين ومن اليسار}

\begin{enumerate}
\item $f(x)=\dfrac3{x-1}$: \quad $\Lm{1^-}f(x)$ و $\Lm{1^+}f(x)$.
\item $g(x)=\dfrac{2x+1}{x+2}$: \quad $\Lm{-2^-}g(x)$ و $\Lm{-2^+}g(x)$.
\item $h(x)=\dfrac{-x+3}{(x-2)^2}$: \quad $\Lm{2}h(x)$.
\item $k(x)=\dfrac{x^2+1}{x^2-4}$: \quad النهايات عند $-2$ وعند $2$ من اليمين ومن اليسار.
\end{enumerate}
\end{tamrin}

\section{القراءة البيانية والمستقيمات المقاربة}
\begin{tamrin}{قراءة بيانية}
في الشكل التالي $(C_f)$ التمثيل البياني لدالة عددية $f$ في معلم متعامد ومتجانس. المستقيمان $x=-1$ و $(D):\ y=x+1$ مقاربان للمنحنى ، و $f(x)=x+1+\dfrac1{x+1}$.
\begin{GR}{-5.2}{5.2}{-6.2}{7.2}{0.42}
\Ticks{-4,-2,2,4}{-4,-2,2,4,6}
\draw[cThm,dashed] (-1,-6.2)--(-1,7.2);
\draw[cRem,thick,domain=-5.2:5.2] plot(\x,{\x+1}); \node[cRem,font=\scriptsize] at (4.3,3.6){$(D)$};
\draw[cDef,very thick,domain=-5.2:-1.18,samples=80] plot(\x,{\x+1+1/(\x+1)});
\draw[cDef,very thick,domain=-0.83:5.2,samples=80] plot(\x,{\x+1+1/(\x+1)});
\fill[cDef] (0,2) circle (2pt) (-2,-2) circle (2pt);
\node[cDef,font=\scriptsize] at (2.5,6){$(C_f)$};
\end{GR}
\textbf{بقراءة بيانية:}
\begin{enumerate}
\item عيّن مجموعة تعريف الدالة $f$.
\item عيّن النهايات: $\Lm{-\infty}f(x)$ ، $\Lm{+\infty}f(x)$ ، $\Lm{-1^-}f(x)$ ، $\Lm{-1^+}f(x)$.
\item اكتب معادلة لكل مستقيم مقارب لـ $(C_f)$.
\item عيّن الوضع النسبي بين $(C_f)$ والمستقيم المقارب المائل له.
\item شكّل جدول تغيرات الدالة $f$.
\item ناقش حسب قيم الوسيط الحقيقي $m$ عدد حلول المعادلة $f(x)=m$.
\end{enumerate}
\textbf{ثم أكّد حسابيًا النتائج 2 ، 3 و 4.}
\end{tamrin}

\begin{tamrin}{تعيين وسطاء من المستقيمات المقاربة}
\begin{enumerate}
\item $g$ الدالة المعرفة على $\R$ بـ: $g(x)=\dfrac{\alpha x^2+\beta x+1}{x^2+1}$ حيث $\alpha$ و $\beta$ عددان حقيقيان. أوجد $\alpha$ و $\beta$ إذا علمت أن $(C_g)$ يشمل النقطة $A(1;2)$ ويقبل مستقيمًا مقاربًا معادلته $y=1$ عند $-\infty$.
\item $f$ الدالة المعرفة على $\R-\{1\}$ بـ: $f(x)=ax+b+\dfrac{c}{x-1}$. أوجد $a$ ، $b$ و $c$ حتى يشمل $(C_f)$ النقطة $B(0;4)$ ، ويقبل مستقيمًا مقاربًا مائلًا معادلته $y=2x+3$ عند $+\infty$.
\end{enumerate}
\end{tamrin}

\section{الاختيار من متعدد وصحيح أم خطأ}
\begin{tamrin}{اختيار من متعدد}
عيّن الاقتراح الصحيح الوحيد في كل حالة مع التعليل:
\begin{enumerate}
\item $\Lm{+\infty}\dfrac{3x^2-x}{1-2x^2}$ تساوي: \ABC{$3$}{$-\dfrac32$}{$+\infty$}
\item $\Lm{2^-}\dfrac{x}{x-2}$ تساوي: \ABC{$+\infty$}{$-\infty$}{$0$}
\item $\Lm{+\infty}\left(\sqrt{x+1}-\sqrt x\right)$ تساوي: \ABC{$0$}{$1$}{$+\infty$}
\item المنحنى الممثل للدالة $x\mapsto\dfrac{3x-2}{x+4}$ يقبل مستقيمًا مقاربًا أفقيًا معادلته: \ABC{$y=-\dfrac12$}{$y=3$}{$x=-4$}
\item المستقيم المقارب المائل للمنحنى الممثل للدالة $x\mapsto\dfrac{x^2+3x+6}{x+2}$ هو: \ABC{$y=x+1$}{$y=x+3$}{$y=x$}
\item $\Lm{3}\dfrac{x^2-9}{x-3}$ تساوي: \ABC{$0$}{$6$}{\ar{غير موجودة}}
\end{enumerate}
\end{tamrin}

\begin{tamrin}{صحيح أم خطأ}
أجب بصحيح أو خطأ مع التبرير:
\begin{enumerate}
\item إذا كان $\Lm{+\infty}f(x)=+\infty$ و $\Lm{+\infty}g(x)=-\infty$ فإن $\Lm{+\infty}\big(f(x)+g(x)\big)=0$.
\item $\Lm{+\infty}\dfrac{2x^3+1}{x^2-5}=+\infty$.
\item $\Lm{-\infty}\sqrt{x^2+1}=-\infty$.
\item إذا كانت $f(x)=-x+1-\dfrac1{x-1}$ فإن المستقيم $y=-x+1$ مقارب مائل لـ $(C_f)$.
\item $\Lm{0}\dfrac1x=+\infty$.
\item الدالة $x\mapsto\dfrac{x^2}{x^2+1}$ يقبل منحناها مستقيمًا مقاربًا عموديًا.
\end{enumerate}
\end{tamrin}

\section{مسائل: دراسة دوال ناطقة}
\begin{tamrin}{مسألة 1: مقارب مائل ومركز تناظر}
نعتبر الدالة $f$ المعرفة على $\R-\{2\}$ بالعبارة: $f(x)=\dfrac{x^2-3x+3}{x-2}$ ، و $(C_f)$ تمثيلها البياني في معلم متعامد ومتجانس $(O;\vec i,\vec j)$.
\begin{enumerate}
\item ادرس نهايات الدالة $f$ عند حدود مجموعة التعريف ، واستنتج أن $(C_f)$ يقبل مستقيمًا مقاربًا يطلب تعيين معادلته.
\item تحقق أنه من أجل كل $x$ من $\R-\{2\}$: $f'(x)=\dfrac{(x-1)(x-3)}{(x-2)^2}$.
\item استنتج اتجاه تغير الدالة $f$ ثم شكّل جدول تغيراتها.
\item أثبت أنه من أجل كل $x$ من $\R-\{2\}$: $f(x)=ax+b+\dfrac{c}{x-2}$ حيث $a$ ، $b$ و $c$ أعداد حقيقية يطلب تعيينها.
\item استنتج أن $(C_f)$ يقبل مستقيمًا مقاربًا مائلًا $(\Delta)$ يطلب تعيين معادلته ، وادرس الوضع النسبي لـ $(C_f)$ و $(\Delta)$.
\item بيّن أن النقطة $\Omega(2;1)$ مركز تناظر للمنحنى $(C_f)$.
\item اكتب معادلة المماس $(T)$ عند النقطة ذات الفاصلة $0$ ، وعيّن نقطة تقاطعه مع $(\Delta)$.
\item احسب $f(0)$ ثم أنشئ $(\Delta)$ ، $(T)$ و $(C_f)$.
\end{enumerate}
\end{tamrin}

\begin{tamrin}{مسألة 2: مناقشة بيانية}
لتكن الدالة $f$ المعرفة على $\R-\{1\}$ كما يلي: $f(x)=\dfrac{-x^2+2x-2}{x-1}$ ، و $(C_f)$ تمثيلها البياني في معلم متعامد ومتجانس.
\begin{enumerate}
\item بيّن أنه من أجل كل $x$ من $\R-\{1\}$: $f(x)=-x+1-\dfrac1{x-1}$.
\item احسب نهايات الدالة $f$ عند أطراف مجالي تعريفها.
\item بيّن أن المستقيم $(\Delta)$ ذا المعادلة $y=-x+1$ مستقيم مقارب مائل لـ $(C_f)$.
\item ادرس وضعية المنحنى $(C_f)$ بالنسبة للمستقيم $(\Delta)$.
\item بيّن أنه من أجل كل $x$ من $\R-\{1\}$: $f'(x)=\dfrac{-x^2+2x}{(x-1)^2}$.
\item ادرس اتجاه تغير الدالة $f$ ثم شكّل جدول تغيراتها.
\item ارسم $(\Delta)$ و $(C_f)$.
\item عيّن بيانيًا قيم الوسيط الحقيقي $m$ بحيث تقبل المعادلة $f(x)=m$ حلين مختلفين في الإشارة.
\end{enumerate}
\end{tamrin}

\begin{tamrin}{مسألة 3: مقاربان ومركز تناظر}
لتكن الدالة العددية $f$ المعرفة على $\R-\{-2\}$ بـ: $f(x)=\dfrac{x^2+3x+6}{x+2}$ ، و $(C_f)$ تمثيلها البياني.
\begin{enumerate}
\item عيّن الأعداد الحقيقية $a$ ، $b$ و $c$ بحيث يكون من أجل كل $x$ من $\R-\{-2\}$: $f(x)=ax+b+\dfrac{c}{x+2}$.
\item احسب نهايات الدالة $f$ عند أطراف مجموعة تعريفها.
\item بيّن أنه من أجل كل $x$ من $\R-\{-2\}$: $f'(x)=\dfrac{x(x+4)}{(x+2)^2}$ ، ثم استنتج اتجاه تغير $f$ وشكّل جدول تغيراتها.
\item \pa بيّن أن $(C_f)$ يقبل مستقيمين مقاربين أحدهما $(\Delta)$ ذو المعادلة $y=x+1$.
\pb ادرس وضعية المنحنى $(C_f)$ بالنسبة إلى المستقيم $(\Delta)$.
\item تحقق أن النقطة $A(-2;-1)$ هي نقطة تقاطع المستقيمين المقاربين ، ثم بيّن أنها مركز تناظر لـ $(C_f)$.
\item ارسم المستقيمات المقاربة والمنحنى $(C_f)$.
\end{enumerate}
\end{tamrin}

\begin{tamrin}{مسألة 4: مماسان وقيمة مطلقة}
لتكن $f$ الدالة المعرفة على $\R-\{-1\}$ بـ: $f(x)=\dfrac{2x^2+x+7}{x+1}$ ، و $(C_f)$ تمثيلها البياني في معلم متعامد ومتجانس.
\begin{enumerate}
\item عيّن الأعداد الحقيقية $a$ ، $b$ و $c$ حيث: $f(x)=ax+b+\dfrac{c}{x+1}$.
\item احسب نهايات الدالة $f$ عند أطراف مجموعة التعريف وفسّر النتائج بيانيًا.
\item بيّن أن المستقيم $(\Delta)$ ذا المعادلة $y=2x-1$ مقارب مائل لـ $(C_f)$ ، وادرس الوضع النسبي بينهما.
\item ادرس اتجاه تغير $f$ وشكّل جدول تغيراتها.
\item بيّن أن النقطة $A(-1;-3)$ مركز تناظر لـ $(C_f)$.
\item اكتب معادلتي المماسين $(T)$ و $(T')$ للمنحنى $(C_f)$ عند النقطتين ذواتي الفاصلتين $-3$ و $1$.
\item لتكن الدالة $g$ المعرفة على $\R-\{-1\}$ بـ: $g(x)=\dfrac{2x^2+x+7}{|x+1|}$.
\pa اكتب عبارة $g(x)$ دون رمز القيمة المطلقة.
\pb بيّن كيف يمكن إنشاء $(C_g)$ انطلاقًا من $(C_f)$.
\end{enumerate}
\end{tamrin}

\begin{tamrin}{مسألة 5: دالة مساعدة من الدرجة الثالثة}
\textbf{I)} لتكن $g$ الدالة المعرفة على $\R$ بـ: $g(x)=x^3-6x^2+13x-8$.
\begin{enumerate}
\item بيّن أن $g$ متزايدة تمامًا على $\R$.
\item احسب $g(1)$ ، ثم استنتج إشارة $g(x)$.
\end{enumerate}
\textbf{II)} نعتبر الدالة $f$ المعرفة على $\R-\{2\}$ بـ: $f(x)=\dfrac{x^3-5x^2+7x-3}{(x-2)^2}$.
\begin{enumerate}[resume]
\item احسب نهايات $f$ وفسّر النتيجة هندسيًا.
\item بيّن أنه من أجل كل $x$ من $\R-\{2\}$: $f'(x)=\dfrac{g(x)}{(x-2)^3}$ ، ثم استنتج اتجاه تغير $f$ وشكّل جدول تغيراتها.
\item بيّن أنه من أجل كل $x$ من $\R-\{2\}$: $f(x)=x-1+\dfrac{1-x}{(x-2)^2}$.
\item بيّن أن المستقيم $(\Delta)$ ذا المعادلة $y=x-1$ مقارب مائل لـ $(C_f)$ ، وادرس وضعية $(C_f)$ بالنسبة لـ $(\Delta)$.
\end{enumerate}
\end{tamrin}

\begin{tamrin}{مسألة 6: لا يوجد مماس يوازي المقارب}
$f$ الدالة المعرفة على $\R-\{-1\}$ بـ: $f(x)=\dfrac{2x^3+3x^2}{(x+1)^2}$ ، و $(C_f)$ تمثيلها البياني.
\begin{enumerate}
\item احسب نهايات $f$ عند أطراف مجموعة التعريف ، وفسّر النتيجة بيانيًا عند $-1$.
\item بيّن أنه من أجل كل $x$ من $\R-\{-1\}$: $f'(x)=\dfrac{2x\left(x^2+3x+3\right)}{(x+1)^3}$ ، ثم ادرس تغيرات $f$.
\item \pa تحقق أن: $f(x)=2x-1+\dfrac1{(x+1)^2}$.
\pb بيّن أن $(\Delta):\ y=2x-1$ مقارب لـ $(C_f)$ عند $-\infty$ و $+\infty$ ، وادرس الوضع النسبي.
\item أثبت أنه لا يوجد أي مماس لـ $(C_f)$ يوازي المستقيم $(\Delta)$.
\item أوجد إحداثيات نقط تقاطع $(C_f)$ مع محور الفواصل.
\end{enumerate}
\end{tamrin}

\section{تمارين نموذجية}
\begin{tamrin}{مسألة نموذجية: مماسان متوازيان ومناقشة}
$f$ دالة معرفة على $\R-\{-2\}$ بـ: $f(x)=\dfrac{x^2+4x+8}{2x+4}$ ، و $(C_f)$ تمثيلها البياني في معلم متعامد ومتجانس.
\begin{enumerate}
\item احسب النهايات عند حدود مجالي تعريف الدالة $f$.
\item عيّن الأعداد الحقيقية $a$ ، $b$ و $c$ بحيث من أجل كل $x$ من $\R-\{-2\}$: $f(x)=ax+b+\dfrac{c}{x+2}$.
\item بيّن أن $(C_f)$ يقبل مستقيمين مقاربين أحدهما مائل $(\Delta)$ معادلته $y=\frac12x+1$ ، وادرس الوضع النسبي.
\item بيّن أنه من أجل كل $x$ من $\R-\{-2\}$: $f'(x)=\dfrac{x^2+4x}{2(x+2)^2}$ ، ثم ادرس تغيرات $f$.
\item بيّن أن $(C_f)$ يقبل مماسين $(T_1)$ و $(T_2)$ ميل كل منهما يساوي $-\frac32$ ، يطلب كتابة معادلة كل منهما.
\item عيّن نقطة تقاطع المستقيمين المقاربين ثم بيّن أنها مركز تناظر لـ $(C_f)$.
\item ناقش بيانيًا حسب قيم الوسيط $m$ عدد وإشارة حلول المعادلة: $x^2+x(4-2m)+8-4m=0$.
\end{enumerate}
\end{tamrin}

\begin{tamrin}{مسألة نموذجية من الواقع: الكلفة المتوسطة}
تنتج مؤسسة $x$ طنًا من مادة غذائية ($x>0$). الكلفة الإجمالية (بآلاف الدنانير) هي $C(x)=x^2+2x+100$. الكلفة المتوسطة للطن الواحد هي: $C_M(x)=\dfrac{C(x)}{x}$.
\begin{enumerate}
\item بيّن أن: $C_M(x)=x+2+\dfrac{100}x$.
\item احسب $\Lm{0^+}C_M(x)$ و $\Lm{+\infty}C_M(x)$ ، وفسّر النتيجتين اقتصاديًا وبيانيًا.
\item بيّن أن المستقيم $(\Delta):\ y=x+2$ مقارب مائل لمنحنى $C_M$ ، وحدد الوضع النسبي.
\item ادرس تغيرات $C_M$ على $]0;+\infty[$.
\item ما هي الكمية التي يجب إنتاجها حتى تكون الكلفة المتوسطة أصغر ما يمكن؟ ما قيمتها؟
\end{enumerate}
\end{tamrin}
