\section{النهايات}
% ------------------------------------------------------------------
\begin{tamrin}{الأشكال غير المعينة في الدوال كثيرات الحدود والناطقة}
احسب النهايات التالية، مع تحديد الشكل غير المعين إن وُجد:
\begin{sq}
\qa $\displaystyle\lim_{x\to+\infty}\left(-2x^{3}+x^{2}-5\right)$
\qb $\displaystyle\lim_{x\to-\infty}\frac{3x^{2}-x+1}{1-2x^{2}}$
\qc $\displaystyle\lim_{x\to+\infty}\frac{x^{3}+1}{x^{2}-4}$
\qd $\displaystyle\lim_{x\to-\infty}\frac{5x+2}{x^{2}+1}$
\qe $\displaystyle\lim_{x\to2}\frac{x^{2}-4}{x^{2}-3x+2}$
\item[(و)] $\displaystyle\lim_{x\to1}\frac{x^{3}-1}{x-1}$
\item[(ز)] $\displaystyle\lim_{\substack{x\to3\\ x<3}}\frac{x+1}{x-3}$ \ و \ $\displaystyle\lim_{\substack{x\to3\\ x>3}}\frac{x+1}{x-3}$
\item[(ح)] $\displaystyle\lim_{x\to0}\left(\frac{1}{x}-\frac{1}{x^{2}+x}\right)$
\end{sq}
\end{tamrin}

% ------------------------------------------------------------------
\begin{tamrin}{إزالة حالات عدم التعيين بالمرافق}
احسب النهايات التالية:
\begin{sq}
\qa $\displaystyle\lim_{x\to+\infty}\left(\sqrt{x^{2}+x}-x\right)$
\qb $\displaystyle\lim_{x\to-\infty}\left(\sqrt{x^{2}+x}+x\right)$
\qc $\displaystyle\lim_{x\to3}\frac{\sqrt{x+1}-2}{x-3}$
\qd $\displaystyle\lim_{x\to+\infty}\left(\sqrt{x+1}-\sqrt{x}\right)$
\qe $\displaystyle\lim_{x\to-\infty}\left(\sqrt{x^{2}+1}+2x\right)$
\item[(و)] $\displaystyle\lim_{x\to0}\frac{x}{\sqrt{x+4}-2}$
\end{sq}
\end{tamrin}

% ------------------------------------------------------------------
\begin{tamrin}{نهايات مثلثية، النهايات بالمقارنة ونهاية دالة مركبة}
\QQ{1} احسب: \quad (أ) $\displaystyle\lim_{x\to0}\frac{\sin 3x}{x}$ \qquad (ب) $\displaystyle\lim_{x\to0}\frac{\sin 2x}{\sin 5x}$ \qquad (جـ) $\displaystyle\lim_{x\to0}\frac{1-\cos x}{x^{2}}$ \qquad (د) $\displaystyle\lim_{x\to0}\frac{1-\cos x}{x}$
\QQ{2} باستعمال الحصر أو المقارنة احسب:
\begin{sq}
\qa $\displaystyle\lim_{x\to+\infty}\frac{2+\cos x}{x}$
\qb $\displaystyle\lim_{x\to+\infty}\frac{x+\sin x}{x+1}$
\qc $\displaystyle\lim_{x\to-\infty}\left(x^{2}+3\sin x\right)$
\end{sq}
\QQ{3} احسب: \quad (أ) $\displaystyle\lim_{x\to+\infty}\sqrt{\frac{4x^{2}+1}{x^{2}+3}}$ \qquad (ب) $\displaystyle\lim_{x\to+\infty}\sin\left(\frac{\pi x}{2x+1}\right)$
\end{tamrin}

\begin{tamrin}{المستقيمات المقاربة والوضع النسبي}
\QQ{1} $f$ الدالة المعرفة على $\R-\{1\}$ بـ: $f(x)=\dfrac{2x^{2}-3x+2}{x-1}$، و $(C_f)$ تمثيلها البياني في معلم متعامد ومتجانس.
\begin{sq}
\qa احسب نهايات $f$ عند أطراف مجموعة تعريفها، وفسّر النتائج هندسيا.
\qb عيّن الأعداد الحقيقية $a$ ، $b$ و $c$ بحيث: $f(x)=ax+b+\dfrac{c}{x-1}$.
\qc استنتج أن $(C_f)$ يقبل مستقيما مقاربا مائلا $(\Delta)$ يطلب تعيين معادلة له، ثم ادرس وضعية $(C_f)$ بالنسبة إلى $(\Delta)$.
\end{sq}
\QQ{2} $g$ الدالة المعرفة على $\R-\{-2\}$ بـ: $g(x)=\dfrac{3x-1}{x+2}$. عيّن المستقيمات المقاربة لمنحناها وادرس الوضع النسبي للمنحنى مع المقارب الأفقي.
\QQ{3} $h$ الدالة المعرفة على $\R$ بـ: $h(x)=\sqrt{x^{2}+2x+5}$. بيّن أن المستقيم $(d_1):\ y=x+1$ مقارب مائل لـ $(C_h)$ عند $+\infty$، وأن $(d_2):\ y=-x-1$ مقارب مائل له عند $-\infty$، ثم حدّد وضعية $(C_h)$ بالنسبة إليهما.
\end{tamrin}

\section{الاستمرارية ومبرهنة القيم المتوسطة}
% ------------------------------------------------------------------
\begin{tamrin}{دوال معرفة بجزأين: تعيين وسيط لتكون الدالة مستمرة}
\QQ{1} $f$ الدالة المعرفة على $[-3;+\infty[$ بـ:
\(f(x)=\left\{\begin{array}{ll}x^{2}+ax & ;\ x\le1\\[3pt] \dfrac{\sqrt{x+3}-2}{x-1} & ;\ x>1\end{array}\right.\) \ حيث $a$ عدد حقيقي.
عيّن $a$ حتى تكون $f$ مستمرة عند $1$.
\QQ{2} $g$ الدالة المعرفة على $\R$ بـ:
\(g(x)=\left\{\begin{array}{ll}\dfrac{\sin 3x}{x} & ;\ x<0\\[3pt] x^{2}+b & ;\ x\ge0\end{array}\right.\). \ عيّن $b$ حتى تكون $g$ مستمرة على $\R$.
\QQ{3} $k$ الدالة المعرفة على $\R$ بـ: $k(x)=\dfrac{x^{2}-4}{x-2}$ إذا كان $x\neq2$، و $k(2)=\lambda$. عيّن $\lambda$ حتى تكون $k$ مستمرة عند $2$.
\end{tamrin}

% ------------------------------------------------------------------
\begin{tamrin}{قراءة الاستمرارية من التمثيل البياني}
الشكل المقابل هو التمثيل البياني لدالة $f$ معرفة على المجال $[-3;4]$ (النقطة المملوءة تنتمي إلى المنحنى والنقطة الفارغة لا تنتمي إليه).
\begin{GR}{-3}{4}{-2}{3}{1}

 \draw[cDef,very thick,domain=-1:1,samples=60] plot (\x,{\x*\x});
 \draw[cDef,very thick] (1,1)--(3,-1);
 \draw[cDef,very thick] (3,2)--(4,2);
 \draw[od] (-1,2) circle (2.5pt);

 \draw[od] (3,2) circle (2.5pt);
 \fill[fd] (4,2) circle (2.5pt);
 \draw[gray,dashed] (-1,0)--(-1,2) (3,0)--(3,2);
 \node[cDef] at (2.4,0.9){$(C_f)$};
\end{GR}
\QQ{1} بقراءة بيانية عيّن: $f(-1)$ ، $\lim\limits_{x\to-1^{-}}f(x)$ ، $\lim\limits_{x\to-1^{+}}f(x)$ ، $f(1)$ ، $f(3)$ ، $\lim\limits_{x\to3^{-}}f(x)$ و $\lim\limits_{x\to3^{+}}f(x)$.
\QQ{2} هل $f$ مستمرة عند $-1$؟ عند $1$؟ عند $3$؟ علّل.
\QQ{3} عيّن أكبر المجالات التي تكون عليها $f$ مستمرة.
\QQ{4}
\begin{sq}
\qa برّر أن المعادلة $f(x)=0$ تقبل حلا في المجال $[1;3]$.
\qb لدينا $f(-1)=1$ و $f(4)=2$ والعدد $1{,}5$ محصور بينهما. هل تقبل المعادلة $f(x)=1{,}5$ حلا في $[-1;4]$ بالضرورة حسب مبرهنة القيم المتوسطة؟ ثم اقرأ حلولها من البيان.
\end{sq}
\end{tamrin}

% ------------------------------------------------------------------
\begin{tamrin}{مبرهنة القيم المتوسطة: الوجود والوحدانية وحصر الحل}
\QQ{1} $f$ الدالة المعرفة على $\R$ بـ: $f(x)=x^{3}+x-1$.
\begin{sq}
\qa ادرس اتجاه تغير $f$ وشكّل جدول تغيراتها.
\qb بيّن أن المعادلة $f(x)=0$ تقبل حلا وحيدا $\alpha$ في $\R$، وأن $0<\alpha<1$.
\qc عيّن حصرا للعدد $\alpha$ سعته $10^{-2}$.
\qd استنتج إشارة $f(x)$ على $\R$، وبيّن أن $\alpha=\dfrac{1}{1+\alpha^{2}}$.
\end{sq}
\QQ{2} $g$ الدالة المعرفة على $\left[0;\dfrac{\pi}{2}\right]$ بـ: $g(x)=\cos x-x$. بيّن أن المعادلة $\cos x=x$ تقبل حلا وحيدا $\beta$ في $\left[0;\dfrac{\pi}{2}\right]$، ثم تحقق أن $0{,}73<\beta<0{,}74$.
\end{tamrin}

% ------------------------------------------------------------------
\begin{tamrin}{صورة مجال بدالة مستمرة وعدد حلول معادلة}
$f$ الدالة المعرفة على $\R$ بـ: $f(x)=x^{3}-3x+1$.
\QQ{1} ادرس تغيرات $f$ وشكّل جدول تغيراتها.
\QQ{2} عيّن صورة كل مجال من المجالات التالية بالدالة $f$: \ $[-1;1]$ ، $[1;+\infty[$ ، $]-\infty;-1]$ ، $[0;2]$ و $]-2;0[$.
\QQ{3} بيّن أن المعادلة $f(x)=0$ تقبل ثلاثة حلول بالضبط، وعيّن مجالا طوله $1$ يحوي كل واحد منها.
\QQ{4} حسب قيم الوسيط الحقيقي $m$، عيّن عدد حلول المعادلة $f(x)=m$.
\end{tamrin}
\section{الاشتقاقية}
% ------------------------------------------------------------------
\begin{tamrin}{العدد المشتق بالتعريف والتقريب التآلفي}
باستعمال التعريف، أحسب العدد المشتق للدالة عند القيمة المعطاة، ثم اكتب معادلة المماس:
\begin{sq}
\qa $f(x)=x^{2}-3x$ \ عند $x_0=2$.
\qb $g(x)=\sqrt{x+3}$ \ عند $x_0=1$.
\qc $h(x)=\dfrac{1}{x}$ \ عند $x_0=-1$.
\end{sq}
\QQ{2} باستعمال التقريب التآلفي للدالة $g$ بجوار $1$، أعط قيمة مقربة للعدد $\sqrt{4{,}02}$.

\begin{tamrin}{دراسة شاملة لدالة صماء ومناقشة بيانية}
$f$ الدالة المعرفة على $[-1;+\infty[$ بـ: $f(x)=x-2\sqrt{x+1}$، و $(C_f)$ تمثيلها البياني في معلم متعامد ومتجانس.
\QQ{1} احسب $\lim\limits_{x\to+\infty}f(x)$.
\QQ{2} ادرس قابلية اشتقاق $f$ على يمين $-1$ وفسّر النتيجة هندسيا.
\QQ{3} بيّن أنه من أجل كل $x>-1$: $f'(x)=\dfrac{\sqrt{x+1}-1}{\sqrt{x+1}}$، ثم ادرس تغيرات $f$ وشكّل جدول تغيراتها.
\QQ{4} عيّن نقط تقاطع $(C_f)$ مع محوري الإحداثيات.
\QQ{5} اكتب معادلة المماس $(T)$ لـ $(C_f)$ عند النقطة ذات الفاصلة $3$.
\QQ{6} أنشئ $(T)$ و $(C_f)$.
\QQ{7} ناقش بيانيا، حسب قيم الوسيط الحقيقي $m$، عدد حلول المعادلة $f(x)=m$.
\end{tamrin}

\begin{tamrin}{عائلة دوال $f_m$: نقطة ثابتة، تمديد بالاستمرار ومقاربات -- شعبة رياضيات}
$m$ وسيط حقيقي. نعتبر الدالة $f_m$ المعرفة على $\R-\{-1\}$ بـ: $f_m(x)=\dfrac{x^{2}+mx+1}{x+1}$، و $(C_m)$ تمثيلها البياني في معلم متعامد ومتجانس.
\QQ{1} بيّن أن جميع المنحنيات $(C_m)$ تشمل نقطة ثابتة $I$ يطلب تعيينها.
\QQ{2} عيّن الأعداد الحقيقية $a$ ، $b$ و $c$ (بدلالة $m$) بحيث: $f_m(x)=ax+b+\dfrac{c}{x+1}$.
\QQ{3}
\begin{sq}
\qa في حالة $m=2$: بسّط $f_2(x)$، وبيّن أن $f_2$ تقبل تمديدا بالاستمرار عند $-1$ يطلب تعريفه. ما طبيعة $(C_2)$؟
\qb في حالة $m\neq2$: احسب $\lim\limits_{x\to-1^{-}}f_m(x)$ و $\lim\limits_{x\to-1^{+}}f_m(x)$ حسب قيم $m$.
\end{sq}
\QQ{4} من أجل $m\neq2$: بيّن أن $(C_m)$ يقبل مستقيما مقاربا مائلا $(\Delta_m)$، وأن نقطة تقاطع المقاربين $\Omega_m$ مركز تناظر لـ $(C_m)$. ما هي مجموعة النقط $\Omega_m$ لما يتغير $m$؟
\QQ{5} احسب $f_m'(x)$، ثم عيّن حسب قيم $m$ ($m\neq2$) عدد القيم الحدية المحلية للدالة $f_m$.
\end{tamrin}

\begin{tamrin}{التمديد بالاستمرار وتعيين وسيطين للاستمرارية والاشتقاقية -- شعبة رياضيات}
\QQ{1} في كل حالة، بيّن أن الدالة تقبل تمديدا بالاستمرار عند $x_0$ وعرّفه، أو بيّن أن ذلك غير ممكن:
\begin{sq}
\qa $f(x)=\dfrac{\sqrt{x+4}-2}{x}$ على $[-4;0[\cup]0;+\infty[$ ، \ $x_0=0$.
\qb $g(x)=\dfrac{1-\cos x}{x^{2}}$ على $\R^{*}$ ، \ $x_0=0$.
\qc $k(x)=\dfrac{x^{3}-8}{x^{2}-4}$ على $\R-\{-2;2\}$ ، عند $x_0=2$ ثم عند $x_0=-2$.
\end{sq}
\QQ{2} $a$ و $b$ عددان حقيقيان، و $\varphi$ الدالة المعرفة على $\R_{+}$ بـ:
\(\varphi(x)=\left\{\begin{array}{ll}x^{2}+ax+b & ;\ 0\le x<1\\ 2\sqrt{x} & ;\ x\ge1\end{array}\right.\)
عيّن $a$ و $b$ حتى تكون $\varphi$ مستمرة وقابلة للاشتقاق عند $1$.
\end{tamrin}

\begin{tamrin}{الحصر والمقارنة مع $|x|$ و $\sin$، والاشتقاقية عند $0$ -- شعبة رياضيات}
\QQ{1} $f$ الدالة المعرفة على $\R^{*}$ بـ: $f(x)=|x|\sin\left(\dfrac1x\right)$.
\begin{sq}
\qa بيّن أنه من أجل كل $x\neq0$: $|f(x)|\le|x|$، ثم استنتج $\lim\limits_{x\to0}f(x)$. هل تقبل $f$ تمديدا بالاستمرار عند $0$؟
\qb احسب $\lim\limits_{x\to+\infty}f(x)$ و $\lim\limits_{x\to-\infty}f(x)$ (ضع $t=\dfrac1x$)، وفسّر النتيجتين هندسيا.
\end{sq}
\QQ{2} $F$ الدالة المعرفة على $\R$ بـ: $F(x)=x^{2}\sin\left(\dfrac1x\right)$ إذا كان $x\neq0$ و $F(0)=0$. بيّن أن $F$ قابلة للاشتقاق عند $0$ وعيّن $F'(0)$.
\QQ{3} احسب: \quad (أ) $\displaystyle\lim_{x\to+\infty}x\left(2+\sin x\right)$ \qquad (ب) $\displaystyle\lim_{x\to-\infty}\frac{x+|x|\cos x}{x^{2}+1}$
\end{tamrin}

\begin{tamrin}{عدد حلول معادلة من الدرجة الثالثة وحلها بالدوال المثلثية -- شعبة رياضيات}
$g$ الدالة المعرفة على $\R$ بـ: $g(x)=x^{3}-3x$.
\QQ{1} ادرس تغيرات $g$، ثم عيّن حسب قيم الوسيط الحقيقي $m$ عدد حلول المعادلة $g(x)=m$.
\QQ{2} نعتبر المعادلة $(E):\ x^{3}-3x-1=0$.
\begin{sq}
\qa بيّن أن $(E)$ تقبل ثلاثة حلول $x_1<x_2<x_3$، وأنها جميعا تنتمي إلى $]-2;2[$؛ عيّن مجالا طوله $1$ يحوي كلا منها.
\qb بيّن أنه من أجل كل عدد حقيقي $t$: $\cos 3t=4\cos^{3}t-3\cos t$.
\qc بوضع $x=2\cos t$ مع $t\in[0;\pi]$، بيّن أن $(E)$ تكافئ $\cos 3t=\dfrac12$، ثم استنتج القيم المضبوطة لـ $x_1$ ، $x_2$ و $x_3$.
\end{sq}
\end{tamrin}

\begin{tamrin}{عائلة دوال $f_m$: النقط المشتركة والمماسات ومحل نقط الانعطاف -- شعبة رياضيات}
$m$ وسيط حقيقي، و $f_m$ الدالة المعرفة على $\R$ بـ: $f_m(x)=x^{3}+m\left(x^{2}-1\right)$، و $(C_m)$ تمثيلها البياني في معلم متعامد ومتجانس.
\QQ{1} بيّن أن جميع المنحنيات $(C_m)$ تشمل نقطتين ثابتتين $A$ و $B$ يطلب تعيينهما ($A$ ذات الفاصلة الموجبة).
\QQ{2} اكتب بدلالة $m$ معادلتي المماسين $(T_A)$ و $(T_B)$ لـ $(C_m)$ عند $A$ و $B$.
\QQ{3} عيّن قيم $m$ التي من أجلها يكون: (أ) $(T_A)$ و $(T_B)$ متوازيين؛ \ (ب) متعامدين؛ \ (جـ) $(T_A)$ يشمل المبدأ $O$.
\QQ{4} بيّن أن $(C_m)$ يقبل نقطة انعطاف وحيدة $I_m$، ثم عيّن المحل الهندسي للنقط $I_m$ لما يتغير $m$ في $\R$.
\QQ{5} عيّن قيم $m$ التي من أجلها تقبل $f_m$ قيمتين حديتين محليتين.
\end{tamrin}
