\section{تعريف الدالة الأسية والعدد $e$}
\begin{tarif}
من أجل كل عدد حقيقي $\LW{x}$ ، المعادلة $\LW{\ln y=x}$ تقبل حلًا وحيدًا $\LW{y}$ موجبًا تمامًا ، نرمز له بـ $\LW{e^x}$.\par
الدالة $\LW{x\mapsto e^x}$ المعرفة على $\LW{\R}$ تسمى \textbf{الدالة الأسية النيبيرية} ، ونرمز لها أيضًا بـ $\LW{\exp}$:
\[\LW{\LW{y=e^x\iff x=\ln y}\qquad(y>0)}\]
\end{tarif}

\begin{khassia}[نتائج مباشرة]
\begin{itemize}
\item من أجل كل $\LW{x\in\R}$: $\LW{e^x>0}$ ، و $\bx{\ln\left(e^x\right)=x}$.
\item من أجل كل $\LW{x>0}$: $\bx{e^{\ln x}=x}$.
\item $\LW{e^0=1}$ (لأن $\LW{\ln1=0}$) ، و $\LW{e^1=e\approx2{,}718}$ (لأن $\LW{\ln e=1}$).
\item الدالة الأسية قابلة للاشتقاق على $\LW{\R}$ و $\bx{\left(e^x\right)'=e^x}$.
\end{itemize}
\end{khassia}

\begin{mithal}
$e^{\ln5}=\bx{5}$ ؛ \quad $\ln\left(e^{-3}\right)=\bx{-3}$ ؛ \quad $e^{2\ln3}=e^{\ln9}=\bx{9}$ ؛ \quad $e^{-\ln2}=e^{\ln\frac12}=\bx{\dfrac12}$.
\end{mithal}

\begin{tabular}{|c|c|c|}\hline

\begin{mithal}[أمثلة سريعة]
\begin{itemize}
\item $e^3\times e^{-5}=e^{3-5}=\bx{e^{-2}}$ ؛ \quad $\dfrac{e^{2x+1}}{e^{x-3}}=e^{(2x+1)-(x-3)}=\bx{e^{x+4}}$.
\item $\left(e^x\right)^3\times e^{-2x}=e^{3x-2x}=\bx{e^x}$ ؛ \quad $\left(e^x+e^{-x}\right)\times e^x=\bx{e^{2x}+1}$.
\end{itemize}
\end{mithal}

\section{تبسيط عبارات أسية}
\begin{tariqa}[تقنيات]
نضرب البسط والمقام في $\LW{e^x}$ (أو $\LW{e^{-x}}$) ، أو نوحد المقامات ، أو نخرج عاملًا مشتركًا.
\end{tariqa}

\begin{tamrin}{تحقق أنه من أجل كل $x$ من $\R$:}
\QQ{1} $\dfrac4{1+e^{-x}}-1=\dfrac{3e^x-1}{1+e^x}$. \qquad \QQ{2} $\dfrac{1-e^{-2x}}{1+e^{-2x}}=\dfrac{e^{2x}-1}{e^{2x}+1}$.
\QQ{3} $e^{-x}-e^{-2x}=\dfrac{e^x-1}{e^{2x}}$. \qquad \QQ{4} $\left(e^x+e^{-x}\right)^2=e^{2x}+2+e^{-2x}$.
\end{tamrin}
\begin{hal}
\QQ{1} نضرب البسط والمقام في $e^x$:
\[\frac4{1+e^{-x}}-1=\frac{4e^x}{e^x+1}-1=\frac{4e^x-(e^x+1)}{e^x+1}=\bx{\frac{3e^x-1}{e^x+1}}\]
\QQ{2} نضرب البسط والمقام في $e^{2x}$:
\[\frac{1-e^{-2x}}{1+e^{-2x}}=\frac{e^{2x}-e^{-2x}e^{2x}}{e^{2x}+e^{-2x}e^{2x}}=\bx{\frac{e^{2x}-1}{e^{2x}+1}}\]
\QQ{3} نوحد المقامات:
\[e^{-x}-e^{-2x}=\frac1{e^x}-\frac1{e^{2x}}=\frac{e^x}{e^{2x}}-\frac1{e^{2x}}=\bx{\frac{e^x-1}{e^{2x}}}\]
\QQ{4} نطبق $(a+b)^2$:
\[\left(e^x+e^{-x}\right)^2=e^{2x}+2e^xe^{-x}+e^{-2x}=\bx{e^{2x}+2+e^{-2x}}\]
\end{hal}

\begin{tamrin}{أكتب على أبسط شكل ممكن (من أجل كل $x$ من $\R$):}
\QQ{1} $\dfrac{e^{4x}}{e^x\times\left(e^{-x}\right)^5}$ \qquad \QQ{2} $\dfrac{\left(e^{x-1}\right)^2\times e^{4x-1}}{\left(e^{2x}\right)^3\times e^{-x+3}}$ \qquad \QQ{3} $\left(e^{2x}-3\right)^2-e^{2x}\left(e^{2x}+9e^{-2x}\right)$
\end{tamrin}
\begin{hal}
\QQ{1}
\[\frac{e^{4x}}{e^x\times e^{-5x}}=\frac{e^{4x}}{e^{-4x}}=e^{4x-(-4x)}=\bx{e^{8x}}\]
\QQ{2}
\[\frac{e^{2x-2}\times e^{4x-1}}{e^{6x}\times e^{-x+3}}=\frac{e^{6x-3}}{e^{5x+3}}=e^{(6x-3)-(5x+3)}=\bx{e^{x-6}}\]
\QQ{3}
\[\left(e^{2x}-3\right)^2-e^{2x}\left(e^{2x}+9e^{-2x}\right)=e^{4x}-6e^{2x}+9-e^{4x}-9=\bx{-6e^{2x}}\]
\end{hal}

\begin{tamrin}{تعيين أعداد حقيقية}
عيّن الأعداد الحقيقية $a$ ، $b$ و $c$ بحيث من أجل كل $x$ من $\R$: $\dfrac{e^{2x}}{e^x+2}=ae^x+b+\dfrac c{e^x+2}$.
\end{tamrin}
\begin{hal}
نوحد المقامات في الطرف الثاني:
\[ae^x+b+\frac c{e^x+2}=\frac{(ae^x+b)(e^x+2)+c}{e^x+2}=\frac{ae^{2x}+(2a+b)e^x+2b+c}{e^x+2}\]
بالمطابقة: $a=1$ ، $2a+b=0$ ، $2b+c=0$ ، ومنه $\bx{a=1\ ;\ b=-2\ ;\ c=4}$.
\end{hal}

\section{المعادلات الأسية}
\begin{tariqa}[ثلاثة أنواع]
\begin{enumerate}
\item $\LW{e^{A(x)}=e^{B(x)}\iff A(x)=B(x)}$.
\item $\LW{e^{A(x)}=k}$: إذا كان $\LW{k\leqslant0}$ فلا حلول ، وإذا كان $\LW{k>0}$ فإن $\LW{A(x)=\ln k}$.
\item $\LW{ae^{2x}+be^x+c=0}$: نضع $\LW{X=e^x}$ ($\LW{X>0}$) فنحصل على $\LW{aX^2+bX+c=0}$ ، نحلها ونقبل الحلول الموجبة فقط ، ثم $\LW{x=\ln X}$.
\end{enumerate}
\end{tariqa}

\begin{mithal}[النوعان 1 و 2]
\QQ{1} $e^{4x}+3=0$: \quad $e^{4x}=-3$ مستحيل لأن $e^{4x}>0$: \quad $\bx{S=\varnothing}$.
\QQ{2} $e^{-2x+10}=1$: \quad $e^{-2x+10}=e^0\ \Rightarrow\ -2x+10=0\ \Rightarrow\ x=5$: \quad $\bx{S=\{5\}}$.
\QQ{3} $e^{5x-3}=\dfrac1{e^{11-3x}}$: \quad $e^{5x-3}=e^{-11+3x}\ \Rightarrow\ 5x-3=-11+3x\ \Rightarrow\ x=-4$: \quad $\bx{S=\{-4\}}$.
\QQ{4} $e^{x^2}=\dfrac{(e^x)^5}{(e^2)^3}$: \quad $e^{x^2}=e^{5x-6}\ \Rightarrow\ x^2-5x+6=0\ \Rightarrow\ (x-2)(x-3)=0$: \quad $\bx{S=\{2;3\}}$.
\QQ{5} $e^{2x-1}=3$: \quad $2x-1=\ln3\ \Rightarrow\ x=\dfrac{1+\ln3}2$: \quad $\bx{S=\left\{\dfrac{1+\ln3}2\right\}}$.
\end{mithal}

\begin{mithal}[النوع 3: معادلات من الدرجة الثانية في $e^x$]
\QQ{1} $e^{2x}-3e^x+2=0$: \quad نضع $X=e^x$: $X^2-3X+2=0$:
\[\Delta=9-8=1\ ;\ X_1=1\ ;\ X_2=2\]
$e^x=1$ يعطي $x=0$ ، و $e^x=2$ يعطي $x=\ln2$: \quad $\bx{S=\{0;\ln2\}}$.
\QQ{2} $e^{2x}+6e^x-7=0$: \quad $X^2+6X-7=0$:
\[\Delta=36+28=64\ ;\ X_1=\frac{-6-8}2=-7\ ;\ X_2=\frac{-6+8}2=1\]
$X=-7$ مرفوض لأن $e^x>0$ ، و $e^x=1$ يعطي $x=0$: \quad $\bx{S=\{0\}}$.
\QQ{3} $2e^{2x}-5e^x+2=0$: \quad $2X^2-5X+2=0$:
\[\Delta=25-16=9\ ;\ X_1=\frac{5-3}4=\frac12\ ;\ X_2=\frac{5+3}4=2\]
$x=\ln\dfrac12=-\ln2$ أو $x=\ln2$: \quad $\bx{S=\{-\ln2;\ln2\}}$.
\QQ{4} $7e^x-5-2e^{-x}=0$: \quad نضرب في $e^x\neq0$: $7e^{2x}-5e^x-2=0$ ، أي $7X^2-5X-2=0$:
\[\Delta=25+56=81\ ;\ X_1=\frac{5-9}{14}=-\frac27\ ;\ X_2=\frac{5+9}{14}=1\]
$X_1$ مرفوض ، و $e^x=1$: \quad $\bx{S=\{0\}}$.
\QQ{5} $e^x-4e^{-x}=0$: \quad نضرب في $e^x$: $e^{2x}=4$ ، أي $\left(e^x\right)^2=4$ و $e^x>0$ ، إذن $e^x=2$: \quad $\bx{S=\{\ln2\}}$.
\QQ{6} $-2e^{x-4}+7e^{-x+4}-5=0$: \quad نضع $t=e^{x-4}>0$ ، و $e^{-x+4}=\dfrac1t$:
\[-2t+\frac7t-5=0\ \Rightarrow\ 2t^2+5t-7=0\ ;\ \Delta=81\ ;\ t_1=1\ ;\ t_2=-\frac72\]
$t_2$ مرفوض ، و $e^{x-4}=1$ يعطي $x=4$: \quad $\bx{S=\{4\}}$.
\end{mithal}

\section{النهايات والنهايات الشهيرة}
\begin{khassia}[نهايات مرجعية]
\begin{center}
\renewcommand{\arraystretch}{2}
\begin{tabular}{|c|c|}\hline
\rowcolor{cSec!12}$\LW{\Lm{-\infty}e^x=0}$ & $\LW{\Lm{+\infty}e^x=+\infty}$\\\hline
$\LW{\Lm{-\infty}xe^x=0}$ & $\LW{\Lm{+\infty}\dfrac{e^x}x=+\infty}$\\\hline

\end{tabular}

\end{khassia}

\begin{tariqa}[رفع حالة عدم التعيين]
نخرج عاملًا مشتركًا مناسبًا ($\LW{x}$ أو $\LW{e^x}$ أو $\LW{e^{2x}}$) لتظهر إحدى النهايات المرجعية ، أو ننشر لتظهر $\LW{xe^x}$.
\end{tariqa}

\begin{mithal}[احسب النهايات]
\QQ{1} $\Lm{+\infty}(e^x-x+1)$: حالة عدم تعيين ، نكتب: $e^x-x+1=x\left(\dfrac{e^x}x-1+\dfrac1x\right)$.
\LD{(e^x-x+1)}{+\infty}{+\infty} $\Car{\Lm{+\infty}x=+\infty\\[6pt]\Lm{+\infty}\left(\dfrac{e^x}x-1+\dfrac1x\right)=+\infty}$\E
\QQ{2}
\LD{(e^x-x+1)}{-\infty}{+\infty} $\Car{\Lm{-\infty}e^x=0\\[6pt]\Lm{-\infty}(-x+1)=+\infty}$\E
\QQ{3} نكتب: $\dfrac{e^{x+2}}x=e^2\times\dfrac{e^x}x$.
\LD{\dfrac{e^{x+2}}x}{+\infty}{+\infty} $\Car{e^2>0\\[6pt]\Lm{+\infty}\dfrac{e^x}x=+\infty}$\E
\QQ{4} نخرج $x$ عاملًا مشتركًا: $\dfrac{e^x-2}{x+3}=\dfrac{\frac{e^x}x-\frac2x}{1+\frac3x}$.
\LD{\dfrac{e^x-2}{x+3}}{+\infty}{+\infty} $\Car{\Lm{+\infty}\left(\dfrac{e^x}x-\dfrac2x\right)=+\infty\\[6pt]\Lm{+\infty}\left(1+\dfrac3x\right)=1}$\E
\QQ{5} نكتب: $(x-3)e^x=xe^x-3e^x$.
\LD{(x-3)e^x}{-\infty}{0} $\Car{\Lm{-\infty}xe^x=0\\[6pt]\Lm{-\infty}3e^x=0}$\E
\QQ{6} نكتب: $(2x-1)e^{-x}=2\times\dfrac x{e^x}-e^{-x}$.
\LD{(2x-1)e^{-x}}{+\infty}{0} $\Car{\Lm{+\infty}\dfrac x{e^x}=0\\[6pt]\Lm{+\infty}e^{-x}=0}$\E
\QQ{7} نخرج $e^x$ عاملًا مشتركًا: $\dfrac{e^x-x}{2e^x+x}=\dfrac{1-\frac x{e^x}}{2+\frac x{e^x}}$.
\LD{\dfrac{e^x-x}{2e^x+x}}{+\infty}{\dfrac12} $\Car{\Lm{+\infty}\left(1-\dfrac x{e^x}\right)=1\\[6pt]\Lm{+\infty}\left(2+\dfrac x{e^x}\right)=2}$\E
\QQ{8} نخرج $e^{2x}$: $(2x+3)e^x-e^{2x}=e^{2x}\left(2\dfrac x{e^x}+3e^{-x}-1\right)$.
\LD{\big[(2x+3)e^x-e^{2x}\big]}{+\infty}{-\infty} $\Car{\Lm{+\infty}e^{2x}=+\infty\\[6pt]\Lm{+\infty}\left(2\dfrac x{e^x}+3e^{-x}-1\right)=-1}$\E
\QQ{9} نهاية دالة مركبة:
\LD{e^{-x+2}}{+\infty}{0} $\Car{\Lm{+\infty}(-x+2)=-\infty\\[6pt]\displaystyle\lim_{X\to-\infty}e^X=0}$\E
\LD{e^{\frac1x}}{0^+}{+\infty} $\Car{\Lm{0^+}\dfrac1x=+\infty\\[6pt]\displaystyle\lim_{X\to+\infty}e^X=+\infty}$\E
\end{mithal}

\begin{khassia}
إذا كانت $\LW{u}$ قابلة للاشتقاق على مجال $\LW{I}$ فإن: $\bx{\left(e^{u(x)}\right)'=u'(x)\,e^{u(x)}}$.\par
وبما أن $\LW{e^{u(x)}>0}$ فللدالتين $\LW{u}$ و $\LW{e^u}$ نفس اتجاه التغير على $\LW{I}$.
\end{khassia}

\begin{mithal}[احسب المشتقة]

\QQ{2} $h(x)=e^{-3x}$: \quad $h'(x)=\bx{-3e^{-3x}}$ ؛ \quad $k(x)=e^{2x-1}$: \quad $k'(x)=\bx{2e^{2x-1}}$.

\QQ{4} $f(x)=(-2x+1)e^{-5x}$:
\[f'(x)=-2e^{-5x}+(-2x+1)(-5)e^{-5x}=(-2+10x-5)e^{-5x}=\bx{(10x-7)e^{-5x}}\]
\QQ{5} $f(x)=-2+\dfrac{x-3}{x-1}e^x$ على $]1;+\infty[$: \quad $\left(\dfrac{x-3}{x-1}\right)'=\dfrac2{(x-1)^2}$:
\[f'(x)=\frac2{(x-1)^2}e^x+\frac{x-3}{x-1}e^x=\frac{2+(x-3)(x-1)}{(x-1)^2}e^x=\bx{\frac{x^2-4x+5}{(x-1)^2}e^x}\]
\QQ{6} $f(x)=\dfrac{-2xe^x}{e^x+1}$: \quad $u=-2xe^x$ ، $u'=-2e^x-2xe^x$ ، $v=e^x+1$ ، $v'=e^x$:
\[f'(x)=\frac{(-2e^x-2xe^x)(e^x+1)+2xe^xe^x}{(e^x+1)^2}=\bx{\frac{-2e^x\left(e^x+x+1\right)}{(e^x+1)^2}}\]
\end{mithal}

\begin{mithal}[معادلة مماس]
$k(x)=e^{2x-1}$ عند $x_0=\dfrac12$: \quad $k\left(\dfrac12\right)=e^0=1$ و $k'\left(\dfrac12\right)=2e^0=2$:
\[y=2\left(x-\frac12\right)+1\ \Rightarrow\ \bx{y=2x}\]
\end{mithal}
