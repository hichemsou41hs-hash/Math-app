\section{العدّ}
\subsection{المبدأ الأساسي للعدّ والعاملي}
\begin{khassia}[المبدأ الأساسي للعدّ]
ليكن حدثان: الأول يتحقق بـ $n$ طريقة مختلفة والثاني بـ $m$ طريقة مختلفة.
\begin{itemize}
\item \textbf{مبدأ الجمع} («أو»): عدد طرق تحقق الأول \textbf{أو} الثاني (دون تداخل) هو $n+m$.
\item \textbf{مبدأ الضرب} («و»): عدد طرق تحقق الأول \textbf{ثم} الثاني هو $n\times m$.
\end{itemize}
\end{khassia}

\begin{tarif}[العاملي]
من أجل كل عدد طبيعي $n\geqslant 1$ نسمي عاملي $n$ العدد: \quad
$\bx{n!=n\times(n-1)\times(n-2)\times\cdots\times 2\times 1}$ \quad ونضع اصطلاحًا $0!=1$.
\end{tarif}

\begin{mithal}
$3!=6$ ، \quad $5!=120$ ، \quad $6!=720$ ، \quad $\dfrac{10!}{8!}=10\times 9=90$.
\par لاختيار قميص من بين $4$ قمصان \textbf{ثم} سروال من بين $3$ سراويل: $4\times 3=12$ طريقة.
\end{mithal}

\subsection{القوائم (الترتيب مهم والتكرار ممكن)}
\begin{tarif}[القائمة]
لتكن $E$ مجموعة منتهية بها $n$ عنصرًا و$p$ عددًا طبيعيًا غير معدوم.
نسمي \textbf{قائمة} ذات $p$ عنصرًا من $E$ كل متتالية مرتبة من $p$ عنصرًا من $E$ (يمكن أن تتكرر العناصر).
\par عدد القوائم ذات $p$ عنصرًا من $E$ هو: \quad $\bx{n^p}$
\end{tarif}
\begin{mithal}
\begin{itemize}
\item عدد الرموز السرية المكونة من $3$ خانات تُختار من الأرقام $\{2;3;5;9\}$: \quad $4^3=64$.
\item عدد طرق سحب $4$ كريات على التوالي \textbf{مع الإرجاع} من كيس به $8$ كريات: \quad $8^4=4096$.
\item عدد طرق الإجابة عشوائيًا على $5$ أسئلة من نوع «صحيح/خطأ»: \quad $2^5=32$.
\end{itemize}
\end{mithal}

\subsection{الترتيبات (الترتيب مهم ودون تكرار)}
\begin{tarif}[الترتيبة]
نسمي \textbf{ترتيبة} ذات $p$ عنصرًا من $E$ (حيث $p\leqslant n$) كل قائمة عناصرها متمايزة مثنى مثنى. عددها:
\[\bx{A_n^p=n(n-1)(n-2)\cdots(n-p+1)=\frac{n!}{(n-p)!}}\]
($p$ عاملًا متناقصًا ابتداءً من $n$).
\end{tarif}
\begin{mithal}
\begin{itemize}
\item عدد الأعداد ذات $5$ أرقام \textbf{مختلفة} المكونة من الأرقام $1;2;\ldots;8$: \quad $A_8^5=8\times7\times6\times5\times4=6720$.
\item عدد طرق اختيار رئيس ونائبه من قسم به $20$ تلميذًا: \quad $A_{20}^2=20\times 19=380$.
\item عدد طرق جلوس $3$ أشخاص على $30$ مقعدًا: \quad $A_{30}^3=30\times29\times28=24360$.
\end{itemize}
\end{mithal}

\subsection{التبديلات}
\begin{tarif}[التبديلة]
في حالة $p=n$، تسمى الترتيبة \textbf{تبديلة} لعناصر $E$. عدد التبديلات هو: \quad $\bx{A_n^n=n!}$
\end{tarif}
\begin{khassia}[تبديلات مع عناصر متكررة (معامل الترتيب)]
عدد طرق ترتيب $n$ عنصرًا، منها $n_1$ عنصرًا متماثلًا من نوع أول، و$n_2$ من نوع ثانٍ، $\ldots$ ، و$n_k$ من نوع أخير ($n_1+n_2+\cdots+n_k=n$) هو:
\[\bx{\frac{n!}{n_1!\times n_2!\times\cdots\times n_k!}}\]
\end{khassia}
\begin{mithal}
\begin{itemize}
\item عدد طرق جلوس $5$ تلاميذ على $5$ كراسٍ في صف واحد: \quad $5!=120$.
\item عدد طرق ترتيب $4$ كريات في صف: زرقاوان (متماثلتان) وحمراء وصفراء:
$\dfrac{4!}{2!\times1!\times1!}=\dfrac{24}{2}=12$.
\item عدد الكلمات (لها معنى أو لا) المكونة من حروف الكلمة «BAC»: \quad $3!=6$.
\end{itemize}
\end{mithal}

\subsection{التوفيقات (الترتيب غير مهم ودون تكرار)}
\begin{tarif}[التوفيقة]
نسمي \textbf{توفيقة} ذات $p$ عنصرًا من $E$ ($0\leqslant p\leqslant n$) كل \textbf{جزء} من $E$ يحوي $p$ عنصرًا. عددها:
\[\bx{C_n^p=\frac{A_n^p}{p!}=\frac{n!}{p!\,(n-p)!}}\]
\end{tarif}
\begin{khassia}[خواص التوفيقات]
من أجل كل عددين طبيعيين $n$ و$p$:
\begin{itemize}
\item $C_n^0=C_n^n=1$ \quad و\quad $C_n^1=C_n^{n-1}=n$.
\item التناظر: $C_n^p=C_n^{n-p}$ \quad ($0\leqslant p\leqslant n$).
\item علاقة باسكال: $C_n^p=C_{n-1}^{p}+C_{n-1}^{p-1}$ \quad ($1\leqslant p\leqslant n-1$).
\end{itemize}
\end{khassia}
\begin{mithal}
\begin{itemize}
\item $C_7^3=\dfrac{7\times6\times5}{3\times2\times1}=35$ \quad ، \quad $C_5^2=\dfrac{5\times 4}{2}=10$ \quad ، \quad $C_{10}^{8}=C_{10}^{2}=45$.
\item عدد اللجان ذات $7$ تلاميذ من قسم به $20$ تلميذًا: \quad $C_{20}^7=\dfrac{20!}{7!\times13!}=77520$.
\item عدد طرق سحب $3$ كريات في آن واحد من كيس به $4$ كريات بيضاء و$5$ حمراء: \quad $C_9^3=84$.
\end{itemize}
\end{mithal}
\begin{molahada}
على الحاسبة: $n!$ بالزر \texttt{x!}، و$A_n^p$ بالزر \texttt{nPr}، و$C_n^p$ بالزر \texttt{nCr}.
مثلًا: \texttt{5 nCr 2 = 10} و\texttt{5 nPr 2 = 20}.
\end{molahada}

\subsection{مثلث باسكال ودستور ثنائي الحد}
\begin{khassia}[مثلث باسكال]
كل عدد في الجدول يساوي مجموع العدد الذي فوقه والعدد الذي على يسار هذا الأخير (علاقة باسكال).
\begin{center}
\begin{tabular}{|c|c|c|c|c|c|c|c|}\hline
\rowcolor{cSec!12} $n\backslash p$ & $0$ & $1$ & $2$ & $3$ & $4$ & $5$ & $6$\\\hline
$0$ & $1$ &&&&&&\\\hline
$1$ & $1$ & $1$ &&&&&\\\hline
$2$ & $1$ & $2$ & $1$ &&&&\\\hline
$3$ & $1$ & $3$ & $3$ & $1$ &&&\\\hline
$4$ & $1$ & $4$ & $6$ & $4$ & $1$ &&\\\hline
$5$ & $1$ & $5$ & $10$ & $10$ & $5$ & $1$ &\\\hline
$6$ & $1$ & $6$ & $15$ & $20$ & $15$ & $6$ & $1$\\\hline
\end{tabular}
\end{center}
\end{khassia}
\begin{khassia}[دستور ثنائي الحد (نيوتن)]
من أجل كل عددين حقيقيين $a$ و$b$ ومن أجل كل عدد طبيعي غير معدوم $n$:
\[\bx{(a+b)^n=\sum_{k=0}^{n}C_n^k\,a^{n-k}\,b^{k}=C_n^0a^n+C_n^1a^{n-1}b+\cdots+C_n^nb^n}\]
\end{khassia}
\begin{mithal}
\begin{align*}
(x-1)^4&=C_4^0x^4+C_4^1x^3(-1)+C_4^2x^2(-1)^2+C_4^3x(-1)^3+C_4^4(-1)^4\\
&=x^4-4x^3+6x^2-4x+1.
\end{align*}
وبأخذ $a=b=1$: \quad $\sum_{k=0}^{n}C_n^k=2^n$ \quad (عدد أجزاء مجموعة بها $n$ عنصرًا)؛ مثلًا $1+5+10+10+5+1=32=2^5$.
\end{mithal}

\section{أنواع السحب}
\begin{tarif}[ثلاثة أنواع للسحب]
يحوي كيس $n$ كرية لا نفرق بينها باللمس، ونسحب منه $p$ كرية:
\begin{itemize}
\item \textbf{في آن واحد} (دفعة واحدة): النتيجة جزء من $p$ كرية، الترتيب غير مهم: عدد الحالات $C_n^p$.
\item \textbf{على التوالي مع الإرجاع}: نعيد الكرية قبل السحب الموالي، النتيجة قائمة: عدد الحالات $n^p$.
\item \textbf{على التوالي دون إرجاع}: لا نعيد الكرية، النتيجة ترتيبة: عدد الحالات $A_n^p$.
\end{itemize}
\end{tarif}

\begin{khassia}[{جدول مقارنة: كيس به $9$ كريات ($4$ بيضاء و$5$ حمراء)، نسحب $3$ كريات}]
\begin{center}
\begin{tabular}{|p{0.22\textwidth}|p{0.22\textwidth}|p{0.12\textwidth}|p{0.08\textwidth}|p{0.21\textwidth}|}\hline
\rowcolor{cSec!12} \textbf{«بيضاوان بالضبط»} & \textbf{«نفس اللون»} & \textbf{$\mathrm{card}\,\Omega$} & \textbf{الترتيب} & \textbf{نوع السحب}\\\hline
$C_4^2\times C_5^1=30$ \newline $P=\frac{30}{84}=\frac{5}{14}$ & $C_4^3+C_5^3=14$ \newline $P=\frac{14}{84}=\frac16$ & $C_9^3=84$ & غير مهم & في آن واحد\\\hline
$3\times 4^2\times 5=240$ \newline $P=\frac{240}{729}=\frac{80}{243}$ & $4^3+5^3=189$ \newline $P=\frac{189}{729}=\frac{7}{27}$ & $9^3=729$ & مهم & على التوالي مع الإرجاع\\\hline
$3\times A_4^2\times A_5^1=180$ \newline $P=\frac{180}{504}=\frac{5}{14}$ & $A_4^3+A_5^3=84$ \newline $P=\frac{84}{504}=\frac16$ & $A_9^3=504$ & مهم & على التوالي دون إرجاع\\\hline
\end{tabular}
\end{center}
\end{khassia}

\begin{molahada}
\begin{itemize}
\item في السحب على التوالي نضرب في عدد \textbf{مواضع} الألوان: «بيضاوان وحمراء» لها $\dfrac{3!}{2!\,1!}=3$ ترتيبات ممكنة: $BBH$ ، $BHB$ ، $HBB$ ($B$: بيضاء، $H$: حمراء).
\item السحب في آن واحد والسحب على التوالي دون إرجاع يعطيان \textbf{نفس الاحتمالات} (مثل $\frac16$ و$\frac{5}{14}$ أعلاه).
\item «على الأقل واحدة» نحسبها بالحادثة العكسية. في السحب في آن واحد، لتكن $W$: «سحب كرية بيضاء على الأقل»، عكسها $\overline{W}$: «الكريات الثلاث حمراء»، إذن:
$P(W)=1-\dfrac{C_5^3}{C_9^3}=1-\dfrac{10}{84}=\dfrac{37}{42}$.
\end{itemize}
\end{molahada}

\begin{mithal}[سحب على التوالي مع الإرجاع]
نفس الكيس، نسحب $3$ كريات على التوالي مع الإرجاع. احتمال «سحب كرية بيضاء على الأقل»:
\[P=1-\frac{5^3}{9^3}=1-\frac{125}{729}=\frac{604}{729}.\]
\end{mithal}

\section{قانون الاحتمال والمتغير العشوائي}
\subsection{المفردات الأساسية}
\begin{tarif}[التجربة العشوائية والحوادث]
\begin{itemize}
\item \textbf{التجربة العشوائية}: تجربة نعرف كل نتائجها الممكنة لكن لا يمكن توقع نتيجتها مسبقًا.
\item \textbf{مجموعة الإمكانيات} $\Omega$ (مجموعة المخارج): مجموعة كل النتائج الممكنة.
\item \textbf{الحادثة}: كل جزء من $\Omega$. \textbf{الحادثة الأولية}: تحوي عنصرًا واحدًا.
\item \textbf{الحادثة الأكيدة}: $\Omega$ نفسها. \textbf{الحادثة المستحيلة}: المجموعة الخالية $\varnothing$.
\item \textbf{الحادثة العكسية} $\overline{A}$: تحوي كل عناصر $\Omega$ التي لا تنتمي إلى $A$.
\item $A\cap B$ («$A$ و$B$»): العناصر المشتركة. \quad $A\cup B$ («$A$ أو $B$»): عناصر $A$ أو $B$ أو كليهما.
\item $A$ و$B$ \textbf{غير متلائمتين} (متنافيتان) معناه $A\cap B=\varnothing$.
\end{itemize}
\end{tarif}

\subsection{الاحتمال وخواصه}
\begin{khassia}[فرضية تساوي الاحتمال]
إذا كانت كل النتائج لها نفس الاحتمال (عبارات مثل: «نرد غير مزيف»، «كريات لا نفرق بينها باللمس»، «عشوائيًا») فإن:
\[\bx{P(A)=\frac{\mathrm{card}\,A}{\mathrm{card}\,\Omega}}\]
حيث $\mathrm{card}\,A$ عدد عناصر $A$ (الحالات الملائمة) و$\mathrm{card}\,\Omega$ عدد الحالات الممكنة.
\end{khassia}
\begin{khassia}[خواص الاحتمال]
من أجل كل حادثتين $A$ و$B$:
\begin{itemize}
\item $0\leqslant P(A)\leqslant 1$ \quad ، \quad $P(\Omega)=1$ \quad ، \quad $P(\varnothing)=0$.
\item $P(\overline{A})=1-P(A)$.
\item $P(A\cup B)=P(A)+P(B)-P(A\cap B)$؛ وإذا كانتا غير متلائمتين: $P(A\cup B)=P(A)+P(B)$.
\item إذا كان $A\subset B$ فإن $P(A)\leqslant P(B)$.
\end{itemize}
\end{khassia}
\begin{mithal}
نرمي نردًا غير مزيف وجوهه مرقمة من $1$ إلى $6$: $\Omega=\{1;2;3;4;5;6\}$.
لتكن الحادثة $A=\{2;4;6\}$ («الرقم زوجي») والحادثة $B=\{4;5;6\}$ («الرقم أكبر من أو يساوي $4$»).
\[P(A)=\frac36=\frac12,\quad P(A\cap B)=P(\{4;6\})=\frac26=\frac13,\quad P(A\cup B)=\frac12+\frac12-\frac13=\frac23 .\]
تحقق: $A\cup B=\{2;4;5;6\}$ و $\frac46=\frac23$ \ok.
\end{mithal}
\begin{molahada}
في نص التمرين: حرف «و» يعني التقاطع $\cap$، وحرف «أو» يعني الاتحاد $\cup$، و«على الأقل» تُحسب غالبًا بالحادثة العكسية.
\end{molahada}

\subsection{المتغير العشوائي وقانون احتماله}
\begin{tarif}[المتغير العشوائي]
\textbf{المتغير العشوائي} $X$ دالة تُرفق بكل نتيجة من $\Omega$ عددًا حقيقيًا. إذا كانت قيمه $x_1<x_2<\cdots<x_n$ فإن
\textbf{قانون احتماله} هو إعطاء كل احتمال $p_i=P(X=x_i)$، ونلخصه في جدول:
\begin{center}
\begin{tabular}{|C|c|c|c|c|C|}\hline
\rowcolor{cSec!12} $x_i$ & $x_1$ & $x_2$ & $\cdots$ & $x_n$ & المجموع\\\hline
$P(X=x_i)$ & $p_1$ & $p_2$ & $\cdots$ & $p_n$ & $1$\\\hline
\end{tabular}
\end{center}
مع $p_1+p_2+\cdots+p_n=1$ (للتحقق دائمًا).
\end{tarif}
\begin{khassia}[الأمل الرياضياتي، التباين، الانحراف المعياري]
\begin{itemize}
\item الأمل الرياضياتي: \quad $\bx{E(X)=\sum_{i=1}^{n}x_i\,p_i=x_1p_1+x_2p_2+\cdots+x_np_n}$
\item التباين: \quad $\bx{V(X)=\sum_{i=1}^{n}p_i\,\big(x_i-E(X)\big)^2=\sum_{i=1}^{n}p_i\,x_i^2-\big(E(X)\big)^2}$
\item الانحراف المعياري: \quad $\bx{\sigma(X)=\sqrt{V(X)}}$
\end{itemize}
ومن أجل كل عددين حقيقيين $a$ و$b$:
\[\bx{E(aX+b)=a\,E(X)+b}\qquad \bx{V(aX+b)=a^2\,V(X)}\qquad \bx{\sigma(aX+b)=|a|\,\sigma(X)}\]
\end{khassia}
\begin{molahada}[تفسير الأمل في لعبة]
$E(X)$ هو متوسط الربح الذي يأمله اللاعب بعد تكرار اللعبة مرات كثيرة:
\begin{itemize}
\item $E(X)=0$: اللعبة \textbf{عادلة}.
\item $E(X)>0$: اللعبة \textbf{مربحة} (في صالح اللاعب).
\item $E(X)<0$: اللعبة \textbf{ليست في صالح اللاعب}.
\end{itemize}
\end{molahada}

\begin{mithal}[قانون احتمال من سحب في آن واحد]
كيس به $4$ كريات بيضاء و$5$ حمراء، نسحب كريتين في آن واحد: $\mathrm{card}\,\Omega=C_9^2=36$.
$X$ عدد الكريات البيضاء المسحوبة، قيمه $0;1;2$:
\[P(X=0)=\frac{C_5^2}{36}=\frac{10}{36},\quad P(X=1)=\frac{C_4^1C_5^1}{36}=\frac{20}{36},\quad P(X=2)=\frac{C_4^2}{36}=\frac{6}{36}.\]
\begin{center}
\begin{tabular}{|C|c|c|c|C|}\hline
\rowcolor{cSec!12} $x_i$ & $0$ & $1$ & $2$ & المجموع\\\hline
$P(X=x_i)$ & $\frac{5}{18}$ & $\frac{5}{9}$ & $\frac{1}{6}$ & $1$\\\hline
\end{tabular}
\end{center}
\[E(X)=0\times\frac{10}{36}+1\times\frac{20}{36}+2\times\frac{6}{36}=\frac{32}{36}=\bx{\frac89}\]
\[V(X)=\Big(0^2\times\frac{10}{36}+1^2\times\frac{20}{36}+2^2\times\frac{6}{36}\Big)-\Big(\frac89\Big)^2=\frac{11}{9}-\frac{64}{81}=\bx{\frac{35}{81}}\ ,\qquad \sigma(X)=\bx{\frac{\sqrt{35}}{9}}\approx 0{,}66\]
\end{mithal}

\begin{mithal}[تعيين وسيط مجهول]
قانون احتمال متغير عشوائي $X$ حيث $a>0$ و$E(X)=1$:
\begin{center}
\begin{tabular}{|C|c|c|c|c|}\hline
\rowcolor{cSec!12} $x_i$ & $-1$ & $0$ & $a$ & $4$\\\hline
$P(X=x_i)$ & $0{,}2$ & $0{,}3$ & $0{,}4$ & $b$\\\hline
\end{tabular}
\end{center}
\begin{itemize}
\item مجموع الاحتمالات $1$: $0{,}2+0{,}3+0{,}4+b=1$ إذن $\bx{b=0{,}1}$.
\item $E(X)=-0{,}2+0+0{,}4a+0{,}4=1$ إذن $0{,}4a=0{,}8$ ومنه $\bx{a=2}$ (وفعلًا $0<2<4$).
\item $V(X)=(1\times0{,}2+0+4\times0{,}4+16\times0{,}1)-1^2=3{,}4-1=\bx{2{,}4}$ \quad و\quad $\sigma(X)=\sqrt{2{,}4}\approx1{,}55$.
\end{itemize}
\end{mithal}

\begin{mithal}[لعبة]
يحوي صندوق $10$ كريات: $5$ سوداء و$3$ خضراء و$2$ حمراء. يسحب لاعب كرية واحدة: يخسر $3$ دنانير إذا كانت سوداء، ويربح $2$ دينار إذا كانت خضراء، و$5$ دنانير إذا كانت حمراء. $X$ الربح الجبري:
\begin{center}
\begin{tabular}{|C|c|c|c|C|}\hline
\rowcolor{cSec!12} $x_i$ & $-3$ & $2$ & $5$ & المجموع\\\hline
$P(X=x_i)$ & $0{,}5$ & $0{,}3$ & $0{,}2$ & $1$\\\hline
\end{tabular}
\end{center}
\begin{itemize}
\item $E(X)=-1{,}5+0{,}6+1=\bx{0{,}1}$ \quad وبما أن $E(X)>0$ فاللعبة في صالح اللاعب.
\item $V(X)=(9\times0{,}5+4\times0{,}3+25\times0{,}2)-0{,}1^2=10{,}7-0{,}01=\bx{10{,}69}$ \quad و\quad $\sigma(X)\approx 3{,}27$.
\item إذا ضاعف المنظم الأرباح وأضاف دينارًا، نضع $Y=2X+1$:
\[E(Y)=2\times0{,}1+1=1{,}2\qquad V(Y)=2^2\times10{,}69=42{,}76\]
\end{itemize}
\end{mithal}

\section{الاحتمال الشرطي وشجرة الاحتمالات}
\subsection{الاحتمال الشرطي}
\begin{tarif}[الاحتمال الشرطي]
لتكن $A$ حادثة حيث $P(A)\neq0$. احتمال $B$ \textbf{علمًا أن} $A$ محققة هو العدد:
\[\bx{P_A(B)=\frac{P(A\cap B)}{P(A)}}\qquad\text{ومنه}\qquad \bx{P(A\cap B)=P(A)\times P_A(B)}\]
\end{tarif}
\begin{molahada}
\begin{itemize}
\item العبارات «علمًا أن»، «إذا كان»، «من بين $\ldots$» تدل على احتمال شرطي.
\item لا نخلط بين $P(A\cap B)$ (الحادثتان معًا) و$P_A(B)$ (نحسب داخل $A$ فقط).
\item $P_A(B)+P_A(\overline{B})=1$.
\end{itemize}
\end{molahada}
\begin{mithal}
نرمي نردًا غير مزيف. $A$: «الرقم زوجي» و$B$: «الرقم أكبر من أو يساوي $4$». لدينا $A\cap B=\{4;6\}$:
\[P_B(A)=\frac{P(A\cap B)}{P(B)}=\frac{\frac26}{\frac36}=\frac23 .\]
(من بين الأرقام $4;5;6$ يوجد رقمان زوجيان.)
\end{mithal}

\subsection{دستور الاحتمالات الكلية}
\begin{tarif}[تجزئة $\Omega$]
نقول إن الحوادث $A_1;A_2;\ldots;A_n$ تشكل \textbf{تجزئة} لـ $\Omega$ إذا كانت غير خالية، وغير متلائمة مثنى مثنى، واتحادها $\Omega$.
(مثال بسيط: $A$ و$\overline{A}$.)
\end{tarif}
\begin{khassia}[دستور الاحتمالات الكلية]
إذا كانت $A_1;A_2;\ldots;A_n$ تجزئة لـ $\Omega$ فإن من أجل كل حادثة $B$:
\[\bx{P(B)=P(A_1\cap B)+P(A_2\cap B)+\cdots+P(A_n\cap B)}\]
\[\bx{P(B)=P(A_1)\,P_{A_1}(B)+P(A_2)\,P_{A_2}(B)+\cdots+P(A_n)\,P_{A_n}(B)}\]
وفي حالة $A$ و$\overline{A}$: \quad $P(B)=P(A)\,P_A(B)+P(\overline{A})\,P_{\overline{A}}(B)$.
\end{khassia}

\begin{mithal}[مصنع بآلتين]
ينتج مصنع قطعًا بآلتين: الآلة $M_1$ تنتج $60\%$ من القطع و$3\%$ من إنتاجها معيب، والآلة $M_2$ تنتج الباقي و$5\%$ من إنتاجها معيب. نختار قطعة عشوائيًا. $D$: «القطعة معيبة».
\begin{center}\begin{LTR}\begin{tikzpicture}[ptree]

    child{node[ev,pr={$P(M_2\cap \overline{D})=0{,}38$}]{$\overline{D}$} edge from parent node[p]{$0{,}95$}}
    child{node[ev,pr={$P(M_2\cap D)=0{,}02$}]{$D$} edge from parent node[p]{$0{,}05$}}
    edge from parent node[p]{$0{,}4$}}
  child{node[ev]{$M_1$}
    child{node[ev,pr={$P(M_1\cap \overline{D})=0{,}582$}]{$\overline{D}$} edge from parent node[p]{$0{,}97$}}
    child{node[ev,pr={$P(M_1\cap D)=0{,}018$}]{$D$} edge from parent node[p]{$0{,}03$}}
    edge from parent node[p]{$0{,}6$}}
;

\item $P(\overline{D})=1-0{,}038=0{,}962$.
\end{itemize}
\end{mithal}

\section{الاستقلالية}
\begin{tarif}[حادثتان مستقلتان]
نقول إن الحادثتين $A$ و$B$ \textbf{مستقلتان} إذا كان:
\[\bx{P(A\cap B)=P(A)\times P(B)}\]
وإذا كان $P(A)\neq0$ فهذا يكافئ $P_A(B)=P(B)$: تحقق $A$ لا يغيّر احتمال $B$.
\end{tarif}
\begin{khassia}
إذا كانت $A$ و$B$ مستقلتين فإن: $A$ و$\overline{B}$ مستقلتان، و$\overline{A}$ و$B$ مستقلتان، و$\overline{A}$ و$\overline{B}$ مستقلتان.
\end{khassia}
\begin{molahada}
لا نخلط بين \textbf{مستقلتين} ($P(A\cap B)=P(A)P(B)$) و\textbf{غير متلائمتين} ($A\cap B=\varnothing$).
حادثتان غير متلائمتين واحتمال كل منهما غير معدوم ليستا مستقلتين أبدًا لأن $P(A\cap B)=0\neq P(A)P(B)$.
\end{molahada}
\begin{mithal}
نرمي نردًا غير مزيف، ونعتبر الحوادث: $A=\{2;4;6\}$ («رقم زوجي»)، و$B=\{1;2\}$، و$C=\{4;5;6\}$ («رقم أكبر من أو يساوي $4$»).
\begin{itemize}
\item $P(A\cap B)=P(\{2\})=\dfrac16$ و$P(A)\times P(B)=\dfrac12\times\dfrac13=\dfrac16$: إذن \bxt{$A$ و$B$ مستقلتان}.
\item $P(A\cap C)=P(\{4;6\})=\dfrac13$ و$P(A)\times P(C)=\dfrac12\times\dfrac12=\dfrac14\neq\dfrac13$: إذن \bxt{$A$ و$C$ غير مستقلتين}.
\end{itemize}
\end{mithal}
\begin{mithal}[تجارب مستقلة]
$30\%$ من تلاميذ ثانوية يملكون هاتفًا نقالًا. نختار $3$ تلاميذ عشوائيًا (الاختيارات مستقلة).
\begin{itemize}
\item احتمال أن يملك الثلاثة هاتفًا: $0{,}3^3=\bx{0{,}027}$.
\item احتمال أن يملك واحد على الأقل هاتفًا: $1-0{,}7^3=1-0{,}343=\bx{0{,}657}$.
\end{itemize}
\end{mithal}

\section{التجارب المتكررة والقانون الثنائي}
\begin{molahada}[للتنبيه]
هذا الجزء مقرر أساسًا في شعبة الرياضيات؛ تحتاجه الشعب الأخرى عند تكرار تجربة بشكل مستقل (سحب مع الإرجاع مثلًا).
\end{molahada}
\begin{tarif}[تجربة برنولي وقانون برنولي]
\textbf{تجربة برنولي} تجربة عشوائية لها مخرجان فقط: \textbf{نجاح} $S$ باحتمال $p$، و\textbf{فشل} $\overline{S}$ باحتمال $1-p$.
المتغير $X$ الذي يأخذ $1$ عند النجاح و$0$ عند الفشل يتبع \textbf{قانون برنولي} ذا الوسيط $p$:
\begin{center}
\begin{tabular}{|C|c|c|C|}\hline
\rowcolor{cSec!12} $x_i$ & $0$ & $1$ & المجموع\\\hline
$P(X=x_i)$ & $1-p$ & $p$ & $1$\\\hline
\end{tabular}
\qquad $E(X)=p$ \quad و\quad $V(X)=p(1-p)$.
\end{center}
\end{tarif}
\begin{khassia}[القانون الثنائي]
نكرر تجربة برنولي $n$ مرة بشكل \textbf{مستقل وفي نفس الظروف} (مخطط برنولي). $X$ عدد النجاحات، قيمه $0;1;\ldots;n$.
نقول إن $X$ يتبع \textbf{القانون الثنائي} ذا الوسيطين $n$ و$p$ ونكتب $X\sim\mathcal{B}(n;p)$:
\[\bx{P(X=k)=C_n^k\,p^k\,(1-p)^{n-k}}\qquad (0\leqslant k\leqslant n)\]
\[\bx{E(X)=np}\qquad \bx{V(X)=np(1-p)}\qquad \bx{\sigma(X)=\sqrt{np(1-p)}}\]
($C_n^k$ هو عدد المسارات في الشجرة التي تحوي $k$ نجاحًا بالضبط.)
\end{khassia}
