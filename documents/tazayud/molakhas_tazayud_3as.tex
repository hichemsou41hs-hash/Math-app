\section{التزايد المقارن للدالتين $x\mapsto e^x$ و $x\mapsto x$}
\begin{mithal}[نشاط]
$f$ الدالة المعرفة على $\R$ بـ: $f(x)=e^x-\dfrac{x^2}2$.
\QQ{1} ادرس اتجاه تغير $f$ على $[0;+\infty[$.
\QQ{2} احسب $f(0)$ ثم استنتج إشارة $f(x)$ على $[0;+\infty[$.
\QQ{3} استنتج أنه من أجل كل $x\in\R_+$: \ $e^x\geqslant\dfrac{x^2}2$.
\QQ{4} احسب $\Lm{+\infty}\dfrac{e^x}x$ و $\Lm{-\infty}xe^x$.
\par\medskip\textbf{الحل:}
\QQ{1} $f^{\prime}(x)=e^x-x$ ، ولدينا $e^x\geqslant x+1>x$ ، إذن $f^{\prime}(x)>0$: \bxt{$f$ متزايدة تمامًا على $[0;+\infty[$}.
\QQ{2} $f(0)=e^0-0=\bx{1}$. \ $f$ متزايدة تمامًا ، إذن من أجل $x\geqslant0$: $f(x)\geqslant f(0)=1>0$.
\QQ{3} $f(x)>0$ يعني $e^x-\dfrac{x^2}2>0$ ، أي \bx{e^x\geqslant\frac{x^2}2}.
\QQ{4} من أجل $x>0$ نقسم على $x$: \quad $\dfrac{e^x}x\geqslant\dfrac x2$
\LD{\dfrac{e^x}x}{+\infty}{+\infty} $\Car{\dfrac{e^x}x\geqslant\dfrac x2\\[2pt]\Lm{+\infty}\dfrac x2=+\infty}$ \quad (مبرهنة المقارنة)
\par نضع $X=-x$ ، فيكون $xe^x=-Xe^{-X}=-\dfrac X{e^X}$:
\LD{xe^x}{-\infty}{0} $\Car{\Lm{-\infty}X=+\infty\\[2pt]\displaystyle\lim_{X\to+\infty}\frac X{e^X}=0}$
\end{mithal}

\begin{khassia}[التزايد المقارن للدالتين $x\mapsto e^x$ و $x\mapsto x$]
\[\bx{\lim_{x\to+\infty}\frac{e^x}x=+\infty}\qquad\qquad\bx{\lim_{x\to-\infty}xe^x=0}\]
\end{khassia}

\begin{molahada}[التفسير البياني]
الدالتان $x\mapsto e^x$ و $x\mapsto x$ تؤولان إلى $+\infty$ لما $x\to+\infty$ ، لكن $e^x$ \textbf{أسرع بكثير}: المنحنى $(C)$ للدالة الأسية يبتعد عن المستقيم $y=x$ ويقع فوقه.
\begin{GR}{-3}{5}{-1}{6}{0.8}
\draw[cRem,very thick,domain=-3:5] plot(\x,\x) node[pos=0.95,right]{$y=x$};
\draw[cThm,very thick,domain=-3:1.79,samples=80] plot(\x,{exp(\x)});
\draw[cDef,very thick,domain=0:3.46,samples=80] plot(\x,{\x*\x/2});
\node[cThm] at (1.2,5.5){$y=e^x$}; \node[cDef] at (4.1,5.2){$y=\frac{x^2}2$};
\fill (0,1) circle(1.5pt) node[left]{$1$};
\end{GR}
\end{molahada}

\begin{mithal}[تطبيق]
\QQ{1} $\Lm{+\infty}\left(e^x-2x\right)$: حالة عدم تعيين من الشكل $+\infty-\infty$. نكتب: \ $e^x-2x=x\left(\dfrac{e^x}x-2\right)$
\LD{\left(e^x-2x\right)}{+\infty}{+\infty} $\Car{\Lm{+\infty}x=+\infty\\[2pt]\Lm{+\infty}\left(\dfrac{e^x}x-2\right)=+\infty}$
\QQ{2} $\Lm{-\infty}(2x+3)e^x$: حالة عدم تعيين من الشكل $\infty\times0$. نكتب: \ $(2x+3)e^x=2xe^x+3e^x$
\LD{(2x+3)e^x}{-\infty}{0} $\Car{\Lm{-\infty}xe^x=0\\[2pt]\Lm{-\infty}e^x=0}$
\end{mithal}

\section{التزايد المقارن للدالتين $x\mapsto\ln x$ و $x\mapsto x$}
\begin{khassia}[التزايد المقارن للدالتين $x\mapsto\ln x$ و $x\mapsto x$]
\[\bx{\lim_{x\to+\infty}\frac{\ln x}x=0}\qquad\qquad\bx{\lim_{\substack{x\to0\\x>0}}x\ln x=0}\]
\end{khassia}
\begin{BOX}{cSol}{البرهان}
\QQ{1} من أجل $x>0$ نضع $X=\ln x$ ، أي $x=e^X$ ، ولما $x\to+\infty$ فإن $X\to+\infty$:
\[\Lm{+\infty}\frac{\ln x}x=\lim_{X\to+\infty}\frac X{e^X}=\lim_{X\to+\infty}\frac1{\frac{e^X}X}=0\]
\QQ{2} نضع $X=\dfrac1x$ ، ولما $x\to0$ بقيم أكبر فإن $X\to+\infty$:
\[\Lmm{0}{x>0}x\ln x=\lim_{X\to+\infty}\frac1X\ln\left(\frac1X\right)=\lim_{X\to+\infty}\left(-\frac{\ln X}X\right)=0\]
\end{BOX}

\begin{molahada}[التفسير البياني]
لما $x\to+\infty$: $\ln x\to+\infty$ لكن \textbf{ببطء شديد} مقارنة بـ $x$: المنحنى $(C)$ للدالة $\ln$ يقع تحت المستقيم $y=x$ ويبتعد عنه.
\begin{GR}{-1}{7}{-2}{5}{0.75}
\draw[cRem,very thick,domain=-1:5] plot(\x,\x) node[pos=0.97,right]{$y=x$};
\draw[cThm,very thick,domain=0.14:7,samples=80] plot(\x,{ln(\x)});
\node[cThm] at (6,1.4){$y=\ln x$};
\fill (1,0) circle(1.5pt) node[below]{$1$};
\end{GR}
\end{molahada}

\begin{mithal}[تطبيق]
\QQ{1} $\Lm{+\infty}\left(x^2-\ln x\right)$: حالة عدم تعيين من الشكل $+\infty-\infty$. نكتب: \ $x^2-\ln x=x^2\left(1-\dfrac{\ln x}x\times\dfrac1x\right)$
\LD{\left(x^2-\ln x\right)}{+\infty}{+\infty} $\Car{\Lm{+\infty}x^2=+\infty\\[2pt]\Lm{+\infty}\dfrac{\ln x}x\times\dfrac1x=0}$
\QQ{2} $\Lmm{0}{x>0}\dfrac1{x\ln x}$: \ من أجل $0<x<1$ يكون $\ln x<0$ ، أي $x\ln x<0$.
\LDy{\dfrac1{x\ln x}}{0}{x>0}{-\infty} $\Car{\Lmm{0}{x>0}x\ln x=0^-}$
\end{mithal}

\section{التعميم: الدوال $x\mapsto x^n$}
\begin{khassia}[من أجل كل عدد طبيعي $n\geqslant1$]
\[\bx{\lim_{x\to+\infty}\frac{e^x}{x^n}=+\infty}\qquad\bx{\lim_{x\to-\infty}x^ne^x=0}\qquad\bx{\lim_{x\to+\infty}\frac{\ln x}{x^n}=0}\qquad\bx{\lim_{\substack{x\to0\\x>0}}x^n\ln x=0}\]
\end{khassia}
\begin{BOX}{cSol}{البرهان}
\QQ{1} $\dfrac{e^x}{x^n}=\left(\dfrac{e^{\frac xn}}{x}\right)^n=\dfrac1{n^n}\left(\dfrac{e^{\frac xn}}{\frac xn}\right)^n$ ، ولما $x\to+\infty$ فإن $X=\dfrac xn\to+\infty$ و $\dfrac{e^X}X\to+\infty$ ، إذن النهاية $+\infty$.
\QQ{2} نضع $X=-x$: \ $x^ne^x=(-1)^n\dfrac{X^n}{e^X}=(-1)^n\dfrac1{\frac{e^X}{X^n}}$ ، ولما $X\to+\infty$ فإن $\dfrac{e^X}{X^n}\to+\infty$ ، إذن النهاية $0$.
\QQ{3} $\dfrac{\ln x}{x^n}=\dfrac1n\times\dfrac{\ln\left(x^n\right)}{x^n}$ ، ولما $x\to+\infty$ فإن $X=x^n\to+\infty$ و $\dfrac{\ln X}X\to0$ ، إذن النهاية $0$.
\QQ{4} $x^n\ln x=\dfrac1n\times x^n\ln\left(x^n\right)$ ، ولما $x\to0^+$ فإن $X=x^n\to0^+$ و $X\ln X\to0$ ، إذن النهاية $0$.
\end{BOX}

\begin{tarif}[الخلاصة]
الدوال $x\mapsto\ln x$ ، $x\mapsto x^n$ ($n\in\N^*$) ، $x\mapsto e^x$ كلها تؤول إلى $+\infty$ لما $x\to+\infty$ ، لكن سلوكها مختلف عند اللانهاية:
\begin{center}\bxt{الدالة الأسية تتفوّق على الدالة قوة ، والدالة قوة تتفوّق على الدالة اللوغاريتمية النيبيرية}\end{center}
في حالة عدم التعيين تكون الغلبة للدالة الأسرع: \quad $\LW{\ln x\ \ll\ x^n\ \ll\ e^x}$ \ (بجوار $+\infty$).
\end{tarif}

\begin{molahada}[التفسير البياني]

\section{تطبيقات محلولة على النهايات}
\begin{tariqa}[إزالة حالة عدم التعيين]
\begin{enumerate}
\item نخرج الحد الأسرع (الغالب) عاملًا مشتركًا.
\item أو نكتب العبارة على شكل إحدى النهايات المرجعية ، أحيانًا بتغيير المتغير ($X=\dfrac1x$ ، $X=e^x$ ، $X=\sqrt x$ ، $X=ax$ ، $\ldots$).
\end{enumerate}
\end{tariqa}

\begin{tmr}
احسب النهايات التالية:
\QQ{1} (أ) $\Lm{+\infty}\dfrac{\ln(x+2)}x$ \qquad (ب) $\Lm{+\infty}\dfrac{\ln x}{\sqrt x}$\par \qquad\ (جـ) $\Lm{+\infty}x\ln\left(1+\dfrac1x\right)$ \qquad (د) $\Lm{-\infty}\dfrac{\ln\left(e^x+1\right)}{e^x}$
\QQ{2} (أ) $\Lm{+\infty}\dfrac{\ln\left(x^2+2\right)}x$ \qquad (ب) $\Lm{-\infty}\dfrac{\ln\left(x^2+2\right)}x$\par \qquad\ (جـ) $\Lm{0}\dfrac{\ln(4x+1)}x$ \qquad (د) $\Lm{0}\dfrac{e^{4x}-1}x$
\QQ{3} (أ) $\Lm{+\infty}\dfrac{e^{x+3}}x$ \qquad (ب) $\Lm{+\infty}\dfrac{e^{2x}}{x^2+3x}$\par \qquad\ (جـ) $\Lm{-\infty}\left(xe^x-x^2\right)$ \qquad (د) $\Lm{-\infty}\left[\left(x^2-3x\right)e^x-3x\right]$
\QQ{4} (أ) $\Lm{-\infty}\left(1+2xe^{4x-3}\right)$ \qquad (ب) $\Lm{+\infty}\left(x^2-2x+3\right)e^{-x}$\par \qquad\ (جـ) $\Lm{+\infty}\dfrac{\ln x+2x}{e^x+3}$ \qquad (د) $\Lm{+\infty}\dfrac{3e^{2x}-3e^x-3}{e^{3x}+4}$
\end{tmr}
\begin{hal}
\QQ{1} \textbf{(أ)} نكتب: \ $\dfrac{\ln(x+2)}x=\dfrac{\ln(x+2)}{x+2}\times\dfrac{x+2}x$
\LD{\dfrac{\ln(x+2)}x}{+\infty}{0} $\Car{\displaystyle\lim_{X\to+\infty}\frac{\ln X}X=0\quad(X=x+2)\\[2pt]\Lm{+\infty}\dfrac{x+2}x=1}$
\par\textbf{(ب)} نضع $X=\sqrt x$ ، فيكون $\ln x=\ln\left(X^2\right)=2\ln X$ ، و $\dfrac{\ln x}{\sqrt x}=2\times\dfrac{\ln X}X$:
\LD{\dfrac{\ln x}{\sqrt x}}{+\infty}{0} $\Car{\Lm{+\infty}\sqrt x=+\infty\\[2pt]\displaystyle\lim_{X\to+\infty}\frac{\ln X}X=0}$
\par\textbf{(جـ)} نضع $X=\dfrac1x$: \ $x\ln\left(1+\dfrac1x\right)=\dfrac{\ln(1+X)}X$
\LD{x\ln\left(1+\dfrac1x\right)}{+\infty}{1} $\Car{\Lm{+\infty}\dfrac1x=0\\[2pt]\displaystyle\lim_{X\to0}\frac{\ln(1+X)}X=1}$
\par\textbf{(د)} نضع $X=e^x$: \ $\dfrac{\ln\left(e^x+1\right)}{e^x}=\dfrac{\ln(1+X)}X$
\LD{\dfrac{\ln\left(e^x+1\right)}{e^x}}{-\infty}{1} $\Car{\Lm{-\infty}e^x=0\\[2pt]\displaystyle\lim_{X\to0}\frac{\ln(1+X)}X=1}$
\QQ{2} \textbf{(أ)} من أجل $x\neq0$: \ $\ln\left(x^2+2\right)=\ln\left[x^2\left(1+\dfrac2{x^2}\right)\right]=2\ln|x|+\ln\left(1+\dfrac2{x^2}\right)$ ، ومن أجل $x>0$:
\[\frac{\ln\left(x^2+2\right)}x=2\,\frac{\ln x}x+\frac1x\ln\left(1+\frac2{x^2}\right)\]
\LD{\dfrac{\ln\left(x^2+2\right)}x}{+\infty}{0} $\Car{\Lm{+\infty}\dfrac{\ln x}x=0\\[2pt]\Lm{+\infty}\dfrac1x\ln\left(1+\dfrac2{x^2}\right)=0}$
\par\textbf{(ب)} من أجل $x<0$: $|x|=-x$ ، و: \ $\dfrac{\ln\left(x^2+2\right)}x=-2\,\dfrac{\ln(-x)}{-x}+\dfrac1x\ln\left(1+\dfrac2{x^2}\right)$
\LD{\dfrac{\ln\left(x^2+2\right)}x}{-\infty}{0} $\Car{\displaystyle\lim_{X\to+\infty}\frac{\ln X}X=0\quad(X=-x)\\[2pt]\Lm{-\infty}\dfrac1x\ln\left(1+\dfrac2{x^2}\right)=0}$
\par\textbf{(جـ)} نكتب: \ $\dfrac{\ln(4x+1)}x=4\times\dfrac{\ln(1+4x)}{4x}$
\LD{\dfrac{\ln(4x+1)}x}{0}{4} $\Car{\Lm{0}4x=0\\[2pt]\displaystyle\lim_{X\to0}\frac{\ln(1+X)}X=1}$
\par\textbf{(د)} نكتب: \ $\dfrac{e^{4x}-1}x=4\times\dfrac{e^{4x}-1}{4x}$
\LD{\dfrac{e^{4x}-1}x}{0}{4} $\Car{\Lm{0}4x=0\\[2pt]\displaystyle\lim_{X\to0}\frac{e^X-1}X=1}$
\QQ{3} \textbf{(أ)} نكتب: \ $\dfrac{e^{x+3}}x=e^3\times\dfrac{e^x}x$
\LD{\dfrac{e^{x+3}}x}{+\infty}{+\infty} $\Car{\Lm{+\infty}\dfrac{e^x}x=+\infty\\[2pt]e^3>0}$
\par\textbf{(ب)} نكتب: \ $\dfrac{e^{2x}}{x^2+3x}=\left(\dfrac{e^x}x\right)^2\times\dfrac1{1+\frac3x}$
\LD{\dfrac{e^{2x}}{x^2+3x}}{+\infty}{+\infty} $\Car{\Lm{+\infty}\left(\dfrac{e^x}x\right)^2=+\infty\\[2pt]\Lm{+\infty}\dfrac1{1+\frac3x}=1}$
\par\textbf{(جـ)}
\LD{\left(xe^x-x^2\right)}{-\infty}{-\infty} $\Car{\Lm{-\infty}xe^x=0\\[2pt]\Lm{-\infty}\left(-x^2\right)=-\infty}$
\par\textbf{(د)} نكتب: \ $\left(x^2-3x\right)e^x-3x=x^2e^x-3xe^x-3x$
\LD{\left[\left(x^2-3x\right)e^x-3x\right]}{-\infty}{+\infty} $\Car{\Lm{-\infty}x^2e^x=\Lm{-\infty}xe^x=0\\[2pt]\Lm{-\infty}(-3x)=+\infty}$
\QQ{4} \textbf{(أ)} نكتب: \ $2xe^{4x-3}=\dfrac2{4e^3}\times(4x)e^{4x}=\dfrac1{2e^3}\times Xe^X$ \ مع $X=4x$:
\LD{\left(1+2xe^{4x-3}\right)}{-\infty}{1} $\Car{\Lm{-\infty}4x=-\infty\\[2pt]\displaystyle\lim_{X\to-\infty}Xe^X=0}$
\par\textbf{(ب)} نكتب: \ $\left(x^2-2x+3\right)e^{-x}=\dfrac{x^2}{e^x}-2\,\dfrac x{e^x}+\dfrac3{e^x}$
\LD{\left(x^2-2x+3\right)e^{-x}}{+\infty}{0} $\Car{\Lm{+\infty}\dfrac{x^2}{e^x}=\Lm{+\infty}\dfrac x{e^x}=0\\[2pt]\Lm{+\infty}\dfrac3{e^x}=0}$
\par\textbf{(جـ)} نكتب: \ $\dfrac{\ln x+2x}{e^x+3}=\dfrac x{e^x}\times\dfrac{\frac{\ln x}x+2}{1+3e^{-x}}$
\LD{\dfrac{\ln x+2x}{e^x+3}}{+\infty}{0} $\Car{\Lm{+\infty}\dfrac x{e^x}=0\\[2pt]\Lm{+\infty}\dfrac{\frac{\ln x}x+2}{1+3e^{-x}}=2}$
\par\textbf{(د)} نقسم البسط والمقام على $e^{3x}$: \ $\dfrac{3e^{2x}-3e^x-3}{e^{3x}+4}=\dfrac{3e^{-x}-3e^{-2x}-3e^{-3x}}{1+4e^{-3x}}$
\LD{\dfrac{3e^{2x}-3e^x-3}{e^{3x}+4}}{+\infty}{0} $\Car{\Lm{+\infty}e^{-x}=\Lm{+\infty}e^{-2x}=\Lm{+\infty}e^{-3x}=0}$
\end{hal}

\section{قوى عدد حقيقي موجب تمامًا}
\begin{molahada}[تمهيد]
ليكن $a$ عددًا حقيقيًا موجبًا تمامًا و $n$ عددًا طبيعيًا. نعلم أن $\ln\left(a^n\right)=n\ln a$ ، وبالتالي $a^n=e^{n\ln a}$.
\par وبما أن $\ln e=1$ فإنه من أجل كل عدد حقيقي $x$: \ $e^x=e^{x\ln e}$.
\end{molahada}
\begin{tarif}

\par يُقرأ $a^b$: «$a$ أس $b$» أو «$a$ قوة $b$».
\end{tarif}
\begin{mithal}
$2^{\sqrt2}=e^{\sqrt2\ln2}\approx\bx{2{,}665}$ \qquad ؛ \qquad $3^{\frac12}=e^{\frac12\ln3}=\sqrt3$ \qquad ؛ \qquad $\left(\dfrac12\right)^{-3}=e^{-3\ln\frac12}=e^{3\ln2}=8$.
\end{mithal}
\begin{khassia}[قواعد الحساب]
من أجل كل عددين حقيقيين موجبين تمامًا $a$ و $b$ ، ومن أجل كل عددين حقيقيين $x$ و $y$:
\[\LW{\ln\left(a^x\right)=x\ln a}\qquad\LW{a^xa^y=a^{x+y}}\qquad\LW{\frac1{a^y}=a^{-y}}\qquad\LW{\frac{a^x}{a^y}=a^{x-y}}\]
\[\LW{\left(a^x\right)^y=a^{xy}}\qquad\LW{(ab)^x=a^xb^x}\qquad\LW{\left(\frac ab\right)^x=\frac{a^x}{b^x}}\]
\end{khassia}

\section{معادلات ومتراجحات}
\begin{tariqa}

\item $a^x=b$ ($b>0$): نُدخل الدالة $\ln$ على الطرفين: $x\ln a=\ln b$ ، أي $x=\dfrac{\ln b}{\ln a}$ ($a\neq1$).

\end{khassia}

\section{تمرين محلول (مسألة)}
\begin{tmr}

\qc استنتج إشارة $h(x)$ على $[0;+\infty[$.
\end{sq}
\textbf{الجزء الثاني:} نعتبر الدالة $f$ المعرفة على $[0;+\infty[$ بـ: $f(x)=\dfrac{3^x-1}{3^x-x\ln3}$ ، و $(C_f)$ تمثيلها البياني في المستوي المنسوب إلى معلم متعامد ومتجانس $(O;\vec i,\vec j)$.

\begin{sq}
\qa بيّن أنه من أجل كل $x\geqslant0$: \ $f(x)=\dfrac{1-3^{-x}}{1-x\ln3\times3^{-x}}$.
\qb استنتج نهاية $f$ عند $+\infty$ ، ثم فسّر النتيجة هندسيًا.
\qc بيّن أنه من أجل كل $x\geqslant0$: \ $f^{\prime}(x)=\dfrac{h(x)\times\ln3}{\left(3^x-x\ln3\right)^2}$ ، ثم شكّل جدول تغيرات $f$.
\end{sq}
\QQ{2}
\begin{sq}
\qa بيّن أنه من أجل كل $x\geqslant0$: \ $f(x)-x\ln3=\dfrac{(1-x\ln3)\,g(x)}{3^x-x\ln3}$.
\qb استنتج الوضع النسبي للمنحنى $(C_f)$ والمستقيم $(D)$ ذي المعادلة $y=x\ln3$.
\end{sq}

\end{sq}
