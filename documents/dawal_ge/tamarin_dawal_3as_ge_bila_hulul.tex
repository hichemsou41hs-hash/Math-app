\section{تمارين على نمط اختبارات الفصل الأول}

\begin{tamrin}{التمرين 1: تحليل كثير حدود وحل متراجحة}
ليكن كثير الحدود $P$ المعرف على $\R$ بـ: $P(x)=x^3-2x^2-5x+6$.
\QQ{1} أحسب $P(1)$. ماذا تستنتج؟
\QQ{2} عيّن الأعداد الحقيقية $a$ ، $b$ و $c$ بحيث من أجل كل عدد حقيقي $x$: $P(x)=(x-1)\left(ax^2+bx+c\right)$.
\QQ{3} حلّل $P(x)$ إلى جداء ثلاثة عوامل من الدرجة الأولى.
\QQ{4} حل في $\R$ المعادلة $P(x)=0$ ، ثم المتراجحة $P(x)\leqslant0$.
\end{tamrin}

\begin{tamrin}{التمرين 2: نهايات ومستقيمات مقاربة}
أحسب النهايات التالية ، ثم فسّر النتائج بيانيًا عندما يكون ذلك ممكنًا:
\QQ{1} $f(x)=-3x^3+2x-7$: \quad $\Lm{-\infty}f(x)$ و $\Lm{+\infty}f(x)$.
\QQ{2} $g(x)=\dfrac{4x^2-1}{2x^2+x+3}$: \quad $\Lm{+\infty}g(x)$.
\QQ{3} $h(x)=\dfrac{2x+1}{x-3}$: \quad $\Lm{3^-}h(x)$ ، $\Lm{3^+}h(x)$ و $\Lm{\pm\infty}h(x)$.
\QQ{4} $k(x)=2x-1+\dfrac3{x+1}$: بيّن أن المستقيم $(\Delta):\ y=2x-1$ مقارب مائل لـ $(C_k)$ ، ثم ادرس الوضع النسبي.
\end{tamrin}

\begin{tamrin}{التمرين 3: حساب المشتقات}
أحسب الدالة المشتقة لكل دالة من الدوال التالية على المجال المعطى:
\QQ{1} $f_1(x)=2x^3-5x^2+4x-1$ على $\R$.
\QQ{2} $f_2(x)=\dfrac{3x-1}{x+2}$ على $]-2;+\infty[$.
\QQ{3} $f_3(x)=x+1-\dfrac4{x^2}$ على $]0;+\infty[$.
\QQ{4} $f_4(x)=\left(2x^2-3\right)^4$ على $\R$.
\QQ{5} $f_5(x)=\sqrt{x^2+3}$ على $\R$.
\end{tamrin}

\begin{tamrin}{التمرين 4: الدالة المساعدة ومبرهنة القيم المتوسطة}
لتكن $g$ الدالة المعرفة على $\R$ بـ: $g(x)=x^3-3x-3$.
\QQ{1} أحسب نهايتي $g$ عند $-\infty$ وعند $+\infty$.
\QQ{2} أدرس اتجاه تغير الدالة $g$ ، ثم شكّل جدول تغيراتها.
\QQ{3}
\begin{sq}
\qa بيّن أن المعادلة $g(x)=0$ تقبل حلًا وحيدًا $\alpha$ في $\R$.
\qb تحقق أن: $2{,}1<\alpha<2{,}11$.
\end{sq}
\QQ{4} استنتج إشارة $g(x)$ حسب قيم $x$.
\end{tamrin}

\begin{tamrin}{التمرين 5: دراسة دالة كثير حدود من الدرجة الثالثة}
لتكن $f$ الدالة المعرفة على $\R$ بـ: $f(x)=x^3-3x^2+4$ ، و $(C_f)$ تمثيلها البياني في معلم متعامد ومتجانس $(O;\vec i,\vec j)$.
\QQ{1} أحسب $\Lm{-\infty}f(x)$ و $\Lm{+\infty}f(x)$.
\QQ{2} أدرس اتجاه تغير الدالة $f$ ، ثم شكّل جدول تغيراتها.
\QQ{3} تحقق أنه من أجل كل عدد حقيقي $x$: $f(x)=(x+1)(x-2)^2$ ، ثم استنتج نقط تقاطع $(C_f)$ مع محور الفواصل.
\QQ{4}
\begin{sq}
\qa بيّن أن النقطة $\Omega(1;2)$ مركز تناظر للمنحنى $(C_f)$.
\qb بيّن أن $\Omega$ نقطة انعطاف لـ $(C_f)$.
\qc أكتب معادلة المماس $(T)$ لـ $(C_f)$ عند النقطة $\Omega$.
\end{sq}
\QQ{5} أرسم $(T)$ و $(C_f)$.
\QQ{6} ناقش بيانيًا ، حسب قيم الوسيط الحقيقي $m$ ، عدد حلول المعادلة: $f(x)=m$.
\end{tamrin}

\begin{tamrin}{التمرين 6: تعيين ثوابت دالة ناطقة ودراستها}
$f$ الدالة المعرفة على $\R-\{2\}$ بـ: $f(x)=ax+b+\dfrac c{x-2}$ حيث $a$ ، $b$ و $c$ أعداد حقيقية ، و $(C_f)$ تمثيلها البياني.
\QQ{1} عيّن $a$ ، $b$ و $c$ علمًا أن $(C_f)$ يشمل النقطة $B\left(0;-\dfrac32\right)$ ، ويقبل عند النقطة $A(3;3)$ مماسًا موازيًا لمحور الفواصل.
\QQ{2} نضع: $f(x)=x-1+\dfrac1{x-2}$.
\begin{sq}
\qa أحسب النهايات عند أطراف مجموعة التعريف ، وفسّر النتائج بيانيًا.
\qb بيّن أن $(\Delta):\ y=x-1$ مقارب مائل لـ $(C_f)$ ، وادرس الوضع النسبي.
\qc بيّن أن: $f'(x)=\dfrac{(x-1)(x-3)}{(x-2)^2}$ ، ثم شكّل جدول تغيرات $f$.
\qd بيّن أن النقطة $\Omega(2;1)$ مركز تناظر لـ $(C_f)$.
\end{sq}
\QQ{3} $g$ الدالة المعرفة على $\R-\{-2;2\}$ بـ: $g(x)=f(|x|)$. بيّن أن $g$ زوجية ، ثم اشرح كيف يمكن رسم $(C_g)$ انطلاقًا من $(C_f)$.
\end{tamrin}

\begin{tamrin}{التمرين 7: مسألة اقتصادية: الكلفة والربح}
تنتج مؤسسة يوميًا كمية $q$ من منتوج (بالأطنان) حيث $q\in[0;10]$. الكلفة الإجمالية (بآلاف الدنانير) هي: $C(q)=q^3-12q^2+60q$.
\QQ{1}
\begin{sq}
\qa أحسب الكلفة الهامشية $C_m(q)=C'(q)$ ، وبيّن أنها موجبة تمامًا. ماذا تستنتج بالنسبة لـ $C$ ؟
\qb أحسب الكلفة المتوسطة $C_M(q)=\dfrac{C(q)}q$ من أجل $q\in]0;10]$ ، ثم عيّن الكمية التي تكون من أجلها الكلفة المتوسطة أصغر ما يمكن.
\end{sq}
\QQ{2} يباع الطن الواحد بـ $39$ ألف دينار. نسمي $B(q)$ الربح الإجمالي.
\begin{sq}
\qa بيّن أن: $B(q)=-q^3+12q^2-21q$.
\qb أدرس اتجاه تغير $B$ على $[0;10]$ ، ثم عيّن الكمية التي تحقق أكبر ربح ، وما قيمة هذا الربح؟
\end{sq}
\end{tamrin}

\section{تمارين البكالوريا -- شعبة تسيير واقتصاد}

\begin{tamrin}{تمرين من بكالوريا 2022 -- الموضوع الأول}
$f$ الدالة العددية المعرفة على $D=\R-\{-2\}$ بـ: $f(x)=\dfrac{x^2+3x+3}{x+2}$ ، و $(C)$ تمثيلها البياني في معلم متعامد ومتجانس $(O;\vec i,\vec j)$.
\QQ{1}
\begin{sq}
\qa أحسب $\Lm{-2^-}f(x)$ و $\Lm{-2^+}f(x)$ ، ثم فسّر النتيجتين بيانيًا.
\qb أحسب $\Lm{-\infty}f(x)$ و $\Lm{+\infty}f(x)$.
\end{sq}
\QQ{2} بيّن أن المستقيم $(\Delta)$ ذا المعادلة $y=x+1$ مقارب مائل لـ $(C)$ ، ثم ادرس وضعية $(C)$ بالنسبة إلى $(\Delta)$.
\QQ{3} بيّن أن النقطة $A(-2;-1)$ مركز تناظر لـ $(C)$.
\QQ{4}
\begin{sq}
\qa بيّن أنه من أجل كل عدد حقيقي $x$ من $D$: $f'(x)=\dfrac{(x+3)(x+1)}{(x+2)^2}$.
\qb أدرس اتجاه تغير الدالة $f$ ، ثم شكّل جدول تغيراتها.
\end{sq}
\QQ{5} أكتب معادلة لـ $(T)$ مماس $(C)$ في النقطة ذات الفاصلة $0$.
\QQ{6} أنشئ $(\Delta)$ ، $(T)$ و $(C)$.
\QQ{7} $g$ الدالة العددية المعرفة على $\R-\{-2;2\}$ بـ: $g(x)=\dfrac{x^2-3|x|+3}{-|x|+2}$ ، و $(C_g)$ تمثيلها البياني في المعلم السابق.
\begin{sq}
\qa بيّن أن $g$ دالة زوجية ، ثم تحقق أنه من أجل كل $x$ من $]-\infty;-2[\cup]-2;0]$: $g(x)=f(x)$.
\qb اشرح كيف يمكن إنشاء $(C_g)$ انطلاقًا من $(C)$ ، ثم أنشئه.
\end{sq}
\end{tamrin}

\begin{tamrin}{تمرين من بكالوريا 2022 -- الموضوع الثاني}
$f$ الدالة المعرفة على $\R-\{1\}$ بـ: $f(x)=-x+1+\dfrac x{(x-1)^2}$ ، و $(C_f)$ تمثيلها البياني في المعلم المتعامد والمتجانس $(O;\vec i,\vec j)$.
