\section{جذر كثير حدود وتحليله}
\begin{tarif}[جذر كثير حدود]
ليكن $f$ كثير حدود درجته أكبر من أو تساوي $1$ ، و $\alpha$ عددًا حقيقيًا. نقول إن $\alpha$ \textbf{جذر} لكثير الحدود $f$ إذا كان: $\bx{f(\alpha)=0}$.
\end{tarif}
\begin{khassia}[مبرهنة التحليل]
إذا كان $\alpha$ جذرًا لكثير الحدود $f$ فإنه يوجد كثير حدود $g$ درجته أقل بواحد من درجة $f$ ، بحيث من أجل كل عدد حقيقي $x$: $\bx{f(x)=(x-\alpha)\,g(x)}$.
\end{khassia}

\begin{tamrin}{تحليل كثير حدود}
ليكن كثير الحدود: $f(x)=x^3-x^2-4x+4$.
\QQ{1} أحسب $f(2)$. ماذا تستنتج؟
\QQ{2} حلل $f(x)$ إلى جداء عوامل من الدرجة الأولى.
\end{tamrin}
\begin{hal}
\QQ{1} $f(2)=8-4-8+4=\bx{0}$ ، إذن $2$ جذر لـ $f$ ، ومنه $f(x)=(x-2)\,g(x)$ حيث $g(x)=ax^2+bx+c$.
\QQ{2} نعيّن $g$ بإحدى الطرق الثلاث:\par
\textbf{الطريقة 1: المطابقة.}
\[(x-2)(ax^2+bx+c)=ax^3+(b-2a)x^2+(c-2b)x-2c\]
بالمطابقة مع $f(x)$: $a=1$ ، $b-2a=-1$ ، $c-2b=-4$ ، $-2c=4$ ، ومنه $\bx{a=1\ ;\ b=1\ ;\ c=-2}$ ، أي $g(x)=x^2+x-2$.\par
\textbf{الطريقة 2: القسمة الإقليدية.} نقسم $x^3-x^2-4x+4$ على $x-2$:
\[\begin{array}{rrrr|l}
x^3&-x^2&-4x&+4&x-2\\\cline{5-5}
-x^3&+2x^2&&&x^2+x-2\\\cline{1-2}
&x^2&-4x&&\\
&-x^2&+2x&&\\\cline{2-3}
&&-2x&+4&\\
&&2x&-4&\\\cline{3-4}
&&&0&
\end{array}\]
\textbf{الطريقة 3: جدول هورنر.}

\textbf{تحقق جبري:} $f(x)=x+m\iff\dfrac{4x}{x^2+1}=m\iff mx^2-4x+m=0$. من أجل $m\neq0$: $\Delta=16-4m^2$ ، موجب تمامًا لما $-2<m<2$ ، ومعدوم لما $m=\pm2$ ، وسالب لما $|m|>2$. جداء الحلين $1>0$ ومجموعهما $\dfrac4m$ له إشارة $m$ \ok\par
\textbf{مناقشة بوسيط:} للمعادلة $f(x)=x+2m-1$ نضع $k=2m-1$:
\begin{itemize}
\item لا حلول لما $k>2$ أو $k<-2$ ، أي $\bx{m>\frac32}$ أو $\bx{m<-\frac12}$.
\item حل وحيد لما $k\in\{-2;0;2\}$ ، أي $\bx{m\in\left\{-\frac12;\frac12;\frac32\right\}}$.
\item حلان موجبان لما $\bx{\frac12<m<\frac32}$ ، وحلان سالبان لما $\bx{-\frac12<m<\frac12}$.
\end{itemize}
\end{mithal}

\begin{mithal}[مناقشة مائلة: مثال ثانٍ]
$f(x)=x+1+\dfrac1{x-1}$. ناقش حسب قيم $m$ عدد حلول $f(x)=x+m$.\par
الحلول هي فواصل نقط تقاطع $(C_f)$ مع $(\Delta_m):\ y=x+m$ الموازية للمقارب $(\Delta):\ y=x+1$. جبريًا:
\[x+1+\frac1{x-1}=x+m\ \Rightarrow\ \frac1{x-1}=m-1\]
\begin{itemize}
\item $m=1$: $(\Delta_m)$ هو المقارب $(\Delta)$: \bxt{لا توجد حلول}.
\item $m\neq1$: \bxt{حل وحيد} $x=1+\dfrac1{m-1}$: موجب إذا كان $m>1$ أو $m<0$ ، معدوم إذا كان $m=0$ ، وسالب إذا كان $0<m<1$.
\end{itemize}
\end{mithal}
