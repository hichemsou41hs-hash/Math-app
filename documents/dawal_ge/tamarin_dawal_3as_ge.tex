\section{تمارين على نمط اختبارات الفصل الأول}

\begin{tamrin}{التمرين 1: تحليل كثير حدود وحل متراجحة}
ليكن كثير الحدود $P$ المعرف على $\R$ بـ: $P(x)=x^3-2x^2-5x+6$.
\QQ{1} أحسب $P(1)$. ماذا تستنتج؟
\QQ{2} عيّن الأعداد الحقيقية $a$ ، $b$ و $c$ بحيث من أجل كل عدد حقيقي $x$: $P(x)=(x-1)\left(ax^2+bx+c\right)$.
\QQ{3} حلّل $P(x)$ إلى جداء ثلاثة عوامل من الدرجة الأولى.
\QQ{4} حل في $\R$ المعادلة $P(x)=0$ ، ثم المتراجحة $P(x)\leqslant0$.
\end{tamrin}
\begin{hal}
\QQ{1} $P(1)=1-2-5+6=\bx{0}$ ، إذن $1$ جذر لكثير الحدود $P$ ، ومنه $P(x)$ يقبل القسمة على $(x-1)$.
\QQ{2} ننشر: $(x-1)(ax^2+bx+c)=ax^3+(b-a)x^2+(c-b)x-c$. بالمطابقة مع $P(x)$:
\[a=1\ ;\ b-a=-2\ ;\ c-b=-5\ ;\ -c=6\ \Rightarrow\ \bx{a=1\ ;\ b=-1\ ;\ c=-6}\]
إذن: $P(x)=(x-1)\left(x^2-x-6\right)$.
\QQ{3} $x^2-x-6$: \quad $\Delta=1+24=25$ ، $x_1=\dfrac{1-5}2=-2$ ، $x_2=\dfrac{1+5}2=3$:
\[\bx{P(x)=(x-1)(x+2)(x-3)}\]
\QQ{4} $P(x)=0$ يعني $x=1$ أو $x=-2$ أو $x=3$: \quad $\bx{S=\{-2;1;3\}}$.

\end{hal}

\begin{tamrin}{التمرين 2: نهايات ومستقيمات مقاربة}
أحسب النهايات التالية ، ثم فسّر النتائج بيانيًا عندما يكون ذلك ممكنًا:
\QQ{1} $f(x)=-3x^3+2x-7$: \quad $\Lm{-\infty}f(x)$ و $\Lm{+\infty}f(x)$.
\QQ{2} $g(x)=\dfrac{4x^2-1}{2x^2+x+3}$: \quad $\Lm{+\infty}g(x)$.
\QQ{3} $h(x)=\dfrac{2x+1}{x-3}$: \quad $\Lm{3^-}h(x)$ ، $\Lm{3^+}h(x)$ و $\Lm{\pm\infty}h(x)$.
\QQ{4} $k(x)=2x-1+\dfrac3{x+1}$: بيّن أن المستقيم $(\Delta):\ y=2x-1$ مقارب مائل لـ $(C_k)$ ، ثم ادرس الوضع النسبي.
\end{tamrin}
\begin{hal}
\QQ{1} نهاية كثير حدود عند $\pm\infty$ هي نهاية حده الأعلى درجة:
\[\Lm{-\infty}f(x)=\Lm{-\infty}\left(-3x^3\right)=\bx{+\infty}\qquad;\qquad\Lm{+\infty}f(x)=\Lm{+\infty}\left(-3x^3\right)=\bx{-\infty}\]
\QQ{2} نهاية دالة ناطقة عند $\pm\infty$ هي نهاية حاصل قسمة الحدين الأعلى درجة:
\[\Lm{+\infty}g(x)=\Lm{+\infty}\frac{4x^2}{2x^2}=\bx{2}\]
إذن المستقيم $y=2$ مقارب أفقي لـ $(C_g)$ بجوار $+\infty$.
\QQ{3}
\LD{h(x)}{3^-}{-\infty} $\Car{\Lm{3}(2x+1)=7\\[6pt]\Lm{3^-}(x-3)=0^-}$\E
\LD{h(x)}{3^+}{+\infty} $\Car{\Lm{3}(2x+1)=7\\[6pt]\Lm{3^+}(x-3)=0^+}$\E
إذن المستقيم $x=3$ مقارب عمودي لـ $(C_h)$.
\[\Lm{\pm\infty}h(x)=\Lm{\pm\infty}\frac{2x}x=\bx{2}\]
إذن المستقيم $y=2$ مقارب أفقي لـ $(C_h)$.
\QQ{4} $k(x)-(2x-1)=\dfrac3{x+1}$ و $\Lm{\pm\infty}\dfrac3{x+1}=0$: \bxt{$(\Delta)$ مقارب مائل لـ $(C_k)$ بجوار $\pm\infty$}.\par

\begin{tamrin}{التمرين 3: حساب المشتقات}
أحسب الدالة المشتقة لكل دالة من الدوال التالية على المجال المعطى:
\QQ{1} $f_1(x)=2x^3-5x^2+4x-1$ على $\R$.
\QQ{2} $f_2(x)=\dfrac{3x-1}{x+2}$ على $]-2;+\infty[$.
\QQ{3} $f_3(x)=x+1-\dfrac4{x^2}$ على $]0;+\infty[$.
\QQ{4} $f_4(x)=\left(2x^2-3\right)^4$ على $\R$.
\QQ{5} $f_5(x)=\sqrt{x^2+3}$ على $\R$.
\end{tamrin}
\begin{hal}
\QQ{1} $f_1'(x)=\bx{6x^2-10x+4}$.
\QQ{2} $u=3x-1$ ، $v=x+2$ ، $u'=3$ ، $v'=1$:
\[f_2'(x)=\frac{3(x+2)-1\times(3x-1)}{(x+2)^2}=\frac{3x+6-3x+1}{(x+2)^2}=\bx{\frac7{(x+2)^2}}\]
\QQ{3} $\left(\dfrac1{x^2}\right)'=-\dfrac2{x^3}$ ، ومنه:
\[f_3'(x)=1-4\times\left(-\frac2{x^3}\right)=1+\frac8{x^3}=\bx{\frac{x^3+8}{x^3}}\]
\QQ{4} $\left(u^n\right)'=n\,u'\,u^{n-1}$ مع $u=2x^2-3$ و $u'=4x$:
\[f_4'(x)=4\times4x\left(2x^2-3\right)^3=\bx{16x\left(2x^2-3\right)^3}\]
\QQ{5} $\left(\sqrt u\right)'=\dfrac{u'}{2\sqrt u}$ ، و $x^2+3>0$:
\[f_5'(x)=\frac{2x}{2\sqrt{x^2+3}}=\bx{\frac x{\sqrt{x^2+3}}}\]
\end{hal}

\begin{tamrin}{التمرين 4: الدالة المساعدة ومبرهنة القيم المتوسطة}
لتكن $g$ الدالة المعرفة على $\R$ بـ: $g(x)=x^3-3x-3$.
\QQ{1} أحسب نهايتي $g$ عند $-\infty$ وعند $+\infty$.
\QQ{2} أدرس اتجاه تغير الدالة $g$ ، ثم شكّل جدول تغيراتها.
\QQ{3}
\begin{sq}
\qa بيّن أن المعادلة $g(x)=0$ تقبل حلًا وحيدًا $\alpha$ في $\R$.
\qb تحقق أن: $2{,}1<\alpha<2{,}11$.
\end{sq}
\QQ{4} استنتج إشارة $g(x)$ حسب قيم $x$.
\end{tamrin}
\begin{hal}
\QQ{1}
\[\Lm{+\infty}g(x)=\Lm{+\infty}x^3=\bx{+\infty}\qquad;\qquad\Lm{-\infty}g(x)=\Lm{-\infty}x^3=\bx{-\infty}\]
\QQ{2} $g'(x)=3x^2-3=3(x-1)(x+1)$:
\begin{itemize}
\item $g'(x)>0$ على $]-\infty;-1[$ وعلى $]1;+\infty[$: \bxt{$g$ متزايدة تمامًا على $]-\infty;-1]$ وعلى $[1;+\infty[$}.
\item $g'(x)<0$ على $]-1;1[$: \bxt{$g$ متناقصة تمامًا على $[-1;1]$}.
\end{itemize}
$g(-1)=-1+3-3=-1$ و $g(1)=1-3-3=-5$.

\begin{tamrin}{تمرين من بكالوريا 2017 (الدورة الاستثنائية) -- الموضوع الثاني}

\textbf{II)} $(\star)$ مؤسسة صناعية تنتج يوميًا كمية $q$ (مقدرة بالطن) من منتوج بكلفة متوسطة $C_M$ (مقدرة بملايين الدنانير) معرفة على $[0;10]$ بـ: $C_M(q)=\dfrac12q^2-q+1-\dfrac12\ln\left(q^2+1\right)$.
\QQ{1} بيّن أنه: من أجل كل عدد حقيقي $q$ من $[0;10]$: $C_M'(q)=\dfrac{g(q)}{q^2+1}$.

\begin{tamrin}{تمارين الاختيار من متعدد}
لكل سؤال جواب واحد فقط صحيح من بين الأجوبة الثلاثة المقترحة ، عيّنه مع التبرير.
\QQ{1} (بكالوريا 2021) $\Lm{+\infty}\left(\dfrac{-2x+1}{1+x+x^2}\right)$ هي:\ABC{$0$}{$-2$}{$1$}
\QQ{2} (بكالوريا 2021) الدالة $h$ المعرفة على $]0;+\infty[$ بـ: $h(x)=2x-1+\dfrac3x$ ، منحناها $(C)$ يقبل مستقيمًا مقاربًا مائلًا معادلته:\ABC{$y=2x-1$}{$y=2x$}{$y=2x+1$}
\QQ{3} (بكالوريا 2021) الدالة $g$ المعرفة على $]2;+\infty[$ بـ: $g(x)=1+\dfrac3{x-2}$. معادلة المماس لـ $(C_g)$ في النقطة ذات الفاصلة $3$ هي:\ABC{$y=3x-5$}{$y=-3x+13$}{$y=-3x+5$}
\QQ{4} (بكالوريا 2015) منحنى الدالة $f$ المعرفة على $\R$ بـ: $f(x)=\left(x^2-1\right)^3$ يقبل مماسًا في النقطة ذات الفاصلة $\sqrt2$ معادلته:\ABC{$y=\sqrt2x+1$}{$y=6\sqrt2x-11$}{$y=6\sqrt2x+1$}
\end{tamrin}
\begin{hal}
\QQ{1} دالة ناطقة: $\Lm{+\infty}\dfrac{-2x}{x^2}=\Lm{+\infty}\dfrac{-2}x=0$. \quad \bxt{الجواب (أ)}.
\QQ{2} $h(x)-(2x-1)=\dfrac3x$ و $\Lm{+\infty}\dfrac3x=0$ ، إذن المقارب المائل $y=2x-1$. \quad \bxt{الجواب (أ)}.
\QQ{3} $g(3)=1+3=4$ ، $g'(x)=-\dfrac3{(x-2)^2}$ ومنه $g'(3)=-3$. المماس: $y=-3(x-3)+4=-3x+13$. \quad \bxt{الجواب (ب)}.
\QQ{4} $f(\sqrt2)=(2-1)^3=1$ ، $f'(x)=3\times2x\left(x^2-1\right)^2=6x\left(x^2-1\right)^2$ ومنه $f'(\sqrt2)=6\sqrt2$.\par
المماس: $y=6\sqrt2\left(x-\sqrt2\right)+1=6\sqrt2x-12+1=6\sqrt2x-11$. \quad \bxt{الجواب (ب)}.
\end{hal}
