\section{الدوال $x\mapsto x^{n}$ حيث $n$ عدد طبيعي غير معدوم}
\begin{khassia}
الدالة $f_n:x\mapsto x^{n}$ معرفة وقابلة للاشتقاق على $\R$ و $f_n'(x)=nx^{n-1}$.
\begin{itemize}
\item إذا كان $n$ \textbf{زوجيًا}: $f_n$ زوجية، متناقصة تمامًا على $]-\infty;0]$ ومتزايدة تمامًا على $[0;+\infty[$، و $\lim\limits_{x\to-\infty}x^{n}=\lim\limits_{x\to+\infty}x^{n}=+\infty$.
\item إذا كان $n$ \textbf{فرديًا}: $f_n$ فردية ومتزايدة تمامًا على $\R$، و $\lim\limits_{x\to-\infty}x^{n}=-\infty$، $\lim\limits_{x\to+\infty}x^{n}=+\infty$.
\end{itemize}
كل المنحنيات تشمل النقطتين $O(0;0)$ و $(1;1)$. على $[0;1]$ لدينا $x^{n+1}\leqslant x^{n}$، وعلى $[1;+\infty[$ لدينا $x^{n+1}\geqslant x^{n}$.
\end{khassia}

\end{molahada}

\section{الجذر النوني لعدد حقيقي موجب}
\begin{tarif}
ليكن $n$ عددًا طبيعيًا حيث $n\geqslant2$. الدالة $x\mapsto x^{n}$ مستمرة ومتزايدة تمامًا على $[0;+\infty[$ وتأخذ قيمها في $[0;+\infty[$؛ إذن من أجل كل عدد حقيقي $a\geqslant0$ يوجد عدد حقيقي موجب وحيد $b$ يحقق $b^{n}=a$.\par
يسمى $b$ \textbf{الجذر النوني} للعدد $a$ ونرمز له $\sqrt[n]{a}$ أو $a^{\frac{1}{n}}$:
\begin{center}\bxt{من أجل $a\geqslant0$ و $b\geqslant0$: \quad $b=\sqrt[n]{a}\iff b^{n}=a$}\end{center}
\end{tarif}

\begin{mithal}
$\sqrt[3]{8}=2$ لأن $2^{3}=8$؛\quad $\sqrt[4]{81}=3$ لأن $3^{4}=81$؛\quad $\sqrt[5]{1}=1$؛\quad $\sqrt[2]{a}=\sqrt{a}$.
\end{mithal}

\begin{khassia}[قواعد الحساب]
من أجل كل عددين حقيقيين موجبين $a$ و $b$، ومن أجل كل عددين طبيعيين $n\geqslant2$ و $p\geqslant 1$:
\begin{align*}
\left(\sqrt[n]{a}\right)^{n}&=a & \sqrt[n]{a^{n}}&=a & \sqrt[n]{ab}&=\sqrt[n]{a}\times\sqrt[n]{b}\\
\sqrt[n]{\frac{a}{b}}&=\frac{\sqrt[n]{a}}{\sqrt[n]{b}}\ (b\neq0) & \sqrt[n]{a^{p}}&=\left(\sqrt[n]{a}\right)^{p} & \sqrt[n]{\sqrt[p]{a}}&=\sqrt[np]{a}
\end{align*}
ومن أجل $a>0$: \quad \bx{\sqrt[n]{a}=a^{\frac{1}{n}}=e^{\frac{1}{n}\ln a}} \quad و\quad $\ln\sqrt[n]{a}=\dfrac{1}{n}\ln a$.
\end{khassia}

\begin{mithal}[تبسيط]
\begin{itemize}
\item $\sqrt[3]{54}=\sqrt[3]{27\times2}=\sqrt[3]{27}\times\sqrt[3]{2}=\bx{3\sqrt[3]{2}}$.
\item $\dfrac{\sqrt[4]{48}}{\sqrt[4]{3}}=\sqrt[4]{\dfrac{48}{3}}=\sqrt[4]{16}=\bx{2}$.
\item $\sqrt[3]{\sqrt{64}}=\sqrt[6]{64}=\sqrt[6]{2^{6}}=\bx{2}$.
\end{itemize}
\end{mithal}

\begin{molahada}
\begin{itemize}
\item الجذر النوني يُعرَّف هنا للأعداد الموجبة فقط: نكتب $\sqrt[3]{8}$ ولا نكتب $\sqrt[3]{-8}$.
\item الدالة $x\mapsto\sqrt[n]{x}$ مستمرة ومتزايدة تمامًا على $[0;+\infty[$، ومنحناها نظير منحنى $x\mapsto x^{n}$ (على $[0;+\infty[$) بالنسبة إلى المستقيم $y=x$.
\end{itemize}
\end{molahada}

\section{حل المعادلة $x^{n}=a$}
\begin{khassia}
ليكن $n\geqslant2$ عددًا طبيعيًا و $a$ عددًا حقيقيًا.
\begin{center}
\begin{tabular}{|C|p{0.36\textwidth}|p{0.36\textwidth}|}\hline
\rowcolor{cSec!12}\textbf{الحالة} & \textbf{$n$ فردي} & \textbf{$n$ زوجي}\\\hline
$a>0$ & حل وحيد: $\sqrt[n]{a}$ & حلان: $\sqrt[n]{a}$ و $-\sqrt[n]{a}$\\\hline
$a=0$ & حل وحيد: $0$ & حل وحيد: $0$\\\hline
$a<0$ & حل وحيد: $-\sqrt[n]{-a}$ & لا توجد حلول\\\hline
\end{tabular}
\end{center}
\end{khassia}

\begin{mithal}
\begin{itemize}
\item $x^{3}=20$: حل وحيد \bx{x=\sqrt[3]{20}\approx2{,}71}.
\item $x^{4}=81$: \bx{x=3} أو \bx{x=-3}.
\item $x^{5}=-32$: حل وحيد \bx{x=-\sqrt[5]{32}=-2}.
\item $x^{6}=-5$: لا توجد حلول لأن $x^{6}\geqslant0$.
\end{itemize}
\end{mithal}

\section{دالة القوة $x\mapsto x^{\alpha}$ حيث $\alpha$ عدد حقيقي}
\begin{tarif}
ليكن $\alpha$ عددًا حقيقيًا. نسمي \textbf{دالة القوة ذات الأس $\alpha$} الدالة المعرفة على $]0;+\infty[$ بـ:
\[ \bx{x^{\alpha}=e^{\alpha\ln x}} \]
\end{tarif}

\begin{molahada}
هذا التعريف ينسجم مع ما سبق: إذا كان $\alpha=n$ طبيعيًا فإن $e^{n\ln x}=\left(e^{\ln x}\right)^{n}=x^{n}$، وإذا كان $\alpha=\frac{1}{n}$ فإن $x^{\frac{1}{n}}=\sqrt[n]{x}$. مثلًا: $x^{\frac{3}{2}}=x\sqrt{x}$ و $x^{-\frac{1}{2}}=\dfrac{1}{\sqrt{x}}$.
\end{molahada}

\begin{khassia}[قواعد الحساب]
من أجل كل عددين حقيقيين موجبين تمامًا $x$ و $y$، ومن أجل كل عددين حقيقيين $\alpha$ و $\beta$:
\begin{align*}
x^{\alpha}\times x^{\beta}&=x^{\alpha+\beta} & \frac{x^{\alpha}}{x^{\beta}}&=x^{\alpha-\beta} & x^{-\alpha}&=\frac{1}{x^{\alpha}}\\
\left(x^{\alpha}\right)^{\beta}&=x^{\alpha\beta} & (xy)^{\alpha}&=x^{\alpha}y^{\alpha} & \ln\left(x^{\alpha}\right)&=\alpha\ln x
\end{align*}
\end{khassia}

\begin{mithal}
\begin{itemize}
\item $8^{\frac{2}{3}}=\left(8^{\frac{1}{3}}\right)^{2}=2^{2}=\bx{4}$\qquad $16^{-\frac{3}{4}}=\dfrac{1}{\left(\sqrt[4]{16}\right)^{3}}=\bx{\dfrac{1}{8}}$.
\item $\dfrac{x^{\frac{1}{2}}\times x^{\frac{1}{3}}}{x^{\frac{1}{6}}}=x^{\frac{1}{2}+\frac{1}{3}-\frac{1}{6}}=\bx{x^{\frac{2}{3}}}$.
\item حل المعادلة $x^{\sqrt{2}}=5$ على $]0;+\infty[$: $\sqrt{2}\ln x=\ln5$ أي \bx{x=e^{\frac{\ln5}{\sqrt{2}}}=5^{\frac{1}{\sqrt{2}}}}.
\end{itemize}
\end{mithal}

\begin{khassia}[الاتجاه والنهايات]
لدينا $\left(x^{\alpha}\right)'=\left(e^{\alpha\ln x}\right)'=\dfrac{\alpha}{x}e^{\alpha\ln x}=\alpha x^{\alpha-1}$ على $]0;+\infty[$.
\begin{itemize}
\item $\alpha>0$: الدالة متزايدة تمامًا، $\lim\limits_{x\to0^{+}}x^{\alpha}=0$ و $\lim\limits_{x\to+\infty}x^{\alpha}=+\infty$.
\item $\alpha<0$: الدالة متناقصة تمامًا، $\lim\limits_{x\to0^{+}}x^{\alpha}=+\infty$ و $\lim\limits_{x\to+\infty}x^{\alpha}=0$.
\item $\alpha=0$: الدالة ثابتة ($x^{0}=1$).
\end{itemize}
\end{khassia}

\begin{molahada}
إذا كان $\alpha>0$ فإن $\lim\limits_{x\to0^{+}}x^{\alpha}=0$؛ لذلك نُمدِّد الدالة بالاستمرار عند $0$ بوضع $0^{\alpha}=0$.
\end{molahada}

\section{الاشتقاق}
\begin{khassia}
\begin{itemize}
\item من أجل كل عدد حقيقي $\alpha$ وكل $x>0$: \bx{\left(x^{\alpha}\right)'=\alpha x^{\alpha-1}}.
\item إذا كانت $u$ قابلة للاشتقاق وموجبة تمامًا على مجال $I$: \bx{\left(u^{\alpha}\right)'=\alpha\,u'\,u^{\alpha-1}}.
\item حالة خاصة ($\alpha=\frac{1}{n}$): من أجل $x>0$: \bx{\left(\sqrt[n]{x}\right)'=\dfrac{1}{n}x^{\frac{1}{n}-1}=\dfrac{1}{n\sqrt[n]{x^{n-1}}}}.
\end{itemize}
\end{khassia}

\begin{mithal}
\begin{itemize}
\item $f(x)=x^{\frac{5}{2}}$: $f'(x)=\dfrac{5}{2}x^{\frac{3}{2}}=\bx{\dfrac{5}{2}x\sqrt{x}}$.
\item $g(x)=\sqrt[3]{x}$ على $]0;+\infty[$: $g'(x)=\dfrac{1}{3}x^{-\frac{2}{3}}=\bx{\dfrac{1}{3\sqrt[3]{x^{2}}}}$.
\item $h(x)=\left(x^{2}+1\right)^{\frac{1}{3}}$ على $\R$: $u=x^{2}+1>0$ و $u'=2x$، إذن $h'(x)=\dfrac{1}{3}\times2x\times\left(x^{2}+1\right)^{-\frac{2}{3}}=\bx{\dfrac{2x}{3\sqrt[3]{\left(x^{2}+1\right)^{2}}}}$.
\end{itemize}
\end{mithal}

\section{التزايد المقارن}
\begin{khassia}[النهايات الأساسية]
من أجل كل عدد حقيقي $\alpha>0$:
\begin{align*}
&\lim_{x\to+\infty}\frac{e^{x}}{x^{\alpha}}=+\infty & &\lim_{x\to+\infty}x^{\alpha}e^{-x}=0\\
&\lim_{x\to+\infty}\frac{\ln x}{x^{\alpha}}=0 & &\lim_{x\to0^{+}}x^{\alpha}\ln x=0
\end{align*}
\end{khassia}

\begin{molahada}[ترتيب السرعة عند $+\infty$]
عند $+\infty$ تتغلب الدالة الأسية على كل دالة قوة، وتتغلب كل دالة قوة ذات أس موجب تمامًا على الدالة $\ln$:
\begin{center}\bxt{$\ln x$ \ \ أبطأ من \ \ $x^{\alpha}$ \ \ أبطأ من \ \ $e^{x}$}\end{center}
مثلًا: $\sqrt{x}$ (أي $x^{0,5}$) تؤول إلى $+\infty$ أسرع من $\ln x$، و $e^{x}$ أسرع من $x^{100}$.
\end{molahada}

\begin{tariqa}[إزالة حالة عدم تعيين]
نخرج الحد الأسرع (الغالب) عاملًا مشتركًا، أو نكتب العبارة على شكل إحدى النهايات الأساسية (أحيانًا بتغيير المتغير).
\end{tariqa}

\begin{mithal}
\begin{itemize}
\item $\lim\limits_{x\to+\infty}\dfrac{e^{x}}{x^{3}}=\bx{+\infty}$\quad و\quad $\lim\limits_{x\to+\infty}x^{5}e^{-x}=\bx{0}$ (حالة خاصة $\alpha=3$ و $\alpha=5$).
\item $\lim\limits_{x\to+\infty}\left(\sqrt{x}-\ln x\right)$: حالة $\infty-\infty$. نكتب $\sqrt{x}-\ln x=\sqrt{x}\left(1-\dfrac{\ln x}{x^{\frac{1}{2}}}\right)$، و $\dfrac{\ln x}{x^{\frac{1}{2}}}\to0$، إذن النهاية \bx{+\infty}.
\item $\lim\limits_{x\to0^{+}}\sqrt[3]{x}\,\ln x=\lim\limits_{x\to0^{+}}x^{\frac{1}{3}}\ln x=\bx{0}$.
\item $\lim\limits_{x\to+\infty}\left(e^{x}-x^{4}\right)=\lim\limits_{x\to+\infty}e^{x}\left(1-x^{4}e^{-x}\right)=\bx{+\infty}$.
\end{itemize}
\end{mithal}

\section{مسائل الاستمثال}
\begin{tariqa}[حل مسألة استمثال]
\begin{enumerate}
\item نختار المتغير $x$ ونعيّن المجال $I$ الذي ينتمي إليه (حسب معطيات المسألة).
\item نعبّر عن المقدار المطلوب جعله أكبر ما يمكن (أو أصغر ما يمكن) بدلالة $x$: نحصل على دالة $f$.
\item ندرس اتجاه تغيّر $f$ على $I$ (إشارة $f'$) ونعيّن القيمة الحدية العظمى (أو الصغرى).
\item نجيب عن السؤال بجملة مع الوحدة.
\end{enumerate}
\end{tariqa}

\begin{mithal}[علبة بدون غطاء]
من صفيحة مربعة طول ضلعها $12\,\text{cm}$ نقطع من كل ركن مربعًا طول ضلعه $x$ ثم نطوي الحواف لنحصل على علبة. ما قيمة $x$ التي تجعل حجم العلبة أكبر ما يمكن؟\par

$V'(x)=(12-2x)^{2}-4x(12-2x)=(12-2x)(12-6x)=12(6-x)(2-x)$.\par
على $]0;6[$ لدينا $6-x>0$، فإشارة $V'(x)$ هي إشارة $2-x$:

\begin{mithal}[علبة أسطوانية سعتها لتر]
نريد صنع علبة أسطوانية مغلقة حجمها $1000\,\text{cm}^{3}$ بأقل كمية من المعدن. نرمز بـ $r$ لنصف قطر القاعدة وبـ $h$ للارتفاع.\par

$S'(r)=4\pi r-\dfrac{2000}{r^{2}}=\dfrac{4\left(\pi r^{3}-500\right)}{r^{2}}$، وهي تنعدم من أجل $r^{3}=\dfrac{500}{\pi}$ أي $r_0=\sqrt[3]{\dfrac{500}{\pi}}\approx5{,}42$.

\section{تمرين محلول: دراسة الدالة $x\mapsto x^{x}$}
\begin{tmr}
$f$ الدالة المعرفة على $]0;+\infty[$ بـ $f(x)=x^{x}$، و $(C_f)$ تمثيلها البياني.
\QQ{1} اكتب $f(x)$ بدلالة الدالتين $\exp$ و $\ln$، ثم احسب نهايتي $f$ عند $0$ وعند $+\infty$.
\QQ{2} ادرس اتجاه تغيّر $f$ وشكّل جدول تغيّراتها.
\QQ{3} أنشئ $(C_f)$.
\end{tmr}
\begin{hal}
\QQ{1} $f(x)=e^{x\ln x}$.\par
عند $0$: $\lim\limits_{x\to0^{+}}x\ln x=0$ (تزايد مقارن)، إذن \bx{\lim_{x\to0^{+}}f(x)=e^{0}=1}.\par
عند $+\infty$: $\lim\limits_{x\to+\infty}x\ln x=+\infty$، إذن \bx{\lim_{x\to+\infty}f(x)=+\infty}.
\QQ{2} $f'(x)=\left(x\ln x\right)'e^{x\ln x}=\left(\ln x+1\right)x^{x}$. بما أن $x^{x}>0$ فإشارة $f'(x)$ هي إشارة $\ln x+1$:
$\ln x+1>0\iff x>e^{-1}$.
