\section{دراسة دوال}
\begin{tamrin}{الدالة $x\mapsto x-3\sqrt[3]{x}$}
$f$ الدالة المعرفة على $[0;+\infty[$ بـ $f(x)=x-3\sqrt[3]{x}$، و $(C_f)$ تمثيلها البياني.
\QQ{1} احسب $\lim\limits_{x\to+\infty}f(x)$.
\QQ{2} بيّن أنه من أجل $x>0$: $f'(x)=\dfrac{\sqrt[3]{x^{2}}-1}{\sqrt[3]{x^{2}}}$، ثم شكّل جدول تغيّرات $f$.
\QQ{3} حل المعادلة $f(x)=0$ واستنتج إشارة $f(x)$، ثم أنشئ $(C_f)$.
\end{tamrin}

\begin{tamrin}{الدالة $x\mapsto\dfrac{\ln x}{\sqrt{x}}$}
$f$ الدالة المعرفة على $]0;+\infty[$ بـ $f(x)=\dfrac{\ln x}{\sqrt{x}}$.
\QQ{1} احسب نهايتي $f$ عند $0$ وعند $+\infty$ وفسّرهما بيانيًا.
\QQ{2} بيّن أن $f'(x)=\dfrac{2-\ln x}{2x\sqrt{x}}$ وشكّل جدول التغيّرات.
\QQ{3} استنتج أنه من أجل كل $x>0$: $\ln x\leqslant\dfrac{2}{e}\sqrt{x}$.
\end{tamrin}
