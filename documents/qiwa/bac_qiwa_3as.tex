\section{الحساب على الجذور والقوى}
\begin{tamrin}{اختر الإجابة الصحيحة مع التعليل}
\QQ{1} العدد $\dfrac{\sqrt[3]{54}}{\sqrt[3]{2}}$ يساوي: \ABC{$3$}{$9$}{$\sqrt[3]{52}$}
\QQ{2} مجموعة حلول المعادلة $x^{4}=16$ في $\R$ هي: \ABC{$\{2\}$}{$\{-2;2\}$}{$\varnothing$}
\QQ{3} العدد $8^{\frac{2}{3}}\times4^{-\frac{1}{2}}$ يساوي: \ABC{$\frac{1}{2}$}{$4$}{$2$}
\QQ{4} من أجل كل $x>0$: $\dfrac{x^{\frac{1}{2}}\times x^{\frac{1}{3}}}{x^{\frac{5}{6}}}$ يساوي: \ABC{$x$}{$1$}{$x^{\frac{1}{6}}$}
\QQ{5} حل المعادلة $x^{\frac{3}{2}}=27$ في $]0;+\infty[$ هو: \ABC{$9$}{$3$}{$18$}
\QQ{6} المقارنة الصحيحة هي: \ABC{$\sqrt[3]{3}<\sqrt{2}$}{$\sqrt[3]{3}=\sqrt{2}$}{$\sqrt[3]{3}>\sqrt{2}$}
\end{tamrin}
\begin{hal}
\QQ{1} \textbf{(أ)}: $\dfrac{\sqrt[3]{54}}{\sqrt[3]{2}}=\sqrt[3]{\dfrac{54}{2}}=\sqrt[3]{27}=\bx{3}$.
\QQ{2} \textbf{(ب)}: $4$ زوجي و $16>0$، إذن $x=\sqrt[4]{16}=2$ أو $x=-2$: \bx{\{-2;2\}}.
\QQ{3} \textbf{(جـ)}: $8^{\frac{2}{3}}=\left(\sqrt[3]{8}\right)^{2}=4$ و $4^{-\frac{1}{2}}=\dfrac{1}{\sqrt{4}}=\dfrac{1}{2}$، إذن الجداء \bx{2}.
\QQ{4} \textbf{(ب)}: $x^{\frac{1}{2}+\frac{1}{3}-\frac{5}{6}}=x^{\frac{3+2-5}{6}}=x^{0}=\bx{1}$.
\QQ{5} \textbf{(أ)}: $x=27^{\frac{2}{3}}=\left(\sqrt[3]{27}\right)^{2}=\bx{9}$.
\QQ{6} \textbf{(جـ)}: $\left(\sqrt[3]{3}\right)^{6}=9>8=\left(\sqrt{2}\right)^{6}$، والدالة $x\mapsto x^{6}$ متزايدة تمامًا على $[0;+\infty[$، إذن \bx{\sqrt[3]{3}>\sqrt{2}}.
\end{hal}

\section{الاشتقاق والنهايات}
\begin{tamrin}{اختر الإجابة الصحيحة مع التعليل}
\QQ{1} مشتقة الدالة $x\mapsto x^{\sqrt{2}}$ على $]0;+\infty[$ هي: \ABC{$x\mapsto x^{\sqrt{2}-1}$}{$x\mapsto\sqrt{2}\,x^{\sqrt{2}-1}$}{$x\mapsto\sqrt{2}\,x^{\sqrt{2}}$}
\QQ{2} مشتقة الدالة $x\mapsto\sqrt[3]{x^{2}+1}$ على $\R$ هي: \ABC{$x\mapsto\dfrac{2x}{3\sqrt[3]{\left(x^{2}+1\right)^{2}}}$}{$x\mapsto\dfrac{1}{3\sqrt[3]{\left(x^{2}+1\right)^{2}}}$}{$x\mapsto\dfrac{2x}{3\sqrt[3]{x^{2}+1}}$}
\QQ{3} $\lim\limits_{x\to+\infty}x^{3}e^{-x}$ تساوي: \ABC{$+\infty$}{$0$}{$1$}
\QQ{4} $\lim\limits_{x\to+\infty}\dfrac{\ln x}{x^{0,1}}$ تساوي: \ABC{$+\infty$}{$1$}{$0$}
\QQ{5} $\lim\limits_{x\to0^{+}}\sqrt{x}\,\ln x$ تساوي: \ABC{$-\infty$}{$0$}{$1$}
\QQ{6} $\lim\limits_{x\to+\infty}\left(e^{x}-x^{10}\right)$ تساوي: \ABC{$-\infty$}{$0$}{$+\infty$}
\end{tamrin}
\begin{hal}
\QQ{1} \textbf{(ب)}: $\left(x^{\alpha}\right)'=\alpha x^{\alpha-1}$ مع $\alpha=\sqrt{2}$: \bx{\sqrt{2}\,x^{\sqrt{2}-1}}.
\QQ{2} \textbf{(أ)}: $u(x)=x^{2}+1>0$، $u'(x)=2x$، و $\left(u^{\frac{1}{3}}\right)'=\dfrac{1}{3}u'u^{-\frac{2}{3}}=\bx{\dfrac{2x}{3\sqrt[3]{\left(x^{2}+1\right)^{2}}}}$.
\QQ{3} \textbf{(ب)}: $\lim\limits_{x\to+\infty}x^{\alpha}e^{-x}=0$ من أجل $\alpha>0$ (تزايد مقارن)، هنا $\alpha=3$: \bx{0}.
\QQ{4} \textbf{(جـ)}: $\lim\limits_{x\to+\infty}\dfrac{\ln x}{x^{\alpha}}=0$ من أجل كل $\alpha>0$ مهما كان صغيرًا، هنا $\alpha=0{,}1$: \bx{0}.
\QQ{5} \textbf{(ب)}: $\sqrt{x}\ln x=x^{\frac{1}{2}}\ln x$ و $\lim\limits_{x\to0^{+}}x^{\alpha}\ln x=0$ ($\alpha>0$): \bx{0}.
\QQ{6} \textbf{(جـ)}: $e^{x}-x^{10}=e^{x}\left(1-x^{10}e^{-x}\right)$ و $x^{10}e^{-x}\to0$، إذن النهاية \bx{+\infty}.
\end{hal}

\section{اتجاه التغيّر والاستمثال}
\begin{tamrin}{اختر الإجابة الصحيحة مع التعليل}
\QQ{1} الدالة $x\mapsto x^{-\frac{2}{3}}$ على $]0;+\infty[$: \ABC{متزايدة تمامًا}{متناقصة تمامًا}{ثابتة}
\QQ{2} القيمة الحدية الصغرى للدالة $x\mapsto x-2\sqrt{x}$ على $[0;+\infty[$ هي: \ABC{$-1$}{$0$}{$-2$}
\QQ{3} القيمة الحدية الصغرى للدالة $x\mapsto x^{x}$ على $]0;+\infty[$ هي: \ABC{$\frac{1}{e}$}{$e^{-\frac{1}{e}}$}{$1$}
\QQ{4} من صفيحة مربعة ضلعها $30\,\text{cm}$ نقطع من كل ركن مربعًا ضلعه $x$ ونطوي الحواف لنصنع علبة بدون غطاء. يكون حجمها أكبر ما يمكن من أجل: \ABC{$x=5$}{$x=10$}{$x=7{,}5$}
\QQ{5} من بين المستطيلات التي مساحتها $100\,\text{m}^{2}$، المستطيل ذو المحيط الأصغر بُعداه: \ABC{$20$ و $5$}{$25$ و $4$}{$10$ و $10$}
\end{tamrin}
\begin{hal}
\QQ{1} \textbf{(ب)}: المشتقة $-\dfrac{2}{3}x^{-\frac{5}{3}}<0$ (الأس $\alpha=-\frac{2}{3}<0$): الدالة \bxt{متناقصة تمامًا}.
\QQ{2} \textbf{(أ)}: $f(x)=x-2\sqrt{x}$، $f'(x)=1-\dfrac{1}{\sqrt{x}}=\dfrac{\sqrt{x}-1}{\sqrt{x}}$ ($x>0$): سالبة على $]0;1[$ وموجبة على $]1;+\infty[$، إذن القيمة الصغرى $f(1)=1-2=\bx{-1}$.
\QQ{3} \textbf{(ب)}: $x^{x}=e^{x\ln x}$، مشتقتها $(\ln x+1)x^{x}$ تنعدم وتغيّر إشارتها عند $x=\frac{1}{e}$، والقيمة الصغرى $e^{\frac{1}{e}\ln\frac{1}{e}}=\bx{e^{-\frac{1}{e}}}$ (انظر الملخص).
\QQ{4} \textbf{(أ)}: $V(x)=x(30-2x)^{2}$ مع $x\in]0;15[$. $V'(x)=(30-2x)(30-6x)=12(15-x)(5-x)$، موجبة على $]0;5[$ وسالبة على $]5;15[$: الحجم أعظمي عند \bx{x=5} وقيمته $V(5)=5\times20^{2}=2000\,\text{cm}^{3}$.
\QQ{5} \textbf{(جـ)}: البعدان $x$ و $\dfrac{100}{x}$، المحيط $P(x)=2x+\dfrac{200}{x}$. $P'(x)=2-\dfrac{200}{x^{2}}=\dfrac{2(x-10)(x+10)}{x^{2}}$: سالبة على $]0;10[$ وموجبة على $]10;+\infty[$، إذن المحيط أصغري عند $x=10$: \bxt{مربع بُعداه $10$ و $10$} ومحيطه $40\,\text{m}$.
\end{hal}
