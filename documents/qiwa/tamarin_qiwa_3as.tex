\section{الحساب على الجذور والقوى}
\begin{tamrin}{تبسيط}
بسّط ما يلي:\quad
$A=\sqrt[3]{16}\times\sqrt[3]{4}$،\quad $B=\dfrac{\sqrt[4]{32}}{\sqrt[4]{2}}$،\quad $C=\sqrt{\sqrt[3]{729}}$،\quad $D=27^{\frac{2}{3}}$،\quad $E=32^{-\frac{3}{5}}$،\quad $F=\left(\sqrt[5]{9}\right)^{\frac{5}{2}}$.
\end{tamrin}
\begin{hal}
\begin{itemize}
\item $A=\sqrt[3]{16\times4}=\sqrt[3]{64}=\sqrt[3]{4^{3}}=\bx{4}$.
\item $B=\sqrt[4]{\dfrac{32}{2}}=\sqrt[4]{16}=\sqrt[4]{2^{4}}=\bx{2}$.
\item $C=\sqrt[6]{729}=\sqrt[6]{3^{6}}=\bx{3}$.
\item $D=\left(27^{\frac{1}{3}}\right)^{2}=3^{2}=\bx{9}$.
\item $E=\left(2^{5}\right)^{-\frac{3}{5}}=2^{-3}=\bx{\dfrac{1}{8}}$.
\item $F=\left(9^{\frac{1}{5}}\right)^{\frac{5}{2}}=9^{\frac{1}{2}}=\bx{3}$.
\end{itemize}
\end{hal}

\begin{tamrin}{مقارنة جذور}
قارن بين العددين في كل حالة: \quad (1) $\sqrt[3]{3}$ و $\sqrt{2}$\qquad (2) $\sqrt[4]{5}$ و $\sqrt[3]{3}$.
\end{tamrin}
\begin{hal}
\begin{tariqa}
العددان موجبان، والدالة $x\mapsto x^{k}$ متزايدة تمامًا على $[0;+\infty[$: نرفعهما إلى قوة تُزيل الجذرين (المضاعف المشترك الأصغر للرتبتين) ثم نقارن.
\end{tariqa}
\QQ{1} $\left(\sqrt[3]{3}\right)^{6}=3^{2}=9$ و $\left(\sqrt{2}\right)^{6}=2^{3}=8$. بما أن $9>8$ فإن \bx{\sqrt[3]{3}>\sqrt{2}}.
\QQ{2} $\left(\sqrt[4]{5}\right)^{12}=5^{3}=125$ و $\left(\sqrt[3]{3}\right)^{12}=3^{4}=81$. إذن \bx{\sqrt[4]{5}>\sqrt[3]{3}}.
\end{hal}

\section{المعادلات}
\begin{tamrin}{معادلات من الشكل $x^{n}=a$}
حل في $\R$ المعادلات:\quad
(1) $x^{3}=-27$\quad (2) $x^{4}=16$\quad (3) $x^{5}=7$\quad (4) $(x-1)^{3}=8$\quad (5) $2x^{4}-162=0$\quad (6) $x^{6}+64=0$.
\end{tamrin}
\begin{hal}
\QQ{1} $n=3$ فردي و $-27<0$: حل وحيد $x=-\sqrt[3]{27}=\bx{-3}$.
\QQ{2} $n=4$ زوجي و $16>0$: $x=\sqrt[4]{16}=2$ أو $x=-2$؛ \bx{S=\{-2;2\}}.
\QQ{3} حل وحيد \bx{x=\sqrt[5]{7}\approx1{,}48}.
\QQ{4} $x-1=\sqrt[3]{8}=2$ إذن \bx{x=3}.
\QQ{5} $x^{4}=81$ إذن \bx{S=\{-3;3\}}.
\QQ{6} $x^{6}=-64<0$ و $6$ زوجي: \bx{S=\varnothing}.
\end{hal}

\begin{tamrin}{معادلات بأس حقيقي}
حل في $]0;+\infty[$ المعادلات:\quad (1) $x^{\frac{3}{2}}=8$\qquad (2) $x^{\sqrt{3}}=e$\qquad (3) $x^{\frac{2}{3}}-3x^{\frac{1}{3}}+2=0$.
\end{tamrin}
\begin{hal}
\QQ{1} نرفع الطرفين إلى القوة $\frac{2}{3}$: $x=8^{\frac{2}{3}}=\left(\sqrt[3]{8}\right)^{2}=\bx{4}$. \textit{تحقق:} $4^{\frac{3}{2}}=\left(\sqrt{4}\right)^{3}=8$ \ok
\QQ{2} $\sqrt{3}\ln x=\ln e=1$ إذن $\ln x=\dfrac{1}{\sqrt{3}}$ أي \bx{x=e^{\frac{1}{\sqrt{3}}}}.
\QQ{3} نضع $t=x^{\frac{1}{3}}$ ($t>0$)، فيكون $x^{\frac{2}{3}}=t^{2}$ والمعادلة تصبح $t^{2}-3t+2=0$، أي $(t-1)(t-2)=0$.\par
$t=1$: $x^{\frac{1}{3}}=1$ أي $x=1$.\quad $t=2$: $x^{\frac{1}{3}}=2$ أي $x=2^{3}=8$.\quad إذن \bx{S=\{1;8\}}.
\end{hal}

\section{الاشتقاق والنهايات}
\begin{tamrin}{حساب المشتقات}
احسب مشتقة كل دالة على المجال المذكور:
\begin{enumerate}[label=(\arabic*)]
\item $f(x)=x^{\pi}$ على $]0;+\infty[$.
\item $g(x)=x^{-\frac{3}{4}}$ على $]0;+\infty[$.
\item $h(x)=\sqrt[3]{2x+1}$ على $\left]-\frac{1}{2};+\infty\right[$.
\item $k(x)=\left(x^{2}+x+1\right)^{\frac{1}{4}}$ على $\R$.
\end{enumerate}
\end{tamrin}
\begin{hal}
\QQ{1} $f'(x)=\bx{\pi x^{\pi-1}}$.
\QQ{2} $g'(x)=-\dfrac{3}{4}x^{-\frac{3}{4}-1}=\bx{-\dfrac{3}{4}x^{-\frac{7}{4}}}$.
\QQ{3} $u(x)=2x+1>0$ على المجال، $u'(x)=2$: $h'(x)=\dfrac{1}{3}\times2\times(2x+1)^{-\frac{2}{3}}=\bx{\dfrac{2}{3\sqrt[3]{(2x+1)^{2}}}}$.
\QQ{4} $u(x)=x^{2}+x+1>0$ على $\R$ (مميزه $\Delta=-3<0$)، $u'(x)=2x+1$:
$k'(x)=\dfrac{1}{4}(2x+1)\left(x^{2}+x+1\right)^{-\frac{3}{4}}=\bx{\dfrac{2x+1}{4\sqrt[4]{\left(x^{2}+x+1\right)^{3}}}}$.
\end{hal}

\begin{tamrin}{نهايات بالتزايد المقارن}
احسب النهايات:
\begin{enumerate}[label=(\arabic*)]
\item $\lim\limits_{x\to+\infty}\dfrac{e^{x}-x^{3}}{x^{2}}$
\item $\lim\limits_{x\to+\infty}x^{2}e^{-\sqrt{x}}$
\item $\lim\limits_{x\to+\infty}\dfrac{\ln x}{\sqrt[3]{x}}$
\item $\lim\limits_{x\to0^{+}}x\left(\ln x\right)^{2}$
\item $\lim\limits_{x\to+\infty}\dfrac{\left(\ln x\right)^{3}}{x}$
\end{enumerate}
\end{tamrin}
\begin{hal}
\QQ{1} $\dfrac{e^{x}-x^{3}}{x^{2}}=\dfrac{e^{x}}{x^{2}}-x$: حالة $\infty-\infty$. نكتب $=x\left(\dfrac{e^{x}}{x^{3}}-1\right)$، و $\dfrac{e^{x}}{x^{3}}\to+\infty$، إذن النهاية \bx{+\infty}.
\QQ{2} نضع $t=\sqrt{x}$ ($t\to+\infty$): $x^{2}e^{-\sqrt{x}}=t^{4}e^{-t}\to\bx{0}$.
\QQ{3} $\dfrac{\ln x}{x^{\frac{1}{3}}}\to\bx{0}$ (حالة $\alpha=\frac{1}{3}>0$).
\QQ{4} $x\left(\ln x\right)^{2}=\left(x^{\frac{1}{2}}\ln x\right)^{2}$ و $\lim\limits_{x\to0^{+}}x^{\frac{1}{2}}\ln x=0$، إذن النهاية \bx{0}.
\QQ{5} نضع $t=\ln x$ أي $x=e^{t}$ ($t\to+\infty$): $\dfrac{(\ln x)^{3}}{x}=\dfrac{t^{3}}{e^{t}}=t^{3}e^{-t}\to\bx{0}$.
\end{hal}

\section{دراسة دوال}
\begin{tamrin}{الدالة $x\mapsto x-3\sqrt[3]{x}$}
$f$ الدالة المعرفة على $[0;+\infty[$ بـ $f(x)=x-3\sqrt[3]{x}$، و $(C_f)$ تمثيلها البياني.
\QQ{1} احسب $\lim\limits_{x\to+\infty}f(x)$.
\QQ{2} بيّن أنه من أجل $x>0$: $f'(x)=\dfrac{\sqrt[3]{x^{2}}-1}{\sqrt[3]{x^{2}}}$، ثم شكّل جدول تغيّرات $f$.
\QQ{3} حل المعادلة $f(x)=0$ واستنتج إشارة $f(x)$، ثم أنشئ $(C_f)$.
\end{tamrin}
\begin{hal}
\QQ{1} من أجل $x>0$: $f(x)=x\left(1-3x^{-\frac{2}{3}}\right)$ و $\lim\limits_{x\to+\infty}x^{-\frac{2}{3}}=0$، إذن \bx{\lim_{x\to+\infty}f(x)=+\infty}.
\QQ{2} $f(x)=x-3x^{\frac{1}{3}}$، إذن $f'(x)=1-3\times\dfrac{1}{3}x^{-\frac{2}{3}}=1-\dfrac{1}{\sqrt[3]{x^{2}}}=\dfrac{\sqrt[3]{x^{2}}-1}{\sqrt[3]{x^{2}}}$.\par
$\sqrt[3]{x^{2}}-1>0\iff x^{2}>1\iff x>1$ (لأن $x>0$). $f(0)=0$ و $f(1)=1-3=-2$.

\begin{tamrin}{الدالة $x\mapsto\dfrac{\ln x}{\sqrt{x}}$}
$f$ الدالة المعرفة على $]0;+\infty[$ بـ $f(x)=\dfrac{\ln x}{\sqrt{x}}$.
\QQ{1} احسب نهايتي $f$ عند $0$ وعند $+\infty$ وفسّرهما بيانيًا.
\QQ{2} بيّن أن $f'(x)=\dfrac{2-\ln x}{2x\sqrt{x}}$ وشكّل جدول التغيّرات.
\QQ{3} استنتج أنه من أجل كل $x>0$: $\ln x\leqslant\dfrac{2}{e}\sqrt{x}$.
\end{tamrin}
\begin{hal}
\QQ{1} عند $0$: $\ln x\to-\infty$ و $\dfrac{1}{\sqrt{x}}\to+\infty$، إذن \bx{\lim_{x\to0^{+}}f(x)=-\infty}: المستقيم $x=0$ مقارب عمودي.\par
عند $+\infty$: $f(x)=\dfrac{\ln x}{x^{\frac{1}{2}}}\to0$ (تزايد مقارن)، إذن \bx{\lim_{x\to+\infty}f(x)=0}: المستقيم $y=0$ مقارب أفقي.
\QQ{2} $f'(x)=\dfrac{\frac{1}{x}\sqrt{x}-\ln x\times\frac{1}{2\sqrt{x}}}{x}=\dfrac{\frac{2-\ln x}{2\sqrt{x}}}{x}=\dfrac{2-\ln x}{2x\sqrt{x}}$.\par
