\Row{بناء المعارف}{%
\Lab{cDef}{تعريف}\par\textbf{(1)} $\Omega=\{e_1;e_2;\dots;e_n\}$ مجموعة الإمكانيات (المجموعة الشاملة).\par\textbf{(2)} \textbf{قانون الاحتمال}: إرفاق كل $e_i$ بعدد $p_i$ حيث $\RR{0\leqslant p_i\leqslant1}$ و $\RR{p_1+p_2+\dots+p_n=1}$.\par\textbf{(3)} احتمال حادثة $A$: \RR{مجموع} \RR{احتمالات} \RR{الإمكانيات} \RR{التي} \RR{تحققها}.\par\textbf{(4)} \textbf{تساوي الاحتمال}: $\RR{P(A)=\dfrac{\text{card}\,A}{\text{card}\,\Omega}}$ (عدد الحالات الملائمة على عدد الحالات الممكنة).}{50 د}{التحقق من المجموع.}
