{\begin{itemize}\item الدوال الأصلية ، التكامل والمساحة.\end{itemize}}%
{\begin{itemize}\item تعريف الدالة $\ln$ كدالة أصلية لـ $t\mapsto\frac1t$ تنعدم عند $1$.\item معرفة الخاصية المميزة $\ln(ab)=\ln a+\ln b$.\end{itemize}}%
{المنهاج ، الكتاب المدرسي ، مراجع ، الحاسبة العلمية والبيانية.}
\begin{Sayr}
\Row{مرحلة الانطلاق}{%
\textbf{التهيئة النفسية:} ما الدالة الأصلية لـ $x^n$ ؟ هل تصلح القاعدة لـ $n=-1$ أي لـ $\dfrac1x$ ؟}{10 د}{سؤال تحدٍّ.}
\Row{التشخيص والاكتشاف}{%
\Lab{cEx}{مناقشة النشاط صفحة 128}\par\ck\ الدالة $t\mapsto\frac1t$ مستمرة على $]0;+\infty[$: تقبل دالة أصلية وحيدة تنعدم عند $1$.\par
\ck\ من أجل $x>1$ هي مساحة الحيز تحت منحني $\frac1t$ بين $1$ و $x$:\par
\par\smallskip\begin{LTR}\centering \begin{tikzpicture}[x=1cm,y=1cm,font=\scriptsize]
\fill[cDef!25] (1,0) -- plot[domain=1:3,samples=40] (\x,{1/\x}) -- (3,0) -- cycle;
\draw[->] (-0.3,0)--(4.4,0) node[right]{$t$}; \draw[->] (0,-0.3)--(0,2.6) node[above]{$y$};
\draw[cDef,thick,domain=0.4:4.2,samples=60] plot (\x,{1/\x}) node[above]{$y=\frac1t$};
\node[below] at (1,0){$1$}; \node[below] at (3,0){$x$}; \node at (1.9,0.25){$\ln x$};
\draw[densely dotted] (1,0)--(1,1) (3,0)--(3,0.333);
\end{tikzpicture}\end{LTR}\par }{35 د}{ربط بالمساحة.}
\Row{بناء المعارف}{%
\Lab{cDef}{تعريف}\ \textbf{الدالة اللوغاريتم النيبيري} $\ln$ هي الدالة الأصلية للدالة $x\mapsto\dfrac1x$ على $]0;+\infty[$ التي تنعدم عند $1$: $\ln1=0$ و $(\ln x)^{\prime}=\dfrac1x$ ؛ وهي $\ln x=\displaystyle\int_1^x\dfrac1t\,dt$.\par\Lab{cThm}{الخاصية المميزة}\ من أجل $a>0$ و $b>0$: $\ln(ab)=\ln a+\ln b$.\par\textbf{البرهان:} $g(x)=\ln(ax)$: $g^{\prime}(x)=\dfrac a{ax}=\dfrac1x$ ، إذن $\ln(ax)=\ln x+c$ ؛ من أجل $x=1$: $c=\ln a$.\par}{50 د}{البرهان بالاشتقاق.}
\Row{}{%
\Lab{cDef}{العدد $e$}\ العدد الوحيد الذي يحقق $\ln e=1$ ؛ $e\approx2{,}718$.\par\Lab{cEx}{مثال}\ بالحاسبة: $\ln2\approx0{,}693$ ، $\ln3\approx1{,}099$ ؛ إذن $\ln6=\ln2+\ln3\approx1{,}792$.\par}{}{}
\Row{}{%
\Lab{cThm!80!black}{حذار}\ $\ln x$ معرف فقط من أجل $x>0$ ؛ و $\ln(a+b)\neq\ln a+\ln b$.}{}{}
\Row{التقويم}{%
\Lab{cEx}{تطبيق}\par 1) احسب بدلالة $\ln2$ و $\ln3$: $\ln12$ ، $\ln18$. 2) تحقق بالحاسبة.\par\textbf{الحل:} $\ln12=2\ln2+\ln3\approx2{,}485$ ؛ $\ln18=\ln2+2\ln3\approx2{,}890$.\par\textbf{واجب منزلي:} تمارين صفحة 137.\par\smallskip\textbf{ملاحظات حول سير الحصة:} \dotfill}{25 د}{تقويم فردي بالحاسبة.}
\end{Sayr}

\ModhHead{29}{3 ت ا}{تحليل}{الدالة اللوغاريتمية}{02 سا}{الخواص الجبرية للدالة $\ln$ ، الحساب العددي}%
{\begin{itemize}\item الخاصية المميزة $\ln(ab)=\ln a+\ln b$.\end{itemize}}%
{\begin{itemize}\item توظيف خواص $\ln$ لتبسيط عبارات وفي الحساب العددي.\end{itemize}}%
{المنهاج ، الكتاب المدرسي ، مراجع ، الحاسبة العلمية والبيانية.}
\begin{Sayr}
\Row{مرحلة الانطلاق}{%
\textbf{التهيئة النفسية:} كيف نستنتج $\ln\dfrac1a$ من $\ln\left(a\times\dfrac1a\right)=\ln1$ ؟}{10 د}{سؤال.}
\Row{التشخيص والاكتشاف}{%
\Lab{cEx}{مناقشة النشاط صفحة 129}\par\ck\ $\ln a+\ln\dfrac1a=\ln1=0$: $\ln\dfrac1a=-\ln a$.\par
\ck\ $\ln a^2=\ln(a\times a)=2\ln a$ ، ثم $\ln a^n=n\ln a$.\par
}{35 د}{استنتاج الخواص.}
\Row{بناء المعارف}{%
\Lab{cThm}{خاصية}\ من أجل $a>0$ ، $b>0$ و $n$ صحيح:\par\ck\ $\ln\dfrac1a=-\ln a$ ؛ $\ln\dfrac ab=\ln a-\ln b$.\par
\ck\ $\ln a^n=n\ln a$ ؛ $\ln\sqrt a=\dfrac12\ln a$.\par
\ck\ $\ln a=\ln b$ تكافئ $a=b$ ؛ $\ln a<\ln b$ تكافئ $a<b$ ($\ln$ متزايدة تماماً).\par
\ck\ $\ln a>0$ تكافئ $a>1$ ؛ $\ln a<0$ تكافئ $0<a<1$.\par
\Lab{cEx}{مثال}\par\ck\ $A=\ln18-\ln2-2\ln3=\ln9-\ln9=0$.\par
\ck\ $B=\ln e^3-\ln\sqrt e=3-\dfrac12=\dfrac52$.\par
}{50 د}{جدول الخواص في الكراس.}
\Row{}{%
\Lab{cThm!80!black}{حذار}\ $(\ln a)^2\neq\ln a^2$.}{}{}
\Row{التقويم}{%
\Lab{cEx}{تطبيق}\par بسّط: $C=\ln50+\ln2-2\ln5$ ؛ $D=\ln\dfrac{e^2}{\sqrt e}$. ثم رتّب: $\ln0{,}5$ ، $0$ ، $\ln3$.\par\textbf{الحل:} $C=\ln100-\ln25=\ln4=2\ln2$ ؛ $D=2-\dfrac12=\dfrac32$ ؛ $\ln0{,}5<0<\ln3$.\par\textbf{واجب منزلي:} تمارين صفحة 137.\par\smallskip\textbf{ملاحظات حول سير الحصة:} \dotfill}{25 د}{تقويم فردي.}
\end{Sayr}

{\begin{itemize}\item خواص $\ln$ ، المعادلات من الدرجة الثانية.\end{itemize}}%
{\begin{itemize}\item حل معادلات ومتراجحات تتضمن $\ln$.\item توظيف $\ln$ في مسائل الفوائد المركبة.\end{itemize}}%
{المنهاج ، الكتاب المدرسي ، مراجع ، الحاسبة العلمية والبيانية.}
\begin{Sayr}
\Row{مرحلة الانطلاق}{%
\textbf{التهيئة النفسية:} رأس مال $1000$ DA بفائدة مركبة $4\%$ سنوياً: بعد كم سنة يتجاوز $1500$ DA ؟}{10 د}{وضعية مالية.}
\Row{التشخيص والاكتشاف}{%
\Lab{cEx}{وضعية مقترحة}\par\ck\ $1000\times1{,}04^n\geqslant1500$ تكافئ $1{,}04^n\geqslant1{,}5$.\par
\ck\ نطبق $\ln$ (متزايدة): $n\ln1{,}04\geqslant\ln1{,}5$ ، $n\geqslant\dfrac{\ln1{,}5}{\ln1{,}04}\approx10{,}34$: بعد $11$ سنة.\par
}{35 د}{شروط الوجود أولاً.}
\Row{بناء المعارف}{%
\Lab{cMeth}{طريقة}\par\ck\ نعيّن أولاً مجموعة الحلول الممكنة (عبارات $\ln$ موجبة تماماً).\par
\ck\ نكتب المعادلة على الشكل $\ln A=\ln B$ ، أي $A=B$ ؛ والمتراجحة $\ln A<\ln B$ ، أي $A<B$.\par
\ck\ نتحقق أن الحلول تنتمي إلى المجموعة.\par
\Lab{cEx}{مثال}\par\ck\ $\ln(2x-1)=0$: $x>\frac12$ ، $2x-1=1$ ، $x=1$.\par
\ck\ $\ln x+\ln(x-1)=\ln6$: $x>1$ ، $x^2-x-6=0$ ، $x=3$ ($-2$ مرفوض).\par
\ck\ $\ln(x-2)\leqslant0$: $x>2$ و $x-2\leqslant1$: $S=]2;3]$.\par
\ck\ $\ln x\geqslant2$: $x\geqslant e^2$ (لأن $2=\ln e^2$).\par
}{50 د}{متغير مساعد.}
\Row{}{%
\Lab{cThm!80!black}{حذار}\ لا ننسى شروط الوجود قبل الحل.}{}{}
\Row{التقويم}{%
\Lab{cEx}{تطبيق}\par حل: $\ln(x+1)=\ln(3-x)$ ؛ $(\ln x)^2-3\ln x+2=0$.\par\textbf{الحل:} $x\in]-1;3[$ ، $x+1=3-x$ ، $x=1$ ؛ نضع $X=\ln x$: $X=1$ أو $X=2$ ، إذن $x=e$ أو $x=e^2$.\par\textbf{واجب منزلي:} تمارين صفحة 137.\par\smallskip\textbf{ملاحظات حول سير الحصة:} \dotfill}{25 د}{تقويم فردي.}
\end{Sayr}

\ModhHead{31}{3 ت ا}{تحليل}{الدالة اللوغاريتمية}{02 سا}{دراسة الدالة $\ln$ وتمثيلها البياني ، النهايات الشهيرة}%
{\begin{itemize}\item المشتقات ، النهايات ، خواص $\ln$.\end{itemize}}%
{\begin{itemize}\item دراسة الدالة $\ln$ وتمثيلها.\item معرفة وتفسير النهايتين $\lim\limits_{x\to +\infty}\frac{\ln x}x=0$ و $\lim\limits_{x\to 0^{+}}x\ln x=0$.\end{itemize}}%
{المنهاج ، الكتاب المدرسي ، مراجع ، الحاسبة العلمية والبيانية.}
\begin{Sayr}
\Row{مرحلة الانطلاق}{%
\textbf{التهيئة النفسية:} هل لـ $\ln x$ حد أعلى؟ احسب $\ln10^{10}$ و $\ln10^{-10}$.}{10 د}{سؤال بالحاسبة.}
\Row{التشخيص والاكتشاف}{%
\Lab{cEx}{وضعية مقترحة}\par\ck\ $\ln10^{n}=n\ln10$ يكبر بلا حدود ؛ $\ln10^{-n}=-n\ln10$ يصغر بلا حدود.\par
\ck\ لكن $\dfrac{\ln x}x$ بالحاسبة من أجل $x=10$ ، $100$ ، $1000$: يقترب من $0$.\par
}{35 د}{جدول التغيرات.}
\Row{بناء المعارف}{%
\Lab{cThm}{دراسة الدالة $\ln$}\par\ck\ معرفة على $]0;+\infty[$ ، $(\ln x)^{\prime}=\dfrac1x>0$: متزايدة تماماً.\par
\ck\ $\lim\limits_{x\to +\infty}\ln x=+\infty$ ؛ $\lim\limits_{x\to 0^{+}}\ln x=-\infty$: محور التراتيب مقارب.\par
\ck\ المماس عند $1$: $y=x-1$ ، و $\ln x\leqslant x-1$.\par
\par\smallskip\begin{LTR}\centering \begin{tikzpicture}[x=0.85cm,y=0.5cm,font=\footnotesize]

\node at (0.7,-0.5){$x$}; \node at (0.7,-1.5){$(\ln x)^{\prime}$}; \node at (0.7,-3.5){$\ln x$};
\node at (1.9,-0.5){$0$}; \node at (5.4,-0.5){$+\infty$}; \node at (3.65,-1.5){$+$};
\node (a) at (1.9,-4.5){$-\infty$}; \node (c) at (5.4,-2.5){$+\infty$}; \draw[->,cDef,thick] (a)--(c);
\end{tikzpicture}\end{LTR}\par \par\smallskip\begin{LTR}\centering \begin{tikzpicture}[x=0.8cm,y=0.8cm,font=\scriptsize]
\draw[->] (-0.5,0)--(7,0) node[right]{$x$}; \draw[->] (0,-3)--(0,2.6) node[above]{$y$};

\end{tikzpicture}\end{LTR}\par \Lab{cThm}{نهايات شهيرة (تُقبل)}\ $\lim\limits_{x\to +\infty}\dfrac{\ln x}x=0$ ؛ $\lim\limits_{x\to 0^{+}}x\ln x=0$.\par}{50 د}{الرسم في معلم.}
\Row{}{%
\Lab{cEx}{مثال}\ $\lim\limits_{x\to +\infty}(x-\ln x)=\lim\limits_{x\to +\infty}x\left(1-\dfrac{\ln x}x\right)=+\infty$ ؛ $\lim\limits_{x\to 0^{+}}x^2\ln x=\lim\limits_{x\to 0^{+}}x\times x\ln x=0$.\par\Lab{cThm!80!black}{حذار}\ الشكل $\infty-\infty$: نخرج العامل الغالب.}{}{}
\Row{التقويم}{%
\Lab{cEx}{تطبيق}\par احسب: $\lim\limits_{x\to +\infty}(2x-3\ln x)$ ؛ $\lim\limits_{x\to 0^{+}}(x\ln x-x)$ ؛ $\lim\limits_{x\to +\infty}\dfrac{\ln x}{x^2}$.\par\textbf{الحل:} $+\infty$ ؛ $0$ ؛ $0$ (لأن $\dfrac{\ln x}{x^2}=\dfrac{\ln x}x\times\dfrac1x$).\par\textbf{واجب منزلي:} تمارين صفحة 137.\par\smallskip\textbf{ملاحظات حول سير الحصة:} \dotfill}{25 د}{تقويم فردي.}
\end{Sayr}

\ModhHead{32}{3 ت ا}{تحليل}{الدالة اللوغاريتمية}{02 سا}{دراسة دوال من الشكل $\ln u$}%
{\begin{itemize}\item دراسة $\ln$ ، مشتقة مركب دالتين.\end{itemize}}%
{\begin{itemize}\item دراسة دالة من الشكل $\ln u$.\item حساب دالة أصلية لـ $\dfrac{u^{\prime}}u$.\end{itemize}}%
{المنهاج ، الكتاب المدرسي ، مراجع ، الحاسبة العلمية والبيانية.}
\begin{Sayr}
\Row{مرحلة الانطلاق}{%
\textbf{التهيئة النفسية:} ما مجموعة تعريف $\ln(2x-4)$ ؟ وما مشتقتها؟}{10 د}{سؤال.}
\Row{التشخيص والاكتشاف}{%
\Lab{cEx}{وضعية مقترحة}\par\ck\ $\ln u$ معرفة حيث $u(x)>0$.\par
\ck\ بمشتقة المركب: $(\ln u)^{\prime}=u^{\prime}\times\dfrac1u$.\par
}{35 د}{مشتقة المركب.}
\Row{بناء المعارف}{%
\Lab{cThm}{خاصية}\par\ck\ $(\ln u)^{\prime}=\dfrac{u^{\prime}}u$: لـ $\ln u$ نفس اتجاه تغير $u$.\par
\ck\ دالة أصلية لـ $\dfrac{u^{\prime}}u$ (مع $u>0$) هي $\ln u$.\par
\ck\ النهايات بنهاية دالة مركبة.\par
\Lab{cEx}{مثال}\ $f(x)=\ln(x^2+1)$ على $\mathbb R$: $f^{\prime}(x)=\dfrac{2x}{x^2+1}$ ؛ $\lim\limits_{x\to \pm\infty}f(x)=+\infty$.\par\smallskip\begin{LTR}\centering \begin{tikzpicture}[x=0.85cm,y=0.5cm,font=\footnotesize]
\draw (0,0) rectangle (7,-5); \draw (0,-1)--(7,-1); \draw (0,-2)--(7,-2); \draw (1.4,0)--(1.4,-5);
\node at (0.7,-0.5){$x$}; \node at (0.7,-1.5){$f^{\prime}(x)$}; \node at (0.7,-3.5){$f(x)$};
\node at (1.9,-0.5){$-\infty$}; \node at (4,-0.5){$0$}; \node at (6.4,-0.5){$+\infty$};
\node at (2.9,-1.5){$-$}; \node at (4,-1.5){$0$}; \node at (5.2,-1.5){$+$};
\node (a) at (1.9,-2.5){$+\infty$}; \node (b) at (4,-4.5){$0$}; \node (c) at (6.4,-2.5){$+\infty$};
\draw[->,cDef,thick] (a)--(b); \draw[->,cDef,thick] (b)--(c);
\end{tikzpicture}\end{LTR}\par \ $\displaystyle\int_0^1\dfrac{2x}{x^2+1}\,dx=\big[\ln(x^2+1)\big]_0^1=\ln2$.\par}{50 د}{جدول التغيرات.}
\Row{}{%
\Lab{cThm!80!black}{حذار}\ نعيّن مجموعة التعريف قبل كل شيء.}{}{}
\Row{التقويم}{%
\Lab{cEx}{تطبيق}\par $g(x)=\ln(2x-4)$. عيّن مجموعة التعريف ، النهايات ، اتجاه التغير ، وحل $g(x)=0$.\par\textbf{الحل:} $]2;+\infty[$ ؛ $\lim\limits_{x\to 2^{+}}g(x)=-\infty$ ، $\lim\limits_{x\to +\infty}g(x)=+\infty$ ؛ $g^{\prime}(x)=\dfrac1{x-2}>0$ ؛ $2x-4=1$: $x=\dfrac52$.\par\textbf{واجب منزلي:} تمارين من صفحة 143 إلى 147.\par\smallskip\textbf{ملاحظات حول سير الحصة:} \dotfill}{25 د}{تقويم فردي.}
\end{Sayr}

\ModhHead{33}{3 ت ا}{تحليل}{الدالة اللوغاريتمية}{02 سا}{الدالة اللوغاريتمية ذات الأساس $a$ ، اللوغاريتم العشري}%
{\begin{itemize}\item خواص $\ln$.\end{itemize}}%
{\begin{itemize}\item معرفة $\log_a$ و $\log$ ، وتوظيف اللوغاريتم العشري (عدد الأرقام ، المقاييس) ، أمثلة من الرياضيات المالية.\end{itemize}}%
{المنهاج ، الكتاب المدرسي ، مراجع ، الحاسبة العلمية والبيانية.}
\begin{Sayr}
\Row{مرحلة الانطلاق}{%
\textbf{التهيئة النفسية:} كم رقماً يكتب به العدد $2^{100}$ ؟}{10 د}{تحدٍّ.}
\Row{التشخيص والاكتشاف}{%
\Lab{cEx}{وضعية مقترحة}\par\ck\ العدد ذو $k$ رقماً يحقق $10^{k-1}\leqslant N<10^k$.\par
\ck\ نحتاج دالة تحوّل $10^k$ إلى $k$: هي $\dfrac{\ln x}{\ln10}$.\par
}{35 د}{الحاسبة.}
\Row{بناء المعارف}{%
\Lab{cDef}{تعريف}\ من أجل $a>0$ و $a\neq1$: $\log_a x=\dfrac{\ln x}{\ln a}$ ($x>0$) ؛ \textbf{اللوغاريتم العشري} $\log x=\dfrac{\ln x}{\ln10}$.\par\Lab{cThm}{خاصية}\ لـ $\log_a$ نفس خواص $\ln$ الجبرية ، و $\log_a a=1$ ، $\log10^n=n$.\par}{50 د}{أمثلة مالية.}
\Row{}{%
\Lab{cEx}{مثال}\par\ck\ $\log2^{100}=100\log2\approx30{,}1$: $10^{30}\leqslant2^{100}<10^{31}$ ، إذن $31$ رقماً.\par
\ck\ المقاييس اللوغاريتمية: شدة الصوت بالديسيبل $L=10\log\dfrac I{I_0}$.\par
\ck\ رأس مال بفائدة مركبة $5\%$ يتضاعف لما $1{,}05^n\geqslant2$: $n\geqslant\dfrac{\ln2}{\ln1{,}05}\approx14{,}2$: بعد $15$ سنة.\par
\Lab{cThm!80!black}{حذار}\ $\log$ في الحاسبة هو اللوغاريتم العشري و $\ln$ هو النيبيري.}{}{}
\Row{التقويم}{%
\Lab{cEx}{تطبيق}\par 1) كم رقماً للعدد $3^{50}$ ؟ 2) احسب $\log_2 32$ و $\log0{,}001$.\par\textbf{الحل:} 1) $50\log3\approx23{,}86$: $24$ رقماً. 2) $5$ ؛ $-3$.\par\textbf{واجب منزلي:} تمارين من صفحة 143 إلى 147.\par\smallskip\textbf{ملاحظات حول سير الحصة:} \dotfill}{25 د}{تقويم فردي.}
\end{Sayr}
