\section{الدوال الأصلية}

% ===================== 1 =====================
\begin{tamrin}{دوال أصلية مباشرة}
في كل حالة من الحالات التالية، عيّن دالة أصلية $F$ للدالة $f$ على المجال $I$:
\begin{sq}
\qa $f(x)=3x^{2}-4x+5$ \quad ، \quad $I=\R$.
\qb $f(x)=\dfrac{1}{x^{2}}+\dfrac{2}{\sqrt{x}}$ \quad ، \quad $I=]0;+\infty[$.
\qc $f(x)=(2x+1)\left(x^{2}+x+3\right)^{3}$ \quad ، \quad $I=\R$.
\qd $f(x)=\dfrac{x}{x^{2}+1}$ \quad ، \quad $I=\R$.
\qe $f(x)=e^{2x-1}$ \quad ، \quad $I=\R$.
\item[(و)] $f(x)=\dfrac{3}{(2x+1)^{2}}$ \quad ، \quad $I=\left]-\dfrac12;+\infty\right[$.
\item[(ز)] $f(x)=\dfrac{\ln x}{x}$ \quad ، \quad $I=]0;+\infty[$.
\item[(ح)] $f(x)=\dfrac{x}{\sqrt{x^{2}+4}}$ \quad ، \quad $I=\R$.
\item[(ط)] $f(x)=\cos(2x)-\sin x$ \quad ، \quad $I=\R$.
\end{sq}
\end{tamrin}

\begin{hal}
\textbf{تذكير:} نتعرّف في كل مرة على شكل من أشكال الجدول: $u'u^{n}$ أصليتها $\dfrac{u^{n+1}}{n+1}$، و $\dfrac{u'}{u}$ أصليتها $\ln|u|$، و $\dfrac{u'}{u^{2}}$ أصليتها $-\dfrac{1}{u}$، و $\dfrac{u'}{\sqrt{u}}$ أصليتها $2\sqrt{u}$، و $u'e^{u}$ أصليتها $e^{u}$. وكل الدوال المعطاة مستمرة على $I$ فهي تقبل دوالا أصلية عليه.

\pa دالة كثير حدود؛ نأخذ أصلية كل حد ($x^{n}$ أصليتها $\dfrac{x^{n+1}}{n+1}$):
\[F(x)=3\cdot\frac{x^{3}}{3}-4\cdot\frac{x^{2}}{2}+5x=\bx{x^{3}-2x^{2}+5x}\]

\pb $\dfrac{1}{x^{2}}$ أصليتها $-\dfrac1x$، و $\dfrac{1}{\sqrt{x}}$ أصليتها $2\sqrt{x}$ على $]0;+\infty[$. إذن:
\[F(x)=-\frac1x+2\cdot2\sqrt{x}=\bx{-\frac{1}{x}+4\sqrt{x}}\]

\pc نضع $u(x)=x^{2}+x+3$، إذن $u'(x)=2x+1$ و $f=u'u^{3}$ (الشكل $u'u^{n}$ مع $n=3$):
\[F(x)=\frac{u^{4}(x)}{4}=\bx{\frac{\left(x^{2}+x+3\right)^{4}}{4}}\]

\pd نضع $u(x)=x^{2}+1$، إذن $u'(x)=2x$ و $u(x)>0$ على $\R$. نكتب $f(x)=\dfrac12\cdot\dfrac{2x}{x^{2}+1}=\dfrac12\cdot\dfrac{u'(x)}{u(x)}$، ومنه:
\[F(x)=\frac12\ln\left(x^{2}+1\right)\ \Rightarrow\ \bx{F(x)=\frac{1}{2}\ln\left(x^{2}+1\right)}\]

\pe نضع $u(x)=2x-1$، $u'(x)=2$. نكتب $f(x)=\dfrac12\cdot 2e^{2x-1}=\dfrac12 u'(x)e^{u(x)}$، ومنه:
\[\bx{F(x)=\frac{1}{2}e^{2x-1}}\]

\par\smallskip\textbf{(و)} نضع $u(x)=2x+1$، $u'(x)=2$، و $u(x)\neq0$ على $I$. نكتب $f(x)=\dfrac32\cdot\dfrac{2}{(2x+1)^{2}}=\dfrac32\cdot\dfrac{u'(x)}{u^{2}(x)}$، ومنه:
\[F(x)=\frac32\cdot\left(-\frac{1}{2x+1}\right)=\bx{-\frac{3}{2(2x+1)}}\]

\par\smallskip\textbf{(ز)} نضع $u(x)=\ln x$، $u'(x)=\dfrac1x$، إذن $f(x)=u'(x)\,u(x)$ (الشكل $u'u^{n}$ مع $n=1$):
\[\bx{F(x)=\frac{(\ln x)^{2}}{2}}\]

\par\smallskip\textbf{(ح)} نضع $u(x)=x^{2}+4>0$، $u'(x)=2x$. نكتب $f(x)=\dfrac12\cdot\dfrac{2x}{\sqrt{x^{2}+4}}=\dfrac12\cdot\dfrac{u'(x)}{\sqrt{u(x)}}$، ومنه:
\[F(x)=\frac12\cdot2\sqrt{x^{2}+4}=\bx{\sqrt{x^{2}+4}}\]

\par\smallskip\textbf{(ط)} $\cos(ax)$ أصليتها $\dfrac1a\sin(ax)$، و $-\sin x$ أصليتها $\cos x$. إذن:
\[\bx{F(x)=\frac{1}{2}\sin(2x)+\cos x}\]
\textbf{تحقق:} في كل حالة نشتق $F$ فنجد $F'(x)=f(x)$.
\end{hal}

% ===================== 2 =====================
\begin{tamrin}{الدالة الأصلية التي تأخذ قيمة معلومة}
\QQ{1} $f$ الدالة المعرفة على المجال $]-1;+\infty[$ بـ: $f(x)=\dfrac{2x+3}{x+1}$.
\begin{sq}
\qa بيّن أنه من أجل كل عدد حقيقي $x$ من $]-1;+\infty[$: $f(x)=2+\dfrac{1}{x+1}$.
\qb عيّن الدالة الأصلية $F$ للدالة $f$ على $]-1;+\infty[$ التي تحقق $F(0)=1$.
\end{sq}
\QQ{2} $g$ الدالة المعرفة على $\R$ بـ: $g(x)=\dfrac{e^{x}}{\left(e^{x}+1\right)^{2}}$.
\begin{sq}
\qa عيّن الدوال الأصلية للدالة $g$ على $\R$.
\qb عيّن الدالة الأصلية $G$ للدالة $g$ على $\R$ التي تنعدم عند $0$.
\end{sq}
\end{tamrin}

\begin{hal}
\QQ{1}
\pa من أجل كل عدد حقيقي $x$ من $]-1;+\infty[$ لدينا $x+1\neq0$، و:
\[2+\frac{1}{x+1}=\frac{2(x+1)+1}{x+1}=\frac{2x+3}{x+1}=f(x)\qquad\text{\ok}\]
\pb أصلية $2$ هي $2x$، و $\dfrac{1}{x+1}$ من الشكل $\dfrac{u'}{u}$ مع $u(x)=x+1>0$ على المجال، فأصليتها $\ln(x+1)$.
إذن الدوال الأصلية لـ $f$ على $]-1;+\infty[$ هي: $F(x)=2x+\ln(x+1)+c$ حيث $c$ ثابت حقيقي.

الشرط $F(0)=1$ يعطي: $0+\ln1+c=1$، أي $c=1$. ومنه:
\[\bx{F(x)=2x+\ln(x+1)+1}\]

\QQ{2}
\pa نضع $u(x)=e^{x}+1$، إذن $u'(x)=e^{x}$ و $u(x)\neq0$، فـ $g=\dfrac{u'}{u^{2}}$ وأصليتها $-\dfrac1u$.
الدوال الأصلية لـ $g$ على $\R$ هي:
\[\bx{G_c(x)=-\frac{1}{e^{x}+1}+c}\quad ;\ c\in\R\]
\pb $G(0)=0$ يعني $-\dfrac{1}{e^{0}+1}+c=0$، أي $-\dfrac12+c=0$، ومنه $c=\dfrac12$:
\[\bx{G(x)=\frac{1}{2}-\frac{1}{e^{x}+1}}\]
\end{hal}

% ===================== 3 =====================
\begin{tamrin}{تعيين دالة أصلية بالمطابقة}
\QQ{1} $f$ الدالة المعرفة على $\R$ بـ: $f(x)=\left(-x^{2}+3x-1\right)e^{-x}$.
\begin{sq}
\qa عيّن الأعداد الحقيقية $a$ و $b$ و $c$ بحيث تكون الدالة $F:x\mapsto\left(ax^{2}+bx+c\right)e^{-x}$ دالة أصلية للدالة $f$ على $\R$.
\qb استنتج الدالة الأصلية للدالة $f$ على $\R$ التي تأخذ القيمة $2$ عند $0$.
\end{sq}
\QQ{2} $g$ الدالة المعرفة على $]-1;+\infty[$ بـ: $g(x)=\dfrac{x^{2}+2x+3}{(x+1)^{2}}$.
\begin{sq}
\qa عيّن العددين الحقيقيين $\alpha$ و $\beta$ بحيث من أجل كل $x$ من $]-1;+\infty[$: $g(x)=\alpha+\dfrac{\beta}{(x+1)^{2}}$.
\qb استنتج دالة أصلية للدالة $g$ على $]-1;+\infty[$.
\end{sq}
\end{tamrin}

\begin{hal}
\QQ{1}
\pa $F$ قابلة للاشتقاق على $\R$ (جداء دالتين قابلتين للاشتقاق). باستعمال $(uv)'=u'v+uv'$ مع $\left(e^{-x}\right)'=-e^{-x}$:
\begin{align*}
F'(x)&=(2ax+b)e^{-x}-\left(ax^{2}+bx+c\right)e^{-x}\\
&=\left(-ax^{2}+(2a-b)x+(b-c)\right)e^{-x}
\end{align*}
تكون $F$ أصلية لـ $f$ إذا وفقط إذا كان $F'(x)=f(x)$ من أجل كل $x$، وبما أن $e^{-x}\neq0$ نطابق المعاملات:
\[\begin{cases}-a=-1\\ 2a-b=3\\ b-c=-1\end{cases}\iff\begin{cases}a=1\\ b=2-3=-1\\ c=b+1=0\end{cases}\]
إذن: $a=1$ ، $b=-1$ ، $c=0$ و \bx{F(x)=\left(x^{2}-x\right)e^{-x}}.

\pb كل الدوال الأصلية لـ $f$ على $\R$ هي $x\mapsto\left(x^{2}-x\right)e^{-x}+k$ مع $k\in\R$.
القيمة $2$ عند $0$: $(0-0)e^{0}+k=2$ أي $k=2$. إذن الدالة المطلوبة:
\[\bx{x\mapsto\left(x^{2}-x\right)e^{-x}+2}\]

\QQ{2}
\pa نلاحظ أن $x^{2}+2x+3=(x+1)^{2}+2$. إذن من أجل كل $x$ من $]-1;+\infty[$:
\[g(x)=\frac{(x+1)^{2}+2}{(x+1)^{2}}=1+\frac{2}{(x+1)^{2}}\]
ومنه \bx{\alpha=1} و \bx{\beta=2}.

\pb أصلية $1$ هي $x$. و $\dfrac{2}{(x+1)^{2}}=2\cdot\dfrac{u'}{u^{2}}$ مع $u(x)=x+1$، فأصليتها $-\dfrac{2}{x+1}$. إذن دالة أصلية لـ $g$ على $]-1;+\infty[$:
\[\bx{G(x)=x-\frac{2}{x+1}}\]
\end{hal}

% ===================== 4 =====================
\begin{tamrin}{دوال أصلية بعد تحويل العبارة}
\QQ{1} $f$ الدالة المعرفة على $\R$ بـ: $f(x)=\dfrac{e^{2x}}{e^{x}+1}$.
\begin{sq}
\qa بيّن أنه من أجل كل عدد حقيقي $x$: $f(x)=e^{x}-\dfrac{e^{x}}{e^{x}+1}$.
\qb استنتج الدالة الأصلية $F$ للدالة $f$ على $\R$ التي تنعدم عند $0$.
\end{sq}
\QQ{2} $h$ الدالة المعرفة على $]0;+\infty[$ بـ: $h(x)=\dfrac{1}{x(x+1)}$.
\begin{sq}
\qa تحقق أنه من أجل كل $x$ من $]0;+\infty[$: $h(x)=\dfrac1x-\dfrac{1}{x+1}$.
\qb استنتج الدالة الأصلية $H$ للدالة $h$ على $]0;+\infty[$ التي تنعدم عند $1$، ثم اكتبها على شكل $\ln$ عبارة واحدة.
\qc أحسب $\lim\limits_{x\to+\infty}H(x)$.
\end{sq}
\end{tamrin}

\begin{hal}
\QQ{1}
\pa نوحّد المقامات:
\[e^{x}-\frac{e^{x}}{e^{x}+1}=\frac{e^{x}\left(e^{x}+1\right)-e^{x}}{e^{x}+1}=\frac{e^{2x}+e^{x}-e^{x}}{e^{x}+1}=\frac{e^{2x}}{e^{x}+1}=f(x)\qquad\text{\ok}\]
\pb أصلية $e^{x}$ هي $e^{x}$. و $\dfrac{e^{x}}{e^{x}+1}$ من الشكل $\dfrac{u'}{u}$ مع $u(x)=e^{x}+1>0$، فأصليتها $\ln\left(e^{x}+1\right)$.
إذن: $F(x)=e^{x}-\ln\left(e^{x}+1\right)+c$ مع $c\in\R$.

$F(0)=0$ يعني $1-\ln2+c=0$، أي $c=\ln2-1$. ومنه:
\[\bx{F(x)=e^{x}-\ln\left(e^{x}+1\right)+\ln2-1}\]

\QQ{2}
\pa من أجل كل $x>0$:
$\dfrac1x-\dfrac{1}{x+1}=\dfrac{(x+1)-x}{x(x+1)}=\dfrac{1}{x(x+1)}=h(x)$ \ok.

\pb على $]0;+\infty[$: أصلية $\dfrac1x$ هي $\ln x$، وأصلية $\dfrac1{x+1}$ هي $\ln(x+1)$ (لأن $x+1>0$). إذن:
\[H(x)=\ln x-\ln(x+1)+c=\ln\left(\frac{x}{x+1}\right)+c\]
$H(1)=0$ يعني $\ln\dfrac12+c=0$، أي $c=\ln2$. ومنه:
\[H(x)=\ln\left(\frac{x}{x+1}\right)+\ln 2=\bx{\ln\left(\frac{2x}{x+1}\right)}\]

\pc $\dfrac{2x}{x+1}=\dfrac{2}{1+\frac1x}$ و $\lim\limits_{x\to+\infty}\dfrac1x=0$، إذن $\lim\limits_{x\to+\infty}\dfrac{2x}{x+1}=2$، وبما أن $\ln$ مستمرة عند $2$:
\[\lim_{x\to+\infty}H(x)=\bx{\ln 2}\]
\end{hal}

\begin{tamrin}{حساب تكاملات مباشرة}

\end{sq}

\begin{hal}

\pc أصلية $e^{2x}$ هي $\dfrac12e^{2x}$، و $e^{2\ln2}=e^{\ln4}=4$:
\[I_3=\left[\frac12e^{2x}\right]_0^{\ln2}=\frac12(4-1)=\bx{\frac{3}{2}}\]
\pd $\dfrac{x}{x^{2}+1}=\dfrac12\cdot\dfrac{2x}{x^{2}+1}$ من الشكل $\dfrac12\cdot\dfrac{u'}{u}$:
\[I_4=\left[\frac12\ln\left(x^{2}+1\right)\right]_0^1=\frac12\ln2-\frac12\ln1=\bx{\frac{\ln 2}{2}}\]
\pe نضع $u(x)=\sin x$ فـ $u'(x)=\cos x$ و $\cos x\sin^{2}x=u'u^{2}$ أصليتها $\dfrac{u^{3}}{3}$:
\[I_5=\left[\frac{\sin^{3}x}{3}\right]_0^{\frac{\pi}{2}}=\frac{1^{3}}{3}-0=\bx{\frac{1}{3}}\]
\par\smallskip\textbf{(و)} أصلية $\dfrac{1}{\sqrt x}$ على $]0;+\infty[$ هي $2\sqrt{x}$:
\[I_6=\big[2\sqrt{x}\big]_1^4=2\times2-2\times1=\bx{2}\]
\end{hal}

\begin{tamrin}{استعمال الزوجية والفردية}
نقبل الخاصيتين التاليتين (لدالة $f$ مستمرة على $[-a;a]$):
إذا كانت $f$ فردية فإن $\displaystyle\int_{-a}^{a}f(x)\,dx=0$، وإذا كانت $f$ زوجية فإن $\displaystyle\int_{-a}^{a}f(x)\,dx=2\int_{0}^{a}f(x)\,dx$.
\QQ{1} أحسب $I=\displaystyle\int_{-1}^{1}x^{3}e^{x^{2}}\,dx$ دون حساب دالة أصلية.
\QQ{2} أحسب $J=\displaystyle\int_{-1}^{1}\left(x^{4}+x^{5}+\sin x\right)dx$.
\QQ{3} أحسب $K=\displaystyle\int_{-2}^{2}\left(|x|+x^{3}\right)dx$.
\QQ{4} $g$ الدالة المعرفة على $]-1;1[$ بـ: $g(x)=\ln\left(\dfrac{1+x}{1-x}\right)$. بيّن أن $g$ فردية، ثم استنتج $\displaystyle\int_{-\frac12}^{\frac12}g(x)\,dx$.
\end{tamrin}

\begin{hal}
\QQ{1} الدالة $f:x\mapsto x^{3}e^{x^{2}}$ مستمرة على $[-1;1]$ (مجال متناظر بالنسبة إلى $0$)، ومن أجل كل $x$:
$f(-x)=(-x)^{3}e^{(-x)^{2}}=-x^{3}e^{x^{2}}=-f(x)$، إذن $f$ فردية، ومنه \bx{I=0}.

\QQ{2} حسب الخطية: $J=\displaystyle\int_{-1}^{1}x^{4}dx+\int_{-1}^{1}x^{5}dx+\int_{-1}^{1}\sin x\,dx$.
\begin{itemize}
\item $x\mapsto x^{5}$ و $x\mapsto\sin x$ فرديتان، فتكاملاهما على $[-1;1]$ معدومان.
\item $x\mapsto x^{4}$ زوجية: $\displaystyle\int_{-1}^{1}x^{4}dx=2\int_0^1x^{4}dx=2\left[\frac{x^{5}}{5}\right]_0^1=\frac25$.
\end{itemize}
إذن \bx{J=\frac{2}{5}}.

\QQ{3} $x\mapsto|x|$ زوجية، و $x\mapsto x^{3}$ فردية. وعلى $[0;2]$ لدينا $|x|=x$، إذن:
\[K=2\int_0^2x\,dx+0=2\left[\frac{x^{2}}{2}\right]_0^2=2\times2=\bx{4}\]

\QQ{4} المجال $]-1;1[$ متناظر بالنسبة إلى $0$، ومن أجل كل $x$ منه:
\[g(-x)=\ln\left(\frac{1-x}{1+x}\right)=\ln\left(\left(\frac{1+x}{1-x}\right)^{-1}\right)=-\ln\left(\frac{1+x}{1-x}\right)=-g(x)\]
إذن $g$ فردية. وهي مستمرة على $\left[-\frac12;\frac12\right]\subset]-1;1[$، ومنه:
\[\bx{\int_{-\frac12}^{\frac12}g(x)\,dx=0}\]
\textbf{تفسير بياني:} المنحنى متناظر بالنسبة إلى المبدأ $O$، فالمساحة فوق محور الفواصل تساوي المساحة تحته، والتكامل (مساحة جبرية) معدوم.
\end{hal}

\section{التكامل بالتجزئة}

\begin{tamrin}{تكامل بالتجزئة مرة واحدة}
باستعمال التكامل بالتجزئة، أحسب:
\begin{sq}
\qa $I=\displaystyle\int_0^1xe^{x}\,dx$
\qb $J=\displaystyle\int_1^e\ln x\,dx$
\qc $K=\displaystyle\int_0^{\frac{\pi}{2}}x\cos x\,dx$
\end{sq}
\end{tamrin}

\begin{hal}
\textbf{القاعدة:} إذا كانت $u$ و $v$ قابلتين للاشتقاق على $[a;b]$ ومشتقتاهما مستمرتين فإن:
\[\int_a^bu(x)v'(x)\,dx=\Big[u(x)v(x)\Big]_a^b-\int_a^bu'(x)v(x)\,dx\]
نختار $u$ الدالة التي «تبسط بالاشتقاق» (كثير حدود، $\ln$)، و $v'$ دالة نعرف أصليتها.

\pa نضع:
$\begin{cases}u(x)=x\\ v'(x)=e^{x}\end{cases}$ \ إذن \ $\begin{cases}u'(x)=1\\ v(x)=e^{x}\end{cases}$
\[I=\Big[xe^{x}\Big]_0^1-\int_0^1e^{x}\,dx=(e-0)-\Big[e^{x}\Big]_0^1=e-(e-1)=\bx{1}\]

\pb نكتب $\ln x=1\times\ln x$ ونضع:
$\begin{cases}u(x)=\ln x\\ v'(x)=1\end{cases}$ \ إذن \ $\begin{cases}u'(x)=\frac1x\\ v(x)=x\end{cases}$
\[J=\Big[x\ln x\Big]_1^e-\int_1^ex\cdot\frac1x\,dx=(e\ln e-1\cdot\ln1)-\int_1^e1\,dx=e-(e-1)=\bx{1}\]
\begin{molahada}
بنفس الطريقة نجد من أجل $x>0$: $\displaystyle\int_1^x\ln t\,dt=x\ln x-x+1$؛ إذن $x\mapsto x\ln x-x$ دالة أصلية للدالة $\ln$ على $]0;+\infty[$.
\end{molahada}

\pc نضع:
$\begin{cases}u(x)=x\\ v'(x)=\cos x\end{cases}$ \ إذن \ $\begin{cases}u'(x)=1\\ v(x)=\sin x\end{cases}$
\[K=\Big[x\sin x\Big]_0^{\frac{\pi}{2}}-\int_0^{\frac{\pi}{2}}\sin x\,dx=\frac{\pi}{2}-\Big[-\cos x\Big]_0^{\frac{\pi}{2}}=\frac{\pi}{2}-(0+1)=\bx{\frac{\pi}{2}-1}\]
\end{hal}

\begin{tamrin}{التكامل بالتجزئة ودالة أصلية}
\QQ{1} أحسب $I=\displaystyle\int_1^ex\ln x\,dx$.
\QQ{2} أحسب $J=\displaystyle\int_0^1(2x+1)e^{-x}\,dx$.
\QQ{3} باستعمال التكامل بالتجزئة، عيّن الدالة الأصلية للدالة $x\mapsto(x+2)e^{x}$ على $\R$ التي تنعدم عند $0$.
\end{tamrin}

\begin{hal}
\QQ{1} نضع:
$\begin{cases}u(x)=\ln x\\ v'(x)=x\end{cases}$ \ إذن \ $\begin{cases}u'(x)=\frac1x\\ v(x)=\frac{x^{2}}{2}\end{cases}$
\begin{align*}
I&=\left[\frac{x^{2}}{2}\ln x\right]_1^e-\int_1^e\frac{x^{2}}{2}\cdot\frac1x\,dx=\frac{e^{2}}{2}-\int_1^e\frac{x}{2}\,dx\\
&=\frac{e^{2}}{2}-\left[\frac{x^{2}}{4}\right]_1^e=\frac{e^{2}}{2}-\frac{e^{2}-1}{4}=\bx{\frac{e^{2}+1}{4}}
\end{align*}

\QQ{2} نضع:
$\begin{cases}u(x)=2x+1\\ v'(x)=e^{-x}\end{cases}$ \ إذن \ $\begin{cases}u'(x)=2\\ v(x)=-e^{-x}\end{cases}$
\begin{align*}
J&=\Big[-(2x+1)e^{-x}\Big]_0^1+\int_0^12e^{-x}\,dx=\left(-3e^{-1}+1\right)+\Big[-2e^{-x}\Big]_0^1\\
&=1-\frac3e+\left(-\frac2e+2\right)=\bx{3-\frac{5}{e}}
\end{align*}

\QQ{3} الدالة الأصلية التي تنعدم عند $0$ هي $F(x)=\displaystyle\int_0^x(t+2)e^{t}\,dt$. نضع:
$\begin{cases}u(t)=t+2\\ v'(t)=e^{t}\end{cases}$ \ إذن \ $\begin{cases}u'(t)=1\\ v(t)=e^{t}\end{cases}$
\begin{align*}
F(x)&=\Big[(t+2)e^{t}\Big]_0^x-\int_0^xe^{t}\,dt=(x+2)e^{x}-2-\left(e^{x}-1\right)
\end{align*}
\[\bx{F(x)=(x+1)e^{x}-1}\]
\textbf{تحقق:} $F'(x)=e^{x}+(x+1)e^{x}=(x+2)e^{x}$ و $F(0)=1-1=0$ \ok.
\end{hal}

\begin{tamrin}{التكامل بالتجزئة مرتين}
\QQ{1} أحسب $I=\displaystyle\int_0^1x^{2}e^{x}\,dx$ (يمكن استعمال $\displaystyle\int_0^1xe^{x}\,dx=1$).
\QQ{2} أحسب $J=\displaystyle\int_1^e(\ln x)^{2}\,dx$، ثم قارن $I$ و $J$.
\QQ{3} أحسب $K=\displaystyle\int_1^ex^{2}\ln x\,dx$.

\begin{hal}

\[I=\Big[x^{2}e^{x}\Big]_0^1-2\int_0^1xe^{x}\,dx=e-2\times1=\bx{e-2}\]
(التكامل $\int_0^1xe^{x}dx=1$ حُسب هو أيضا بالتجزئة في التمرين 9).

\QQ{2} نضع:
$\begin{cases}u(x)=(\ln x)^{2}\\ v'(x)=1\end{cases}$ \ إذن \ $\begin{cases}u'(x)=\frac{2\ln x}{x}\\ v(x)=x\end{cases}$
\[J=\Big[x(\ln x)^{2}\Big]_1^e-\int_1^ex\cdot\frac{2\ln x}{x}\,dx=e-2\int_1^e\ln x\,dx\]
وقد وجدنا بالتجزئة (التمرين 9) أن $\displaystyle\int_1^e\ln x\,dx=1$، إذن \bx{J=e-2}.

\textbf{المقارنة:} \bx{I=J=e-2\approx0{,}72}.

\QQ{3} نضع:
$\begin{cases}u(x)=\ln x\\ v'(x)=x^{2}\end{cases}$ \ إذن \ $\begin{cases}u'(x)=\frac1x\\ v(x)=\frac{x^{3}}{3}\end{cases}$
\begin{align*}
K&=\left[\frac{x^{3}}{3}\ln x\right]_1^e-\int_1^e\frac{x^{2}}{3}\,dx=\frac{e^{3}}{3}-\left[\frac{x^{3}}{9}\right]_1^e=\frac{e^{3}}{3}-\frac{e^{3}-1}{9}=\bx{\frac{2e^{3}+1}{9}}
\end{align*}
\end{hal}

\section{اختيار من متعدد وصحيح أم خطأ}

% ===================== 16 =====================
\begin{tamrin}{اختيار من متعدد}
في كل حالة من الحالات التالية، اختر الجواب الصحيح الوحيد من بين الأجوبة الثلاثة المقترحة مع التعليل.

\QQ{1} دالة أصلية للدالة $x\mapsto2xe^{x^{2}}$ على $\R$ هي:
\ABC{$x\mapsto e^{x^{2}}$}{$x\mapsto x^{2}e^{x^{2}}$}{$x\mapsto2e^{x^{2}}$}

\QQ{2} $\displaystyle\int_0^1(2x+1)\,dx$ يساوي:
\ABC{$1$}{$2$}{$3$}

\QQ{3} $\displaystyle\int_1^e\frac{1}{x}\,dx$ يساوي:
\ABC{$e$}{$0$}{$1$}

\QQ{4} القيمة المتوسطة للدالة $x\mapsto x^{2}$ على المجال $[0;3]$ هي:
\ABC{$3$}{$9$}{$1$}

\QQ{5} $\displaystyle\int_{-1}^{1}\left(x^{3}+1\right)dx$ يساوي:
\ABC{$0$}{$2$}{$1$}

\QQ{6} $\displaystyle\int_0^{\ln3}e^{x}\,dx$ يساوي:
\ABC{$3$}{$\ln 3$}{$2$}
\end{tamrin}

\begin{hal}
\QQ{1} \textbf{الجواب الصحيح: (أ).}
نضع $u(x)=x^{2}$، $u'(x)=2x$: الدالة من الشكل $u'e^{u}$، أصليتها $e^{u}$، أي \bx{x\mapsto e^{x^{2}}}.
(تحقق: $\left(e^{x^{2}}\right)'=2xe^{x^{2}}$، بينما مشتقة $2e^{x^{2}}$ هي $4xe^{x^{2}}$.)

\QQ{2} \textbf{الجواب الصحيح: (ب).}
$\displaystyle\int_0^1(2x+1)\,dx=\Big[x^{2}+x\Big]_0^1=1+1=\bx{2}$.

\QQ{3} \textbf{الجواب الصحيح: (جـ).}
على $[1;e]$: $\displaystyle\int_1^e\frac1x\,dx=\Big[\ln x\Big]_1^e=\ln e-\ln1=\bx{1}$.

\QQ{4} \textbf{الجواب الصحيح: (أ).}
$\mu=\dfrac{1}{3-0}\displaystyle\int_0^3x^{2}\,dx=\frac13\left[\frac{x^{3}}{3}\right]_0^3=\frac13\times9=\bx{3}$.

\QQ{5} \textbf{الجواب الصحيح: (ب).}
$x\mapsto x^{3}$ فردية فتكاملها على $[-1;1]$ معدوم، و $\displaystyle\int_{-1}^{1}1\,dx=2$، ومنه النتيجة \bx{2}.

\QQ{6} \textbf{الجواب الصحيح: (جـ).}
$\displaystyle\int_0^{\ln3}e^{x}\,dx=\Big[e^{x}\Big]_0^{\ln3}=e^{\ln3}-e^{0}=3-1=\bx{2}$.
\end{hal}

% ===================== 17 =====================
\begin{tamrin}{صحيح أم خطأ}
أجب بـ «صحيح» أو «خطأ» مع التعليل: برّر كل عبارة صحيحة، وانقض كل عبارة خاطئة بمثال مضاد.
\QQ{1} كل دالة مستمرة على مجال $I$ تقبل دوالا أصلية على $I$.
\QQ{2} إذا كانت $F$ و $G$ دالتين أصليتين لنفس الدالة $f$ على مجال $I$ فإن $F-G$ دالة ثابتة على $I$.
\QQ{3} $\displaystyle\int_0^1xe^{x}\,dx=\left(\int_0^1x\,dx\right)\times\left(\int_0^1e^{x}\,dx\right)$.
\QQ{4} إذا كانت $f$ و $g$ مستمرتين على $[a;b]$ مع $a<b$، و $f(x)\leqslant g(x)$ من أجل كل $x$ من $[a;b]$، فإن $\displaystyle\int_a^bf(x)\,dx\leqslant\int_a^bg(x)\,dx$.
\QQ{5} الدالة $x\mapsto\ln\left(x^{2}+1\right)$ دالة أصلية للدالة $x\mapsto\dfrac{1}{x^{2}+1}$ على $\R$.
\QQ{6} إذا كانت $f$ مستمرة على $[a;b]$ و $\displaystyle\int_a^bf(x)\,dx\geqslant0$ فإن $f(x)\geqslant0$ من أجل كل $x$ من $[a;b]$.
\end{tamrin}

\begin{hal}
\QQ{1} \textbf{صحيح.} هذه مبرهنة الدرس: كل دالة مستمرة على مجال تقبل دوالا أصلية عليه (منها $x\mapsto\int_a^xf(t)\,dt$ حيث $a\in I$).

\QQ{2} \textbf{صحيح.} $(F-G)'=F'-G'=f-f=0$ على المجال $I$، والدالة التي مشتقتها معدومة على \textbf{مجال} ثابتة.

\QQ{3} \textbf{خطأ.} التكامل لا يحوّل الجداء إلى جداء. بالتجزئة (التمرين 9): $\displaystyle\int_0^1xe^{x}dx=1$، بينما
$\left(\displaystyle\int_0^1x\,dx\right)\left(\int_0^1e^{x}dx\right)=\dfrac12(e-1)\approx0{,}86\neq1$.

\QQ{4} \textbf{صحيح.} $g-f\geqslant0$ ومستمرة على $[a;b]$ مع $a<b$، فحسب الإيجابية $\displaystyle\int_a^b\big(g(x)-f(x)\big)dx\geqslant0$، وبالخطية نحصل على المطلوب.

\QQ{5} \textbf{خطأ.} $\left(\ln\left(x^{2}+1\right)\right)'=\dfrac{2x}{x^{2}+1}$، وهي تختلف عن $\dfrac{1}{x^{2}+1}$ (مثلا عند $x=0$: $0\neq1$).

\QQ{6} \textbf{خطأ.} مثال مضاد: $f(x)=x$ على $[-1;2]$: $\displaystyle\int_{-1}^{2}x\,dx=\left[\frac{x^{2}}{2}\right]_{-1}^{2}=2-\frac12=\frac32\geqslant0$، لكن $f(-1)=-1<0$.
(العكس صحيح: إذا كانت $f\geqslant0$ على $[a;b]$ مع $a<b$ فإن التكامل موجب.)
\end{hal}

\section{تمارين شعبة رياضيات}

% ===================== 18 =====================
\begin{tamrin}{متتالية معرفة بتكامل -- شعبة رياضيات}
من أجل كل عدد طبيعي $n$ نضع: $u_n=\displaystyle\int_0^1\frac{x^{n}}{1+x}\,dx$.
\QQ{1} أحسب $u_0$ و $u_1$.
\QQ{2} بيّن أن المتتالية $(u_n)$ متناقصة.
\QQ{3}
\begin{sq}
\qa بيّن أنه من أجل كل عدد طبيعي $n$: $0\leqslant u_n\leqslant\dfrac{1}{n+1}$.
\qb استنتج أن $(u_n)$ متقاربة، وعيّن نهايتها.
\end{sq}
\QQ{4}
\begin{sq}
\qa بيّن أنه من أجل كل عدد طبيعي $n$: $u_n+u_{n+1}=\dfrac{1}{n+1}$، ثم استنتج $u_2$.
\qb استنتج أنه من أجل كل عدد طبيعي $n\geqslant1$: $\dfrac{1}{2(n+1)}\leqslant u_n\leqslant\dfrac{1}{2n}$.
\qc أحسب $\limn n\,u_n$.
\end{sq}
\end{tamrin}

\begin{hal}
\QQ{1} $u_0=\displaystyle\int_0^1\frac{1}{1+x}\,dx=\Big[\ln(1+x)\Big]_0^1=\bx{\ln2}$.

$\dfrac{x}{1+x}=\dfrac{(1+x)-1}{1+x}=1-\dfrac{1}{1+x}$، إذن:
$u_1=\displaystyle\int_0^1\left(1-\frac{1}{1+x}\right)dx=\Big[x-\ln(1+x)\Big]_0^1=\bx{1-\ln2}$.

\QQ{2} من أجل كل $n\in\N$، حسب الخطية:
\[u_{n+1}-u_n=\int_0^1\frac{x^{n+1}-x^{n}}{1+x}\,dx=\int_0^1\frac{x^{n}(x-1)}{1+x}\,dx\]
من أجل $x\in[0;1]$: $x^{n}\geqslant0$ و $x-1\leqslant0$ و $1+x>0$، إذن $\dfrac{x^{n}(x-1)}{1+x}\leqslant0$، وبما أن $0<1$ فالتكامل سالب: $u_{n+1}-u_n\leqslant0$.
إذن \bxt{المتتالية $(u_n)$ متناقصة}.

\QQ{3}
\pa من أجل $x\in[0;1]$: $1\leqslant1+x$ ، إذن $0\leqslant\dfrac{1}{1+x}\leqslant1$، وبالضرب في $x^{n}\geqslant0$:
\[0\leqslant\frac{x^{n}}{1+x}\leqslant x^{n}\]
بالمكاملة على $[0;1]$ (الترتيب محفوظ): $0\leqslant u_n\leqslant\displaystyle\int_0^1x^{n}dx=\left[\frac{x^{n+1}}{n+1}\right]_0^1$، ومنه:
\[\bx{0\leqslant u_n\leqslant\frac{1}{n+1}}\]
\pb $\limn\dfrac{1}{n+1}=0$، وحسب مبرهنة الحصر (الدركي): \bx{\limn u_n=0}، إذن $(u_n)$ متقاربة.
(ويمكن أيضا: متناقصة ومحدودة من الأسفل بـ $0$ فهي متقاربة.)

\QQ{4}
\pa حسب الخطية:
\[u_n+u_{n+1}=\int_0^1\frac{x^{n}+x^{n+1}}{1+x}\,dx=\int_0^1\frac{x^{n}(1+x)}{1+x}\,dx=\int_0^1x^{n}dx=\bx{\frac{1}{n+1}}\]
من أجل $n=1$: $u_1+u_2=\dfrac12$، إذن $u_2=\dfrac12-(1-\ln2)=\bx{\ln2-\frac{1}{2}}$.

\pb $(u_n)$ متناقصة إذن $u_{n+1}\leqslant u_n$ من أجل كل $n$.
\begin{itemize}
\item من جهة: $\dfrac{1}{n+1}=u_n+u_{n+1}\leqslant u_n+u_n=2u_n$، ومنه $u_n\geqslant\dfrac{1}{2(n+1)}$.
\item من جهة أخرى، من أجل $n\geqslant1$ نطبق المساواة على الرتبة $n-1$: $u_{n-1}+u_n=\dfrac1n$، و $u_n\leqslant u_{n-1}$، إذن $2u_n\leqslant u_{n-1}+u_n=\dfrac1n$، ومنه $u_n\leqslant\dfrac{1}{2n}$.
\end{itemize}
\[\bx{\frac{1}{2(n+1)}\leqslant u_n\leqslant\frac{1}{2n}}\quad;\ n\geqslant1\]
\pc بالضرب في $n>0$: $\dfrac{n}{2(n+1)}\leqslant n\,u_n\leqslant\dfrac12$.
و $\limn\dfrac{n}{2(n+1)}=\limn\dfrac{1}{2\left(1+\frac1n\right)}=\dfrac12$، إذن حسب الحصر:
\[\bx{\limn n\,u_n=\frac{1}{2}}\]
\textbf{تحقق عددي:} $u_1\approx0{,}307$ و $u_2\approx0{,}193$، وكلاهما في الحصر $\left[\frac14;\frac12\right]$ ثم $\left[\frac16;\frac14\right]$.
\end{hal}

\begin{tamrin}{دالة معرفة بتكامل -- شعبة رياضيات}
نعتبر الدالة $F$ المعرفة على $\R$ بـ: $F(x)=\displaystyle\int_0^xe^{-t^{2}}\,dt$، و $(C_F)$ تمثيلها البياني في معلم متعامد ومتجانس.
\QQ{1} برّر أن $F$ قابلة للاشتقاق على $\R$، أحسب $F'(x)$، ثم استنتج اتجاه تغير $F$. أحسب $F(0)$.
\QQ{2} نضع من أجل كل $x$ من $\R$: $G(x)=F(x)+F(-x)$. أحسب $G'(x)$ ثم استنتج أن $F$ دالة فردية.
\QQ{3}
\begin{sq}
\qa بيّن أنه من أجل كل $t\geqslant1$: $e^{-t^{2}}\leqslant e^{-t}$.
\qb استنتج أنه من أجل كل $x\geqslant1$: $F(x)\leqslant F(1)+\dfrac1e-e^{-x}$.
\qc بيّن أن $F(1)\leqslant1$، ثم استنتج أنه من أجل كل $x\geqslant0$: $F(x)<1+\dfrac1e$.
\qd نقبل أن الدالة المتزايدة والمحدودة من الأعلى على $[0;+\infty[$ تقبل نهاية منتهية $\ell$ عند $+\infty$. ماذا تستنتج؟ وما هي النهاية عند $-\infty$؟
\end{sq}
\QQ{4}
\begin{sq}
\qa أكتب معادلة المماس $(T)$ لـ $(C_F)$ عند المبدأ $O$.
\qb بيّن أنه من أجل كل $x\geqslant0$: $F(x)\leqslant x$، ثم استنتج الوضع النسبي لـ $(C_F)$ و $(T)$ على $\R$.
\end{sq}
\QQ{5} شكّل جدول تغيرات $F$، ثم أنشئ $(T)$ و $(C_F)$ (نأخذ $\ell\approx0{,}89$).
\end{tamrin}

\begin{hal}
\QQ{1} الدالة $f:t\mapsto e^{-t^{2}}$ مستمرة على $\R$، إذن $F:x\mapsto\displaystyle\int_0^xf(t)\,dt$ هي الدالة الأصلية لـ $f$ على $\R$ التي تنعدم عند $0$. ومنه $F$ قابلة للاشتقاق على $\R$ و:
\[\bx{F'(x)=e^{-x^{2}}}\qquad;\qquad \bx{F(0)=\int_0^0e^{-t^{2}}dt=0}\]
بما أن $F'(x)=e^{-x^{2}}>0$ من أجل كل $x$، فإن \bxt{$F$ متزايدة تماما على $\R$}.

\QQ{2} $G$ قابلة للاشتقاق على $\R$، وباستعمال مشتقة الدالة المركبة $\big(F(-x)\big)'=-F'(-x)$:
\[G'(x)=F'(x)-F'(-x)=e^{-x^{2}}-e^{-(-x)^{2}}=0\]
إذن $G$ ثابتة على $\R$ (مجال)، و $G(0)=2F(0)=0$، ومنه $G(x)=0$ أي $F(-x)=-F(x)$ من أجل كل $x$.
و $\R$ متناظر بالنسبة إلى $0$، إذن \bxt{$F$ دالة فردية} و $(C_F)$ متناظر بالنسبة إلى المبدأ $O$.

\QQ{3}
\pa من أجل $t\geqslant1$: $t^{2}-t=t(t-1)\geqslant0$، أي $-t^{2}\leqslant-t$، والدالة $\exp$ متزايدة تماما، إذن \bx{e^{-t^{2}}\leqslant e^{-t}}.

\pb حسب علاقة شال، من أجل $x\geqslant1$: $F(x)=F(1)+\displaystyle\int_1^xe^{-t^{2}}dt$. وبمكاملة المتباينة (أ) على $[1;x]$ (مع $1\leqslant x$):
\[\int_1^xe^{-t^{2}}dt\leqslant\int_1^xe^{-t}dt=\Big[-e^{-t}\Big]_1^x=\frac1e-e^{-x}\]
ومنه: \bx{F(x)\leqslant F(1)+\frac{1}{e}-e^{-x}}.

\pc من أجل $t\in[0;1]$: $e^{-t^{2}}\leqslant1$، فبالمكاملة: $F(1)\leqslant\displaystyle\int_0^11\,dt=1$.
\begin{itemize}
\item إذا كان $x\geqslant1$: من (ب) و $e^{-x}>0$: $F(x)<F(1)+\dfrac1e\leqslant1+\dfrac1e$.
\item إذا كان $0\leqslant x\leqslant1$: $F$ متزايدة إذن $F(x)\leqslant F(1)\leqslant1<1+\dfrac1e$.
\end{itemize}
إذن من أجل كل $x\geqslant0$: \bx{F(x)<1+\frac{1}{e}}.

\pd $F$ متزايدة على $[0;+\infty[$ ومحدودة من الأعلى بـ $1+\frac1e$، إذن تقبل نهاية منتهية: $\lim\limits_{x\to+\infty}F(x)=\ell$ مع $0<\ell\leqslant1+\frac1e$.
هندسيا: المستقيم ذو المعادلة $y=\ell$ مقارب أفقي لـ $(C_F)$ عند $+\infty$.
وبما أن $F$ فردية: $\lim\limits_{x\to-\infty}F(x)=\lim\limits_{x\to+\infty}\big(-F(x)\big)=\bx{-\ell}$، والمستقيم $y=-\ell$ مقارب أفقي عند $-\infty$.

\begin{tamrin}{مساحة بدلالة وسيط ونهايتها -- شعبة رياضيات}
$f$ الدالة المعرفة على $\R$ بـ: $f(x)=x-1+\dfrac{2}{e^{x}+1}$، و $(C_f)$ تمثيلها البياني في معلم متعامد ومتجانس حيث $\|\vec i\|=2\,\text{cm}$.
\QQ{1} بيّن أن $f$ فردية، وأن $f$ متزايدة تماما على $\R$.
\QQ{2} بيّن أن المستقيم $(\Delta):y=x-1$ مقارب مائل لـ $(C_f)$ عند $+\infty$، وأن $(\Delta'):y=x+1$ مقارب مائل عند $-\infty$، ثم أدرس الوضع النسبي لـ $(C_f)$ و $(\Delta)$.
\QQ{3} تحقق أنه من أجل كل عدد حقيقي $x$: $\dfrac{2}{e^{x}+1}=\dfrac{2e^{-x}}{1+e^{-x}}$.
\QQ{4} $\lambda$ عدد حقيقي موجب تماما. نسمي $\mathcal{A}(\lambda)$ مساحة الحيز المستوي المحدد بـ $(C_f)$ و $(\Delta)$ والمستقيمين اللذين معادلتاهما $x=0$ و $x=\lambda$.
\begin{sq}
\qa أحسب $\mathcal{A}(\lambda)$ بوحدة المساحة ثم بـ $\text{cm}^{2}$.
\qb أحسب $\lim\limits_{\lambda\to+\infty}\mathcal{A}(\lambda)$ وفسّر النتيجة.
\qc عيّن قيمة $\lambda$ التي يكون من أجلها $\mathcal{A}(\lambda)=\ln2\ \mathrm{u.a}$.
\end{sq}
\end{tamrin}

\begin{hal}
\QQ{1} $\R$ متناظر بالنسبة إلى $0$. من أجل كل $x$، بضرب البسط والمقام في $e^{x}$:
\begin{align*}
f(-x)&=-x-1+\frac{2}{e^{-x}+1}=-x-1+\frac{2e^{x}}{1+e^{x}}=-x-1+\frac{2\left(e^{x}+1\right)-2}{e^{x}+1}\\
&=-x-1+2-\frac{2}{e^{x}+1}=-\left(x-1+\frac{2}{e^{x}+1}\right)=-f(x)
\end{align*}
إذن \bxt{$f$ دالة فردية}.

$f$ قابلة للاشتقاق على $\R$، و $\left(\dfrac{2}{e^{x}+1}\right)'=-\dfrac{2e^{x}}{\left(e^{x}+1\right)^{2}}$، إذن:
\[f'(x)=1-\frac{2e^{x}}{\left(e^{x}+1\right)^{2}}=\frac{e^{2x}+2e^{x}+1-2e^{x}}{\left(e^{x}+1\right)^{2}}=\frac{e^{2x}+1}{\left(e^{x}+1\right)^{2}}>0\]
إذن \bxt{$f$ متزايدة تماما على $\R$}.

\QQ{2} $f(x)-(x-1)=\dfrac{2}{e^{x}+1}$ و $\lim\limits_{x\to+\infty}e^{x}=+\infty$، إذن $\lim\limits_{x\to+\infty}\big[f(x)-(x-1)\big]=0$: $(\Delta)$ مقارب مائل عند $+\infty$.

$f(x)-(x+1)=\dfrac{2}{e^{x}+1}-2$ و $\lim\limits_{x\to-\infty}e^{x}=0$، إذن $\lim\limits_{x\to-\infty}\big[f(x)-(x+1)\big]=2-2=0$: $(\Delta')$ مقارب مائل عند $-\infty$.

حسب جدول الوضع النسبي، $\big|f(x)-(x-1)\big|=\dfrac{2}{e^{x}+1}$، وباستعمال السؤال 3 مع $u(x)=1+e^{-x}>0$ و $u'(x)=-e^{-x}$: $\dfrac{2e^{-x}}{1+e^{-x}}=-2\dfrac{u'(x)}{u(x)}$، أصليته $-2\ln\left(1+e^{-x}\right)$:
\begin{align*}
\mathcal{A}(\lambda)&=\int_0^{\lambda}\frac{2e^{-x}}{1+e^{-x}}\,dx=\Big[-2\ln\left(1+e^{-x}\right)\Big]_0^{\lambda}=-2\ln\left(1+e^{-\lambda}\right)+2\ln2
\end{align*}
\[\bx{\mathcal{A}(\lambda)=2\ln2-2\ln\left(1+e^{-\lambda}\right)\ \mathrm{u.a}}\]
$1\,\mathrm{u.a}=2\times2=4\,\text{cm}^{2}$، ومنه:
\[\bx{\mathcal{A}(\lambda)=8\ln2-8\ln\left(1+e^{-\lambda}\right)\ \mathrm{cm}^{2}}\]

\pb $\lim\limits_{\lambda\to+\infty}e^{-\lambda}=0$ و $\ln$ مستمرة عند $1$، إذن $\lim\limits_{\lambda\to+\infty}\ln\left(1+e^{-\lambda}\right)=\ln1=0$، ومنه:
\[\bx{\lim_{\lambda\to+\infty}\mathcal{A}(\lambda)=2\ln2\ \mathrm{u.a}=8\ln2\ \mathrm{cm}^{2}\approx5{,}55\,\mathrm{cm}^{2}}\]

\pc $\mathcal{A}(\lambda)=\ln2\iff2\ln2-2\ln\left(1+e^{-\lambda}\right)=\ln2\iff\ln\left(1+e^{-\lambda}\right)=\dfrac{\ln2}{2}=\ln\sqrt2$.
بما أن $\ln$ متباينة: $1+e^{-\lambda}=\sqrt2$، أي $e^{-\lambda}=\sqrt2-1>0$، ومنه $\lambda=-\ln\left(\sqrt2-1\right)$.
و $\left(\sqrt2-1\right)\left(\sqrt2+1\right)=1$، إذن $-\ln\left(\sqrt2-1\right)=\ln\left(\sqrt2+1\right)$:
\[\bx{\lambda=\ln\left(\sqrt{2}+1\right)\approx0{,}88}\]
(وهو موجب تماما فهو مقبول؛ عنده تساوي المساحة نصف المساحة الكلية $2\ln2$.)
