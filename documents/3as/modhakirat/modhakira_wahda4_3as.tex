\ModhHead{23}{3 ع ت}{تحليل}{التزايد المقارن ودراسة الدوال}{02 سا}{قوى عدد حقيقي موجب تماماً ، الدالة $x\mapsto a^x$}%
{\begin{itemize}\item الدالتان الأسية واللوغاريتمية.\end{itemize}}%
{\begin{itemize}\item تعريف $a^x$ واستعمال خواصه ، ودراسة الدالة $x\mapsto a^x$.\end{itemize}}%
{المنهاج ، الكتاب المدرسي ، مراجع ، الحاسبة البيانية.}
\begin{Sayr}
\Row{مرحلة الانطلاق}{%
\textbf{التهيئة النفسية:} نعرف $2^3$ و $2^{-1}$ و $2^{\frac12}$: فما معنى $2^{\sqrt2}$ ؟}{10 د}{سؤال تحفيزي.}
\Row{التشخيص والاكتشاف}{%
\Lab{cEx}{مناقشة نشاط 01 صفحة 120}\par\ck\ من أجل $n$ صحيح: $a^n=e^{n\ln a}$ ؛ نمدد هذا الدستور إلى كل عدد حقيقي $x$.\par
}{35 د}{نشاط الكتاب.}
\Row{بناء المعارف}{%
\Lab{cDef}{تعريف}\ من أجل $a>0$ و $x$ حقيقي: $a^x=e^{x\ln a}$.\par\Lab{cThm}{خاصية}\ $a^{x+y}=a^xa^y$ ؛ $(a^x)^y=a^{xy}$ ؛ $(ab)^x=a^xb^x$ ؛ $\ln(a^x)=x\ln a$.\par}{50 د}{جدول الخواص.}
\Row{}{%
\Lab{cThm}{الدالة $x\mapsto a^x$}\ مشتقتها $(\ln a)\,a^x$: متزايدة تماماً إذا كان $a>1$ ، ومتناقصة تماماً إذا كان $0<a<1$.\par\Lab{cEx}{مثال}\ $2^x=8\Leftrightarrow x=3$ ؛ $3^{2x}=5\Leftrightarrow2x\ln3=\ln5\Leftrightarrow x=\dfrac{\ln5}{2\ln3}$.\par}{}{}
\Row{}{%
\Lab{cThm!80!black}{حذار}\ $a^x$ معرف من أجل $a>0$ فقط.}{}{}
\Row{التقويم}{%
\Lab{cEx}{تطبيق}\par حل في $\mathbb R$: $\left(\frac12\right)^x\geqslant\frac18$.\par\textbf{الحل:} $\left(\frac12\right)^x\geqslant\left(\frac12\right)^3$ والدالة متناقصة: $x\leqslant3$.\par\textbf{واجب منزلي:} تمارين من باب التزايد المقارن.\par\smallskip\textbf{ملاحظات حول سير الحصة:} \dotfill}{25 د}{تقويم فردي.}
\end{Sayr}

\ModhHead{25}{3 ع ت}{تحليل}{التزايد المقارن ودراسة الدوال}{02 سا}{التزايد المقارن}%
{\begin{itemize}\item نهايات $e^x$ و $\ln x$ ، الحصر.\end{itemize}}%
{\begin{itemize}\item معرفة النهايات المرجعية للتزايد المقارن وتوظيفها في رفع حالات عدم التعيين.\end{itemize}}%
{المنهاج ، الكتاب المدرسي ، مراجع ، الحاسبة البيانية.}
\begin{Sayr}
\Row{مرحلة الانطلاق}{%
\textbf{التهيئة النفسية:} من «يكبر أسرع» لما $x\to+\infty$: $e^x$ أم $x^{10}$ ؟ جرب $x=10$ ، $50$ ، $100$.}{10 د}{تحدٍّ بالحاسبة.}
\Row{التشخيص والاكتشاف}{%
\Lab{cEx}{وضعية مقترحة}\par\ck\ جدول قيم بالحاسبة: في البداية $x^{10}$ أكبر ، ثم $e^x$ يتجاوزه نهائياً.\par
\ck\ $g(x)=e^x-\frac{x^2}2$: $g^{\prime\prime}(x)=e^x-1$ ، ونستنتج $e^x>\frac{x^2}2$ لما $x>0$ ، إذن $\dfrac{e^x}x>\dfrac x2$.\par
}{35 د}{برهان مقارنة.}
