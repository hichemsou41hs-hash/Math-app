\begin{Sayr}
\Row{مرحلة الانطلاق}{%
\textbf{التهيئة النفسية:} كم تشابهاً مباشراً يحول $A$ إلى $A^{\prime}$ و $B$ إلى $B^{\prime}$ ؟}{10 د}{سؤال.}
\Row{التشخيص والاكتشاف}{%
\Lab{cEx}{وضعية مقترحة}\par\ck\ نبحث عن $a$ و $b$ حيث $z_{A^{\prime}}=az_A+b$ و $z_{B^{\prime}}=az_B+b$: جملة لها حل وحيد.\par
}{35 د}{جملة معادلتين.}
\Row{بناء المعارف}{%
\Lab{cThm}{خاصية}\par\ck\ التشابه المباشر يحافظ على نسب المسافات والزوايا الموجهة والاستقامية والتوازي.\par
\ck\ من أجل $A\neq B$ و $A^{\prime}\neq B^{\prime}$ يوجد تشابه مباشر وحيد: $a=\dfrac{z_{B^{\prime}}-z_{A^{\prime}}}{z_B-z_A}$ ، $b=z_{A^{\prime}}-az_A$.\par
\Lab{cEx}{مثال}\ $A(0)\mapsto A^{\prime}(1)$ ، $B(1)\mapsto B^{\prime}(1+i)$: $a=i$ ، $b=1$: $z^{\prime}=iz+1$ ، دوران زاويته $\frac\pi2$ ومركزه $\dfrac{1+i}2$.\par}{50 د}{رسم الصور.}
