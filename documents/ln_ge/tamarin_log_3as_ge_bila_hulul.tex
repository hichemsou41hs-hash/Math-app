\section{تمارين من الاختبارات (البكالوريا التجريبية)}

\begin{tamrin}{بكالوريا تجريبية 2026 -- تسيير واقتصاد (بوقيرات) -- التمرين الثالث}
ليكن في $\R$ كثير الحدود $P(x)$ حيث: $P(x)=2x^3-x^2-4050x+2025$.
\QQ{1} تحقق أنه من أجل كل عدد حقيقي $x$: $P(x)=(2x-1)\left(x^2-2025\right)$ ، ثم حل في $\R$ المعادلة $P(x)=0$.
\QQ{2} استنتج حلول المعادلتين:
\begin{sq}
\qa $2(\log x)^3-(\log x)^2-4050\log x+2025=0$ حيث $x>0$.
\qb \EXP\ $2e^{3x}-e^{2x}-4050e^x+2025=0$.
\end{sq}
\QQ{3} حل في المجال $]0;+\infty[$ المتراجحة: $(2\ln x-1)\left((\ln x)^2-2025\right)\geqslant0$.
\end{tamrin}

\begin{tamrin}{بكالوريا تجريبية 2026 -- تسيير واقتصاد (بوقيرات) -- التمرين الرابع}
\textbf{I)} الدالة $g$ معرفة على $]0;+\infty[$ بـ: $g(x)=3x^3-2+4\ln x$ ، وهي متزايدة تمامًا من $-\infty$ إلى $+\infty$.
\QQ{1} بيّن أن المعادلة $g(x)=0$ تقبل حلًا وحيدًا $\alpha$ حيث: $0{,}9<\alpha<1$.
\QQ{2} استنتج إشارة $g(x)$ حسب قيم $x$ من $]0;+\infty[$.\par
\textbf{II)} $f$ الدالة المعرفة على $]0;+\infty[$ بـ: $f(x)=3x-2-\dfrac{2\ln x}{x^2}$ ، و $(C_f)$ تمثيلها البياني (وحدة الطول $2\,cm$).
\QQ{1} أحسب $\Lm{0^+}f(x)$ و $\Lm{+\infty}f(x)$. (يعطى: $\Lm{+\infty}\dfrac{\ln x}{x^2}=0$).
\QQ{2}
\begin{sq}
\qa بيّن أن المستقيم $(\Delta)$ ذا المعادلة $y=3x-2$ مقارب مائل للمنحنى $(C_f)$.
\qb ادرس وضعية المنحنى $(C_f)$ بالنسبة إلى المستقيم $(\Delta)$.
\end{sq}
\QQ{3}
\begin{sq}
\qa بيّن أنه من أجل كل $x$ من $]0;+\infty[$: $f'(x)=\dfrac{g(x)}{x^3}$.
\qb استنتج اتجاه تغير الدالة $f$ ثم شكّل جدول تغيراتها.
\qc ليكن $(T)$ مماس $(C_f)$ عند النقطة ذات الفاصلة $x_0$ ، معادلته $y=3x-2-e^{-1}$. أحسب $x_0$.
\end{sq}
\QQ{4} ارسم كلًا من $(\Delta)$ و $(C_f)$. (نأخذ $f(\alpha)\approx0{,}9$).
\QQ{5} $(\star)$ $H$ الدالة المعرفة على $]0;+\infty[$ بـ: $H(x)=\dfrac{1+\ln x}{x}$.
\begin{sq}
\qa بيّن أن $H$ دالة أصلية للدالة $x\mapsto-\dfrac{\ln x}{x^2}$.
\qb أحسب بـ $cm^2$ مساحة الحيز المحدد بالمنحنى $(C_f)$ والمستقيم $(\Delta)$ والمستقيمين $x=1$ و $x=e$.
\end{sq}
\end{tamrin}

\begin{tamrin}{بكالوريا تجريبية 2026 -- تسيير واقتصاد (تلمسان) -- التمرين الرابع}
\QQ{1} الدالة $g$ معرفة على $]0;+\infty[$ بـ: $g(x)=2x^3-1+2\ln(2x)$ ، وهي متزايدة تمامًا من $-\infty$ إلى $+\infty$.
\begin{sq}
\qa بيّن أن المعادلة $g(x)=0$ تقبل حلًا وحيدًا $\alpha$ حيث $0{,}6<\alpha<0{,}7$.
\qb استنتج إشارة $g(x)$ حسب قيم $x$.
\end{sq}
\QQ{2} الدالة $f$ معرفة على $]0;+\infty[$ بـ: $f(x)=2x-1-\dfrac{\ln(2x)}{x^2}$ ، و $(C)$ تمثيلها البياني.
\begin{sq}
\qa أحسب $\Lm{0^+}f(x)$ و $\Lm{+\infty}f(x)$. (يعطى: $\Lm{+\infty}\dfrac{\ln(2x)}{x^2}=0$).
\qb بيّن أن المستقيم $(\Delta)$ ذا المعادلة $y=2x-1$ مقارب للمنحنى $(C)$.
\qc ادرس وضعية المنحنى $(C)$ بالنسبة إلى المستقيم $(\Delta)$.
\end{sq}
\QQ{3} بيّن أنه من أجل كل $x$ من $]0;+\infty[$: $f'(x)=\dfrac{g(x)}{x^3}$ ، ثم شكّل جدول تغيرات $f$.
\QQ{4} أرسم $(\Delta)$ و $(C)$.
\QQ{5} $(\star)$ الدالة $H$ معرفة على $]0;+\infty[$ بـ: $H(x)=\dfrac{1+\ln x}{x}$.
\begin{sq}
\qa بيّن أن $H$ أصلية للدالة $h:x\mapsto-\dfrac{\ln x}{x^2}$ على $]0;+\infty[$.
\qb أحسب مساحة الحيز المستوي المحدد بالمنحنى $(C)$ ، محور الفواصل والمستقيمين $x=1$ و $x=e$.
\end{sq}
\end{tamrin}

\begin{tamrin}{بكالوريا تجريبية}
\textbf{I)} الدالة العددية $g$ معرفة على المجال $]0;+\infty[$ بـ: $g(x)=x\ln x-1$.

\textbf{II)} الدالة العددية $f$ معرفة على المجال $]0;+\infty[$ بـ: $f(x)=(x-1)(\ln x-1)$ ، و $(C_f)$ تمثيلها البياني.

\begin{sq}

\QQ{2}
\begin{sq}

\QQ{3}

\end{sq}
\end{tamrin}

\section{مسائل البكالوريا -- شعبة تسيير واقتصاد}

\begin{tamrin}{بكالوريا 2026 -- الموضوع الأول -- التمرين الثالث}
يمثل الجدول التالي تغيرات الدالة $f$ المعرفة على $\R$ بـ: $f(x)=\ln\left(2x^2+ax+b\right)$ حيث $a$ و $b$ عددان حقيقيان.
\begin{TAB}
\tkzTabInit[lgt=1.6,espcl=3]{$x$/1,$f(x)$/2}{$-\infty$,$-\frac12$,$+\infty$}
\tkzTabVar{+/$+\infty$,-/$-\ln2$,+/$+\infty$}
\end{TAB}
\QQ{1}
\begin{sq}
\qa أحسب $f'(x)$ بدلالة $a$ و $b$.
\qb انطلاقًا من جدول التغيرات ، عيّن العددين $a$ و $b$.
\end{sq}
\QQ{2} نفرض أن: $f(x)=\ln\left(2x^2+2x+1\right)$.
\begin{sq}
\qa حل في $\R$ المعادلة: $f(x)=0$.
\qb حدّد إشارة كل من $f(x)$ و $f'(x)$.
\end{sq}
\QQ{3} $g$ الدالة المعرفة على $]0;+\infty[$ بـ: $g(x)=\dfrac1{f(x)}$. ادرس اتجاه تغير الدالة $g$ ثم شكّل جدول تغيراتها.
\end{tamrin}

\begin{tamrin}{بكالوريا 2025 -- الموضوع الأول -- التمرين الرابع}
$f$ الدالة المعرفة بـ: $f(0)=0$ ، ومن أجل كل $x$ من $]0;+\infty[$: $f(x)=\dfrac14x^2(2\ln x-3)$ ، و $(C_f)$ تمثيلها البياني.
\QQ{1} أحسب $\Lm{+\infty}f(x)$.
\QQ{2}
