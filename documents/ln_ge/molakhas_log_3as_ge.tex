\section{التعريف: الدالة الأصلية لمقلوب $x$}
\begin{tarif}
الدالة $\LW{x\mapsto\dfrac1x}$ مستمرة على $\LW{]0;+\infty[}$ ، فهي تقبل دوالًا أصلية عليه.\par
\textbf{الدالة اللوغاريتمية النيبيرية} ، ونرمز لها بـ $\LW{\ln}$ ، هي \textbf{الدالة الأصلية للدالة $\LW{x\mapsto\dfrac1x}$ على $\LW{]0;+\infty[}$ والتي تنعدم عند $\LW{1}$}:
من أجل كل $x>0$:
\[\LW{(\ln x)'=\frac1x}\qquad;\qquad\LW{\ln1=0}\]
\end{tarif}

\begin{khassia}[نتائج مباشرة]
\begin{itemize}
\item مجموعة تعريف $\LW{\ln}$ هي $\LW{]0;+\infty[}$ ، و $\LW{\ln}$ قابلة للاشتقاق (إذن مستمرة) عليها.
\item $\LW{(\ln x)'=\dfrac1x>0}$: الدالة $\LW{\ln}$ \textbf{متزايدة تمامًا} على $\LW{]0;+\infty[}$.
\item العدد $\LW{e\approx2{,}718}$ هو العدد الذي يحقق $\bx{\ln e=1}$.
\item إشارة $\LW{\ln x}$: \quad $\LW{\ln x<0}$ إذا كان $\LW{0<x<1}$ ؛ \quad $\LW{\ln1=0}$ ؛ \quad $\LW{\ln x>0}$ إذا كان $\LW{x>1}$.
\end{itemize}
\end{khassia}

\begin{mithal}[بالآلة الحاسبة]
$\ln2\approx\bx{0{,}69}$ ؛ \quad $\ln3\approx\bx{1{,}10}$ ؛ \quad $\ln\dfrac12\approx\bx{-0{,}69}$ ؛ \quad $\ln10\approx\bx{2{,}30}$.
\end{mithal}

\section{الخواص الجبرية}
\begin{khassia}[من أجل كل عددين حقيقيين موجبين تمامًا $a$ و $b$ ، وكل عدد صحيح $n$]
\begin{center}
\renewcommand{\arraystretch}{1.9}
\begin{tabular}{|c|c|}\hline
\rowcolor{cSec!12}$\LW{\ln\dfrac ab=\ln a-\ln b}$ & $\LW{\ln(ab)=\ln a+\ln b}$\\\hline
$\LW{\ln a^n=n\ln a}$ & $\LW{\ln\dfrac1a=-\ln a}$\\\hline
$\LW{\ln e^n=n}$ & $\LW{\ln\sqrt a=\dfrac12\ln a}$\\\hline
\end{tabular}
\end{center}
\end{khassia}

\begin{molahada}[برهان الخاصية الأساسية]
نثبت $a>0$ ونضع $h(x)=\ln(ax)-\ln x$ على $]0;+\infty[$: $h'(x)=\dfrac a{ax}-\dfrac1x=0$ ، إذن $h$ ثابتة ، و $h(1)=\ln a$. ومنه $\ln(ax)=\ln a+\ln x$.
\end{molahada}

\begin{khassia}[نتائج: المقارنة]
من أجل $\LW{a>0}$ و $\LW{b>0}$ (لأن $\LW{\ln}$ متزايدة تمامًا):
\[\LW{\LW{\ln a=\ln b\iff a=b\qquad\qquad\ln a<\ln b\iff a<b}}\]
ومن أجل كل عدد حقيقي $\LW{k}$: $\LW{k=k\ln e=\ln e^k}$ ، إذن $\bx{\ln x=k\iff x=e^k}$.
\end{khassia}

\section{تبسيط عبارات لوغاريتمية}
\begin{tariqa}[كتابة عدد بدلالة $\ln2$ و $\ln3$]
نحلل الأعداد إلى جداء عوامل أولية ، ثم نطبق $\LW{\ln(ab)=\ln a+\ln b}$ و $\LW{\ln a^n=n\ln a}$.
\end{tariqa}

\begin{mithal}[اكتب بدلالة $\ln2$ و $\ln3$]
\begin{itemize}
\item $\ln18=\ln(2\times3^2)=\bx{\ln2+2\ln3}$.
\item $\ln\dfrac{16}9=\ln2^4-\ln3^2=\bx{4\ln2-2\ln3}$.
\item $\ln\sqrt{12}=\dfrac12\ln(2^2\times3)=\bx{\ln2+\dfrac12\ln3}$.
\item $A=\ln12-\ln3+\ln\dfrac14=\ln\dfrac{12}{3\times4}=\ln1=\bx{0}$.
\end{itemize}
\end{mithal}

\begin{mithal}[أعداد فيها $e$]
\begin{itemize}
\item $\ln e^3=\bx{3}$ ؛ \quad $\ln\dfrac1e=-\ln e=\bx{-1}$ ؛ \quad $\ln\sqrt e=\dfrac12\ln e=\bx{\dfrac12}$.
\item $B=\ln(e^2\sqrt e)=\ln e^2+\ln\sqrt e=2+\dfrac12=\bx{\dfrac52}$.
\item $C=\ln\left(\sqrt5+1\right)+\ln\left(\sqrt5-1\right)=\ln\left(5-1\right)=\ln4=\bx{2\ln2}$.
\end{itemize}
\end{mithal}

\section{المعادلات اللوغاريتمية}
\begin{tariqa}[حل معادلة لوغاريتمية]
\begin{enumerate}
\item نعيّن مجموعة الحلول الممكنة $\LW{D}$ (العبارات داخل $\LW{\ln}$ موجبة تمامًا).
\item نكتب المعادلة على الشكل $\LW{\ln A=\ln B}$ (أي $\LW{A=B}$) ، أو $\LW{\ln A=k}$ (أي $\LW{A=e^k}$).
\item نحتفظ بالحلول التي تنتمي إلى $\LW{D}$.
\end{enumerate}
\end{tariqa}

\begin{mithal}[حل في $\R$]
\QQ{1} $\ln(2x-1)=\ln(x+3)$: \quad $D$: $2x-1>0$ و $x+3>0$ ، أي $D=\left]\dfrac12;+\infty\right[$.
\[2x-1=x+3\ \Rightarrow\ x=4\in D\qquad;\qquad\bx{S=\{4\}}\]
\QQ{2} $\ln(3x-2)=0$: \quad $D=\left]\dfrac23;+\infty\right[$:
\[\ln(3x-2)=\ln1\ \Rightarrow\ 3x-2=1\ \Rightarrow\ x=1\in D\qquad;\qquad\bx{S=\{1\}}\]
\QQ{3} $\ln(x+1)=4$: \quad $D=]-1;+\infty[$:
\[\ln(x+1)=\ln e^4\ \Rightarrow\ x+1=e^4\ \Rightarrow\ x=e^4-1\in D\qquad;\qquad\bx{S=\left\{e^4-1\right\}}\]
\QQ{4} $\ln x+\ln(x-1)=\ln6$: \quad $D=]1;+\infty[$:
\[\ln\big(x(x-1)\big)=\ln6\ \Rightarrow\ x^2-x-6=0\ \Rightarrow\ (x-3)(x+2)=0\]
$x=3\in D$ أو $x=-2\notin D$ ، إذن $\bx{S=\{3\}}$.
\QQ{5} \textbf{(بكالوريا 2018)} $\ln(x+2)+\ln(8-x)=\ln16$: \quad $D=]-2;8[$:
\[(x+2)(8-x)=16\ \Rightarrow\ -x^2+6x+16=16\ \Rightarrow\ x(6-x)=0\]
$x=0$ أو $x=6$ ، وكلاهما في $D$: $\bx{S=\{0;6\}}$.
\QQ{6} \textbf{(بكالوريا 2016)} $(2x-4)\ln x=0$: \quad $D=]0;+\infty[$: جداء معدوم:
\[2x-4=0\ \Rightarrow\ x=2\qquad;\qquad\ln x=0\ \Rightarrow\ x=1\qquad;\qquad\bx{S=\{1;2\}}\]
\end{mithal}

\begin{molahada}
$\ln x+\ln(x-1)$ و $\ln\big(x(x-1)\big)$ ليستا لهما نفس مجموعة التعريف: نعيّن $D$ من المعادلة الأصلية \textbf{قبل} أي تحويل.
\end{molahada}

\section{معادلات ومتراجحات من الدرجة الثانية في $\ln x$}
\begin{tariqa}[المعادلة $a(\ln x)^2+b\ln x+c=0$]
\begin{enumerate}
\item $D=]0;+\infty[$ ، ونضع $X=\ln x$ فنحصل على $aX^2+bX+c=0$.
\item نحسب $\Delta=b^2-4ac$ ونعيّن $X_1$ و $X_2$ (إذا كان $\Delta\geqslant0$).
\item نرجع إلى $x$: $\ln x=X_1$ يعطي $x=e^{X_1}$ ، و $\ln x=X_2$ يعطي $x=e^{X_2}$.
\end{enumerate}
\end{tariqa}

\begin{mithal}[حل في $\R$]
\QQ{1} \textbf{(بكالوريا 2010)} $(\ln x)^2+2\ln x-3=0$: \quad نضع $X=\ln x$:
\[X^2+2X-3=0\ ;\ \Delta=4+12=16\ ;\ X_1=\frac{-2-4}2=-3\ ;\ X_2=\frac{-2+4}2=1\]
$\ln x=-3$ يعطي $x=e^{-3}$ ، و $\ln x=1$ يعطي $x=e$: \quad $\bx{S=\left\{e^{-3};e\right\}}$.
\QQ{2} $(\ln x)^2-3\ln x+2=0$: \quad $X^2-3X+2=0$:
\[\Delta=9-8=1\ ;\ X_1=\frac{3-1}2=1\ ;\ X_2=\frac{3+1}2=2\]
$x=e$ أو $x=e^2$: \quad $\bx{S=\left\{e;e^2\right\}}$.
\QQ{3} $2(\ln x)^2-\ln x-1=0$: \quad $2X^2-X-1=0$:
\[\Delta=1+8=9\ ;\ X_1=\frac{1-3}4=-\frac12\ ;\ X_2=\frac{1+3}4=1\]
$x=e^{-\frac12}=\dfrac1{\sqrt e}$ أو $x=e$: \quad $\bx{S=\left\{\dfrac1{\sqrt e};e\right\}}$.

\section{المشتقات ، الدالة $\ln u$}
\begin{khassia}
\begin{itemize}
\item $\LW{(\ln x)'=\dfrac1x}$ على $\LW{]0;+\infty[}$.
\item إذا كانت $\LW{u}$ قابلة للاشتقاق و\textbf{موجبة تمامًا} على مجال $\LW{I}$ فإن: $\bx{\big(\ln u\big)'=\dfrac{u'}u}$.
\item للدالتين $\LW{u}$ و $\LW{\ln u}$ نفس اتجاه التغير على $\LW{I}$.
\end{itemize}
\end{khassia}

\begin{mithal}[احسب المشتقة]
\QQ{1} $f(x)=x-1-\ln x$: \quad $f'(x)=1-\dfrac1x=\bx{\dfrac{x-1}x}$.
\QQ{2} $g(x)=x\ln x-x$: \quad $g'(x)=\ln x+x\times\dfrac1x-1=\bx{\ln x}$.
\QQ{3} $h(x)=\dfrac{\ln x}x$: \quad $h'(x)=\dfrac{\frac1x\times x-\ln x}{x^2}=\bx{\dfrac{1-\ln x}{x^2}}$.
\QQ{4} $k(x)=(\ln x)^2$: \quad $k'(x)=2\ln x\times\dfrac1x=\bx{\dfrac{2\ln x}x}$.
\QQ{5} $m(x)=\ln(x^2+1)$ على $\R$ ($x^2+1>0$): \quad $m'(x)=\bx{\dfrac{2x}{x^2+1}}$.
\QQ{6} $p(x)=\ln\left(\dfrac{x-1}{x+1}\right)$ على $]1;+\infty[$: \quad $u'(x)=\dfrac2{(x+1)^2}$:
\[p'(x)=\frac{u'(x)}{u(x)}=\frac2{(x+1)^2}\times\frac{x+1}{x-1}=\bx{\frac2{x^2-1}}\]
\end{mithal}

\begin{mithal}[اتجاه تغير $\ln u$]

\end{mithal}

\begin{mithal}[إيداع بفائدة مركبة]
